    function addMesh(parent, geometry, mat, pos, scale, rotation) {
      const mesh = new THREE.Mesh(geometry, mat);

    function setSegment(mesh, a, b) {
      const start = Array.isArray(a) ? v3(...a) : a;
      const end = Array.isArray(b) ? v3(...b) : b;
      const delta = new THREE.Vector3().subVectors(end, start);
      mesh.position.copy(start).add(end).multiplyScalar(.5);
      mesh.scale.y = delta.length();
      mesh.quaternion.setFromUnitVectors(Y_AXIS, delta.normalize());
    }

      const mesh = addMesh(parent, new THREE.TubeGeometry(curve, Math.max(24, points.length * 8), radius, 8, closed), mat);
      return { mesh, curve };
    }
    function loftX(sections, sides = 16) {
      const vertices = [], indices = [];
      for (const s of sections) {
        for (let j = 0; j < sides; j++) {
          const a = j / sides * Math.PI * 2;
          vertices.push(s.x, s.y + Math.sin(a) * s.h, Math.cos(a) * s.w);
        }
      }
      for (let i = 0; i < sections.length - 1; i++) {
        for (let j = 0; j < sides; j++) {
          const n = (j + 1) % sides;
          const a = i * sides + j, b = i * sides + n, c = (i + 1) * sides + n, d = (i + 1) * sides + j;
          indices.push(a, b, d, b, c, d);
        }
      }
      for (let j = 1; j < sides - 1; j++) indices.push(0, j + 1, j);
      const last = (sections.length - 1) * sides;
      for (let j = 1; j < sides - 1; j++) indices.push(last, last + j, last + j + 1);
      const geo = new THREE.BufferGeometry();
      geo.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));
      geo.setIndex(indices);
      geo.computeVertexNormals();
      return geo;
    }

    // Reusable two-bone IK keeps each webbed foot planted on its moving pedal.
    function solveKnee(hip, foot, l1=.79, l2=.82) {
      const delta=new THREE.Vector3().subVectors(foot,hip);
      const d=Math.min(delta.length(),l1+l2-.001);
      const dir=delta.normalize();
      const a=(l1*l1-l2*l2+d*d)/(2*d);
      const h=Math.sqrt(Math.max(0,l1*l1-a*a));
      const mid=hip.clone().addScaledVector(dir,a);
      const bend=new THREE.Vector3(-dir.y,dir.x,0).normalize();
      return mid.addScaledVector(bend,h);
    }

      if(cameraMode!=='free') {
        desiredCamera.copy(riderPos).add(tempOffset);
        const k=immediate?1:1-Math.exp(-dt*2.3);
        camera.position.lerp(desiredCamera,k); controls.target.lerp(desiredTarget,k);
      } else {
        const delta=riderPos.clone().sub(previousRiderPos);
        camera.position.add(delta); controls.target.add(delta);
      }
      controls.update();
    }

    let firstFrame=true;
    function animate() {
      requestAnimationFrame(animate);
      const dt=Math.min(clock.getDelta(),.05);
      updateRider(dt);
      updateCamera(dt,firstFrame); firstFrame=false;
      previousRiderPos.copy(riderPos);
      if(waterShader) waterShader.uniforms.uTime.value=simTime*.8;
      trees.forEach((t,i)=>{t.rotation.z=Math.sin(simTime*.8+i)*.008;});
      clouds.forEach((c,i)=>{c.position.y+=Math.sin(simTime*.35+i)*.0007;c.rotation.y=Math.sin(simTime*.08+i)*.04;});
      renderer.render(scene,camera);
    }
    updateRider(0); updateCamera(.016,true); animate();
    requestAnimationFrame(()=>document.querySelector('#loader').classList.add('hidden'));

    window.addEventListener('resize',()=>{
      camera.aspect=innerWidth/innerHeight;
      camera.fov=camera.aspect<.85?46:35;
      camera.updateProjectionMatrix();
      renderer.setPixelRatio(Math.min(window.devicePixelRatio,2)); renderer.setSize(innerWidth,innerHeight);
    });
    renderer.domElement.addEventListener('webglcontextlost',e=>{
      e.preventDefault(); document.querySelector('#loader').classList.remove('hidden');
      document.querySelector('#load-text').textContent='WEBGL PAUSED · RELOAD TO RESUME';
    });
  </script>

      }
