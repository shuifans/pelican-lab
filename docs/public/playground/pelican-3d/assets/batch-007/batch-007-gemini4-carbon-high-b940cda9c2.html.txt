    import * as THREE from 'three';
    import { OrbitControls } from 'three/addons/controls/OrbitControls.js';
    import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js';
    import { RenderPass } from 'three/addons/postprocessing/RenderPass.js';
    import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js';
    import { OutputPass } from 'three/addons/postprocessing/OutputPass.js';

    // =========================================================================
    // 1. DETERMINISTIC NOISE & PROCEDURAL DATATEXTURE GENERATORS
    //    (Strictly NO SVG and NO 2D Canvas anywhere — pure mathematical arrays)
    // =========================================================================
    function hash2(x, y) {
      const n = Math.sin(x * 127.1 + y * 311.7) * 43758.5453123;
      return n - Math.floor(n);
    }

    function valueNoise2D(x, y) {
      const xi = Math.floor(x);
      const yi = Math.floor(y);
      const xf = x - xi;
      const yf = y - yi;
      const u = xf * xf * (3 - 2 * xf);
      const v = yf * yf * (3 - 2 * yf);
      const a = hash2(xi, yi);
      const b = hash2(xi + 1, yi);
      const c = hash2(xi, yi + 1);
      const d = hash2(xi + 1, yi + 1);
      return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v;
    }

    function fbm2D(x, y, octaves = 4) {
      let val = 0;
      let amp = 0.5;
      let freq = 1;
      for (let i = 0; i < octaves; i++) {
        val += amp * valueNoise2D(x * freq, y * freq);
        freq *= 2.02;
        amp *= 0.5;
      }
      return val;
    }

    function makeDataTexture(width, height, pixelFn, isSRGB = true) {
      const data = new Uint8Array(width * height * 4);
      const rgba = [0, 0, 0, 255];
      for (let y = 0; y < height; y++) {
        const v = y / height;
        for (let x = 0; x < width; x++) {
          const u = x / width;
          pixelFn(u, v, rgba);
          const idx = (y * width + x) * 4;
          data[idx] = Math.max(0, Math.min(255, rgba[0] | 0));
          data[idx + 1] = Math.max(0, Math.min(255, rgba[1] | 0));
          data[idx + 2] = Math.max(0, Math.min(255, rgba[2] | 0));
          data[idx + 3] = Math.max(0, Math.min(255, rgba[3] | 0));
        }
      }
      const tex = new THREE.DataTexture(data, width, height, THREE.RGBAFormat);
      tex.wrapS = THREE.RepeatWrapping;
      tex.wrapT = THREE.RepeatWrapping;
      tex.colorSpace = isSRGB ? THREE.SRGBColorSpace : THREE.NoColorSpace;
      tex.magFilter = THREE.LinearFilter;
      tex.minFilter = THREE.LinearMipmapLinearFilter;
      tex.generateMipmaps = true;
      tex.needsUpdate = true;
      return tex;
    }

    // Procedural weathered cedar wood grain DataTexture
    function createWoodTextures() {
      const albedo = makeDataTexture(256, 256, (u, v, out) => {
        const warp = fbm2D(u * 4.0, v * 16.0, 4) * 0.25;
        const rings = Math.sin((v * 26.0 + warp * 14.0) * Math.PI);
        const fine = fbm2D(u * 48.0, v * 8.0, 3);
        const grain = rings * 0.5 + 0.5;
        const tone = 0.62 + grain * 0.22 + fine * 0.16;
        out[0] = 178 * tone;
        out[1] = 136 * tone;
        out[2] = 96 * tone;
        out[3] = 255;
      }, true);

      const bump = makeDataTexture(256, 256, (u, v, out) => {
        const warp = fbm2D(u * 4.0, v * 16.0, 3) * 0.25;
        const rings = Math.sin((v * 26.0 + warp * 14.0) * Math.PI) * 0.5 + 0.5;
        const fine = fbm2D(u * 60.0, v * 10.0, 2);
        const h = (rings * 0.65 + fine * 0.35) * 255;
        out[0] = h;
        out[1] = h;
        out[2] = h;
        out[3] = 255;
      }, false);

      return { albedo, bump };
    }

    // Procedural woven rattan wicker texture for the bicycle basket
    function createWickerTextures() {
      const albedo = makeDataTexture(128, 128, (u, v, out) => {
        const gx = u * 12.0;
        const gy = v * 12.0;
        const ix = Math.floor(gx);
        const iy = Math.floor(gy);
        const fx = gx - ix;
        const fy = gy - iy;
        const over = (ix + iy) % 2 === 0;
        const strand = over
          ? Math.sin(fy * Math.PI) * (0.75 + 0.25 * Math.sin(fx * Math.PI))
          : Math.sin(fx * Math.PI) * (0.75 + 0.25 * Math.sin(fy * Math.PI));
        const n = fbm2D(u * 32, v * 32, 2) * 0.12;
        const shade = 0.48 + strand * 0.52 + n;
        out[0] = 212 * shade;
        out[1] = 162 * shade;
        out[2] = 98 * shade;
        out[3] = 255;
      }, true);

      const bump = makeDataTexture(128, 128, (u, v, out) => {
        const gx = u * 12.0;
        const gy = v * 12.0;
        const ix = Math.floor(gx);
        const iy = Math.floor(gy);
        const fx = gx - ix;
        const fy = gy - iy;
        const over = (ix + iy) % 2 === 0;
        const h = (over ? Math.sin(fy * Math.PI) : Math.sin(fx * Math.PI)) * 255;
        out[0] = h;
        out[1] = h;
        out[2] = h;
        out[3] = 255;
      }, false);

      albedo.repeat.set(2, 1.5);
      bump.repeat.set(2, 1.5);
      return { albedo, bump };
    }

    // Procedural leather grain texture for vintage saddle & handlebar grips
    function createLeatherTexture() {
      const tex = makeDataTexture(128, 128, (u, v, out) => {
        const n1 = fbm2D(u * 28, v * 28, 4);
        const n2 = valueNoise2D(u * 64, v * 64);
        const shade = 0.72 + n1 * 0.22 + n2 * 0.1;
        out[0] = 152 * shade;
        out[1] = 82 * shade;
        out[2] = 40 * shade;
        out[3] = 255;
      }, true);
      tex.repeat.set(2, 2);
      return tex;
    }

    // Procedural subtle feather barbule sheen bump texture
    function createFeatherBumpTexture() {
      const tex = makeDataTexture(128, 128, (u, v, out) => {
        const barbs = Math.sin((u * 40.0 + Math.abs(v - 0.5) * 18.0) * Math.PI) * 0.5 + 0.5;
        const soft = fbm2D(u * 14.0, v * 14.0, 3);
        const h = (barbs * 0.45 + soft * 0.55) * 255;
        out[0] = h;
        out[1] = h;
        out[2] = h;
        out[3] = 255;
      }, false);
      tex.repeat.set(3, 3);
      return tex;
    }

    // Procedural sandy beach grain texture
    function createSandTexture() {
      const tex = makeDataTexture(256, 256, (u, v, out) => {
        const ripples = Math.sin((v * 20.0 + fbm2D(u * 5.0, v * 5.0, 3) * 2.5) * Math.PI) * 0.5 + 0.5;
        const speckle = valueNoise2D(u * 90.0, v * 90.0);
        const shade = 0.84 + ripples * 0.1 + speckle * 0.08;
        out[0] = 232 * shade;
        out[1] = 208 * shade;
        out[2] = 166 * shade;
        out[3] = 255;
      }, true);
      tex.repeat.set(8, 3);
      return tex;
    }

    // =========================================================================
    // 2. PROCEDURAL WEB AUDIO SYNTHESIZER (No external audio files)
    // =========================================================================
    class SoundFX {
      constructor() {
        this.ctx = null;
      }

      ensure() {
        if (!this.ctx) {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          if (AudioCtx) this.ctx = new AudioCtx();
        }
        if (this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
        return this.ctx;
      }

      ringBell() {
        const ctx = this.ensure();
        if (!ctx) return;
        const now = ctx.currentTime;
        // Classic bicycle twin-strike brass bell ("brring-brring!")
        [0, 0.07, 0.22, 0.29].forEach((offset, idx) => {
          const t = now + offset;
          const partials = [2180, 4390, 6120];
          partials.forEach((freq, pIdx) => {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(freq * (idx % 2 === 0 ? 1 : 1.015), t);
            const amp = (idx === 3 ? 0.14 : 0.09) / (pIdx + 1);
            gain.gain.setValueAtTime(0.0001, t);
            gain.gain.exponentialRampToValueAtTime(amp, t + 0.006);
            gain.gain.exponentialRampToValueAtTime(0.0001, t + (pIdx === 0 ? 0.95 : 0.42));
            osc.connect(gain).connect(ctx.destination);
            osc.start(t);
            osc.stop(t + 1.0);
          });
        });
      }

      pelicanChirp() {
        const ctx = this.ensure();
        if (!ctx) return;
        const now = ctx.currentTime;
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(340, now);
        osc.frequency.exponentialRampToValueAtTime(520, now + 0.09);
        osc.frequency.exponentialRampToValueAtTime(270, now + 0.24);
        gain.gain.setValueAtTime(0.001, now);
        gain.gain.linearRampToValueAtTime(0.11, now + 0.03);

    // 7. EXACT 3D TWO-BONE INVERSE KINEMATICS (IK) ENGINE

    const _ikDiff = new THREE.Vector3();
    const _ikDir = new THREE.Vector3();
    const _ikPole = new THREE.Vector3();
    const _ikBend = new THREE.Vector3();
    const _basisX = new THREE.Vector3();
    const _basisY = new THREE.Vector3();
    const _basisZ = new THREE.Vector3();
    const _basisMat = new THREE.Matrix4();

    /**
     * Solves 2-bone IK in a common coordinate frame:
     * Given rootPos, targetPos, polePos and bone lengths L1, L2,
     * writes the solved middle joint into outJoint and clamped end into outEnd.
     */
    function solveTwoBoneIK(rootPos, targetPos, polePos, L1, L2, outJoint, outEnd) {
      _ikDiff.subVectors(targetPos, rootPos);
      const rawDist = _ikDiff.length();
      const minDist = Math.abs(L1 - L2) + 0.004;
      const maxDist = L1 + L2 - 0.004;
      const dist = THREE.MathUtils.clamp(rawDist, minDist, maxDist);

      if (rawDist > 1e-6) {
        _ikDir.copy(_ikDiff).divideScalar(rawDist);
      } else {
        _ikDir.set(0, -1, 0);
      }

      const a = (L1 * L1 - L2 * L2 + dist * dist) / (2 * dist);
      const h = Math.sqrt(Math.max(0, L1 * L1 - a * a));

      _ikPole.subVectors(polePos, rootPos);
      _ikBend.copy(_ikPole).addScaledVector(_ikDir, -_ikPole.dot(_ikDir));
      if (_ikBend.lengthSq() < 1e-6) {
        _ikBend.set(1, 0, 0).addScaledVector(_ikDir, -_ikDir.x);
      }
      _ikBend.normalize();

      outJoint.copy(rootPos).addScaledVector(_ikDir, a).addScaledVector(_ikBend, h);
      outEnd.copy(rootPos).addScaledVector(_ikDir, dist);
    }

    /**
     * Orients a limb Object3D (whose geometry runs from (0,0,0) down along local -Y,
     * with local +X facing forwardRef) cleanly from startPos to endPos with zero twist/flip.
     */
    function orientBoneGroup(group, startPos, endPos, forwardRef) {
      _basisY.subVectors(startPos, endPos); // local +Y points from end back to start
      const len = _basisY.length();
      if (len < 1e-5) return;
      _basisY.divideScalar(len);

      // boneDown = -_basisY; right (+Z) = boneDown x forwardRef = forwardRef x _basisY
      _basisZ.crossVectors(forwardRef, _basisY);
      if (_basisZ.lengthSq() < 1e-6) {
        _basisZ.set(0, 0, 1);
      } else {
        _basisZ.normalize();
      }
      _basisX.crossVectors(_basisY, _basisZ).normalize();

      _basisMat.makeBasis(_basisX, _basisY, _basisZ);
      group.position.copy(startPos);
      group.quaternion.setFromRotationMatrix(_basisMat);
    }

    function createCurvedFender(radius, width, startAng, endAng, mat) {
      const arcSegs = 36;
      const profSegs = 8;
      const positions = [];
      const indices = [];

      for (let i = 0; i <= arcSegs; i++) {
        const u = i / arcSegs;
        const ang = THREE.MathUtils.lerp(startAng, endAng, u);
        // Gentle flare at the two ends of the mudguard
        const endTaper = Math.sin(u * Math.PI);
        const w = width * (0.86 + 0.14 * endTaper);
        const ca = Math.cos(ang);
        const sa = Math.sin(ang);

        for (let j = 0; j <= profSegs; j++) {
          const v = j / profSegs; // 0..1 across width (-w/2 .. +w/2)
          const pAng = (v - 0.5) * Math.PI * 0.88;
          const z = Math.sin(pAng) * (w * 0.5);
          const r = radius - (1 - Math.cos(pAng)) * 0.024;
          positions.push(ca * r, sa * r, z);
        }
      }

      for (let i = 0; i < arcSegs; i++) {
        for (let j = 0; j < profSegs; j++) {
          const a = i * (profSegs + 1) + j;
          const b = a + profSegs + 1;
          indices.push(a, a + 1, b);
          indices.push(a + 1, b + 1, b);
        }
      }

      const geo = new THREE.BufferGeometry();
      geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
      geo.setIndex(indices);
      geo.computeVertexNormals();

      const fenderMat = mat.clone();
      fenderMat.side = THREE.DoubleSide;
      const mesh = new THREE.Mesh(geo, fenderMat);
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      return mesh;

    // Reusable soft volumetric cone ShaderMaterial (fades smoothly along length & grazing edges!)
    function createVolumetricBeamMaterial(hexColor, initialOpacity = 0.0) {
      return new THREE.ShaderMaterial({
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
        side: THREE.DoubleSide,
        uniforms: {
          uColor: { value: new THREE.Color(hexColor) },
          uOpacity: { value: initialOpacity }
        },
        vertexShader: `
          varying float vAlong;
          varying vec3 vNormalW;
          varying vec3 vWorldPos;
          void main() {
            vAlong = uv.y; // ConeGeometry UV.y: 1 at apex, 0 at open base
            vec4 wp = modelMatrix * vec4(position, 1.0);
            vWorldPos = wp.xyz;
            vNormalW = normalize(mat3(modelMatrix) * normal);
            gl_Position = projectionMatrix * viewMatrix * wp;
          }
        `,
        fragmentShader: `
          uniform vec3 uColor;
          uniform float uOpacity;
          varying float vAlong;
          varying vec3 vNormalW;
          varying vec3 vWorldPos;
          void main() {
            vec3 viewDir = normalize(cameraPosition - vWorldPos);
            float edgeSoft = pow(abs(dot(normalize(vNormalW), viewDir)), 1.55);
            float longitudinal = pow(clamp(vAlong, 0.0, 1.0), 2.1) * smoothstep(1.0, 0.90, vAlong);
            float alpha = uOpacity * edgeSoft * longitudinal;
            gl_FragColor = vec4(uColor, alpha);
          }
        `
      });
    }

          uvs.push(t, v);

    function updateThroatPouch(jiggleY, jiggleZ, gulpSwelling) {
      const arr = pouchGeo.attributes.position.array;
      for (let i = 0; i <= POUCH_NX; i++) {
        const u = Math.max(0, i / POUCH_NX); // 0 = throat base, 1 = beak tip
        const x = THREE.MathUtils.lerp(-0.02, BEAK_LEN * 0.97, u);
        const rimY = THREE.MathUtils.lerp(-0.015, -0.022, u);
        const halfW = THREE.MathUtils.lerp(0.046, 0.004, Math.pow(u, 0.86))
          * (1 + gulpSwelling * 0.42 * Math.sin(u * Math.PI));

        // Classic deep belly profile of the pelican gular sac (deepest in rear-middle ~u=0.34)
        const bagProfile = Math.sin(Math.pow(u, 0.62) * Math.PI);
        const depth = (0.128 + gulpSwelling * 0.085 + jiggleY) * bagProfile;

        for (let j = 0; j <= POUCH_NV; j++) {
          const v = j / POUCH_NV; // 0 = left rim (-Z), 0.5 = bottom keel, 1 = right rim (+Z)
          const ang = v * Math.PI;
          const cz = -Math.cos(ang);
          const sag = Math.pow(Math.max(0, Math.sin(ang)), 0.82);
          // Slight natural fullness bulge + lateral sway jiggle
          const bulgeW = halfW + Math.sin(ang) * (0.012 + gulpSwelling * 0.026) * bagProfile;
          const swayZ = jiggleZ * sag * bagProfile;

          const idx = (i * (POUCH_NV + 1) + j) * 3;
          arr[idx] = x - sag * bagProfile * 0.024;
          arr[idx + 1] = rimY - sag * depth;
          arr[idx + 2] = cz * bulgeW + swayZ;
        }
      }
      pouchGeo.attributes.position.needsUpdate = true;
      pouchGeo.computeVertexNormals();
    }

    // Direct 3D Raycast Clicking on Pelican / Basket / Bell / Lamp
    const raycaster = new THREE.Raycaster();
    const pointerNDC = new THREE.Vector2();
    let pointerDownPos = { x: 0, y: 0 };

    renderer.domElement.addEventListener('pointerdown', (e) => {
      pointerDownPos.x = e.clientX;
      pointerDownPos.y = e.clientY;
    });

    renderer.domElement.addEventListener('pointerup', (e) => {
      const moveDist = Math.hypot(e.clientX - pointerDownPos.x, e.clientY - pointerDownPos.y);
      if (moveDist > 6) return; // Ignore orbit drags
      pointerNDC.x = (e.clientX / window.innerWidth) * 2 - 1;
      pointerNDC.y = -(e.clientY / window.innerHeight) * 2 + 1;
      raycaster.setFromCamera(pointerNDC, camera);
      const hits = raycaster.intersectObjects([cyclistRoot], true);
      for (const hit of hits) {
        let cur = hit.object;
        while (cur) {
          const tag = cur.userData && cur.userData.interact;
          if (tag === 'bell') { triggerBell(); return; }
          if (tag === 'lamp') { setHeadlamp(!state.lampOn); return; }
          if (tag === 'fish') { triggerFishToss(); return; }

    let lastTimeMs = null;
    let elapsed = 0;

    function animate(nowMs = 0) {
      requestAnimationFrame(animate);

      if (lastTimeMs === null) lastTimeMs = nowMs;
      const rawDt = (nowMs - lastTimeMs) * 0.001;
      lastTimeMs = nowMs;
      const dt = rawDt > 0 ? Math.min(rawDt, 0.05) : 1 / 60;
      elapsed += dt;

      if (initialFishQueued && elapsed > 0.35) {
        initialFishQueued = false;
        triggerFishToss();
      }

      // 1. Speed & Steering Input Integration
      if (state.keys['KeyW'] || state.keys['ArrowUp']) {
        state.targetSpeedKmh = Math.min(28, state.targetSpeedKmh + 10 * dt);
        speedSlider.value = state.targetSpeedKmh.toFixed(1);
      }
      if (state.keys['KeyS'] || state.keys['ArrowDown']) {
        state.targetSpeedKmh = Math.max(0, state.targetSpeedKmh - 12 * dt);
        speedSlider.value = state.targetSpeedKmh.toFixed(1);
      }

      let steerInput = 0;
      if (state.keys['KeyA'] || state.keys['ArrowLeft']) steerInput -= 1;
      if (state.keys['KeyD'] || state.keys['ArrowRight']) steerInput += 1;

      state.turboBoostKmh = Math.max(0, state.turboBoostKmh - 4.2 * dt);
      const desiredKmh = Math.min(36, state.targetSpeedKmh + state.turboBoostKmh);
      state.speedKmh += (desiredKmh - state.speedKmh) * Math.min(1, dt * 3.8);

      const linearSpeed = state.speedKmh / 3.6; // m/s
      const speedNorm = THREE.MathUtils.clamp(linearSpeed / 8.0, 0, 1.35);

      // Exact physical wheel & crank synchronization: v = omega * R_wheel
      const wheelAngularVel = linearSpeed / WHEEL_RADIUS;
      state.wheelAngle += wheelAngularVel * dt;
      state.crankAngle = state.wheelAngle / GEAR_RATIO;
      state.distance += linearSpeed * dt;

      // Gentle steering across the wide wooden boardwalk [-1.65m .. +1.65m]
      const autoWeave = steerInput === 0 ? Math.sin(elapsed * 0.55) * 0.035 * Math.min(1, speedNorm) : 0;
      const targetSteer = -steerInput * 0.34 + autoWeave;
      // Prevent steering off boardwalk edges
      let boundedSteer = targetSteer;
      if (state.lateralZ < -1.55 && boundedSteer > 0) boundedSteer *= 0.15;
      if (state.lateralZ > 1.55 && boundedSteer < 0) boundedSteer *= 0.15;

      state.steerAngle += (boundedSteer - state.steerAngle) * Math.min(1, dt * 6.5);
      // Note: positive steerAngle around +Y turns front wheel toward -Z (left), moving lateralZ accordingly
      const prevZ = state.lateralZ;
      state.lateralZ = THREE.MathUtils.clamp(
        state.lateralZ - Math.sin(state.steerAngle) * linearSpeed * 0.48 * dt,
        -1.68,
        1.68
      );
      const dz = state.lateralZ - prevZ;
      if (state.cameraMode === 'orbit') {
        camera.position.z += dz;
        controls.target.z += dz;
      }

      // Authentic bicycle lean banking into turns + subtle cadence counterbalance roll!
      const cadenceRoll = Math.sin(state.crankAngle) * 0.028 * Math.min(1, speedNorm);
      const turnBank = state.steerAngle * Math.min(1.1, speedNorm * 0.55);
      state.bankAngle += (turnBank + cadenceRoll - state.bankAngle) * Math.min(1, dt * 7.5);

      // Bunny-hop vertical spring-gravity physics
      if (state.hopY > 0 || state.hopVel !== 0) {
        state.hopY += state.hopVel * dt;
        state.hopVel -= 9.8 * dt;
        if (state.hopY <= 0) {
          state.hopY = 0;
          if (Math.abs(state.hopVel) > 0.6) {
            state.saddleSpringBounce = -0.028;
            state.pouchVelY -= 0.42;
          }
          state.hopVel = 0;
        }
      }

      // Subtle micro-vibration from wooden boardwalk plank joints
      const plankFreq = linearSpeed / PLANK_STEP;
      const plankBump = Math.max(0, Math.sin(state.distance * (Math.PI * 2 / PLANK_STEP))) * 0.0035 * Math.min(1, speedNorm);

      // Apply Cyclist Root Position & Roll/Pitch
      cyclistRoot.position.set(0, state.hopY + plankBump, state.lateralZ);
      bikeGroup.rotation.x = state.bankAngle;
      bikeGroup.rotation.z = THREE.MathUtils.clamp(state.hopVel * 0.045, -0.14, 0.16);
      steeringGroup.rotation.y = state.steerAngle;

      // 2. Rotate Wheels, Crankset, Self-Leveling Pedals & Roller Chain
      rearWheel.rotation.z = -state.wheelAngle;
      frontWheel.rotation.z = -state.wheelAngle;
      chainringGroup.rotation.z = -state.crankAngle;
      crankGroup.rotation.z = -state.crankAngle;
      updateChainLinks(state.crankAngle);

      // Counter-rotate pedals in world space + natural cycling "ankling" flex!
      const rightAnkling = -0.05 - 0.14 * Math.sin(state.crankAngle);
      const leftAnkling = -0.05 - 0.14 * Math.sin(state.crankAngle + Math.PI);
      rightPedalRig.pedalCarrier.rotation.z = state.crankAngle + rightAnkling;
      leftPedalRig.pedalCarrier.rotation.z = state.crankAngle - Math.PI + leftAnkling;

      // 3. Pelican Body Cadence Bounce, Hip Sway, Saddle Spring Compression & Aero Lean
      state.saddleSpringBounce += (0 - state.saddleSpringBounce) * Math.min(1, dt * 10);
      const cadenceBounce = Math.sin(state.crankAngle * 2 - 0.4) * 0.019 * Math.min(1.2, speedNorm);
      const breath = Math.sin(elapsed * 2.4) * 0.005;

      saddleGroup.position.y = 0.99 + cadenceBounce * 0.35 + state.saddleSpringBounce;
      const springScaleY = THREE.MathUtils.clamp(1 + (cadenceBounce * 0.35 + state.saddleSpringBounce) * 8, 0.68, 1.25);
      saddleSprings.forEach((sp) => { sp.scale.y = springScaleY; });

      // Torso leans forward naturally with speed (aero tuck during turbo sprint!)
      const forwardLean = -0.22 - Math.min(0.20, speedNorm * 0.14);
      torsoGroup.position.set(
        -0.29 + Math.min(0.03, speedNorm * 0.02),
        1.155 + cadenceBounce + breath + state.saddleSpringBounce * 0.8,
        Math.sin(state.crankAngle) * 0.012 * Math.min(1, speedNorm)
      );
      torsoGroup.rotation.set(
        Math.sin(state.crankAngle) * 0.055 * Math.min(1, speedNorm), // hip rock toward power leg
        Math.sin(state.crankAngle) * 0.038 * Math.min(1, speedNorm) + state.steerAngle * 0.18,
        forwardLean + Math.cos(state.crankAngle * 2) * 0.018 * Math.min(1, speedNorm)
      );

      // Tail feathers flutter lively with pedaling & wind
      tailGroup.rotation.z = 0.08 + Math.sin(state.crankAngle * 2 + 0.8) * 0.07 * Math.min(1.2, speedNorm + 0.2);

      // 4. Fish-Toss Parabolic Trajectory & Pelican Gulp Reaction!
      let targetJaw = 0.06 + Math.sin(elapsed * 2.1) * 0.015 + (state.turboBoostKmh > 0.5 ? 0.12 : 0);
      let headPitchExtra = 0;

      if (state.fishFlight) {
        const ff = state.fishFlight;
        ff.t += dt / ff.duration;
        pelicanGroup.updateMatrixWorld(true);
        mouthCatchTarget.getWorldPosition(_vMouthWorld);

        if (ff.t < 1.0) {
          _vFishPos.lerpVectors(ff.startPos, _vMouthWorld, ff.t);
          _vFishPos.y += Math.sin(ff.t * Math.PI) * 0.52;
          flyingFish.position.copy(_vFishPos);
          flyingFish.rotation.set(ff.t * 5.2, -0.3, -0.6 + ff.t * 3.4);
          // Pelican opens big hinged mandible wide & tilts head up to catch!
          targetJaw = Math.sin(Math.pow(ff.t, 0.7) * Math.PI * 0.85) * 0.58;
          headPitchExtra = 0.26 * Math.sin(ff.t * Math.PI);
        } else {
          // Caught! Gulp, throat-pouch swell, and turbo sprint!
          flyingFish.visible = false;
          state.fishFlight = null;
          sfx.gulpSound();
          state.gulpSwelling = 1.0;
          state.pouchVelY -= 0.55;
          state.turboBoostKmh = 11.5;
          state.happyChirpTimer = 0.9;
        }
      }

      if (state.happyChirpTimer > 0) {
        state.happyChirpTimer = Math.max(0, state.happyChirpTimer - dt);
        targetJaw = Math.max(targetJaw, Math.abs(Math.sin(state.happyChirpTimer * 18)) * 0.34);
        headPitchExtra += Math.sin(state.happyChirpTimer * 12) * 0.11;
      }

      state.jawOpen += (targetJaw - state.jawOpen) * Math.min(1, dt * 14);
      state.gulpSwelling = Math.max(0, state.gulpSwelling - dt * 1.45);

      // 5. Avian Head Stabilization, Gaze Steering, Blinking & Throat Pouch Soft Physics
      const idleLookYaw = linearSpeed < 0.4 ? Math.sin(elapsed * 1.1) * 0.28 : 0;
      const headBobY = Math.sin(state.crankAngle * 2 - 1.15) * 0.014 * Math.min(1.2, speedNorm);
      const headThrustX = Math.cos(state.crankAngle * 2 - 0.6) * 0.014 * Math.min(1.2, speedNorm);

      headGroup.position.set(
        -0.025 + Math.min(0.055, speedNorm * 0.035) + headThrustX,
        1.615 + headBobY + cadenceBounce * 0.45 + state.gulpSwelling * 0.03,
        Math.sin(state.crankAngle) * 0.008 * Math.min(1, speedNorm) - state.steerAngle * 0.07
      );
      headGroup.rotation.set(
        -state.steerAngle * 0.12,
        state.steerAngle * 0.55 + idleLookYaw,
        -0.06 + headPitchExtra + state.gulpSwelling * 0.18
      );
      lowerJawGroup.rotation.z = -state.jawOpen;

      // Blinking eyelids (natural periodic blink + happy squint during fish gulp!)
      const blinkCycle = elapsed % 4.3;
      const naturalBlink = blinkCycle > 4.12 ? Math.sin(((blinkCycle - 4.12) / 0.18) * Math.PI) : 0;
      const squint = Math.max(naturalBlink, state.gulpSwelling * 0.75);
      eyelids.forEach((lid) => {
        lid.rotation.z = -0.55 + squint * 0.62;
      });

      // Crest plumes fluttering in the coastal headwind
      crestPlumes.forEach((cp) => {
        cp.mesh.rotation.z = cp.baseZ + Math.sin(elapsed * (7 + cp.idx * 1.4) + cp.idx) * (0.04 + speedNorm * 0.11);
      });

      // Throat pouch spring-damper jiggle
      const pouchForceY = -cadenceBounce * 6.2 - plankBump * 18.0;
      state.pouchVelY += (pouchForceY - 58.0 * state.pouchDispY - 7.2 * state.pouchVelY) * dt;
      state.pouchDispY = THREE.MathUtils.clamp(state.pouchDispY + state.pouchVelY * dt, -0.045, 0.055);

      const pouchForceZ = -state.bankAngle * 0.35 + Math.cos(state.crankAngle) * 0.08 * speedNorm;
      state.pouchVelZ += (pouchForceZ - 48.0 * state.pouchDispZ - 6.5 * state.pouchVelZ) * dt;
      state.pouchDispZ = THREE.MathUtils.clamp(state.pouchDispZ + state.pouchVelZ * dt, -0.03, 0.03);

      updateThroatPouch(state.pouchDispY, state.pouchDispZ, state.gulpSwelling);

      if (state.bellTimer > 0) {
        state.bellTimer = Math.max(0, state.bellTimer - dt);
        bellDome.rotation.z = Math.sin(elapsed * 75) * 0.14 * (state.bellTimer / 0.55);
      } else {
        bellDome.rotation.z = 0;
      }

      // 6. Update World Matrices BEFORE Solving IK -> Zero 1-Frame Lag!
      cyclistRoot.updateMatrixWorld(true);

      // Dynamic S-Neck connecting moving chest collar to moving head!
      neckBaseAnchor.getWorldPosition(_vNeckBase);
      pelicanGroup.worldToLocal(_vNeckBase);
      _vHeadBase.copy(headGroup.position).add(new THREE.Vector3(-0.025, -0.045, 0));
      updateDynamicNeck(_vNeckBase, _vHeadBase, 1.0 - Math.min(0.28, speedNorm * 0.18));

      if (state.outfitOn) {
        neckCurve.getPoint(0.26, _vScarfPt);
        neckCurve.getTangent(0.26, _vScarfTan);
        scarfCollar.position.copy(_vScarfPt);
        scarfCollar.quaternion.setFromUnitVectors(_vUpRef, _vScarfTan);

        [scarfTailA, scarfTailB].forEach((tail, tIdx) => {
          tail.mesh.position.copy(_vScarfPt).add(new THREE.Vector3(
            -0.068,
            -0.004 + (tIdx === 0 ? 0.008 : -0.008),
            tIdx === 0 ? 0.032 : 0.056
          ));
          tail.mesh.rotation.y = (tIdx === 0 ? -0.18 : -0.34) - state.steerAngle * 0.25;
          const arr = tail.geo.attributes.position.array;
          const base = tail.basePos;
          const droop = THREE.MathUtils.lerp(0.62, 0.08, Math.min(1, speedNorm));
          for (let k = 0; k < arr.length; k += 3) {
            const bx = base[k]; // 0 .. -0.32
            const u = -bx / 0.32;
            const wave = Math.sin(elapsed * (8 + speedNorm * 10) - u * 5.2 + tIdx * 1.6) * (0.014 + speedNorm * 0.035) * u;
            arr[k] = bx * (1 - droop * 0.22);
            arr[k + 1] = base[k + 1] - u * u * droop * 0.22 + wave;
            arr[k + 2] = wave * 0.7;
          }
          tail.geo.attributes.position.needsUpdate = true;
          tail.geo.computeVertexNormals();
        });
      }

      // 7. Solve Exact Two-Bone Leg IK (Hip -> Knee -> Pedal Ankle) + Webbed Foot Alignment!
      leftHipAnchor.getWorldPosition(_vHipL);
      rightHipAnchor.getWorldPosition(_vHipR);
      leftPedalRig.ankleTarget.getWorldPosition(_vAnkleL);
      rightPedalRig.ankleTarget.getWorldPosition(_vAnkleR);

      pelicanGroup.worldToLocal(_vHipL);
      pelicanGroup.worldToLocal(_vHipR);
      pelicanGroup.worldToLocal(_vAnkleL);
      pelicanGroup.worldToLocal(_vAnkleR);

      _vPoleKneeR.set(_vHipR.x + 1.1, _vHipR.y + 0.18, 0.165);

      solveTwoBoneIK(_vHipR, _vAnkleR, _vPoleKneeR, L_THIGH, L_SHIN, _vKneeR, _vEndR);

      orientBoneGroup(leftLeg.lowerLeg, _vKneeL, _vEndL, _vForwardX);
      orientBoneGroup(rightLeg.upperLeg, _vHipR, _vKneeR, _vForwardX);
      orientBoneGroup(rightLeg.lowerLeg, _vKneeR, _vEndR, _vForwardX);

      leftPedalRig.footMount.getWorldPosition(_vFootPosL);
      pelicanGroup.worldToLocal(_vFootPosL);

      rightLeg.footGroup.position.copy(_vFootPosR);

      leftShoulderAnchor.getWorldPosition(_vShoulderL);
      rightShoulderAnchor.getWorldPosition(_vShoulderR);

      rightGripTarget.getWorldPosition(_vGripR);

      pelicanGroup.worldToLocal(_vShoulderR);
      pelicanGroup.worldToLocal(_vGripL);

      const wingFlare = state.happyChirpTimer > 0 ? Math.sin(elapsed * 22) * 0.08 : 0;

      _vPoleElbowR.set(_vShoulderR.x - 0.36, _vShoulderR.y - 0.14 + wingFlare, 0.58 + wingFlare);

      solveTwoBoneIK(_vShoulderR, _vGripR, _vPoleElbowR, L_UPPER_WING, L_FORE_WING, _vElbowR, _vWingEndR);

      orientBoneGroup(leftWing.upperGroup, _vShoulderL, _vElbowL, _vForwardX);
      orientBoneGroup(leftWing.foreGroup, _vElbowL, _vWingEndL, _vForwardX);
      orientBoneGroup(rightWing.upperGroup, _vShoulderR, _vElbowR, _vForwardX);
      orientBoneGroup(rightWing.foreGroup, _vElbowR, _vWingEndR, _vForwardX);

        const baseX = -TRACK_SPAN * 0.5 + i * PLANK_STEP;
        const wx = wrapCoord(baseX - state.distance, TRACK_SPAN);
        _boardDummy.position.set(wx, -0.038 + plankJitterY[i], 0.58);
        _boardDummy.rotation.set(0, 0, 0);
        _boardDummy.updateMatrix();
        plankInstanced.setMatrixAt(i, _boardDummy.matrix);

      plankInstanced.instanceMatrix.needsUpdate = true;

      const halfDashes = DASH_COUNT / 2;
      const dashStep = TRACK_SPAN / halfDashes;
      for (let i = 0; i < halfDashes; i++) {
        const wx = wrapCoord(-TRACK_SPAN * 0.5 + i * dashStep - state.distance, TRACK_SPAN);
        _boardDummy.position.set(wx, 0.003, -1.92);
        _boardDummy.updateMatrix();
        dashInstanced.setMatrixAt(i * 2, _boardDummy.matrix);

        _boardDummy.position.set(wx, 0.003, 2.95);
        _boardDummy.updateMatrix();
        dashInstanced.setMatrixAt(i * 2 + 1, _boardDummy.matrix);
      }
      dashInstanced.instanceMatrix.needsUpdate = true;

      for (let i = 0; i < scrollingProps.length; i++) {
        const p = scrollingProps[i];
        p.obj.position.x = wrapCoord(p.baseX - state.distance, TRACK_SPAN);
      }

      for (let i = 0; i < swayingPalms.length; i++) {
        const sp = swayingPalms[i];
        sp.crown.rotation.z = Math.sin(elapsed * 1.4 + sp.phase) * 0.055;
        sp.crown.rotation.x = Math.cos(elapsed * 1.1 + sp.phase) * 0.04;
      }

      // Ocean, Sky & Lighthouse
      skyUniforms.uTime.value = elapsed;
      oceanUniforms.uTime.value = elapsed;
      oceanUniforms.uScrollX.value = state.distance;
      beaconPivot.rotation.y = elapsed * 0.55;

      // Bobbing Sailboats & Drifting Clouds
      for (let i = 0; i < sailboats.length; i++) {
        const sb = sailboats[i];
        sb.boat.position.x = wrapCoord(sb.baseX - state.distance * 0.08 + elapsed * 0.35, 110);
        sb.boat.position.y = -0.68 + Math.sin(elapsed * 1.5 + sb.phase) * 0.08;
        sb.boat.rotation.z = Math.sin(elapsed * 1.3 + sb.phase) * 0.06;
        sb.boat.rotation.x = Math.cos(elapsed * 1.1 + sb.phase) * 0.05;
      }

      for (let i = 0; i < clouds.length; i++) {
        const cl = clouds[i];
        cl.cg.position.x = wrapCoord(cl.cg.position.x - (linearSpeed * 0.04 + cl.speed) * dt, 150);
      }

      // Flapping Coastal Seagulls
      for (let i = 0; i < seagulls.length; i++) {
        const sg = seagulls[i];
        sg.scatterBoost = Math.max(0, sg.scatterBoost - dt * 0.8);
        const spd = sg.speed * (1 + sg.scatterBoost * 1.8);
        sg.phase += spd * dt;
        const gx = sg.centerX + Math.cos(sg.phase) * sg.radius;
        const gz = sg.centerZ + Math.sin(sg.phase) * sg.radius * 0.7;
        const gy = sg.centerY + Math.sin(sg.phase * 2) * 0.45 + sg.scatterBoost * 0.8;
        sg.gull.position.set(gx, gy, gz);
        sg.gull.rotation.y = -sg.phase - Math.PI * 0.5;
        const flap = Math.sin(elapsed * (7.5 + sg.scatterBoost * 8) + i * 1.4) * 0.58;
        sg.wingL.rotation.x = flap;
        sg.wingR.rotation.x = flap;
      }

      // Wind-streak particles
      const wArr = windGeo.attributes.position.array;
      const windSpeed = linearSpeed * 1.55 + 0.5;
      for (let i = 0; i < WIND_COUNT; i++) {
        const idx = i * 3;
        wArr[idx] -= windSpeed * dt;
        if (wArr[idx] < -5.0) {
          wArr[idx] = 5.0;
          wArr[idx + 1] = 0.25 + Math.random() * 2.3;
          wArr[idx + 2] = state.lateralZ + (Math.random() - 0.5) * 4.2;
        }
      }
      windGeo.attributes.position.needsUpdate = true;
      windMat.uniforms.uOpacity.value = THREE.MathUtils.smoothstep(state.speedKmh, 8, 28) * 0.52;

      // 10. Camera Mode Tracking
      if (state.cameraMode === 'orbit') {
        controls.update();
      } else if (state.cameraMode === 'side') {
        _vCamDesired.set(-0.08, 0.82 + state.hopY * 0.5, state.lateralZ + 2.55);
        _vCamLook.set(-0.08, 0.76 + state.hopY * 0.5, state.lateralZ);
        camera.position.lerp(_vCamDesired, Math.min(1, dt * 6));
        camera.lookAt(_vCamLook);
      } else if (state.cameraMode === 'hero') {
        _vCamDesired.set(2.25, 0.68 + state.hopY * 0.4, state.lateralZ + 1.48);
        _vCamLook.set(-0.05, 1.08 + state.hopY * 0.6, state.lateralZ - 0.1);
        camera.position.lerp(_vCamDesired, Math.min(1, dt * 5.5));
        camera.lookAt(_vCamLook);
      } else if (state.cameraMode === 'pov') {
        // Rider POV right beside/above the pelican's feathered head looking past its giant orange beak!
        headGroup.getWorldPosition(_vCamDesired);
        _vCamDesired.add(new THREE.Vector3(-0.18, 0.16, 0.14));
        _vCamLook.copy(_vCamDesired).add(new THREE.Vector3(3.6, -0.38, -state.steerAngle * 1.4));
        camera.position.lerp(_vCamDesired, Math.min(1, dt * 14));
        camera.lookAt(_vCamLook);
      }

      // 11. Live Telemetry HUD Update
      const cadenceRpm = (wheelAngularVel / GEAR_RATIO) * (60 / (Math.PI * 2));
      hudSpeed.textContent = state.speedKmh.toFixed(1);
      hudRpm.textContent = Math.round(cadenceRpm);
      hudDist.textContent = Math.floor(state.distance);

      composer.render();
    }

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
      composer.setSize(window.innerWidth, window.innerHeight);
    });

    animate();
  </script>
