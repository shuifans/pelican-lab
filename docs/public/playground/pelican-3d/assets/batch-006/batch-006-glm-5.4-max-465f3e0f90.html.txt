// chain: closed path around chainring (A) and cog (B), links travel along it in sync with the crank
const chainPts = [];
(function buildChainPath() {
  const cA = new THREE.Vector2(BB.z, BB.y), cB = new THREE.Vector2(rearAxle.z, rearAxle.y), rA = 0.10, rB = 0.045;
  const D = cB.clone().sub(cA), Dh = D.clone().normalize();
  const a = Math.acos(THREE.MathUtils.clamp((rA - rB) / D.length(), -1, 1));
  const rot = (v, ang) => new THREE.Vector2(v.x * Math.cos(ang) - v.y * Math.sin(ang), v.x * Math.sin(ang) + v.y * Math.cos(ang));
  const n1 = rot(Dh, a), n2 = rot(Dh, -a);
  const wrap = (x) => Math.atan2(Math.sin(x), Math.cos(x));
  const arc = (c, r, a0, amid, a1) => {
    const s = wrap(amid - a0) >= 0 ? 1 : -1;
    let sweep = wrap(a1 - a0); if (Math.sign(sweep) !== s) sweep = s * (2 * Math.PI - Math.abs(sweep));
    const n = Math.max(2, Math.ceil(Math.abs(sweep) * r / 0.01));
    for (let i = 0; i < n; i++) { const t = a0 + sweep * i / n; chainPts.push(new THREE.Vector2(c.x + r * Math.cos(t), c.y + r * Math.sin(t))); }
  };
  const angN1 = Math.atan2(n1.y, n1.x), angN2 = Math.atan2(n2.y, n2.x), angBack = Math.atan2(-Dh.y, -Dh.x), angFwd = Math.atan2(Dh.y, Dh.x);
  arc(cA, rA, angN1, angBack, angN2);

const _f = new THREE.Vector3(), _p = new THREE.Vector3(), _k = new THREE.Vector3(), _e = new THREE.Vector3(), _w = new THREE.Vector3();
function solveIK(root, end, pole, L1, L2, upper, lower, joint) {
  _f.subVectors(end, root);
  const dist = THREE.MathUtils.clamp(_f.length(), Math.abs(L1 - L2) + 1e-3, L1 + L2 - 1e-3);
  _f.normalize();
  _e.copy(root).addScaledVector(_f, dist);
  const cosA = THREE.MathUtils.clamp((L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist), -1, 1), sinA = Math.sqrt(1 - cosA * cosA);
  _p.subVectors(pole, root).addScaledVector(_f, -_p.dot(_f));
  if (_p.lengthSq() < 1e-8) _p.set(0, 0, 1); else _p.normalize();
  _k.copy(root).addScaledVector(_f, L1 * cosA).addScaledVector(_p, L1 * sinA);
  aimBone(upper, root, _k); aimBone(lower, _k, _e);
  joint.position.copy(_k);
}

let audioCtx = null;

function tone(freq, start, dur, gain, type = 'sine') {
  const ctx = audio(), o = ctx.createOscillator(), g = ctx.createGain();
  o.type = type; o.frequency.setValueAtTime(freq, start);
  g.gain.setValueAtTime(0.0001, start);
  g.gain.exponentialRampToValueAtTime(gain, start + 0.01);
  g.gain.exponentialRampToValueAtTime(0.0001, start + dur);
  o.connect(g).connect(ctx.destination); o.start(start); o.stop(start + dur + 0.05);
}

  f.type = 'lowpass'; f.frequency.value = 900;
  g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(0.25, t + 0.05); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.6);
  o.connect(f).connect(g).connect(ctx.destination); o.start(t); o.stop(t + 0.65);
}

function animate() {

  const raw = Math.min(clock.getDelta(), 0.05), dt = state.paused ? 0 : raw, t = (state.t += dt);
  state.speed += (state.target - state.speed) * Math.min(1, raw * 1.5);
  const omegaWheel = state.speed / R, omegaCrank = omegaWheel / GEAR;
  state.crank += omegaCrank * dt;
  state.wheel += omegaWheel * dt;
  chainOffset -= omegaCrank * 0.10 * dt;
  if (state.squawk > 0) state.squawk = Math.min(1, state.squawk + raw / 0.9);
  if (state.nod > 0) state.nod = Math.min(1, state.nod + raw / 0.5);

  updateChain();
  updateRider();
  updateScenery(dt, t);
  updateCamera(raw, t);
  renderer.render(scene, camera);
}
