    function tube(points, radius, material, parent, closed = false, segments = 48) {
      const curve = new THREE.CatmullRomCurve3(points.map(p => new THREE.Vector3(...p)), closed, 'centripetal');

    function taperedTube(points, radii, material, parent, segments = 32, radial = 12) {
      const curve = new THREE.CatmullRomCurve3(points.map(p => new THREE.Vector3(...p)), false, 'centripetal');
      const frames = curve.computeFrenetFrames(segments, false);
      const positions = [];
      const indices = [];
      for (let i = 0; i <= segments; i++) {
        const t = i / segments;
        const p = curve.getPointAt(t);
        const rIndex = t * (radii.length - 1);
        const r0 = Math.floor(rIndex);
        const r1 = Math.min(r0 + 1, radii.length - 1);
        const radius = THREE.MathUtils.lerp(radii[r0], radii[r1], rIndex - r0);
        for (let j = 0; j <= radial; j++) {
          const angle = j / radial * TAU;
          const normal = frames.normals[i].clone().multiplyScalar(Math.cos(angle) * radius);
          const binormal = frames.binormals[i].clone().multiplyScalar(Math.sin(angle) * radius);
          const v = p.clone().add(normal).add(binormal);
          positions.push(v.x, v.y, v.z);
          if (i < segments && j < radial) {
            const a = i * (radial + 1) + j;
            const b = a + radial + 1;
            indices.push(a, b, a + 1, b, b + 1, a + 1);
          }
        }
      }
      const geometry = new THREE.BufferGeometry();
      geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
      geometry.setIndex(indices);
      geometry.computeVertexNormals();
      return addMesh(geometry, material, parent);
    }

    function solveLeg(leg, hip, ankle) {
      const l1 = 0.64;
      const l2 = 0.67;
      const dx = ankle.x - hip.x;
      const dy = ankle.y - hip.y;
      const rawDistance = Math.hypot(dx, dy);
      const distance = clamp(rawDistance, 0.12, l1 + l2 - 0.001);
      const ux = dx / (rawDistance || 1);
      const uy = dy / (rawDistance || 1);
      const along = (l1*l1 - l2*l2 + distance*distance) / (2*distance);
      const height = Math.sqrt(Math.max(0, l1*l1 - along*along));
      let px = -uy;
      let py = ux;
      if (px < 0) { px = -px; py = -py; }
      leg.kneePos.set(hip.x + ux*along + px*height, hip.y + uy*along + py*height, hip.z);
      updateBone(leg.upper, hip, leg.kneePos, 0.052);
      updateBone(leg.lower, leg.kneePos, ankle, 0.041);
      leg.knee.position.copy(leg.kneePos);
      leg.ankle.position.copy(ankle);
    }

    function playBell() {
      try {
        audioContext ||= new (window.AudioContext || window.webkitAudioContext)();
        if (audioContext.state === 'suspended') audioContext.resume();
        const now = audioContext.currentTime;
        [1568, 2093, 3136].forEach((frequency, i) => {
          const oscillator = audioContext.createOscillator();
          const gain = audioContext.createGain();
          oscillator.type = 'sine';
          oscillator.frequency.setValueAtTime(frequency, now);
          oscillator.frequency.exponentialRampToValueAtTime(frequency * 0.985, now + 0.8);
          gain.gain.setValueAtTime(0.0001, now);
          gain.gain.exponentialRampToValueAtTime(0.11 / (i + 1), now + 0.008);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + 1.25);
          oscillator.connect(gain).connect(audioContext.destination);
          oscillator.start(now);
          oscillator.stop(now + 1.3);
        });
      } catch (error) {
        console.warn('Audio is unavailable:', error);
      }
    }

    function updateBike(dt) {
      if (!state.paused) {
        state.speed = damp(state.speed, state.targetSpeed, 2.2, dt);
        const movement = state.speed * dt;
        state.travel += movement;
        state.wheelAngle -= movement / 0.64;
        const wheelRate = state.speed / 0.64;
        state.cadence = wheelRate * 2.15;
        state.crankAngle -= state.cadence * dt;
        state.elapsed += dt;
      } else {
        state.cadence = 0;
      }
      const bikeX = ((state.travel + 42) % 84) - 42;
      const dx = bikeX - lastBikeX;
      rig.position.x = bikeX;
      camera.position.x += dx;
      controls.target.x += dx;
      lastBikeX = bikeX;

      rearWheel.rotation.z = state.wheelAngle;
      frontWheel.rotation.z = state.wheelAngle;
      crank.rotation.z = state.crankAngle;
      const bob = Math.sin(state.crankAngle * 2) * 0.013 * clamp(state.speed / 2.4, 0, 1);

      wingGroups[0].rotation.z = -0.3 + state.bellPulse * 0.24;
      wingGroups[1].rotation.z = -0.3 - state.bellPulse * 0.24;

      for (let i = 0; i < legs.length; i++) {

      }
      const positions = windGeometry.attributes.position.array;
      const flow = state.speed * dt * 1.7;
      for (let i = 0; i < windCount; i++) {
        positions[i*3] -= flow;
        positions[i*3+1] += Math.sin(state.elapsed*1.8+i)*dt*0.025;
        if (positions[i*3] < -7) positions[i*3] = 7;
      }
      windGeometry.attributes.position.needsUpdate = true;
      wind.position.x = rig.position.x;
      windMaterial.opacity = clamp(state.speed / 5, 0, 1) * 0.42;
      sunLight.position.x = rig.position.x - 18;
      sunLight.target.position.x = rig.position.x;
      sunLight.target.position.y = 1;
    }

    function updateCamera(dt) {
      desiredTarget.set(rig.position.x, 1.65, 0);
      if (state.transition) {
        cameraGoal.copy(views[state.view]).add(new THREE.Vector3(rig.position.x, 0.025, 0));
        camera.position.lerp(cameraGoal, 1 - Math.exp(-dt * 3.4));
        controls.target.lerp(desiredTarget, 1 - Math.exp(-dt * 4));
        if (camera.position.distanceTo(cameraGoal) < 0.025) state.transition = false;
      } else {
        controls.target.x = damp(controls.target.x, rig.position.x, 8, dt);
        controls.target.y = damp(controls.target.y, 1.65, 3, dt);
        controls.target.z = damp(controls.target.z, 0, 3, dt);
      }
      controls.update();
    }

    const clock = new THREE.Clock();
    let firstFrame = true;
    function animate() {
      requestAnimationFrame(animate);
      const dt = Math.min(clock.getDelta(), 0.05);
      updateBike(dt);
      updateAtmosphere(dt);
      updateWorld(dt);
      updateCamera(dt);
      renderer.render(scene, camera);
      if (firstFrame) {
        firstFrame = false;
        window.__pelicanReady = true;
        document.getElementById('loading').classList.add('hidden');
      }
    }

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    animate();
  </script>
