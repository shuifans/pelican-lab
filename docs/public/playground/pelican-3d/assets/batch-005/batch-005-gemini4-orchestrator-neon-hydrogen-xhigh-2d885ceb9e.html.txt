    const R_WHEEL = 0.36;
    const R_TIRE_TUBE = 0.032;
    const R_TIRE_MAJOR = R_WHEEL - R_TIRE_TUBE; // 0.328 -> outer radius exactly 0.360
    const R_RIM_MAJOR = 0.292;
    const R_RIM_TUBE = 0.012;

    const RA = new THREE.Vector3(-0.58, 0.36, 0);
    const FA = new THREE.Vector3(0.62, 0.36, 0);
    const BB = new THREE.Vector3(-0.08, 0.31, 0);
    const ST = new THREE.Vector3(-0.24, 0.80, 0);
    const SEAT_DIR = new THREE.Vector3().subVectors(ST, BB).normalize();
    const SP = ST.clone().addScaledVector(SEAT_DIR, 0.13); // ~(-0.280, 0.924, 0)
    const HTB = new THREE.Vector3(0.42, 0.72, 0);
    const HTT = new THREE.Vector3(0.36, 0.90, 0);
    const HEAD_ANGLE = Math.atan2(0.06, 0.18); // ~0.32175 rad

    const FA_LOCAL = new THREE.Vector3()
      .subVectors(FA, HTB)
      .applyAxisAngle(new THREE.Vector3(0, 0, 1), -HEAD_ANGLE);

    const CRANK_LEN = 0.155;
    const CRANK_Z = 0.088;
    const PEDAL_Z = 0.132;
    const R_CHAINRING = 0.095;
    const R_COG = 0.058;
    const TEETH_FRONT = 36;
    const TEETH_REAR = 22;
    const GEAR_RATIO = TEETH_FRONT / TEETH_REAR; // 1.63636

    const LEG_L1 = 0.39;
    const LEG_L2 = 0.39;
    const WING_L1 = 0.26;
    const WING_L2 = 0.26;

    const Z_AXIS = new THREE.Vector3(0, 0, 1);

    // =========================================================================
    // 2. PROCEDURAL NOISE TEXTURES (PURE UINT8ARRAY -> THREE.DataTexture)
    // =========================================================================
    function hash2D(ix, iy, seed) {
      let n = Math.imul(ix, 374761393) + Math.imul(iy, 668265263) + Math.imul(seed, 1442695041);
      n = (n ^ (n >>> 13)) * 1274126177;
      return ((n ^ (n >>> 16)) >>> 0) / 4294967295;
    }

    function tileableValueNoise(u, v, period, seed) {
      const x = u * period;
      const y = v * period;
      const x0 = Math.floor(x);
      const y0 = Math.floor(y);
      const tx = x - x0;
      const ty = y - y0;
      const sx = tx * tx * (3 - 2 * tx);
      const sy = ty * ty * (3 - 2 * ty);
      const ix0 = ((x0 % period) + period) % period;
      const iy0 = ((y0 % period) + period) % period;
      const ix1 = (ix0 + 1) % period;
      const iy1 = (iy0 + 1) % period;

      const a = hash2D(ix0, iy0, seed);
      const b = hash2D(ix1, iy0, seed);
      const c = hash2D(ix0, iy1, seed);
      const d = hash2D(ix1, iy1, seed);

      const ab = a + (b - a) * sx;
      const cd = c + (d - c) * sx;
      return ab + (cd - ab) * sy;
    }

    function createNoiseDataTexture(size, octaves, contrast, baseVal, repeatX, repeatY, seed = 17) {
      const data = new Uint8Array(size * size * 4);
      for (let y = 0; y < size; y++) {
        const v = y / size;
        for (let x = 0; x < size; x++) {
          const u = x / size;
          let amp = 0.5;
          let sum = 0;
          let norm = 0;
          let period = 8;
          for (let o = 0; o < octaves; o++) {
            sum += amp * tileableValueNoise(u, v, period, seed + o * 31);
            norm += amp;
            amp *= 0.5;
            period *= 2;
          }
          const n = sum / norm - 0.5;
          const val = Math.max(0, Math.min(255, Math.round(baseVal + n * contrast * 255)));
          const idx = (y * size + x) * 4;
          data[idx] = val;
          data[idx + 1] = val;
          data[idx + 2] = val;
          data[idx + 3] = 255;
        }
      }
      const tex = new THREE.DataTexture(data, size, size, THREE.RGBAFormat);
      tex.wrapS = THREE.RepeatWrapping;
      tex.wrapT = THREE.RepeatWrapping;
      tex.repeat.set(repeatX, repeatY);
      tex.needsUpdate = true;
      return tex;
    }

    function createWoodPlankTexture(size = 128, repeatX = 1, repeatY = 1) {
      const data = new Uint8Array(size * size * 4);
      for (let y = 0; y < size; y++) {
        const v = y / size;
        for (let x = 0; x < size; x++) {
          const u = x / size;
          const grain = Math.sin((v * 22 + tileableValueNoise(u, v, 8, 91) * 2.4) * Math.PI * 2) * 0.5 + 0.5;
          const fine = tileableValueNoise(u, v, 32, 13);
          const shade = 0.72 + grain * 0.18 + fine * 0.10;
          const idx = (y * size + x) * 4;
          data[idx] = Math.round(182 * shade);
          data[idx + 1] = Math.round(138 * shade);
          data[idx + 2] = Math.round(98 * shade);
          data[idx + 3] = 255;
        }
      }
      const tex = new THREE.DataTexture(data, size, size, THREE.RGBAFormat);
      tex.colorSpace = THREE.SRGBColorSpace;
      tex.wrapS = THREE.RepeatWrapping;
      tex.wrapT = THREE.RepeatWrapping;
      tex.repeat.set(repeatX, repeatY);
      tex.needsUpdate = true;
      return tex;
    }

    // =========================================================================
    // 3. PROCEDURAL GEOMETRY & KINEMATICS HELPERS
    // =========================================================================
    function createTubeBetween(p1, p2, radius, material, radialSegs = 12) {
      const curve = new THREE.LineCurve3(p1.clone(), p2.clone());
      const geo = new THREE.TubeGeometry(curve, 1, radius, radialSegs, false);
      const mesh = new THREE.Mesh(geo, material);
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      return mesh;
    }

    function createCurvedTube(points, radius, material, tubularSegs = 24, radialSegs = 12) {
      const curve = new THREE.CatmullRomCurve3(points);
      const geo = new THREE.TubeGeometry(curve, tubularSegs, radius, radialSegs, false);
      const mesh = new THREE.Mesh(geo, material);
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      return mesh;
    }

    function createCollarAt(p1, p2, t, radius, length, material) {
      const pos = new THREE.Vector3().lerpVectors(p1, p2, t);
      const dir = new THREE.Vector3().subVectors(p2, p1).normalize();
      const mesh = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, length, 14), material);
      mesh.position.copy(pos);
      mesh.quaternion.setFromUnitVectors(WORLD_UP, dir);
      mesh.castShadow = true;
      return mesh;
    }

    function createGearMesh(teeth, pitchRadius, thickness, material, holeCount = 5) {
      const outerR = pitchRadius + 0.0055;
      const rootR = pitchRadius - 0.0065;
      const shape = new THREE.Shape();
      const steps = teeth * 4;
      for (let i = 0; i <= steps; i++) {
        const a = (i / steps) * Math.PI * 2;
        const phase = i % 4;
        const r = phase === 1 || phase === 2 ? outerR : rootR;
        const x = Math.cos(a) * r;
        const y = Math.sin(a) * r;
        if (i === 0) shape.moveTo(x, y);
        else shape.lineTo(x, y);
      }
      shape.closePath();

      const centerHole = new THREE.Path();
      centerHole.absarc(0, 0, pitchRadius * 0.22, 0, Math.PI * 2, true);
      shape.holes.push(centerHole);

      if (holeCount > 0 && pitchRadius > 0.065) {
        const ringR = pitchRadius * 0.56;
        const holeR = pitchRadius * 0.16;
        for (let h = 0; h < holeCount; h++) {
          const ha = (h / holeCount) * Math.PI * 2;
          const hole = new THREE.Path();
          hole.absarc(Math.cos(ha) * ringR, Math.sin(ha) * ringR, holeR, 0, Math.PI * 2, true);
          shape.holes.push(hole);
        }
      }

      const geo = new THREE.ExtrudeGeometry(shape, {
        depth: thickness,
        bevelEnabled: true,
        bevelThickness: 0.001,
        bevelSize: 0.0008,
        bevelSegments: 1,
        curveSegments: 1
      });
      geo.translate(0, 0, -thickness * 0.5);
      const mesh = new THREE.Mesh(geo, material);
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      return mesh;
    }

    // Orthonormal twist-free bone orientation: local +Y points from start -> end,
    // local +Z is orthogonalized against poleDir.
    const _boneY = new THREE.Vector3();
    const _boneX = new THREE.Vector3();
    const _boneZ = new THREE.Vector3();
    const _boneMat = new THREE.Matrix4();

    function alignBone(group, start, end, poleDir) {
      group.position.copy(start);
      _boneY.subVectors(end, start);
      const len = _boneY.length();
      if (len < 1e-5) return;
      _boneY.multiplyScalar(1 / len);

      _boneX.crossVectors(_boneY, poleDir);
      if (_boneX.lengthSq() < 1e-6) {
        _boneX.set(1, 0, 0);
        if (Math.abs(_boneY.dot(_boneX)) > 0.9) _boneX.set(0, 0, 1);
        _boneX.crossVectors(_boneY, _boneX);
      }
      _boneX.normalize();
      _boneZ.crossVectors(_boneX, _boneY).normalize();

      _boneMat.makeBasis(_boneX, _boneY, _boneZ);
      group.quaternion.setFromRotationMatrix(_boneMat);
    }

    // Exact 3D analytical 2-bone IK using law of cosines.
    const _ikDir = new THREE.Vector3();
    const _ikBend = new THREE.Vector3();

    function solveTwoBoneIK(rootPos, targetPos, L1, L2, poleDir, outMid) {
      _ikDir.subVectors(targetPos, rootPos);
      const rawDist = _ikDir.length();
      const minD = Math.abs(L1 - L2) + 0.006;
      const maxD = L1 + L2 - 0.006;
      const dist = Math.max(minD, Math.min(maxD, rawDist));

      if (rawDist < 1e-6) {
        _ikDir.set(0, -1, 0);
      } else {
        _ikDir.multiplyScalar(1 / rawDist);
      }

      const cosA = Math.max(-1, Math.min(1, (L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist)));
      const sinA = Math.sqrt(Math.max(0, 1 - cosA * cosA));

      // Project poleDir perpendicular to bone axis
      _ikBend.copy(poleDir).addScaledVector(_ikDir, -poleDir.dot(_ikDir));
      if (_ikBend.lengthSq() < 1e-6) {
        _ikBend.set(1, 0, 0);
        _ikBend.addScaledVector(_ikDir, -_ikDir.x);
      }
      _ikBend.normalize();

      outMid.copy(rootPos)
        .addScaledVector(_ikDir, L1 * cosA)
        .addScaledVector(_ikBend, L1 * sinA);
      return outMid;
    }

        }
      });

          const crossDir = (i % 4 < 2 ? 1 : -1) * 0.32;

      // Independent pedal assemblies (positioned in bikeRig space directly from crankAngle)
      function buildPedal() {
        const pg = new THREE.Group();
        const body = new THREE.Mesh(new THREE.BoxGeometry(0.092, 0.018, 0.068), steelDarkMat);
        body.castShadow = true;
        pg.add(body);
        for (const ry of [-0.010, 0.010]) {
          const rubber = new THREE.Mesh(new THREE.BoxGeometry(0.084, 0.005, 0.056), tireMat);
          rubber.position.y = ry;
          pg.add(rubber);
        }
        const reflectorMat = new THREE.MeshStandardMaterial({
          color: '#f59e0b',
          emissive: '#f59e0b',
          emissiveIntensity: 0.25,
          roughness: 0.3,
        });
        for (const rx of [-0.047, 0.047]) {
          const ref = new THREE.Mesh(new THREE.BoxGeometry(0.004, 0.010, 0.036), reflectorMat);
          ref.position.x = rx;
          pg.add(ref);
        }
        return pg;
      }

      const pedalR = buildPedal();
      const pedalL = buildPedal();
      bikeRig.add(pedalR, pedalL);

      // Closed-loop exact geometry chain path for 48 roller chain links
      const dxC = BB.x - RA.x; // +0.50
      const dyC = BB.y - RA.y; // -0.05
      const distC = Math.hypot(dxC, dyC);
      const centerAngle = Math.atan2(dyC, dxC); // -0.09967
      const tangOffset = Math.asin((R_CHAINRING - R_COG) / distC); // +0.0737

      // Angles on circles for top and bottom external tangents
      const topAngle = centerAngle + Math.PI * 0.5 - tangOffset;
      const botAngle = centerAngle - Math.PI * 0.5 + tangOffset;

      const P1T = new THREE.Vector3(BB.x + R_CHAINRING * Math.cos(topAngle), BB.y + R_CHAINRING * Math.sin(topAngle), 0.058);
      const P2T = new THREE.Vector3(RA.x + R_COG * Math.cos(topAngle), RA.y + R_COG * Math.sin(topAngle), 0.048);
      const P2B = new THREE.Vector3(RA.x + R_COG * Math.cos(botAngle), RA.y + R_COG * Math.sin(botAngle), 0.048);
      const P1B = new THREE.Vector3(BB.x + R_CHAINRING * Math.cos(botAngle), BB.y + R_CHAINRING * Math.sin(botAngle), 0.058);

      const lenTop = P1T.distanceTo(P2T);
      const lenBot = P2B.distanceTo(P1B);
      let rearArcSpan = botAngle - topAngle;
      if (rearArcSpan < 0) rearArcSpan += Math.PI * 2; // goes around left/rear side
      const lenRearArc = rearArcSpan * R_COG;
      let frontArcSpan = topAngle - botAngle;
      if (frontArcSpan < 0) frontArcSpan += Math.PI * 2; // Wait: going clockwise around front: botAngle -> topAngle goes CCW! Let's check:
      // topAngle ~ +1.40, botAngle ~ -1.60. CCW from topAngle (+1.40) to botAngle (-1.60 + 2pi = 4.68) is +3.28 rad (goes through pi = left/rear)!
      // CCW from botAngle (-1.60) to topAngle (+1.40) is +3.00 rad (goes through 0 = right/front)!
      // Moving along the chain when pedaling forward (clockwise around Z):
      // Front arc moves clockwise: topAngle -> botAngle (through 0), top run moves P2T -> P1T (toward front), rear arc moves P2B -> P2T (through pi, clockwise), bottom run moves P1B -> P2B (toward rear).
      const frontArcCW = topAngle - botAngle; // ~3.00 rad
      const lenFrontArc = frontArcCW * R_CHAINRING;
      const CHAIN_PERIMETER = lenTop + lenFrontArc + lenBot + lenRearArc;

      function sampleChainPoint(sRaw, outPos) {
        let s = ((sRaw % CHAIN_PERIMETER) + CHAIN_PERIMETER) % CHAIN_PERIMETER;
        if (s < lenTop) {
          const t = s / lenTop;
          outPos.lerpVectors(P2T, P1T, t);
          return Math.atan2(P1T.y - P2T.y, P1T.x - P2T.x);
        }
        s -= lenTop;
        if (s < lenFrontArc) {
          const a = topAngle - (s / lenFrontArc) * frontArcCW;
          outPos.set(BB.x + R_CHAINRING * Math.cos(a), BB.y + R_CHAINRING * Math.sin(a), 0.058);
          return a - Math.PI * 0.5;
        }
        s -= lenFrontArc;
        if (s < lenBot) {
          const t = s / lenBot;
          outPos.lerpVectors(P1B, P2B, t);
          return Math.atan2(P2B.y - P1B.y, P2B.x - P1B.x);
        }
        s -= lenBot;
        const rearArcCW = Math.PI * 2 - frontArcCW; // ~3.28 rad (around pi)
        const a = botAngle - (s / lenRearArc) * rearArcCW;
        outPos.set(RA.x + R_COG * Math.cos(a), RA.y + R_COG * Math.sin(a), 0.048);
        return a - Math.PI * 0.5;
      }

      const CHAIN_LINKS = 48;
      const chainLinkGeo = new THREE.BoxGeometry(0.017, 0.0085, 0.009);
      const chainInstanced = new THREE.InstancedMesh(chainLinkGeo, steelDarkMat, CHAIN_LINKS);
      chainInstanced.castShadow = true;
      bikeRig.add(chainInstanced);

      const _chainDummy = new THREE.Object3D();
      const _chainPos = new THREE.Vector3();
      function updateChain(chainPhaseDist) {
        for (let i = 0; i < CHAIN_LINKS; i++) {
          const s = (i / CHAIN_LINKS) * CHAIN_PERIMETER + chainPhaseDist;
          const ang = sampleChainPoint(s, _chainPos);
          _chainDummy.position.copy(_chainPos);
          _chainDummy.rotation.set(0, 0, ang);
          _chainDummy.scale.set(1, i % 2 === 0 ? 1.0 : 0.85, i % 2 === 0 ? 1.0 : 0.78);
          _chainDummy.updateMatrix();
          chainInstanced.setMatrixAt(i, _chainDummy.matrix);
        }
        chainInstanced.instanceMatrix.needsUpdate = true;
      }
      updateChain(0);

      return {
        bikeRig,
        frameGroup,
        steerAxisGroup,
        steerPivot,
        frontWheel,
        rearWheel,
        chainringGroup,
        crankPivot,
        pedalR,
        pedalL,
        gripTargetL,
        gripTargetR,
        bellGroup,
        bellStrikerPivot,
        headlampLensMat,
        tailLightMat,
        headSpot,
        updateChain,
      };
    }

    // =========================================================================
    // 7. PROCEDURAL COASTAL BOARDWALK ENVIRONMENT (GLSL SKY, GLSL OCEAN, PROPS)
    // =========================================================================
    function buildEnvironment() {
      // 7.1 Custom GLSL Sky Dome
      const skyUniforms = {
        uTopColor: { value: new THREE.Color('#2d86c4') },
        uMidColor: { value: new THREE.Color('#7bc9e3') },
        uHorizonColor: { value: new THREE.Color('#e6f4f1') },
        uSunColor: { value: new THREE.Color('#fff5d6') },
        uSunDir: { value: new THREE.Vector3(0.55, 0.48, -0.68).normalize() },
        uStarIntensity: { value: 0.0 },
        uTime: { value: 0.0 },
      };

      const skyMat = new THREE.ShaderMaterial({
        uniforms: skyUniforms,
        side: THREE.BackSide,
        depthWrite: false,
        vertexShader: /* glsl */ `
          varying vec3 vWorldDir;
          void main() {
            vec4 wp = modelMatrix * vec4(position, 1.0);
            vWorldDir = normalize(position);
            gl_Position = projectionMatrix * viewMatrix * wp;
          }
        `,
        fragmentShader: /* glsl */ `
          uniform vec3 uTopColor;
          uniform vec3 uMidColor;
          uniform vec3 uHorizonColor;
          uniform vec3 uSunColor;
          uniform vec3 uSunDir;
          uniform float uStarIntensity;
          uniform float uTime;
          varying vec3 vWorldDir;

          float hash3(vec3 p) {
            p = fract(p * vec3(443.897, 441.423, 437.195));
            p += dot(p, p.yzx + 19.19);
            return fract((p.x + p.y) * p.z);
          }

          void main() {
            vec3 dir = normalize(vWorldDir);
            float h = clamp(dir.y, 0.0, 1.0);
            vec3 col = mix(uHorizonColor, uMidColor, smoothstep(0.0, 0.26, h));
            col = mix(col, uTopColor, smoothstep(0.22, 0.85, h));

            float sunDot = max(dot(dir, normalize(uSunDir)), 0.0);
            col += uSunColor * pow(sunDot, 12.0) * 0.28;
            col += uSunColor * pow(sunDot, 160.0) * 0.95;
            col += uSunColor * smoothstep(0.9985, 0.9996, sunDot) * 1.4;

            if (uStarIntensity > 0.01 && dir.y > 0.06) {
              vec3 cell = floor(dir * 190.0);
              float rnd = hash3(cell);
              if (rnd > 0.992) {
                float twinkle = 0.65 + 0.35 * sin(uTime * 2.4 + rnd * 60.0);
                col += vec3(0.92, 0.96, 1.0) * uStarIntensity * twinkle * smoothstep(0.06, 0.3, dir.y);
              }
            }
            gl_FragColor = vec4(col, 1.0);
          }
        `,
      });
      const skyDome = new THREE.Mesh(new THREE.SphereGeometry(280, 32, 24), skyMat);
      scene.add(skyDome);

      // 7.2 Custom GLSL Animated Ocean
      const oceanUniforms = {
        uTime: { value: 0 },
        uDistance: { value: 0 },
        uShallowColor: { value: new THREE.Color('#27b4b8') },
        uDeepColor: { value: new THREE.Color('#0c4f78') },
        uSkyHorizon: { value: new THREE.Color('#c8eef2') },
        uSunColor: { value: new THREE.Color('#fff2cf') },
        uSunDir: { value: new THREE.Vector3(0.55, 0.48, -0.68).normalize() },
      };

      const oceanGeo = new THREE.PlaneGeometry(340, 190, 140, 80);
      oceanGeo.rotateX(-Math.PI * 0.5);
      const oceanMat = new THREE.ShaderMaterial({
        uniforms: oceanUniforms,
        vertexShader: /* glsl */ `
          uniform float uTime;
          uniform float uDistance;
          varying vec3 vWorldPos;
          varying vec3 vNormalEst;
          varying float vWaveH;

          float waveHeight(vec2 p, float t) {
            float w = 0.0;
            w += sin(p.x * 0.22 + t * 1.35 + p.y * 0.08) * 0.12;
            w += sin(p.x * 0.45 - t * 1.75 + p.y * 0.31) * 0.065;
            w += sin(p.y * 0.65 + t * 2.1) * 0.045;
            return w;
          }

          void main() {
            vec3 pos = position;
            vec2 worldXZ = vec2(pos.x + uDistance, pos.z - 98.0);
            float shoreAtten = smoothstep(0.0, 10.0, abs(pos.z + 95.0 - 95.0));
            // Plane is centered at z = -98, width z = 190 => near edge z = -3, far edge z = -193
            float zWorld = pos.z - 98.0;
            float atten = smoothstep(-4.2, -16.0, zWorld);
            float h = waveHeight(worldXZ, uTime) * atten;
            pos.y += h;

            float eps = 0.25;
            float hx = waveHeight(worldXZ + vec2(eps, 0.0), uTime) * atten;
            float hz = waveHeight(worldXZ + vec2(0.0, eps), uTime) * atten;
            vec3 n = normalize(vec3(h - hx, eps, h - hz));

            vec4 wp = modelMatrix * vec4(pos, 1.0);
            vWorldPos = wp.xyz;
            vNormalEst = n;
            vWaveH = h;
            gl_Position = projectionMatrix * viewMatrix * wp;
          }
        `,
        fragmentShader: /* glsl */ `
          uniform float uTime;
          uniform float uDistance;
          uniform vec3 uShallowColor;
          uniform vec3 uDeepColor;
          uniform vec3 uSkyHorizon;
          uniform vec3 uSunColor;
          uniform vec3 uSunDir;
          varying vec3 vWorldPos;
          varying vec3 vNormalEst;
          varying float vWaveH;

          void main() {
            vec3 N = normalize(vNormalEst);
            vec3 V = normalize(cameraPosition - vWorldPos);
            float distFromShore = max(0.0, -vWorldPos.z - 4.0);
            float depthT = smoothstep(0.0, 42.0, distFromShore);

            vec3 waterCol = mix(uShallowColor, uDeepColor, depthT);

            // Fresnel sky reflection
            float fresnel = pow(1.0 - max(dot(N, V), 0.0), 4.0);
            waterCol = mix(waterCol, uSkyHorizon, clamp(fresnel * 0.72, 0.0, 0.85));

            // Sparkling sun specular path
            vec3 H = normalize(normalize(uSunDir) + V);
            float spec = pow(max(dot(N, H), 0.0), 140.0);
            waterCol += uSunColor * spec * 1.45;

            // Animated shoreline surf foam
            float surfWave = sin(vWorldPos.x * 0.18 + uDistance * 0.18 + uTime * 1.8) * 0.6;
            float foamZone = smoothstep(2.4 + surfWave, 0.0, distFromShore);
            float foamRipple = sin(distFromShore * 3.5 - uTime * 3.2 + sin(vWorldPos.x * 0.35 + uDistance * 0.35)) * 0.5 + 0.5;
            float crestFoam = smoothstep(0.11, 0.19, vWaveH) * 0.35;
            float foam = clamp(foamZone * (0.45 + 0.55 * foamRipple) + crestFoam, 0.0, 1.0);
            waterCol = mix(waterCol, vec3(0.95, 0.98, 1.0), foam * 0.75);

            // Distance fog blend
            float fogDist = length(cameraPosition - vWorldPos);
            float fogAmt = 1.0 - exp(-pow(fogDist * 0.0046, 2.0));
            waterCol = mix(waterCol, uSkyHorizon, clamp(fogAmt, 0.0, 1.0));

            gl_FragColor = vec4(waterCol, 1.0);
          }
        `,
      });
      const oceanMesh = new THREE.Mesh(oceanGeo, oceanMat);
      oceanMesh.position.set(0, -0.34, -98);
      scene.add(oceanMesh);

      // 7.3 Road Surface, Curbs, Sandy Beach Slope & Inland Dunes
      const roadMat = new THREE.MeshStandardMaterial({
        color: '#505a62',
        roughness: 0.88,
        metalness: 0.04,
        bumpMap: roadBumpTex,
        bumpScale: 0.008,
      });
      const roadMesh = new THREE.Mesh(new THREE.PlaneGeometry(340, 3.8), roadMat);
      roadMesh.rotation.x = -Math.PI * 0.5;
      roadMesh.position.set(0, 0, 0.3);
      roadMesh.receiveShadow = true;
      scene.add(roadMesh);

      const curbMat = new THREE.MeshStandardMaterial({ color: '#dcd7c9', roughness: 0.85 });
      for (const cz of [-1.62, 2.22]) {
        const curb = new THREE.Mesh(new THREE.BoxGeometry(340, 0.09, 0.16), curbMat);
        curb.position.set(0, 0.035, cz);
        curb.receiveShadow = true;
        scene.add(curb);
      }

      const sandMat = new THREE.MeshStandardMaterial({
        color: '#e5cf9f',
        roughness: 0.94,
        metalness: 0.0,
        bumpMap: sandBumpTex,
        bumpScale: 0.015,
      });

      // Sloped beach from boardwalk down into ocean
      const beachSlope = new THREE.Mesh(new THREE.PlaneGeometry(340, 6.2), sandMat);
      beachSlope.rotation.x = -Math.PI * 0.5;
      beachSlope.rotation.order = 'YXZ';
      beachSlope.position.set(0, -0.22, -4.6);
      beachSlope.rotation.set(-Math.PI * 0.5 + 0.075, 0, 0);
      beachSlope.receiveShadow = true;
      scene.add(beachSlope);

      // Inland sand & grassy verge
      const inlandSand = new THREE.Mesh(new THREE.PlaneGeometry(340, 48), sandMat);
      inlandSand.rotation.x = -Math.PI * 0.5;
      inlandSand.position.set(0, -0.02, 26.2);
      inlandSand.receiveShadow = true;

      function buildPalmTree(height, curveX, curveZ, seed) {
        const tg = new THREE.Group();
        const segs = 9;
        let curX = 0;
        let curZ = 0;
        for (let s = 0; s < segs; s++) {
          const t0 = s / segs;
          const t1 = (s + 1) / segs;
          const nx = Math.pow(t1, 1.6) * curveX;
          const nz = Math.pow(t1, 1.6) * curveZ;
          const r0 = 0.17 * (1 - t0 * 0.52);
          const r1 = 0.17 * (1 - t1 * 0.52);
          const seg = new THREE.Mesh(new THREE.CylinderGeometry(r1, r0, height / segs * 1.08, 9), trunkMat);
          seg.position.set((curX + nx) * 0.5, (t0 + t1) * 0.5 * height, (curZ + nz) * 0.5);
          seg.rotation.z = -Math.atan2(nx - curX, height / segs);
          seg.rotation.x = Math.atan2(nz - curZ, height / segs);
          seg.castShadow = true;
          tg.add(seg);
          curX = nx;
          curZ = nz;
        }

        const crown = new THREE.Group();
        crown.position.set(curveX, height, curveZ);
        tg.add(crown);

        // Coconuts cluster
        for (let c = 0; c < 3; c++) {
          const ca = c * 2.1 + seed;
          const nut = new THREE.Mesh(new THREE.SphereGeometry(0.11, 8, 6), trunkMat);
          nut.position.set(Math.cos(ca) * 0.14, -0.06, Math.sin(ca) * 0.14);
          crown.add(nut);
        }

        // Radial arched palm fronds
        const frondCount = 9;
        for (let f = 0; f < frondCount; f++) {
          const fa = (f / frondCount) * Math.PI * 2 + seed * 0.7;
          const frondHolder = new THREE.Group();
          frondHolder.rotation.y = fa;
          crown.add(frondHolder);

          const fLen = 1.65 + (f % 3) * 0.25;
          const blade = new THREE.Mesh(
            new THREE.ConeGeometry(0.28, fLen, 4),
            f % 2 === 0 ? frondMat : frondLightMat
          );
          blade.position.set(fLen * 0.45, 0.08 - (f % 2) * 0.12, 0);
          blade.rotation.z = -Math.PI * 0.66;
          blade.scale.set(0.16, 1, 1.35);
          blade.castShadow = true;
          frondHolder.add(blade);
        }

        return { group: tg, crown };
      }

    let audioCtx = null;
    function playBicycleBellSound() {
      try {
        const AudioCtor = window.AudioContext || window.webkitAudioContext;
        if (!AudioCtor) return;
        if (!audioCtx) audioCtx = new AudioCtor();
        if (audioCtx.state === 'suspended') audioCtx.resume();

        const triggerChime = (offset) => {
          const t0 = audioCtx.currentTime + offset;
          const partials = [2180, 4395, 6540];
          partials.forEach((freq, idx) => {
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(freq, t0);
            const peak = 0.16 / (idx + 1);
            gain.gain.setValueAtTime(0.0001, t0);
            gain.gain.exponentialRampToValueAtTime(peak, t0 + 0.008);
            gain.gain.exponentialRampToValueAtTime(0.0001, t0 + (idx === 0 ? 0.65 : 0.35));
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start(t0);
            osc.stop(t0 + 0.7);
          });
        };
        triggerChime(0);
        triggerChime(0.16);
      } catch (_) {
        // Ignore audio restrictions silently
      }
    }

    // Raycast clicks on bell or pelican (distinguish click vs orbit drag)
    const raycaster = new THREE.Raycaster();
    const pointerNDC = new THREE.Vector2();
    let pointerDownPos = { x: 0, y: 0 };

    renderer.domElement.addEventListener('pointerdown', (e) => {
      pointerDownPos = { x: e.clientX, y: e.clientY };
    });

    renderer.domElement.addEventListener('pointerup', (e) => {
      const moveDist = Math.hypot(e.clientX - pointerDownPos.x, e.clientY - pointerDownPos.y);
      if (moveDist > 6) return;
      const rect = renderer.domElement.getBoundingClientRect();
      pointerNDC.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      pointerNDC.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
      raycaster.setFromCamera(pointerNDC, camera);
      const hits = raycaster.intersectObject(bike.bikeRig, true);
      if (hits.length > 0) {
        const tag = hits[0].object.userData.interactive;
        if (tag === 'bell') triggerBell(false);
        else if (tag === 'pelican') triggerBell(true);
      }
    });

    // Reusable scratch vectors for zero-GC per-frame IK
    const _hipWorld = new THREE.Vector3();
    const _shWorld = new THREE.Vector3();
    const _gripWorld = new THREE.Vector3();
    const _hipLocal = new THREE.Vector3();
    const _shLocal = new THREE.Vector3();
    const _gripLocal = new THREE.Vector3();
    const _pedalPos = new THREE.Vector3();
    const _ankleTarget = new THREE.Vector3();
    const _kneePos = new THREE.Vector3();
    const _elbowPos = new THREE.Vector3();
    const _kneePole = new THREE.Vector3();
    const _elbowPole = new THREE.Vector3();
    const _tmpVec = new THREE.Vector3();
    const _sunDirScratch = new THREE.Vector3();
    let prevLateralZ = state.lateralZ;
    let lastClock = performance.now();

    function updateMoodLighting(dt) {
      const m = MOODS[currentMoodState.target];
      const f = Math.min(1, dt * 2.2);

      env.skyUniforms.uTopColor.value.lerp(m.top, f);
      env.skyUniforms.uMidColor.value.lerp(m.mid, f);
      env.skyUniforms.uHorizonColor.value.lerp(m.horizon, f);
      env.skyUniforms.uSunColor.value.lerp(m.sunColor, f);
      env.skyUniforms.uSunDir.value.lerp(m.sunDir, f).normalize();
      env.skyUniforms.uStarIntensity.value += (m.star - env.skyUniforms.uStarIntensity.value) * f;

      env.oceanUniforms.uShallowColor.value.lerp(m.shallow, f);
      env.oceanUniforms.uDeepColor.value.lerp(m.deep, f);
      env.oceanUniforms.uSkyHorizon.value.lerp(m.horizon, f);
      env.oceanUniforms.uSunColor.value.lerp(m.sunColor, f);
      env.oceanUniforms.uSunDir.value.copy(env.skyUniforms.uSunDir.value);

      sunLight.color.lerp(m.sunColor, f);
      sunLight.intensity += (m.sunIntensity - sunLight.intensity) * f;
      _sunDirScratch.copy(env.skyUniforms.uSunDir.value).multiplyScalar(15);
      sunLight.position.copy(_sunDirScratch);

      hemiLight.color.lerp(m.hemiSky, f);
      hemiLight.groundColor.lerp(m.hemiGround, f);
      hemiLight.intensity += (m.hemiIntensity - hemiLight.intensity) * f;

      scene.fog.color.lerp(m.fog, f);
      renderer.toneMappingExposure += (m.exposure - renderer.toneMappingExposure) * f;

      currentMoodState.lampIntensity += (m.lamp - currentMoodState.lampIntensity) * f;
      currentMoodState.lanternGlow += (m.lanternGlow - currentMoodState.lanternGlow) * f;
      bike.headSpot.intensity += (m.headSpot - bike.headSpot.intensity) * f;
      bike.headlampLensMat.emissiveIntensity += (m.headLens - bike.headlampLensMat.emissiveIntensity) * f;
    }

    function solveLegIK(hipSocket, pedalGroup, legRig, side, crankAngleSide) {
      hipSocket.getWorldPosition(_hipWorld);
      _hipLocal.copy(_hipWorld);
      bike.bikeRig.worldToLocal(_hipLocal);

      // Realistic ankle pitch: toes dip gently on push stroke, level at top/bottom
      const footPitch = Math.sin(crankAngleSide) * 0.14 - 0.05;
      pedalGroup.rotation.z = footPitch;

      // Place ball of foot flush on top surface of pedal
      legRig.footGroup.position.set(
        pedalGroup.position.x + 0.006,
        pedalGroup.position.y + 0.014,
        pedalGroup.position.z
      );
      legRig.footGroup.rotation.set(0, 0, footPitch);

      // Compute exact ankle target inside bikeRig space from footAnkleLocal
      _tmpVec.copy(legRig.footAnkleLocal).applyAxisAngle(Z_AXIS, footPitch);
      _ankleTarget.copy(legRig.footGroup.position).add(_tmpVec);

      // Forward-and-slightly-outward knee pole vector keeps knees cleanly outside frame tubes
      _kneePole.set(1.0, 0.24, side * 0.22);
      solveTwoBoneIK(_hipLocal, _ankleTarget, LEG_L1, LEG_L2, _kneePole, _kneePos);

      alignBone(legRig.thighGroup, _hipLocal, _kneePos, _kneePole);
      alignBone(legRig.shinGroup, _kneePos, _ankleTarget, _kneePole);
    }

    function solveWingIK(shoulderSocket, gripTarget, wingRig, side) {
      shoulderSocket.getWorldPosition(_shWorld);
      _shLocal.copy(_shWorld);
      bike.bikeRig.worldToLocal(_shLocal);

      gripTarget.getWorldPosition(_gripWorld);
      _gripLocal.copy(_gripWorld);
      bike.bikeRig.worldToLocal(_gripLocal);

      _elbowPole.set(-0.28, -0.58, side * 0.85);
      solveTwoBoneIK(_shLocal, _gripLocal, WING_L1, WING_L2, _elbowPole, _elbowPos);

      alignBone(wingRig.upperGroup, _shLocal, _elbowPos, _elbowPole);
      alignBone(wingRig.forearmGroup, _elbowPos, _gripLocal, _elbowPole);
    }

    function animate(now) {
      requestAnimationFrame(animate);
      const dt = Math.min(0.05, (now - lastClock) * 0.001);
      lastClock = now;

      if (!state.paused) {
        state.time += dt;

        // Keyboard speed adjustment
        if (state.keys.has('w') || state.keys.has('arrowup')) {
          state.targetSpeed = Math.min(8.0, state.targetSpeed + dt * 3.0);
          speedSlider.value = state.targetSpeed.toFixed(1);
          speedValEl.textContent = `${state.targetSpeed.toFixed(1)} m/s`;
        }
        if (state.keys.has('s') || state.keys.has('arrowdown')) {
          state.targetSpeed = Math.max(0.0, state.targetSpeed - dt * 3.0);
          speedSlider.value = state.targetSpeed.toFixed(1);
          speedValEl.textContent = `${state.targetSpeed.toFixed(1)} m/s`;
        }

        // Steering input (manual overrides auto-cruise temporarily or blends)
        let targetSteer = 0;
        const turnL = state.keys.has('a') || state.keys.has('arrowleft');
        const turnR = state.keys.has('d') || state.keys.has('arrowright');
        if (turnL || turnR) {
          targetSteer = (turnL ? 1 : 0) * 0.24 + (turnR ? -1 : 0) * 0.24;
        } else if (state.autoCruise && state.speed > 0.2) {
          targetSteer = Math.sin(state.time * 0.48) * 0.055 + Math.sin(state.time * 1.15) * 0.022;
        }
        state.steerAngle += (targetSteer - state.steerAngle) * Math.min(1, dt * 6.5);

        // Speed & distance physics
        state.speed += (state.targetSpeed - state.speed) * Math.min(1, dt * 4.0);
        const dx = state.speed * dt;
        state.distance += dx;

        const dWheel = dx / R_WHEEL;
        state.wheelAngle += dWheel;

        if (state.isPedaling) {
          state.crankAngle += dWheel / GEAR_RATIO;
        } else {
          // Coasting: smoothly settle pedals to comfortable horizontal cruising stance
          const mod = ((state.crankAngle % (Math.PI * 2)) + Math.PI * 2) % (Math.PI * 2);
          const nearestRest = mod < Math.PI ? 0.15 : Math.PI + 0.15;
          const diff = nearestRest - mod;
          state.crankAngle += diff * Math.min(1, dt * 3.2);
        }

        // Lateral road position from steering angle (+steerAngle rotates front wheel left -> moves -Z)
        state.lateralZ -= Math.sin(state.steerAngle) * dx * 0.95;
        if (state.lateralZ > 0.95) {
          state.lateralZ = 0.95;
          state.steerAngle = Math.max(state.steerAngle, 0);
        } else if (state.lateralZ < -0.85) {
          state.lateralZ = -0.85;
          state.steerAngle = Math.min(state.steerAngle, 0);
        }

        // Lean dynamically into turns + subtle pedaling cadence roll
        const pedalingRoll = state.isPedaling && state.speed > 0.2
          ? Math.sin(state.crankAngle) * 0.022
          : 0;
        const turnLean = -state.steerAngle * Math.min(1, state.speed / 3.5) * 0.55;
        state.leanAngle += (turnLean + pedalingRoll - state.leanAngle) * Math.min(1, dt * 8.0);

        // Bell animation timer
        if (state.bellTimer > 0) {
          state.bellTimer = Math.max(0, state.bellTimer - dt);
          bike.bellStrikerPivot.rotation.y = Math.sin(state.time * 42) * 0.45 * (state.bellTimer / 0.72);
        } else {
          bike.bellStrikerPivot.rotation.y *= 0.85;
        }
        if (state.pelicanReactTimer > 0) {
          state.pelicanReactTimer = Math.max(0, state.pelicanReactTimer - dt);
        }

        // Blink timer
        state.blinkTimer -= dt;
        if (state.blinkTimer <= 0) {
          state.blinkProgress += dt * 7.5;
          if (state.blinkProgress >= Math.PI) {
            state.blinkProgress = 0;
            state.blinkTimer = 2.4 + Math.random() * 3.0;
          }
        }
        const blinkAmt = state.blinkProgress > 0 ? Math.sin(state.blinkProgress) : 0;
        for (const lid of pelican.eyelids) {
          lid.rotation.x = -Math.PI * 0.48 + blinkAmt * Math.PI * 0.48;
        }
      }

      // STEP 1: Update bikeRig transform & steering
      const speedFactor = Math.min(1, state.speed / 4.0);
      const roadMicroBounce = state.speed > 0.1
        ? Math.sin(state.wheelAngle * 2.0) * 0.003 * speedFactor
        : 0;
      bike.bikeRig.position.set(0, roadMicroBounce, state.lateralZ);
      bike.bikeRig.rotation.set(state.leanAngle, -state.steerAngle * 0.25, 0);
      bike.steerPivot.rotation.y = state.steerAngle;

      // STEP 2: Spin wheels, chainring, cranks, pedals, and 48 chain links
      bike.frontWheel.rotation.z = -state.wheelAngle;
      bike.rearWheel.rotation.z = -state.wheelAngle;
      bike.chainringGroup.rotation.z = -state.crankAngle;
      bike.crankPivot.rotation.z = -state.crankAngle;
      bike.updateChain(state.crankAngle * R_CHAINRING);

      const phiR = state.crankAngle;
      const phiL = state.crankAngle + Math.PI;
      bike.pedalR.position.set(
        BB.x + CRANK_LEN * Math.cos(phiR),
        BB.y - CRANK_LEN * Math.sin(phiR),
        PEDAL_Z
      );
      bike.pedalL.position.set(
        BB.x + CRANK_LEN * Math.cos(phiL),
        BB.y - CRANK_LEN * Math.sin(phiL),
        -PEDAL_Z
      );

      // STEP 3: Animate Pelican torso, tail, S-neck, head, bill, pouch & scarf
      const cadenceBounce = state.isPedaling && state.speed > 0.15
        ? Math.cos(state.crankAngle * 2.0) * 0.008 * speedFactor
        : 0;
      const hipRock = state.isPedaling && state.speed > 0.15
        ? Math.sin(state.crankAngle) * 0.048
        : 0;
      const forwardLean = -0.04 - speedFactor * 0.09;

      pelican.pelicanBodyGroup.position.y = 0.985 + cadenceBounce;
      pelican.pelicanBodyGroup.rotation.set(hipRock, 0, forwardLean);

      pelican.tailGroup.rotation.z = Math.sin(state.crankAngle * 2.0) * 0.05 - 0.06;
      pelican.tailGroup.rotation.y = Math.sin(state.time * 1.4) * 0.05;

      // Neck counter-stabilizes head & looks into turns / reacts when bell rings
      const reactEnv = state.pelicanReactTimer > 0 ? Math.sin((state.pelicanReactTimer / 1.15) * Math.PI) : 0;
      const curiousYaw = Math.sin(state.time * 0.55) * 0.12 + state.steerAngle * 0.65;
      const headBob = Math.sin(state.crankAngle * 2.0 + 0.5) * 0.025 * speedFactor;

      pelican.neckBase.rotation.z = -forwardLean * 0.4 + headBob * 0.5 + reactEnv * 0.14;
      pelican.neckMid.rotation.z = headBob * 0.6;
      pelican.neckUpper.rotation.z = -headBob * 0.5;
      pelican.headGroup.rotation.set(
        -hipRock * 0.4,
        curiousYaw + reactEnv * 0.22 * Math.sin(state.time * 9.0),
        -forwardLean * 0.55 - headBob * 0.7 + reactEnv * 0.16
      );

      // Bill opening & gular pouch spring bounce
      const jawOpen = reactEnv * 0.24 + Math.max(0, Math.sin(state.time * 0.75) - 0.88) * 0.45;
      pelican.lowerBillGroup.rotation.z = -jawOpen;

      const headY = cadenceBounce + headBob * 0.2;
      const headAccel = (headY - state.prevHeadY) / Math.max(0.008, dt);
      state.prevHeadY = headY;
      if (!state.paused) {
        const springF = -135 * state.pouchDisp - 11.5 * state.pouchVel - headAccel * 0.22 + reactEnv * 0.9;
        state.pouchVel += springF * dt;
        state.pouchDisp = Math.max(-0.025, Math.min(0.045, state.pouchDisp + state.pouchVel * dt));
      }
      pelican.pouchPivot.scale.set(1, 1 + state.pouchDisp * 6.5 + jawOpen * 1.1, 1);
      pelican.pouchPivot.rotation.z = -state.speed * 0.032 + state.pouchVel * 0.12;

      // Crest feather & striped scarf wind flutter
      pelican.crestGroup.rotation.z = -0.05 * speedFactor + Math.sin(state.time * (5.5 + state.speed * 1.4)) * 0.09;
      for (let s = 0; s < pelican.scarfTailSegments.length; s++) {
        const seg = pelican.scarfTailSegments[s];
        const wave = Math.sin(state.time * (5.5 + state.speed * 1.6) - s * 0.85);
        const waveZ = Math.cos(state.time * (4.2 + state.speed * 1.2) - s * 0.7);
        seg.rotation.z = (0.14 - speedFactor * 0.16) + wave * (0.09 + speedFactor * 0.08);
        seg.rotation.y = waveZ * (0.10 + speedFactor * 0.09);
      }

      // STEP 4: Update bikeRig world matrices before reading sockets
      bike.bikeRig.updateMatrixWorld(true);

      // STEP 5: Solve exact 2-bone IK for both legs & both wings in bikeRig space
      solveLegIK(pelican.hipSocketL, bike.pedalL, pelican.legL, -1, phiL);
      solveLegIK(pelican.hipSocketR, bike.pedalR, pelican.legR, 1, phiR);

      solveWingIK(pelican.shoulderSocketL, bike.gripTargetL, pelican.wingL, -1);
      solveWingIK(pelican.shoulderSocketR, bike.gripTargetR, pelican.wingR, 1);

      // STEP 6: Update coastal environment & mood lighting
      updateMoodLighting(dt);
      env.updateEnvironment(state.time, state.distance, currentMoodState);

      // STEP 7: Camera tracking & OrbitControls
      const deltaLatZ = state.lateralZ - prevLateralZ;
      prevLateralZ = state.lateralZ;
      controls.target.z += deltaLatZ;
      camera.position.z += deltaLatZ;

      if (cameraLerpTimer > 0) {
        cameraLerpTimer = Math.max(0, cameraLerpTimer - dt);
        const p = CAMERA_PRESETS[activePreset];
        desiredCamPos.copy(p.offset).add(bike.bikeRig.position);
        desiredCamTarget.copy(p.target).add(bike.bikeRig.position);
        const alpha = Math.min(1, dt * 6.5);
        camera.position.lerp(desiredCamPos, alpha);
        controls.target.lerp(desiredCamTarget, alpha);
      }
      controls.update();

      // Update telemetry readouts
      readoutSpeedEl.textContent = (state.speed * 3.6).toFixed(1);
      const rpm = state.isPedaling ? (state.speed / (2 * Math.PI * R_WHEEL * GEAR_RATIO)) * 60 : 0;
      readoutRpmEl.textContent = Math.round(rpm);
      readoutDistEl.textContent = Math.floor(state.distance);

      renderer.render(scene, camera);
    }

    requestAnimationFrame(animate);
  </script>
