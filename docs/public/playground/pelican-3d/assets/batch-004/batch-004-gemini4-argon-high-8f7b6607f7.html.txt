    import * as THREE from 'three';

    // ============================================================================
    // 1. Web Audio 纯程序化音效合成器（无外部音频文件）
    // ============================================================================
    class SoundFX {
      constructor() {
        this.ctx = null;
      }

      init() {
        if (!this.ctx) {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          if (AudioCtx) this.ctx = new AudioCtx();
        }
        if (this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

      playBikeBell() {
        this.init();
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        // 双击清脆黄铜车铃声 "叮-铃铃！"
        const strikes = [0, 0.07, 0.22, 0.29];
        const freqs = [2150, 3220, 4580];
        strikes.forEach((offset, idx) => {
          freqs.forEach((f, fi) => {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = fi === 0 ? 'sine' : 'triangle';
            osc.frequency.setValueAtTime(f * (1 + (idx % 2) * 0.015), now + offset);

            const peak = (0.14 / (fi + 1)) * (idx >= 2 ? 1.15 : 0.85);
            gain.gain.setValueAtTime(0.0001, now + offset);
            gain.gain.exponentialRampToValueAtTime(peak, now + offset + 0.008);
            gain.gain.exponentialRampToValueAtTime(0.0001, now + offset + 0.65);

            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(now + offset);
            osc.stop(now + offset + 0.68);
          });
        });
      }

      playFishGulp() {
        this.init();
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        // 喉囊接鱼咕咚声
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(190, now);
        osc.frequency.exponentialRampToValueAtTime(95, now + 0.11);
        osc.frequency.exponentialRampToValueAtTime(340, now + 0.26);

        gain.gain.setValueAtTime(0.001, now);
        gain.gain.linearRampToValueAtTime(0.22, now + 0.04);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.30);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.32);
      }

      playWingHop() {
        this.init();
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(140, now);
        osc.frequency.exponentialRampToValueAtTime(420, now + 0.18);
        osc.frequency.exponentialRampToValueAtTime(220, now + 0.35);

        gain.gain.setValueAtTime(0.001, now);
        gain.gain.linearRampToValueAtTime(0.18, now + 0.05);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.38);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(now);
        osc.stop(now + 0.4);
      }

    // 3. 几何辅助工具函数（纯 3D 程序化构建连杆、弧形管与羽毛）
    // ============================================================================
    const UP_AXIS = new THREE.Vector3(0, 1, 0);
    const _tmpVecA = new THREE.Vector3();
    const _tmpVecB = new THREE.Vector3();
    const _tmpQuat = new THREE.Quaternion();

    function createCylinderBetween(p1, p2, rTop, rBottom, material, radialSeg = 14) {
      const dir = new THREE.Vector3().subVectors(p2, p1);
      const len = dir.length();
      const geo = new THREE.CylinderGeometry(rTop, rBottom, len, radialSeg);
      const mesh = new THREE.Mesh(geo, material);
      mesh.position.copy(p1).add(p2).multiplyScalar(0.5);
      mesh.quaternion.setFromUnitVectors(UP_AXIS, dir.normalize());
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      return mesh;
    }

    // 将一个起点在原点、沿 +Y 方向延伸的骨骼组对齐到 pStart -> pEnd
    function alignBoneLimb(limbGroup, lengthMesh, pStart, pEnd, fixedLength = null) {
      limbGroup.position.copy(pStart);
      _tmpVecA.subVectors(pEnd, pStart);
      const dist = _tmpVecA.length();
      if (dist > 0.0001) {
        _tmpVecA.divideScalar(dist);
        _tmpQuat.setFromUnitVectors(UP_AXIS, _tmpVecA);
        limbGroup.quaternion.copy(_tmpQuat);
        if (lengthMesh && !fixedLength) {
          lengthMesh.scale.y = dist;
        }
      }
    }

    // 双骨骼解析逆运动学求解器 (Two-Bone Analytic IK Solver)
    // 输入髋/肩起点 pA、脚踝/手腕目标点 pB、大腿长 L1、小腿长 L2、弯曲朝向提示 bendHint
    // 输出精确的膝盖/手肘 3D 坐标 pJoint
    function solveTwoBoneIK(pA, pB, L1, L2, bendHint, outJoint) {
      _tmpVecA.subVectors(pB, pA);
      let d = _tmpVecA.length();
      const minD = Math.abs(L1 - L2) + 0.015;
      const maxD = L1 + L2 - 0.015;
      const clampedD = THREE.MathUtils.clamp(d, minD, maxD);

      const dir = _tmpVecA.clone().normalize();
      // 余弦定理求投影长度 a 与法向高度 h
      const a = (L1 * L1 - L2 * L2 + clampedD * clampedD) / (2 * clampedD);
      const h = Math.sqrt(Math.max(0, L1 * L1 - a * a));

      // 计算与 dir 正交的弯曲平面法向
      const proj = dir.clone().multiplyScalar(bendHint.dot(dir));
      const bendNormal = _tmpVecB.subVectors(bendHint, proj);
      if (bendNormal.lengthSq() < 0.0001) {
        bendNormal.set(1, 0, 0);
      } else {
        bendNormal.normalize();
      }

      outJoint.copy(pA)
        .addScaledVector(dir, a)
        .addScaledVector(bendNormal, h);
      return outJoint;
    }

        vec3 col = mix(horizonColor, topColor, pow(h, 0.55));
        if (dir.y < 0.0) {
          col = mix(horizonColor, bottomColor, clamp(-dir.y * 3.0, 0.0, 1.0));
        }
        float sunAmt = max(dot(dir, normalize(sunDir)), 0.0);
        col += sunColor * pow(sunAmt, 36.0) * 0.65;
        col += sunColor * smoothstep(0.9985, 0.9998, sunAmt) * 1.4;
        gl_FragColor = vec4(col, 1.0);
      }
    `;

    function createSpokedWheel() {

      wheelGroup.add(rim);

      hub.rotation.x = Math.PI / 2;

      flangeL.rotation.x = Math.PI / 2;
      flangeL.position.z = -0.08;
      wheelGroup.add(flangeL);

      const flangeR = flangeL.clone();
      flangeR.position.z = 0.08;
      wheelGroup.add(flangeR);

      // 24 根交叉编织金属辐条 + 黄铜气嘴
      const spokeCount = 24;
      const innerR = WHEEL_R - 0.16;
      for (let i = 0; i < spokeCount; i++) {
        const a1 = (i / spokeCount) * Math.PI * 2;
        const crossAngle = a1 + (i % 2 === 0 ? 0.34 : -0.34);
        const zSide = i % 2 === 0 ? 0.078 : -0.078;
        const pHub = new THREE.Vector3(Math.cos(crossAngle) * 0.105, Math.sin(crossAngle) * 0.105, zSide);
        const pRim = new THREE.Vector3(Math.cos(a1) * innerR, Math.sin(a1) * innerR, 0);
        const spoke = createCylinderBetween(pHub, pRim, 0.008, 0.008, MATS.chrome, 6);
        wheelGroup.add(spoke);
      }

      // 黄铜气门嘴

      valve.position.set(0, -(WHEEL_R - 0.21), 0);

    const clock = new THREE.Clock();

    // 预分配复用向量避免 GC

    const invBikeLeanMat = new THREE.Matrix4();

    function animate() {
      requestAnimationFrame(animate);
      const dt = Math.min(clock.getDelta(), 0.05);
      const elapsed = clock.getElapsedTime();

      // 1. 键盘加减速与转向输入处理
      if (state.keys.w) {
        state.speedMultiplier = Math.min(2.4, state.speedMultiplier + dt * 0.9);
        speedSlider.value = state.speedMultiplier.toFixed(2);
      }
      if (state.keys.s) {
        state.speedMultiplier = Math.max(0, state.speedMultiplier - dt * 0.9);
        speedSlider.value = state.speedMultiplier.toFixed(2);
      }

      if (state.keys.a) {
        state.steerTarget = 0.42;
      } else if (state.keys.d) {
        state.steerTarget = -0.42;
      } else {
        // 自然骑行微幅平衡摆动
        state.steerTarget = Math.sin(elapsed * 1.4) * 0.055 * Math.min(1, state.speedMultiplier);
      }

      state.steerCurrent = THREE.MathUtils.lerp(state.steerCurrent, state.steerTarget, dt * 6.5);

      // 横向车道位移与压弯倾角
      state.lateralZ = THREE.MathUtils.clamp(
        state.lateralZ - state.steerCurrent * state.speedMultiplier * dt * 1.6,
        -1.55,
        1.55
      );
      bikeRoot.position.z = state.lateralZ;
      bikeRoot.rotation.y = state.steerCurrent * 0.22;

      // 2. 曲柄踏频、齿轮传动比与车轮线速度同步
      const hopBoost = state.hopTimer > 0 ? 1.35 : 1.0;
      const crankOmega = state.speedMultiplier * 3.8 * hopBoost; // rad/s
      state.crankAngle = (state.crankAngle + crankOmega * dt) % (Math.PI * 2);

      const gearRatio = 2.15; // 牙盘/飞轮传动比
      const wheelOmega = crankOmega * gearRatio;
      state.wheelAngle = (state.wheelAngle + wheelOmega * dt) % (Math.PI * 2);

      // 地面向后滚动的线速度 v = wheelOmega * WHEEL_R (保证轮胎与地面 1:1 零滑移)
      const linearVelocity = wheelOmega * WHEEL_R;
      state.scrollOffset += linearVelocity * dt;

      // 更新顶部遥测面板
      const kmh = (linearVelocity * 2.2).toFixed(1);
      const rpm = Math.round((crankOmega / (Math.PI * 2)) * 60);
      hudSpeed.textContent = `${kmh} km/h`;
      hudCadence.textContent = `${rpm} RPM`;

      // 3. 旋转车轮、曲柄、牙盘、飞轮与滚动链节
      // 注意：沿 +X 前进时，绕 Z 轴顺时针旋转即 rotation.z = -angle
      rearWheel.rotation.z = -state.wheelAngle;
      frontWheel.rotation.z = -state.wheelAngle;
      crankAssembly.rotation.z = -state.crankAngle;
      chainringGroup.rotation.z = -state.crankAngle;
      rearCog.rotation.z = -state.wheelAngle;

      // 更新链条滚子动画
      const chainPhase = (state.crankAngle * 0.8) % 1;
      chainRollers.forEach((item) => {
        const half = 8;
        const isTop = item.index < half;
        const idx = item.index % half;
        let t = (idx + (isTop ? chainPhase : -chainPhase + 1)) / half;
        t = ((t % 1) + 1) % 1;
        item.mesh.position.lerpVectors(
          isTop ? pChainRearTop : pChainRearBot,
          isTop ? pChainFrontTop : pChainFrontBot,
          t
        );
      });

      // 4. 左右脚踏板空间位置（始终保持水平，随曲柄相位做轻微脚踝俯仰）
      const thetaR = state.crankAngle;
      const thetaL = state.crankAngle + Math.PI;

      rightPedalGroup.position.set(
        P_BB.x + Math.cos(thetaR) * CRANK_LEN,
        P_BB.y - Math.sin(thetaR) * CRANK_LEN,
        0.24
      );
      rightPedalGroup.rotation.z = Math.sin(thetaR) * 0.16;

      leftPedalGroup.position.set(
        P_BB.x + Math.cos(thetaL) * CRANK_LEN,
        P_BB.y - Math.sin(thetaL) * CRANK_LEN,
        -0.24
      );
      leftPedalGroup.rotation.z = Math.sin(thetaL) * 0.16;

      // 5. 前叉与车把转向 + 车身踩踏摇摆 + 振翅弹跳
      steeringGroup.rotation.y = state.steerCurrent;

      let hopHeight = 0;
      let hopPitch = 0;
      if (state.hopTimer > 0) {
        state.hopTimer = Math.max(0, state.hopTimer - dt * 1.9);
        const s = Math.sin(state.hopTimer * Math.PI);
        hopHeight = s * 0.55;
        hopPitch = Math.sin(state.hopTimer * Math.PI * 2) * 0.12;
      }

      const pedalSway = Math.sin(state.crankAngle) * 0.032 * Math.min(1.4, state.speedMultiplier);
      const turnLean = -state.steerCurrent * 0.35 * Math.min(1.5, state.speedMultiplier);
      bikeLeanGroup.position.y = hopHeight;
      bikeLeanGroup.rotation.x = turnLean + pedalSway;
      bikeLeanGroup.rotation.z = hopPitch;

      // 6. 鹈鹕躯干、尾羽、长脖子、头部与大喉囊物理协同运动
      // 每半圈踩踏（左右脚各发力一次）身体上下律动一次 (2 * crankAngle)
      const cadenceBounce = Math.sin(state.crankAngle * 2) * 0.065 * Math.min(1.5, state.speedMultiplier);
      const hipRock = Math.cos(state.crankAngle) * 0.055;

      torsoGroup.position.y = cadenceBounce;
      torsoGroup.rotation.x = hipRock;
      torsoGroup.rotation.z = -0.05 * state.speedMultiplier + Math.cos(state.crankAngle * 2) * 0.035;

      // 尾羽随踩踏节拍欢快上下翘动
      tailGroup.rotation.z = Math.sin(state.crankAngle * 2 - 0.6) * 0.18;
      tailGroup.rotation.y = Math.cos(state.crankAngle) * 0.14;

      // S 形脖颈与头部前倾平衡 & 转弯注视
      neckSeg1.rotation.z = Math.sin(state.crankAngle * 2 - 0.4) * 0.06;
      neckSeg2.rotation.z = -Math.sin(state.crankAngle * 2) * 0.07;
      headGroup.rotation.y = state.steerCurrent * 0.65;
      headGroup.rotation.z = Math.sin(state.crankAngle * 2 + 0.5) * 0.05 + 0.04 * state.speedMultiplier;

      // 羽冠与红领巾随骑行海风飘扬
      crestGroup.rotation.z = -0.12 * state.speedMultiplier + Math.sin(elapsed * 9) * 0.08;
      const windSpeed = 3.5 + state.speedMultiplier * 7.5;
      scarfTailSegments.forEach((seg, idx) => {
        seg.rotation.y = Math.sin(elapsed * windSpeed - idx * 0.7) * (0.14 + idx * 0.04);
        seg.rotation.z = (0.12 * state.speedMultiplier) + Math.cos(elapsed * (windSpeed * 0.8) - idx * 0.6) * 0.1;
      });

      // 嘴巴张合与大喉囊弹性物理果冻抖动
      state.mouthOpen = Math.max(0, state.mouthOpen - dt * 1.4);
      state.pouchInflate = Math.max(0, state.pouchInflate - dt * 0.85);
      state.pouchJigglePhase += dt * (8 + state.speedMultiplier * 5);

      const jawAngle = state.mouthOpen * 0.46 + Math.max(0, Math.sin(state.crankAngle * 2)) * 0.04;
      upperBeakGroup.rotation.z = jawAngle * 0.45;
      lowerJawGroup.rotation.z = -jawAngle;

      const pouchBounce = Math.sin(state.crankAngle * 2 - 1.1) * 0.09 * Math.min(1.5, state.speedMultiplier);
      const gulpSwell = 1 + state.pouchInflate * 0.55 + Math.sin(state.pouchJigglePhase * 2.2) * state.pouchInflate * 0.18;
      gularPouchGroup.scale.set(
        1 + pouchBounce * 0.4 + state.pouchInflate * 0.25,
        gulpSwell - pouchBounce * 0.7,
        gulpSwell
      );
      gularPouchGroup.rotation.x = Math.cos(state.crankAngle) * 0.12;

      // 自然周期眨眼
      const blinkWave = Math.sin(elapsed * 1.7) > 0.97;
      eyelids.forEach((lid) => {
        lid.visible = blinkWave;
        if (blinkWave) lid.scale.y = 1.0;
      });

      // 7. 双腿与双翅解析逆运动学 (Two-Bone IK Solving in bikeLeanGroup Space)
      bikeLeanGroup.updateMatrixWorld(true);
      invBikeLeanMat.copy(bikeLeanGroup.matrixWorld).invert();

      // 计算髋关节在 bikeLeanGroup 局部坐标系中的位置
      pHipR.set(0.08, 0.12, 0.22).applyMatrix4(torsoGroup.matrixWorld).applyMatrix4(invBikeLeanMat);
      pHipL.set(0.08, 0.12, -0.22).applyMatrix4(torsoGroup.matrixWorld).applyMatrix4(invBikeLeanMat);

      // 脚踝目标点位于踏板上方
      pAnkleR.copy(rightPedalGroup.position).add(new THREE.Vector3(-0.04, 0.065, 0));
      pAnkleL.copy(leftPedalGroup.position).add(new THREE.Vector3(-0.04, 0.065, 0));

      // 求解左右膝关节 3D 位置并对齐大腿、膝盖、小腿、脚蹼
      solveTwoBoneIK(pHipR, pAnkleR, LEG_L1, LEG_L2, bendLegR, pKneeR);
      solveTwoBoneIK(pHipL, pAnkleL, LEG_L1, LEG_L2, bendLegL, pKneeL);

      alignBoneLimb(rightLeg.thighGroup, null, pHipR, pKneeR, true);
      rightLeg.kneeSphere.position.copy(pKneeR);
      alignBoneLimb(rightLeg.shinGroup, null, pKneeR, pAnkleR, true);
      rightLeg.footGroup.position.copy(pAnkleR);
      rightLeg.footGroup.rotation.z = rightPedalGroup.rotation.z;

      alignBoneLimb(leftLeg.thighGroup, null, pHipL, pKneeL, true);
      leftLeg.kneeSphere.position.copy(pKneeL);
      alignBoneLimb(leftLeg.shinGroup, null, pKneeL, pAnkleL, true);
      leftLeg.footGroup.position.copy(pAnkleL);
      leftLeg.footGroup.rotation.z = leftPedalGroup.rotation.z;

      // 计算左右肩关节与车把握把在 bikeLeanGroup 空间中的位置，求解翅膀双骨骼 IK
      pShoulderR.set(0.32, 0.58, 0.36).applyMatrix4(torsoGroup.matrixWorld).applyMatrix4(invBikeLeanMat);
      pShoulderL.set(0.32, 0.58, -0.36).applyMatrix4(torsoGroup.matrixWorld).applyMatrix4(invBikeLeanMat);

      gripRightAnchor.getWorldPosition(pGripR);
      pGripR.applyMatrix4(invBikeLeanMat);
      gripLeftAnchor.getWorldPosition(pGripL);
      pGripL.applyMatrix4(invBikeLeanMat);

      solveTwoBoneIK(pShoulderR, pGripR, WING_L1, WING_L2, bendWingR, pElbowR);
      solveTwoBoneIK(pShoulderL, pGripL, WING_L1, WING_L2, bendWingL, pElbowL);

      alignBoneLimb(rightWing.upperWingGroup, null, pShoulderR, pElbowR, true);
      rightWing.elbowSphere.position.copy(pElbowR);
      alignBoneLimb(rightWing.foreWingGroup, null, pElbowR, pGripR, true);
      rightWing.wingHand.position.copy(pGripR);

      alignBoneLimb(leftWing.upperWingGroup, null, pShoulderL, pElbowL, true);
      leftWing.elbowSphere.position.copy(pElbowL);
      alignBoneLimb(leftWing.foreWingGroup, null, pElbowL, pGripL, true);
      leftWing.wingHand.position.copy(pGripL);

      // 更新 IK 骨骼可视化层
      if (ikDebugGroup.visible) {
        debugLegR.update(pHipR, pKneeR, pAnkleR);
        debugLegL.update(pHipL, pKneeL, pAnkleL);
        debugWingR.update(pShoulderR, pElbowR, pGripR);
        debugWingL.update(pShoulderL, pElbowL, pGripL);
      }

      // 8. 车铃声波扩散环与抛物线投喂小鱼动画
      if (state.bellWaveTimer > 0) {
        state.bellWaveTimer = Math.max(0, state.bellWaveTimer - dt * 1.8);
        const s = 1 + (1 - state.bellWaveTimer) * 4.5;
        bellWaveRing.scale.setScalar(s);
        bellRingMat.opacity = state.bellWaveTimer * 0.85;
        bellWaveRing.lookAt(camera.position);
      } else {
        bellRingMat.opacity = 0;
      }

      if (flyingFishes.length > 0) {
        const mouthWorld = new THREE.Vector3();
        gularPouchGroup.getWorldPosition(mouthWorld);
        mouthWorld.add(new THREE.Vector3(0.32, 0.16, 0));

        for (let i = flyingFishes.length - 1; i >= 0; i--) {
          const f = flyingFishes[i];
          f.t += dt * 1.65;
          state.mouthOpen = Math.max(state.mouthOpen, 0.9);
          if (f.t >= 1.0) {
            scene.remove(f.mesh);
            flyingFishes.splice(i, 1);
            state.pouchInflate = 1.0;
            state.fishCount++;
            hudFish.textContent = `${state.fishCount} 条`;
            sfx.playFishGulp();
          } else {
            f.mesh.position.lerpVectors(f.start, mouthWorld, f.t);
            f.mesh.position.y += Math.sin(f.t * Math.PI) * 1.95;
            f.mesh.rotation.z = f.t * Math.PI * 3;
            f.mesh.rotation.y = Math.PI;
          }
        }
      }

      // 9. 环境滚动、海浪顶点动画、灯塔光束旋转与海鸥盘旋
      scrollItems.forEach((obj) => {
        obj.position.x -= linearVelocity * dt;
        if (obj.position.x < -SCROLL_SPAN * 0.5) {
          obj.position.x += SCROLL_SPAN;
        }
      });

      const oceanPos = oceanGeo.attributes.position;
      for (let i = 0; i < oceanPos.count; i++) {
        const ox = oceanPos.getX(i);
        const oz = oceanPos.getZ(i);
        const wave =
          Math.sin(ox * 0.22 + elapsed * 2.1) * 0.18 +
          Math.cos(oz * 0.28 + elapsed * 1.7) * 0.14;
        oceanPos.setY(i, oceanBaseY[i] + wave);
      }
      oceanPos.needsUpdate = true;

      beaconPivot.rotation.y = elapsed * 0.85;

      clouds.forEach((c, idx) => {
        c.position.x -= (0.25 + idx * 0.04) * dt;
        if (c.position.x < -45) c.position.x = 45;
      });

      seagulls.forEach((g, idx) => {
        const a = elapsed * 0.45 + g.phase;
        g.group.position.set(
          Math.cos(a) * g.radius - 6,
          10 + Math.sin(elapsed * 0.8 + idx) * 1.2,
          Math.sin(a) * g.radius - 16
        );
        g.group.rotation.y = -a;
        const flap = Math.sin(elapsed * 6.5 + g.phase) * 0.42;
        g.wingL.rotation.x = flap;
        g.wingR.rotation.x = -flap;
      });

      // 10. 摄像机机位平滑追踪
      if (state.camMode === 'free') {
        controls.target.lerp(new THREE.Vector3(0.1, 2.25 + hopHeight * 0.5, state.lateralZ * 0.6), 0.1);
        controls.update();
      } else if (state.camMode === 'side') {
        const targetPos = new THREE.Vector3(0.0, 2.05, state.lateralZ + 5.6);
        const lookAtPos = new THREE.Vector3(0.0, 1.95 + hopHeight * 0.4, state.lateralZ);
        camera.position.lerp(targetPos, 0.08);
        camera.lookAt(lookAtPos);
      } else if (state.camMode === 'front') {
        const targetPos = new THREE.Vector3(4.8, 1.85, state.lateralZ + 2.2);
        const lookAtPos = new THREE.Vector3(0.2, 2.45 + hopHeight * 0.4, state.lateralZ);
        camera.position.lerp(targetPos, 0.08);
        camera.lookAt(lookAtPos);
      } else if (state.camMode === 'pov') {
        const headWorld = new THREE.Vector3();
        headGroup.getWorldPosition(headWorld);
        const targetPos = headWorld.clone().add(new THREE.Vector3(-0.65, 0.42, 0));
        const lookAtPos = headWorld.clone().add(new THREE.Vector3(4.5, -0.35, -state.steerCurrent * 1.8));
        camera.position.lerp(targetPos, 0.14);
        camera.lookAt(lookAtPos);
      }

      renderer.render(scene, camera);
    }

    animate();
  </script>
