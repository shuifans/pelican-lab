    import * as THREE from 'three';
    import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

    // =========================================================================
    // 1. CORE SCENE, RENDERER, CAMERA & PROCEDURAL REFLECTION ENVIRONMENT
    // =========================================================================
    const container = document.getElementById('viewport');
    const scene = new THREE.Scene();
    scene.fog = new THREE.FogExp2(0xe6b995, 0.0082);

    const renderer = new THREE.WebGLRenderer({
      antialias: true,
      powerPreference: 'high-performance'
    });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.08;
    renderer.outputColorSpace = THREE.SRGBColorSpace;
    container.appendChild(renderer.domElement);

    const camera = new THREE.PerspectiveCamera(42, window.innerWidth / window.innerHeight, 0.08, 320);
    camera.position.set(2.45, 1.52, 2.75);

    const controls = new OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;
    controls.dampingFactor = 0.065;
    controls.target.set(0.02, 0.88, 0.0);
    controls.minDistance = 0.85;
    controls.maxDistance = 13.5;
    controls.maxPolarAngle = Math.PI * 0.492;
    controls.minPolarAngle = 0.08;

    // Procedural PMREM environment map (WebGL-only, zero 2D canvas, zero external HDRI)
    const pmremGenerator = new THREE.PMREMGenerator(renderer);
    pmremGenerator.compileEquirectangularShader();

    function buildProceduralEnvMap(topColor, midColor, bottomColor, sunColor, sunPos) {
      const envScene = new THREE.Scene();
      const envSkyMat = new THREE.ShaderMaterial({
        side: THREE.BackSide,
        uniforms: {
          uTop: { value: new THREE.Color(topColor) },
          uMid: { value: new THREE.Color(midColor) },
          uBot: { value: new THREE.Color(bottomColor) },
          uSunCol: { value: new THREE.Color(sunColor) },
          uSunDir: { value: sunPos.clone().normalize() }
        },
        vertexShader: `
          varying vec3 vDir;
          void main() {
            vDir = normalize(position);
            gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
          }
        `,
        fragmentShader: `
          varying vec3 vDir;
          uniform vec3 uTop, uMid, uBot, uSunCol, uSunDir;
          void main() {
            float h = vDir.y;
            vec3 col = h >= 0.0
              ? mix(uMid, uTop, pow(clamp(h, 0.0, 1.0), 0.65))
              : mix(uMid, uBot, pow(clamp(-h, 0.0, 1.0), 0.5));
            float s = max(dot(normalize(vDir), uSunDir), 0.0);
            col += uSunCol * (pow(s, 36.0) * 1.8 + pow(s, 6.0) * 0.35);
            gl_FragColor = vec4(col, 1.0);
          }
        `
      });
      const envSphere = new THREE.Mesh(new THREE.SphereGeometry(20, 24, 16), envSkyMat);
      envScene.add(envSphere);
      const rt = pmremGenerator.fromScene(envScene, 0.03);
      envSphere.geometry.dispose();
      envSkyMat.dispose();
      return rt;
    }

    let currentEnvRT = null;

    // =========================================================================
    // 2. PURE MATH PROCEDURAL DataTexture GENERATORS (ZERO 2D CANVAS, ZERO SVG)
    // =========================================================================
    function hash2(x, y) {
      const n = Math.sin(x * 127.1 + y * 311.7) * 43758.5453123;
      return n - Math.floor(n);
    }

    function valueNoise(x, y) {
      const xi = Math.floor(x);
      const yi = Math.floor(y);
      const xf = x - xi;
      const yf = y - yi;
      const u = xf * xf * (3 - 2 * xf);
      const v = yf * yf * (3 - 2 * yf);
      const a = hash2(xi, yi);
      const b = hash2(xi + 1, yi);
      const c = hash2(xi, yi + 1);
      const d = hash2(xi + 1, yi + 1);
      return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v;
    }

    function fbm(x, y, octaves = 4) {
      let val = 0;
      let amp = 0.5;
      let freq = 1;
      for (let i = 0; i < octaves; i++) {
        val += amp * valueNoise(x * freq, y * freq);
        freq *= 2.02;
        amp *= 0.5;
      }
      return val;
    }

    // Road coastal bike-path texture + matching bump map
    function createRoadTextures() {
      const w = 512;
      const h = 512;
      const colorData = new Uint8Array(w * h * 4);
      const bumpData = new Uint8Array(w * h * 4);

      for (let y = 0; y < h; y++) {
        const v = y / h; // across road width (0 = seawall side, 1 = inland side)
        for (let x = 0; x < w; x++) {
          const u = x / w; // along road length (tiles seamlessly)
          const idx = (y * w + x) * 4;

          // Seamless noise along u using periodic sampling
          const angle = u * Math.PI * 2;
          const n1 = fbm(Math.cos(angle) * 2.5 + 10, Math.sin(angle) * 2.5 + v * 16, 4);
          const n2 = hash2(x * 0.7, y * 0.7);
          const grain = (n1 - 0.5) * 24 + (n2 - 0.5) * 14;

          // Warm sunlit coastal promenade tarmac with light stone aggregate
          let r = 82 + grain;
          let g = 86 + grain * 0.96;
          let b = 93 + grain * 0.92;
          let bump = 128 + grain * 2.4;

          // Sandstone stone-paver promenade borders along outer edges
          if (v < 0.11 || v > 0.89) {
            const tileU = (u * 12) % 1;
            const tileV = ((v < 0.11 ? v : 1 - v) / 0.11 * 2) % 1;
            const seam = (tileU < 0.04 || tileU > 0.96 || tileV < 0.08 || tileV > 0.92) ? 1 : 0;
            const paverVar = (hash2(Math.floor(u * 12), Math.floor(v * 20)) - 0.5) * 18;
            r = seam ? 128 : 206 + paverVar + grain * 0.45;
            g = seam ? 121 : 193 + paverVar + grain * 0.45;
            b = seam ? 110 : 172 + paverVar + grain * 0.45;
            bump = seam ? 60 : 175 + grain;
          } else {
            // Subtle teal-tinted coastal bike-lane stripe near curb
            if (Math.abs(v - 0.145) < 0.011 || Math.abs(v - 0.855) < 0.011) {
              r = 236 + grain * 0.25;
              g = 232 + grain * 0.25;
              b = 220 + grain * 0.25;
              bump = 148 + grain * 0.8;
            }
            // Dashed golden-cream center line
            if (Math.abs(v - 0.5) < 0.013 && ((u * 4) % 1) < 0.52) {
              r = 244 + grain * 0.2;
              g = 205 + grain * 0.2;
              b = 128 + grain * 0.2;
              bump = 152 + grain * 0.8;
            }
          }

          colorData[idx] = Math.max(0, Math.min(255, r));
          colorData[idx + 1] = Math.max(0, Math.min(255, g));
          colorData[idx + 2] = Math.max(0, Math.min(255, b));
          colorData[idx + 3] = 255;

          const bv = Math.max(0, Math.min(255, bump));
          bumpData[idx] = bv;
          bumpData[idx + 1] = bv;
          bumpData[idx + 2] = bv;
          bumpData[idx + 3] = 255;
        }
      }

      const map = new THREE.DataTexture(colorData, w, h, THREE.RGBAFormat);
      map.colorSpace = THREE.SRGBColorSpace;
      map.wrapS = THREE.RepeatWrapping;
      map.wrapT = THREE.ClampToEdgeWrapping;
      map.repeat.set(16, 1);
      map.anisotropy = 8;
      map.needsUpdate = true;

      const bumpMap = new THREE.DataTexture(bumpData, w, h, THREE.RGBAFormat);
      bumpMap.wrapS = THREE.RepeatWrapping;
      bumpMap.wrapT = THREE.ClampToEdgeWrapping;
      bumpMap.repeat.set(16, 1);
      bumpMap.needsUpdate = true;

      return { map, bumpMap };
    }

    // Wicker weave texture + bump map for the front bicycle basket
    function createWickerTextures() {
      const size = 256;
      const col = new Uint8Array(size * size * 4);
      const bmp = new Uint8Array(size * size * 4);
      const freqU = 16;
      const freqV = 24;
      for (let y = 0; y < size; y++) {
        for (let x = 0; x < size; x++) {
          const u = x / size;
          const v = y / size;
          const cellX = Math.floor(u * freqU);
          const cellY = Math.floor(v * freqV);
          const lx = (u * freqU) % 1;
          const ly = (v * freqV) % 1;
          const over = (cellX + cellY) % 2 === 0;
          const strandH = Math.sin(ly * Math.PI);
          const strandV = Math.sin(lx * Math.PI);
          const hVal = over
            ? Math.pow(strandH, 0.65) * (0.65 + 0.35 * Math.sin(lx * Math.PI))
            : Math.pow(strandV, 0.65) * (0.65 + 0.35 * Math.sin(ly * Math.PI));
          const tone = 0.58 + 0.42 * hVal + (hash2(x * 0.5, y * 0.5) - 0.5) * 0.08;
          const idx = (y * size + x) * 4;
          col[idx] = Math.min(255, Math.max(0, 214 * tone));
          col[idx + 1] = Math.min(255, Math.max(0, 166 * tone));
          col[idx + 2] = Math.min(255, Math.max(0, 102 * tone));
          col[idx + 3] = 255;
          const b = Math.min(255, Math.max(0, hVal * 255));
          bmp[idx] = b; bmp[idx + 1] = b; bmp[idx + 2] = b; bmp[idx + 3] = 255;
        }
      }
      const map = new THREE.DataTexture(col, size, size, THREE.RGBAFormat);
      map.colorSpace = THREE.SRGBColorSpace;
      map.wrapS = map.wrapT = THREE.RepeatWrapping;
      map.repeat.set(2, 1.5);
      map.needsUpdate = true;

      const bumpMap = new THREE.DataTexture(bmp, size, size, THREE.RGBAFormat);
      bumpMap.wrapS = bumpMap.wrapT = THREE.RepeatWrapping;
      bumpMap.repeat.set(2, 1.5);
      bumpMap.needsUpdate = true;
      return { map, bumpMap };
    }

    // Tire tread chevron bump texture
    function createTireTreadBump() {
      const w = 128;
      const h = 256;
      const data = new Uint8Array(w * h * 4);
      for (let y = 0; y < h; y++) {
        const v = y / h;
        for (let x = 0; x < w; x++) {
          const u = x / w;
          const distCenter = Math.abs(u - 0.5) * 2;
          const chev = Math.sin((v * 48 + distCenter * 2.2) * Math.PI * 2);
          const rib = Math.abs(u - 0.5) < 0.08 ? 0.85 : (chev > 0 ? 0.9 : 0.25);
          const val = Math.floor((rib * 0.85 + hash2(x, y) * 0.15) * 255);
          const idx = (y * w + x) * 4;
          data[idx] = data[idx + 1] = data[idx + 2] = val;
          data[idx + 3] = 255;
        }
      }
      const tex = new THREE.DataTexture(data, w, h, THREE.RGBAFormat);
      tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
      tex.repeat.set(1, 6);
      tex.needsUpdate = true;
      return tex;
    }

    // Procedural wood grain texture (for harbor bollards & palm bark)
    function createWoodTexture(baseR, baseG, baseB) {
      const size = 256;
      const data = new Uint8Array(size * size * 4);
      for (let y = 0; y < size; y++) {
        for (let x = 0; x < size; x++) {
          const u = x / size;
          const v = y / size;
          const warp = fbm(u * 6, v * 12, 3) * 0.35;
          const ring = Math.sin((u * 14 + warp * 8) * Math.PI * 2) * 0.5 + 0.5;
          const grain = fbm(u * 28, v * 6, 3);
          const shade = 0.72 + ring * 0.22 + (grain - 0.5) * 0.16;
          const idx = (y * size + x) * 4;
          data[idx] = Math.min(255, Math.max(0, baseR * shade));
          data[idx + 1] = Math.min(255, Math.max(0, baseG * shade));
          data[idx + 2] = Math.min(255, Math.max(0, baseB * shade));
          data[idx + 3] = 255;
        }
      }
      const tex = new THREE.DataTexture(data, size, size, THREE.RGBAFormat);
      tex.colorSpace = THREE.SRGBColorSpace;
      tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
      tex.needsUpdate = true;
      return tex;
    }

    function playBicycleBellAndCall() {
      const ctx = ensureAudio();
      if (!ctx) return;
      const now = ctx.currentTime;

      // Double brass bicycle bell "ding-ding!"
      [0, 0.16].forEach((offset, i) => {
        const t = now + offset;
        const freqs = [1568, 3136, 4698, 2349];
        const gains = [0.16, 0.085, 0.04, 0.045];
        freqs.forEach((f, j) => {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'sine';
          osc.frequency.setValueAtTime(f * (i === 1 ? 1.015 : 1.0), t);
          gain.gain.setValueAtTime(0.0001, t);
          gain.gain.linearRampToValueAtTime(gains[j], t + 0.006);
          gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.95 / (j + 1));
          osc.connect(gain).connect(ctx.destination);
          osc.start(t);
          osc.stop(t + 1.0);
        });
      });

      // Playful warm pelican vocal croak/squawk shortly after the bell
      const tCall = now + 0.24;
      const osc = ctx.createOscillator();
      const mod = ctx.createOscillator();
      const modGain = ctx.createGain();
      const filter = ctx.createBiquadFilter();
      const env = ctx.createGain();

      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(240, tCall);
      osc.frequency.exponentialRampToValueAtTime(340, tCall + 0.09);
      osc.frequency.exponentialRampToValueAtTime(195, tCall + 0.34);

      mod.frequency.setValueAtTime(34, tCall);
      modGain.gain.setValueAtTime(26, tCall);
      mod.connect(modGain).connect(osc.frequency);

      filter.type = 'bandpass';
      filter.frequency.setValueAtTime(780, tCall);
      filter.Q.setValueAtTime(3.2, tCall);

      env.gain.setValueAtTime(0.0001, tCall);
      env.gain.linearRampToValueAtTime(0.075, tCall + 0.04);
      env.gain.exponentialRampToValueAtTime(0.0001, tCall + 0.38);

      osc.connect(filter).connect(env).connect(ctx.destination);
      osc.start(tCall);
      mod.start(tCall);
      osc.stop(tCall + 0.4);
      mod.stop(tCall + 0.4);
    }

      const dir = new THREE.Vector3().subVectors(pB, pA);

    // --- 7D. Chrome Fenders, Rear Luggage Rack & Taillight ---
    function createFender(center, startAngle, endAngle, radius = R_WHEEL + 0.026) {
      const group = new THREE.Group();
      const segs = 28;
      const archSegs = 8;
      const halfWidth = 0.036;
      const positions = [];
      const uvs = [];
      const indices = [];

      for (let i = 0; i <= segs; i++) {
        const t = i / segs;
        const a = startAngle + (endAngle - startAngle) * t;
        const ca = Math.cos(a);
        const sa = Math.sin(a);
        for (let j = 0; j <= archSegs; j++) {
          const s = j / archSegs;
          const phi = (s - 0.5) * Math.PI * 0.92;
          const rOff = Math.cos(phi) * 0.015;
          const z = Math.sin(phi) * halfWidth;
          const r = radius + rOff;
          positions.push(center.x + ca * r, center.y + sa * r, z);
          uvs.push(t, s);
        }
      }
      for (let i = 0; i < segs; i++) {
        for (let j = 0; j < archSegs; j++) {
          const a = i * (archSegs + 1) + j;
          const b = a + (archSegs + 1);
          indices.push(a, b, a + 1, a + 1, b, b + 1);
        }
      }
      const geo = new THREE.BufferGeometry();
      geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
      geo.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
      geo.setIndex(indices);
      geo.computeVertexNormals();
      const fenderMat = bikePaintMat.clone();
      fenderMat.side = THREE.DoubleSide;
      const mesh = new THREE.Mesh(geo, fenderMat);
      mesh.castShadow = true;
      mesh.receiveShadow = true;
      group.add(mesh);
      return group;

    // Helper: soft volumetric beam ShaderMaterial (smooth longitudinal & edge Fresnel fade)
    function createSoftBeamMaterial(hexColor) {
      return new THREE.ShaderMaterial({
        transparent: true,
        depthWrite: false,
        side: THREE.DoubleSide,
        blending: THREE.AdditiveBlending,
        uniforms: {
          uColor: { value: new THREE.Color(hexColor) },
          uOpacity: { value: 0.0 }
        },
        vertexShader: `
          varying vec2 vUv;
          varying vec3 vNormalView;
          varying vec3 vViewPos;
          void main() {
            vUv = uv;
            vec4 mvPos = modelViewMatrix * vec4(position, 1.0);
            vViewPos = -mvPos.xyz;
            vNormalView = normalize(normalMatrix * normal);
            gl_Position = projectionMatrix * mvPos;
          }
        `,
        fragmentShader: `
          uniform vec3 uColor;
          uniform float uOpacity;
          varying vec2 vUv;
          varying vec3 vNormalView;
          varying vec3 vViewPos;
          void main() {
            float along = clamp(vUv.y, 0.0, 1.0); // 1 at cone tip, 0 at open far base
            float fadeDist = pow(along, 1.85) * smoothstep(1.0, 0.92, along);
            float edge = pow(abs(dot(normalize(vNormalView), normalize(vViewPos))), 1.45);
            float alpha = fadeDist * edge * uOpacity;
            gl_FragColor = vec4(uColor, alpha);
          }
        `
      });
    }

    function sampleChainPoint(s, outPos) {
      let d = ((s % chainPerim) + chainPerim) % chainPerim;
      let angle = 0;
      if (d < frontArcLen) {
        // Front chainring clockwise arc from aTop down to aBot
        const a = aTop - d / R_CHAINRING;
        outPos.set(BB_POS.x + Math.cos(a) * R_CHAINRING, BB_POS.y + Math.sin(a) * R_CHAINRING, chainringZ);
        angle = a - Math.PI / 2;
      } else if (d < frontArcLen + chainTanLen) {
        // Bottom straight run from front bottom to rear bottom
        const t = (d - frontArcLen) / chainTanLen;
        outPos.set(
          pFrontBot.x + (pRearBot.x - pFrontBot.x) * t,
          pFrontBot.y + (pRearBot.y - pFrontBot.y) * t,
          chainringZ
        );
        angle = Math.atan2(pRearBot.y - pFrontBot.y, pRearBot.x - pFrontBot.x);
      } else if (d < frontArcLen + chainTanLen + rearArcLen) {
        // Rear cog clockwise arc from aBot down to aTop - 2PI
        const a = aBot - (d - frontArcLen - chainTanLen) / R_COG;
        outPos.set(REAR_AXLE.x + Math.cos(a) * R_COG, REAR_AXLE.y + Math.sin(a) * R_COG, chainringZ);
        angle = a - Math.PI / 2;
      } else {
        // Top straight run from rear top to front top
        const t = (d - frontArcLen - chainTanLen - rearArcLen) / chainTanLen;
        outPos.set(
          pRearTop.x + (pFrontTop.x - pRearTop.x) * t,
          pRearTop.y + (pFrontTop.y - pRearTop.y) * t,
          chainringZ
        );
        angle = Math.atan2(pFrontTop.y - pRearTop.y, pFrontTop.x - pRearTop.x);
      }
      return angle;
    }

        _chainMat4.compose(_chainPos, _chainQuat, _chainScale);
        chainInst.setMatrixAt(i, _chainMat4);
      }
      chainInst.instanceMatrix.needsUpdate = true;
    }

    const _ikToTarget = new THREE.Vector3();
    const _ikDir = new THREE.Vector3();
    const _ikBend = new THREE.Vector3();
    const _ikMid = new THREE.Vector3();

    const _ikX = new THREE.Vector3();
    const _ikZ = new THREE.Vector3();
    const _ikBasis = new THREE.Matrix4();

    function orientWingSegment(obj, boneDir, trailHint) {
      // Strictly right-handed orthonormal frame (det = +1):
      // +Y = boneDir (along limb toward elbow/grip)
      // +X = orthogonalized trailHint (where trailing dark flight feathers fan gracefully!)
      // +Z = X x Y
      _ikX.copy(trailHint).addScaledVector(boneDir, -trailHint.dot(boneDir));
      if (_ikX.lengthSq() < 1e-6) _ikX.set(-1, 0, 0);
      _ikX.normalize();
      _ikZ.crossVectors(_ikX, boneDir).normalize();
      _ikBasis.makeBasis(_ikX, boneDir, _ikZ);
      obj.quaternion.setFromRotationMatrix(_ikBasis);
    }

    function solveTwoBoneIK(rootPos, targetPos, L1, L2, poleDir, upperObj, lowerObj, wingTrailHint = null) {
      _ikToTarget.subVectors(targetPos, rootPos);
      const rawDist = _ikToTarget.length();
      const minD = Math.abs(L1 - L2) + 0.008;
      const maxD = L1 + L2 - 0.006;
      const dist = Math.max(minD, Math.min(maxD, rawDist));

      _ikDir.copy(_ikToTarget).divideScalar(rawDist || 1);
      const cosA = Math.max(-1, Math.min(1, (L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist)));
      const along = L1 * cosA;
      const perp = L1 * Math.sqrt(Math.max(0, 1 - cosA * cosA));

      // Orthogonalize pole vector against root->target direction
      _ikBend.copy(poleDir).addScaledVector(_ikDir, -poleDir.dot(_ikDir));
      if (_ikBend.lengthSq() < 1e-6) _ikBend.set(1, 0, 0);
      _ikBend.normalize();

      _ikMid.copy(rootPos).addScaledVector(_ikDir, along).addScaledVector(_ikBend, perp);

      // Position & orient upper limb (rootPos -> midPos)
      upperObj.position.copy(rootPos);
      _ikSegDir.subVectors(_ikMid, rootPos).normalize();
      if (wingTrailHint) {
        orientWingSegment(upperObj, _ikSegDir, wingTrailHint);
      } else {
        upperObj.quaternion.setFromUnitVectors(_upAxis, _ikSegDir);
      }

      // Position & orient lower limb (midPos -> targetPos)
      lowerObj.position.copy(_ikMid);
      _ikSegDir.subVectors(targetPos, _ikMid).normalize();
      if (wingTrailHint) {
        orientWingSegment(lowerObj, _ikSegDir, wingTrailHint);
      } else {
        lowerObj.quaternion.setFromUnitVectors(_upAxis, _ikSegDir);
      }

      return _ikMid;
    }

    function updatePelicanAndBicycle(dt) {
      // 1. Keyboard speed & steering adjustments
      if (state.keys.w) {
        state.targetSpeedMult = Math.min(2.2, state.targetSpeedMult + dt * 0.9);
        sliderSpeed.value = state.targetSpeedMult.toFixed(2);
        lblSpeedMult.textContent = `${state.targetSpeedMult.toFixed(2)}×`;
      }
      if (state.keys.s) {
        state.targetSpeedMult = Math.max(0.0, state.targetSpeedMult - dt * 1.1);
        sliderSpeed.value = state.targetSpeedMult.toFixed(2);
        lblSpeedMult.textContent = `${state.targetSpeedMult.toFixed(2)}×`;
      }

      state.speedMult += (state.targetSpeedMult - state.speedMult) * (1 - Math.exp(-dt * 4.5));

      let desiredSteerInput = 0;
      if (state.autoSteer) {
        // Gentle organic coastal cruising weave
        desiredSteerInput = Math.sin(elapsed * 0.42) * 0.24 + Math.sin(elapsed * 0.93) * 0.11;
      } else {
        if (state.keys.a) desiredSteerInput += 0.85;
        if (state.keys.d) desiredSteerInput -= 0.85;
        if (!state.keys.a && !state.keys.d) desiredSteerInput = state.manualSteer;
      }

      // 2. Coupled Drivetrain Kinematics (Crank -> Chain -> Rear Cog -> Wheels -> Ground Speed!)
      const BASE_CADENCE_RAD = Math.PI * 2 * 1.0; // 60 RPM at 1.0x speed
      const wheelOmega = BASE_CADENCE_RAD * GEAR_RATIO * state.speedMult;
      const linearSpeed = wheelOmega * R_WHEEL; // exact ground velocity (m/s), ~4.39 m/s at 1.0x

      let crankOmega = 0;
      if (!state.coasting) {
        crankOmega = BASE_CADENCE_RAD * state.speedMult;
        state.crankAngle = (state.crankAngle + crankOmega * dt) % (Math.PI * 2);
      } else {
        // Coasting freewheel mode: smoothly ease crank toward relaxed horizontal cruise stance!
        const targetCoastAngle = 0.22;
        let diff = targetCoastAngle - state.crankAngle;
        while (diff > Math.PI) diff -= Math.PI * 2;
        while (diff < -Math.PI) diff += Math.PI * 2;
        state.crankAngle += diff * (1 - Math.exp(-dt * 5.0));

        // Authentic freewheel ratchet click sound while coasting!
        if (linearSpeed > 0.6) {
          state.freewheelAcc += wheelOmega * dt;
          if (state.freewheelAcc > 0.42) {
            state.freewheelAcc = 0;
            playFreewheelClick();
          }
        }
      }

      // Forward rolling rotates around +Z by NEGATIVE angle
      state.wheelAngle -= wheelOmega * dt;
      frontWheel.rotation.z = state.wheelAngle;
      rearWheel.rotation.z = state.wheelAngle;

      crankGroup.rotation.z = -state.crankAngle;
      state.chainOffset = (state.chainOffset + crankOmega * R_CHAINRING * dt) % chainPerim;
      updateChainVisual(state.chainOffset);

      // 3. Steering, Banking & Lane Lateral Motion
      const maxSteerRad = 0.36;
      const targetSteer = desiredSteerInput * maxSteerRad;
      state.steerAngle += (targetSteer - state.steerAngle) * (1 - Math.exp(-dt * 7.0));
      steerPivot.rotation.y = state.steerAngle;

      // Physically intuitive lean/bank into turns (left steer [+Y] shifts toward -Z, leaning bike around X by -angle!)
      const speedScale = Math.min(1.0, linearSpeed / 3.5);
      // Add subtle authentic rhythmic pedaling counter-sway synchronized with leg power strokes!
      const pedalSway = (!state.coasting && state.speedMult > 0.05)
        ? Math.sin(state.crankAngle) * 0.026 * Math.min(1.3, state.speedMult)
        : 0;
      const targetLean = -state.steerAngle * 0.58 * speedScale + pedalSway;
      state.leanAngle += (targetLean - state.leanAngle) * (1 - Math.exp(-dt * 8.0));
      bikeLeanGroup.rotation.x = state.leanAngle;

      // Move smoothly laterally across bike path width [-1.55, +1.55]
      const lateralVel = -Math.sin(state.steerAngle) * linearSpeed * 0.62;
      state.laneZ += lateralVel * dt;
      // Gentle self-centering boundary force so bike stays cleanly on the coastal path
      if (state.autoSteer) {
        state.laneZ *= (1 - Math.min(0.4, dt * 0.35));
      }
      state.laneZ = Math.max(-1.55, Math.min(1.55, state.laneZ));
      bikeRig.position.z = state.laneZ;
      bikeRig.rotation.y = state.steerAngle * 0.24 * speedScale;

      // Subtle micro-vibration from coastal pavement texture
      const roadBump = Math.sin(state.roadScroll * 3.8) * 0.004 * speedScale
                     + Math.sin(state.roadScroll * 9.5) * 0.002 * speedScale;
      bikeFrameGroup.position.y = roadBump;

      // 4. Coordinated Torso Posture, Hip Rock, Saddle Bob & Tail Wag
      const theta = state.crankAngle;
      const powerFactor = state.coasting ? 0 : Math.min(1.4, state.speedMult);
      // Twice-per-revolution vertical cadence bounce (once per downstroke!)
      const cadenceBob = Math.sin(theta * 2 - 0.5) * 0.011 * powerFactor;
      const hipRoll = -Math.sin(theta) * 0.048 * powerFactor; // pelvis dips slightly on pushing side
      const torsoYaw = Math.cos(theta) * 0.035 * powerFactor + state.steerAngle * 0.28;
      // Aero forward lean scales naturally with cycling speed (upright cruise -> sporty sprint!)
      const forwardLean = -0.17 - Math.min(0.24, state.speedMult * 0.11);

      torsoPivot.position.set(-0.32, 1.035 + cadenceBob, Math.sin(theta) * 0.012 * powerFactor);
      torsoPivot.rotation.set(hipRoll, torsoYaw, forwardLean);

      // Saddle slightly compresses in sync with rider weight
      saddleGroup.position.y = SADDLE_MOUNT.y + 0.035 + cadenceBob * 0.45;

      // Tail feathers wag & bob rhythmically
      tailPivot.rotation.z = Math.sin(theta * 2 - 1.2) * 0.075 * powerFactor + Math.sin(elapsed * 2.5) * 0.025;
      tailPivot.rotation.y = -hipRoll * 1.4;

      // 5. Head Stabilization, Look-Into-Turn, Beak Squawk, Blinking & Gular Pouch Spring Physics
      if (state.squawkTimer > 0) state.squawkTimer = Math.max(0, state.squawkTimer - dt);
      if (state.bellTimer > 0) state.bellTimer = Math.max(0, state.bellTimer - dt);

      // Brass bell striker animation
      if (state.bellTimer > 0) {
        bellHammerPivot.rotation.y = Math.sin(state.bellTimer * 44) * 0.52;
      } else {
        bellHammerPivot.rotation.y *= 0.82;
      }

      const squawkOpen = state.squawkTimer > 0
        ? Math.sin((state.squawkTimer / 0.72) * Math.PI) * 0.34
        : 0;

      // Birds stabilize their head against body bounce (vestibulo-ocular reflex!)
      const headBobY = -cadenceBob * 0.65 + Math.sin(elapsed * 2.2) * 0.008 + squawkOpen * 0.04;
      const headForwardX = 0.078 + Math.min(0.055, state.speedMult * 0.025) + Math.sin(theta * 2) * 0.007 * powerFactor;
      const headLookYaw = state.steerAngle * 0.85 + Math.sin(elapsed * 0.7) * 0.12;
      const headLookPitch = -forwardLean * 0.62 - 0.06 + squawkOpen * 0.52 - Math.sin(theta * 2 - 0.8) * 0.025 * powerFactor;

      _headTargetPos.set(headForwardX, 0.795 + headBobY, -state.steerAngle * 0.04);
      headGroup.position.copy(_headTargetPos);
      headGroup.rotation.set(-hipRoll * 0.65, headLookYaw, headLookPitch);

      // Live S-curved neck skin update!
      updateNeckGeometry(_headTargetPos, headLookYaw);

      // Lower jaw open/close during squawk + subtle breathing
      lowerJawPivot.rotation.z = -(squawkOpen + (Math.sin(elapsed * 3.1) * 0.5 + 0.5) * 0.022);

      // Spring-damper elastic bounce for the iconic pelican gular pouch!
      const pouchDrive = -cadenceBob * 1.4 - roadBump * 2.2;
      const springForce = -115.0 * (state.pouchDisp - pouchDrive) - 11.5 * state.pouchVel;
      state.pouchVel += springForce * dt;
      state.pouchDisp += state.pouchVel * dt;
      const pouchPuff = squawkOpen * 0.45 + Math.max(0, -state.pouchDisp) * 1.6;
      deformGularPouch(Math.max(-0.045, Math.min(0.045, state.pouchDisp)), pouchPuff);

      // Natural pelican blinking
      state.blinkTimer -= dt;
      if (state.blinkTimer <= 0) {
        state.blinkPhase += dt * 11.5;
        if (state.blinkPhase >= Math.PI) {
          state.blinkPhase = 0;
          state.blinkTimer = 2.4 + Math.random() * 3.6;
        }
      }
      const blinkAmt = Math.sin(state.blinkPhase);
      for (let i = 0; i < eyelidUpperMeshes.length; i++) {
        const u = eyelidUpperMeshes[i];
        const l = eyelidLowerMeshes[i];
        u.mesh.rotation.x = u.side * (-0.78 + blinkAmt * 0.78);
        l.mesh.rotation.x = l.side * (0.74 - blinkAmt * 0.74);
      }

      // Crest feathers & nautical scarf flutter in the wind!
      const windSpeed = 0.25 + state.speedMult;
      for (let i = 0; i < crestFeathers.length; i++) {
        crestFeathers[i].rotation.z = -0.22 - i * 0.14 + Math.sin(elapsed * (9 + windSpeed * 6) + i * 0.9) * 0.065 * windSpeed;
      }
      updateScarfRibbon(scarfTailA, elapsed, state.speedMult, 0.0);
      updateScarfRibbon(scarfTailB, elapsed, state.speedMult, 1.85);

      // 6. Update Matrices so Anchor World Positions Are 100% Current Before Solving IK!
      bikeRig.updateMatrixWorld(true);
      _invPelicanMat.copy(pelicanRoot.matrixWorld).invert();

      // Convert anchor points into pelicanRoot (= bikeFrameGroup) local space
      hipAnchorR.getWorldPosition(_hipR).applyMatrix4(_invPelicanMat);
      hipAnchorL.getWorldPosition(_hipL).applyMatrix4(_invPelicanMat);
      shoulderAnchorR.getWorldPosition(_shR).applyMatrix4(_invPelicanMat);
      shoulderAnchorL.getWorldPosition(_shL).applyMatrix4(_invPelicanMat);
      gripAnchorR.getWorldPosition(_gripR).applyMatrix4(_invPelicanMat);
      gripAnchorL.getWorldPosition(_gripL).applyMatrix4(_invPelicanMat);

      // When ringing the bell, right wing-tip gracefully reaches over to tap the brass bell lever!
      if (state.bellTimer > 0.12) {
        const reach = Math.sin(Math.min(1, (state.bellTimer - 0.12) / 0.5) * Math.PI);
        bellGroup.getWorldPosition(_bellWorld).applyMatrix4(_invPelicanMat);
        _bellWorld.add(new THREE.Vector3(-0.025, 0.032, 0.02));
        _gripR.lerp(_bellWorld, reach * 0.92);
      }

      // 7. Exact Pedal Positions, Realistic Ankling Pitch & Locked Webbed Feet + 2-Bone Leg IK!
      const thetaR = theta;
      const thetaL = theta + Math.PI;

      _pedalPosR.set(BB_POS.x + Math.cos(thetaR) * CRANK_LEN, BB_POS.y - Math.sin(thetaR) * CRANK_LEN, PEDAL_Z);
      _pedalPosL.set(BB_POS.x + Math.cos(thetaL) * CRANK_LEN, BB_POS.y - Math.sin(thetaL) * CRANK_LEN, -PEDAL_Z);

      // Authentic cycling "ankling" biomechanics (heel drops slightly on power downstroke, toe points gently at bottom)
      const footPitchR = state.coasting ? -0.05 : (-0.14 * Math.sin(thetaR - 0.45) - 0.04);
      const footPitchL = state.coasting ? -0.05 : (-0.14 * Math.sin(thetaL - 0.45) - 0.04);

      pedalRight.position.copy(_pedalPosR);
      pedalRight.rotation.z = footPitchR;
      pedalLeft.position.copy(_pedalPosL);
      pedalLeft.rotation.z = footPitchL;

      // Seat webbed feet directly onto the top face of each rotating pedal!
      const footOffsetY = 0.014;
      legR.foot.position.set(
        _pedalPosR.x - Math.sin(footPitchR) * footOffsetY,
        _pedalPosR.y + Math.cos(footPitchR) * footOffsetY,
        PEDAL_Z
      );
      legR.foot.rotation.z = footPitchR;

      legL.foot.position.set(
        _pedalPosL.x - Math.sin(footPitchL) * footOffsetY,
        _pedalPosL.y + Math.cos(footPitchL) * footOffsetY,
        -PEDAL_Z
      );
      legL.foot.rotation.z = footPitchL;

      // Exact ankle joint center at the rear arch of each webbed foot
      const ankLocalX = -0.026;
      const ankLocalY = 0.030;
      _ankleR.set(
        legR.foot.position.x + Math.cos(footPitchR) * ankLocalX - Math.sin(footPitchR) * ankLocalY,
        legR.foot.position.y + Math.sin(footPitchR) * ankLocalX + Math.cos(footPitchR) * ankLocalY,
        PEDAL_Z
      );
      _ankleL.set(
        legL.foot.position.x + Math.cos(footPitchL) * ankLocalX - Math.sin(footPitchL) * ankLocalY,
        legL.foot.position.y + Math.sin(footPitchL) * ankLocalX + Math.cos(footPitchL) * ankLocalY,
        -PEDAL_Z
      );

      // Solve exact 2-Bone IK for both legs (Zero foot slip, smooth anatomical knee motion!)
      solveTwoBoneIK(_hipR, _ankleR, LEG_L1, LEG_L2, _poleLegR, legR.upper, legR.lower, null);
      solveTwoBoneIK(_hipL, _ankleL, LEG_L1, LEG_L2, _poleLegL, legL.upper, legL.lower, null);

      // Solve exact 2-Bone IK for both wings (Strictly right-handed basis locked onto steering handlebars!)
      solveTwoBoneIK(_shR, _gripR, WING_L1, WING_L2, _poleWingR, wingR.upper, wingR.lower, _trailWingR);
      solveTwoBoneIK(_shL, _gripL, WING_L1, WING_L2, _poleWingL, wingL.upper, wingL.lower, _trailWingL);

      // 8. Infinite World Scrolling Perfectly Matched to Exact Tire Linear Speed!
      const dx = linearSpeed * dt;
      state.roadScroll += dx;

      // Scroll road UV seamlessly (ROAD_LEN = 128, repeat.x = 16 -> 1 UV tile = 8 meters)
      const tileMeters = ROAD_LEN / roadTextures.map.repeat.x;
      const uvOffset = (state.roadScroll / tileMeters) % 1;
      roadTextures.map.offset.x = uvOffset;
      roadTextures.bumpMap.offset.x = uvOffset;

      // Sync ocean world scroll
      oceanUniforms.uScrollX.value = state.roadScroll * 0.72;
      oceanUniforms.uTime.value = elapsed;
      skyUniforms.uTime.value = elapsed;

      // Recycle harbor bollards & ropes seamlessly
      for (let i = 0; i < bollardPool.length; i++) {
        const b = bollardPool[i];
        b.position.x -= dx;
        if (b.position.x < -BOLLARD_SPAN * 0.5) b.position.x += BOLLARD_SPAN;
      }

      // Recycle roadside palms, pines, lampposts & flowers seamlessly
      for (let i = 0; i < roadsidePool.length; i++) {
        const s = roadsidePool[i];
        s.position.x -= dx;
        if (s.position.x < -PROP_SPAN * 0.5) s.position.x += PROP_SPAN;
      }

      // Gentle breeze sway on palm crowns
      for (let i = 0; i < swayingFrondCrowns.length; i++) {
        const item = swayingFrondCrowns[i];
        item.crown.rotation.z = Math.sin(elapsed * 1.45 + item.phase) * 0.065;
        item.crown.rotation.x = Math.cos(elapsed * 1.15 + item.phase) * 0.045;
      }

      // Rotate lighthouse beacon & subtle far parallax
      lhBeamPivot.rotation.y = elapsed * 0.85;
      headlandGroup.position.x -= dx * 0.045;
      if (headlandGroup.position.x < -78) headlandGroup.position.x = 78;

      // Sailboats gliding & bobbing on ocean waves
      for (let i = 0; i < sailboats.length; i++) {
        const sb = sailboats[i];
        sb.mesh.position.x += (sb.speed - linearSpeed * 0.16) * dt;
        if (sb.mesh.position.x < -68) sb.mesh.position.x = 68;
        if (sb.mesh.position.x > 68) sb.mesh.position.x = -68;
        sb.mesh.position.y = -1.04 + Math.sin(elapsed * 1.6 + sb.phase) * 0.14;
        sb.mesh.rotation.z = Math.sin(elapsed * 1.4 + sb.phase) * 0.07;
        sb.mesh.rotation.x = Math.cos(elapsed * 1.1 + sb.phase) * 0.05;
      }

      // Drifting 3D clouds
      for (let i = 0; i < clouds.length; i++) {
        const cl = clouds[i];
        cl.mesh.position.x -= (cl.speed + linearSpeed * 0.07) * dt;
        if (cl.mesh.position.x < -110) cl.mesh.position.x = 110;
      }

      // Flapping distant coastal seabirds
      for (let i = 0; i < seabirds.length; i++) {
        const b = seabirds[i];
        b.group.position.x += (b.speed - linearSpeed * 0.22) * dt;
        if (b.group.position.x < -48) b.group.position.x = 48;
        if (b.group.position.x > 48) b.group.position.x = -48;
        const flap = Math.sin(elapsed * 6.8 + b.phase);
        b.wL.rotation.x = flap * 0.62;
        b.wR.rotation.x = -flap * 0.62;
        b.group.position.y = b.baseY + Math.sin(elapsed * 1.9 + b.phase) * 0.45;
      }

      // Breeze motes streaming past the bicycle
      const mArr = moteGeo.attributes.position.array;
      const moteSpeed = Math.max(0.8, linearSpeed * 1.15);
      for (let i = 0; i < MOTE_COUNT; i++) {
        mArr[i * 3] -= moteSpeed * dt;
        mArr[i * 3 + 1] += Math.sin(elapsed * 2.5 + i) * 0.08 * dt;
        if (mArr[i * 3] < -13) {
          mArr[i * 3] = 13;
          mArr[i * 3 + 1] = 0.25 + Math.random() * 3.5;
        }
      }
      moteGeo.attributes.position.needsUpdate = true;

      // 9. Update Telemetry HUD
      const kmh = linearSpeed * 3.6;
      const rpm = state.coasting ? 0 : (crankOmega / (Math.PI * 2)) * 60;
      const deg = Math.round(((state.crankAngle * 180) / Math.PI + 360) % 360);
      elSpeed.innerHTML = `${kmh.toFixed(1)}<span>km/h</span>`;
      elRpm.innerHTML = `${Math.round(rpm)}<span>RPM</span>`;
      elAngle.innerHTML = `${deg}<span>°</span>`;
    }

    function updateCamera(dt) {
      if (camTransition) {
        camTransition.t += dt / camTransition.duration;
        const p = Math.min(1, camTransition.t);
        // Smooth cubic ease-in-out
        const ease = p < 0.5 ? 4 * p * p * p : 1 - Math.pow(-2 * p + 2, 3) / 2;
        camera.position.lerpVectors(camTransition.startPos, camTransition.endPos, ease);
        controls.target.lerpVectors(camTransition.startTarget, camTransition.endTarget, ease);
        if (p >= 1) camTransition = null;
      } else if (state.orbitShowcase) {
        const radius = 3.45;
        const a = elapsed * 0.34;
        camera.position.set(
          Math.cos(a) * radius,
          1.35 + Math.sin(a * 1.5) * 0.38,
          Math.sin(a) * radius + bikeRig.position.z * 0.4
        );
        controls.target.set(0.02, 0.90, bikeRig.position.z * 0.5);
      } else {
        // Gently follow lateral bike weave so the pelican remains well-framed
        controls.target.z += (bikeRig.position.z * 0.45 - controls.target.z) * (1 - Math.exp(-dt * 2.5));
      }
      controls.update();
    }

    function animate() {
      requestAnimationFrame(animate);
      const now = performance.now() * 0.001;
      const dt = Math.min(0.05, Math.max(0.001, now - lastTime));
      lastTime = now;
      elapsed += dt;

      applyTimeTheme(dt);
      updatePelicanAndBicycle(dt);
      updateCamera(dt);

      renderer.render(scene, camera);
    }

    // Responsive resize
    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    animate();
  </script>
