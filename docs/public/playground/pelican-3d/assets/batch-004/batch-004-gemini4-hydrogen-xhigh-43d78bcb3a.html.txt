        return new Promise((resolve, reject) => {
          const s = document.createElement('script');
          s.src = src;
          s.async = false;
          s.onload = resolve;
          s.onerror = () => reject(new Error('Failed: ' + src));
          document.head.appendChild(s);
        });
      }

      async function loadThreeWithFallback() {
        const providers = [
          'https://cdn.jsdelivr.net/npm/three@0.147.0',
          'https://unpkg.com/three@0.147.0'
        ];
        for (let i = 0; i < providers.length; i++) {
          const base = providers[i];
          try {
            statusEl.textContent = i === 0 ? '加载 Three.js 渲染核心...' : '切换备用源加载 Three.js...';
            await loadScript(base + '/build/three.min.js');
            await loadScript(base + '/examples/js/controls/OrbitControls.js');
            if (window.THREE && window.THREE.OrbitControls) {
              return true;
            }
          } catch (err) {
            console.warn(err.message);
          }
        }
        return false;
      }

      loadThreeWithFallback().then((ok) => {

    function initPelicanCyclingScene() {
      'use strict';

      const container = document.getElementById('canvas-container');
      const loadingOverlay = document.getElementById('loading-overlay');
      const toastEl = document.getElementById('toast');

      let toastTimer = null;
      function showToast(msg) {
        toastEl.textContent = msg;
        toastEl.classList.add('show');
        clearTimeout(toastTimer);
        toastTimer = setTimeout(() => toastEl.classList.remove('show'), 2200);
      }

      // ============================================================
      // 1. DETERMINISTIC MATH & PROCEDURAL TEXTURES (DataTexture)
      // ============================================================
      function hash2D(ix, iy) {
        let n = Math.imul(ix, 374761393) + Math.imul(iy, 668265263);
        n = (n ^ (n >>> 13)) >>> 0;
        n = Math.imul(n, 1274126177) >>> 0;
        return ((n ^ (n >>> 16)) >>> 0) / 4294967295;
      }

      function valueNoise2D(x, y) {
        const ix = Math.floor(x);
        const iy = Math.floor(y);
        const fx = x - ix;
        const fy = y - iy;
        const ux = fx * fx * (3 - 2 * fx);
        const uy = fy * fy * (3 - 2 * fy);
        const a = hash2D(ix, iy);
        const b = hash2D(ix + 1, iy);
        const c = hash2D(ix, iy + 1);
        const d = hash2D(ix + 1, iy + 1);
        return a + (b - a) * ux + (c - a) * uy + (a - b - c + d) * ux * uy;
      }

      function fbm2D(x, y, octaves = 4) {
        let val = 0;
        let amp = 0.5;
        let freq = 1;
        for (let i = 0; i < octaves; i++) {
          val += amp * valueNoise2D(x * freq, y * freq);
          freq *= 2.02;
          amp *= 0.5;
        }
        return val;
      }

      function makeDataTexture(data, w, h, repeatX = 1, repeatY = 1, isSRGB = false) {
        const tex = new THREE.DataTexture(data, w, h, THREE.RGBAFormat);
        tex.wrapS = THREE.RepeatWrapping;
        tex.wrapT = THREE.RepeatWrapping;
        tex.repeat.set(repeatX, repeatY);
        tex.magFilter = THREE.LinearFilter;
        tex.minFilter = THREE.LinearMipmapLinearFilter;
        tex.generateMipmaps = true;
        tex.encoding = isSRGB ? THREE.sRGBEncoding : THREE.LinearEncoding;
        tex.anisotropy = 8;
        tex.needsUpdate = true;
        return tex;
      }

      function createCobbleTextures() {
        const S = 256;
        const col = new Uint8Array(S * S * 4);
        const bmp = new Uint8Array(S * S * 4);
        const rows = 8;
        const cols = 6;
        for (let y = 0; y < S; y++) {
          for (let x = 0; x < S; x++) {
            const u = x / S;
            const v = y / S;
            const row = Math.floor(v * rows);
            const rowF = (v * rows) - row;
            const offsetU = (row % 2) * 0.5;
            const colIdx = Math.floor((u * cols + offsetU) % cols);
            const colF = (u * cols + offsetU) - Math.floor(u * cols + offsetU);

            const edgeU = Math.min(colF, 1 - colF);
            const edgeV = Math.min(rowF, 1 - rowF);
            const seam = Math.min(edgeU * 1.15, edgeV * 1.45);
            const stoneMask = THREE.MathUtils.smoothstep(seam, 0.05, 0.17);

            const stoneSeed = hash2D(colIdx + row * 17, row * 31);
            const grain = fbm2D(u * 24 + stoneSeed * 10, v * 24, 3);
            const micro = hash2D(x, y) * 0.09;

            const baseR = 172 + (stoneSeed - 0.5) * 38 + (grain - 0.5) * 26;
            const baseG = 160 + (stoneSeed - 0.5) * 32 + (grain - 0.5) * 22;
            const baseB = 144 + (stoneSeed - 0.5) * 28 + (grain - 0.5) * 20;

            const mortarR = 92, mortarG = 88, mortarB = 82;
            const r = THREE.MathUtils.clamp(mortarR + (baseR - mortarR) * stoneMask - micro * 28, 0, 255);
            const g = THREE.MathUtils.clamp(mortarG + (baseG - mortarG) * stoneMask - micro * 26, 0, 255);
            const b = THREE.MathUtils.clamp(mortarB + (baseB - mortarB) * stoneMask - micro * 24, 0, 255);

            const idx = (y * S + x) * 4;
            col[idx] = r; col[idx + 1] = g; col[idx + 2] = b; col[idx + 3] = 255;

            const hVal = THREE.MathUtils.clamp((stoneMask * (0.76 + grain * 0.24) + 0.08) * 255, 0, 255);
            bmp[idx] = hVal; bmp[idx + 1] = hVal; bmp[idx + 2] = hVal; bmp[idx + 3] = 255;
          }
        }
        return {
          map: makeDataTexture(col, S, S, 1, 1, true),
          bumpMap: makeDataTexture(bmp, S, S, 1, 1, false)
        };
      }

      function createWoodTextures() {
        const S = 256;
        const col = new Uint8Array(S * S * 4);
        const bmp = new Uint8Array(S * S * 4);
        const planks = 6;
        for (let y = 0; y < S; y++) {
          for (let x = 0; x < S; x++) {
            const u = x / S;
            const v = y / S;
            const pIdx = Math.floor(u * planks);
            const pf = (u * planks) - pIdx;
            const gap = Math.min(pf, 1 - pf);
            const plankMask = THREE.MathUtils.smoothstep(gap, 0.015, 0.06);
            const pRand = hash2D(pIdx, 7);

            const warp = fbm2D(u * 6 + pRand * 5, v * 3, 3) * 2.4;
            const rings = 0.5 + 0.5 * Math.sin((u * 42 + warp * 5.0 + pRand * 12));
            const fiber = fbm2D(u * 48, v * 8, 2);

            const r = THREE.MathUtils.clamp((115 + pRand * 24 + rings * 28 + fiber * 16) * (0.45 + 0.55 * plankMask), 0, 255);
            const g = THREE.MathUtils.clamp((84 + pRand * 18 + rings * 20 + fiber * 12) * (0.45 + 0.55 * plankMask), 0, 255);
            const b = THREE.MathUtils.clamp((58 + pRand * 12 + rings * 14 + fiber * 9) * (0.45 + 0.55 * plankMask), 0, 255);

            const idx = (y * S + x) * 4;
            col[idx] = r; col[idx + 1] = g; col[idx + 2] = b; col[idx + 3] = 255;
            const hVal = THREE.MathUtils.clamp((plankMask * (0.6 + rings * 0.25 + fiber * 0.15)) * 255, 0, 255);
            bmp[idx] = hVal; bmp[idx + 1] = hVal; bmp[idx + 2] = hVal; bmp[idx + 3] = 255;
          }
        }
        return {
          map: makeDataTexture(col, S, S, 1, 1, true),
          bumpMap: makeDataTexture(bmp, S, S, 1, 1, false)
        };
      }

      function createLeatherTextures() {
        const S = 128;
        const col = new Uint8Array(S * S * 4);
        const bmp = new Uint8Array(S * S * 4);
        for (let y = 0; y < S; y++) {
          for (let x = 0; x < S; x++) {
            const u = x / S, v = y / S;
            const pebble = fbm2D(u * 26, v * 26, 3);
            const fine = hash2D(x, y) * 0.12;
            const shade = 0.78 + pebble * 0.28 - fine;
            const idx = (y * S + x) * 4;
            col[idx] = THREE.MathUtils.clamp(142 * shade, 0, 255);
            col[idx + 1] = THREE.MathUtils.clamp(76 * shade, 0, 255);
            col[idx + 2] = THREE.MathUtils.clamp(38 * shade, 0, 255);
            col[idx + 3] = 255;
            const hVal = THREE.MathUtils.clamp((0.35 + pebble * 0.65) * 255, 0, 255);
            bmp[idx] = hVal; bmp[idx + 1] = hVal; bmp[idx + 2] = hVal; bmp[idx + 3] = 255;
          }
        }
        return {
          map: makeDataTexture(col, S, S, 2, 2, true),
          bumpMap: makeDataTexture(bmp, S, S, 2, 2, false)
        };
      }

      function createTireTreadBump() {
        const S = 128;
        const bmp = new Uint8Array(S * S * 4);
        for (let y = 0; y < S; y++) {
          for (let x = 0; x < S; x++) {
            const u = x / S;
            const v = y / S;
            const centerDist = Math.abs(Math.sin(u * Math.PI * 2));
            const chevron = Math.sin((v * 36 + Math.abs(u - 0.5) * 14) * Math.PI * 2);
            const tread = centerDist > 0.65 ? (chevron > 0 ? 225 : 110) : 180;
            const idx = (y * S + x) * 4;
            bmp[idx] = tread; bmp[idx + 1] = tread; bmp[idx + 2] = tread; bmp[idx + 3] = 255;
          }
        }
        return makeDataTexture(bmp, S, S, 1, 16, false);
      }

      function createScaleBump() {
        const S = 128;
        const bmp = new Uint8Array(S * S * 4);
        for (let y = 0; y < S; y++) {
          for (let x = 0; x < S; x++) {
            const u = (x / S) * 10;
            const v = (y / S) * 16;
            const cell = Math.abs(Math.sin(u * Math.PI)) * Math.abs(Math.sin((v + (Math.floor(u) % 2) * 0.5) * Math.PI));
            const val = THREE.MathUtils.clamp((0.35 + Math.pow(cell, 0.45) * 0.65) * 255, 0, 255);
            const idx = (y * S + x) * 4;
            bmp[idx] = val; bmp[idx + 1] = val; bmp[idx + 2] = val; bmp[idx + 3] = 255;
          }
        }
        return makeDataTexture(bmp, S, S, 2, 4, false);
      }

      function createRadialGlowTexture() {
        const S = 64;
        const data = new Uint8Array(S * S * 4);
        for (let y = 0; y < S; y++) {
          for (let x = 0; x < S; x++) {
            const dx = (x + 0.5) / S * 2 - 1;
            const dy = (y + 0.5) / S * 2 - 1;
            const r = Math.sqrt(dx * dx + dy * dy);
            const a = THREE.MathUtils.clamp(Math.exp(-r * r * 4.5) * (1 - THREE.MathUtils.smoothstep(r, 0.6, 1.0)), 0, 1);
            const idx = (y * S + x) * 4;
            data[idx] = 255; data[idx + 1] = 248; data[idx + 2] = 230;
            data[idx + 3] = Math.round(a * 255);
          }
        }
        return makeDataTexture(data, S, S, 1, 1, true);
      }

      // 2. ANALYTIC 2-BONE INVERSE KINEMATICS & ORTHONORMAL BASIS
      // ============================================================
      const _ikDiff = new THREE.Vector3();
      const _ikFwd = new THREE.Vector3();
      const _ikBend = new THREE.Vector3();
      const _ikFallback = new THREE.Vector3();

      function solveTwoBoneIK(root, target, len1, len2, poleDir, outMid) {
        _ikDiff.subVectors(target, root);
        const rawDist = _ikDiff.length();
        const minD = Math.abs(len1 - len2) + 0.004;
        const maxD = len1 + len2 - 0.004;
        const d = THREE.MathUtils.clamp(rawDist, minD, maxD);

        if (rawDist > 1e-5) {
          _ikFwd.copy(_ikDiff).divideScalar(rawDist);
        } else {
          _ikFwd.set(0, -1, 0);
        }

        const cosA = THREE.MathUtils.clamp((len1 * len1 + d * d - len2 * len2) / (2 * len1 * d), -1, 1);
        const sinA = Math.sqrt(Math.max(0, 1 - cosA * cosA));

        _ikBend.copy(poleDir).addScaledVector(_ikFwd, -poleDir.dot(_ikFwd));
        if (_ikBend.lengthSq() < 1e-6) {
          _ikFallback.set(0, 0, 1);
          if (Math.abs(_ikFwd.z) > 0.9) _ikFallback.set(1, 0, 0);
          _ikBend.crossVectors(_ikFallback, _ikFwd);
        }
        _ikBend.normalize();

        outMid.copy(root)
          .addScaledVector(_ikFwd, len1 * cosA)
          .addScaledVector(_ikBend, len1 * sinA);
        return d;
      }

      const _bUp = new THREE.Vector3();
      const _bFwd = new THREE.Vector3();
      const _bRight = new THREE.Vector3();
      const _bMat = new THREE.Matrix4();

      function orientBoneAlongY(mesh, start, end, rightHint) {
        _bUp.subVectors(end, start);
        const len = _bUp.length();
        if (len < 1e-5) return;
        _bUp.divideScalar(len);
        _bFwd.crossVectors(rightHint, _bUp);
        if (_bFwd.lengthSq() < 1e-6) {
          _bFwd.set(0, 0, 1).cross(_bUp);
        }
        _bFwd.normalize();
        _bRight.crossVectors(_bUp, _bFwd).normalize();
        _bMat.makeBasis(_bRight, _bUp, _bFwd);
        mesh.position.copy(start);
        mesh.quaternion.setFromRotationMatrix(_bMat);
      }

      // Wing-specific basis: +Y = bone axis (start -> end), +Z = dorsalHint (outward/upward away from torso)
      function orientWingAlongY(mesh, start, end, dorsalHint) {
        _bUp.subVectors(end, start);
        const len = _bUp.length();
        if (len < 1e-5) return;
        _bUp.divideScalar(len);
        _bRight.crossVectors(_bUp, dorsalHint);
        if (_bRight.lengthSq() < 1e-6) {
          _bRight.set(1, 0, 0);
        }
        _bRight.normalize();
        _bFwd.crossVectors(_bRight, _bUp).normalize();
        _bMat.makeBasis(_bRight, _bUp, _bFwd);
        mesh.position.copy(start);
        mesh.quaternion.setFromRotationMatrix(_bMat);
      }

      class ProceduralAudio {
        constructor() {
          this.ctx = null;
          this.enabled = false;
          this.windGain = null;
          this.windFilter = null;
          this.lastClickStep = 0;
        }

        toggle() {
          if (!this.ctx) {
            const AC = window.AudioContext || window.webkitAudioContext;
            if (!AC) return false;
            this.ctx = new AC();
            this._setupWind();
          }
          if (this.ctx.state === 'suspended') {
            this.ctx.resume();
          }
          this.enabled = !this.enabled;
          if (this.windGain) {
            this.windGain.gain.setTargetAtTime(this.enabled ? 0.025 : 0.0, this.ctx.currentTime, 0.15);
          }
          return this.enabled;
        }

        _setupWind() {
          const bufLen = this.ctx.sampleRate * 2;
          const buf = this.ctx.createBuffer(1, bufLen, this.ctx.sampleRate);
          const ch = buf.getChannelData(0);
          let b0 = 0, b1 = 0, b2 = 0;
          for (let i = 0; i < bufLen; i++) {
            const white = Math.random() * 2 - 1;
            b0 = 0.997 * b0 + white * 0.029;
            b1 = 0.985 * b1 + white * 0.032;
            b2 = 0.950 * b2 + white * 0.048;
            ch[i] = (b0 + b1 + b2 + white * 0.04) * 0.35;
          }
          const src = this.ctx.createBufferSource();
          src.buffer = buf;
          src.loop = true;

          const filter = this.ctx.createBiquadFilter();
          filter.type = 'bandpass';
          filter.frequency.value = 340;
          filter.Q.value = 0.7;

          const gain = this.ctx.createGain();
          gain.gain.value = 0;

          src.connect(filter);
          filter.connect(gain);
          gain.connect(this.ctx.destination);
          src.start();
          this.windFilter = filter;
          this.windGain = gain;
        }

        updateWind(speed) {
          if (!this.enabled || !this.ctx || !this.windGain) return;
          const now = this.ctx.currentTime;
          const norm = THREE.MathUtils.clamp(speed / 9.5, 0, 1);
          this.windGain.gain.setTargetAtTime(0.012 + norm * 0.048, now, 0.12);
          this.windFilter.frequency.setTargetAtTime(220 + norm * 520, now, 0.12);
        }

        playBell() {
          if (!this.enabled || !this.ctx) return;
          const now = this.ctx.currentTime;
          const partials = [
            { freq: 1860, amp: 0.14, decay: 1.15 },
            { freq: 3745, amp: 0.08, decay: 0.75 },
            { freq: 5290, amp: 0.04, decay: 0.42 }
          ];
          partials.forEach((p) => {
            const osc = this.ctx.createOscillator();
            const g = this.ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(p.freq, now);
            g.gain.setValueAtTime(0.0001, now);
            g.gain.linearRampToValueAtTime(p.amp, now + 0.004);
            g.gain.exponentialRampToValueAtTime(0.0001, now + p.decay);
            osc.connect(g);
            g.connect(this.ctx.destination);
            osc.start(now);
            osc.stop(now + p.decay + 0.05);
          });
        }

        playSquawk() {
          if (!this.enabled || !this.ctx) return;
          const now = this.ctx.currentTime;
          const osc = this.ctx.createOscillator();
          const filt = this.ctx.createBiquadFilter();
          const g = this.ctx.createGain();
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(210, now);
          osc.frequency.exponentialRampToValueAtTime(145, now + 0.16);
          osc.frequency.exponentialRampToValueAtTime(180, now + 0.26);

          filt.type = 'bandpass';
          filt.frequency.setValueAtTime(620, now);
          filt.frequency.exponentialRampToValueAtTime(420, now + 0.25);
          filt.Q.value = 3.2;

          g.gain.setValueAtTime(0.001, now);
          g.gain.linearRampToValueAtTime(0.09, now + 0.03);
          g.gain.exponentialRampToValueAtTime(0.0001, now + 0.28);

          osc.connect(filt);
          filt.connect(g);
          g.connect(this.ctx.destination);
          osc.start(now);
          osc.stop(now + 0.3);
        }

        playFreehubClick() {
          if (!this.enabled || !this.ctx) return;
          const now = this.ctx.currentTime;
          const osc = this.ctx.createOscillator();
          const g = this.ctx.createGain();
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(3400, now);
          osc.frequency.exponentialRampToValueAtTime(1100, now + 0.014);
          g.gain.setValueAtTime(0.025, now);
          g.gain.exponentialRampToValueAtTime(0.0001, now + 0.016);
          osc.connect(g);
          g.connect(this.ctx.destination);
          osc.start(now);
          osc.stop(now + 0.02);
        }
      }

      for (let i = 0; i < LOOKUP_SAMPLES; i++) {
        const u = i / LOOKUP_SAMPLES;
        const p = roadCurve.getPointAt(u);
        const t = roadCurve.getTangentAt(u).normalize();
        roadTable.push({ pos: p, tan: t });
      }
      for (let i = 0; i < LOOKUP_SAMPLES; i++) {
        const prev = roadTable[(i - 4 + LOOKUP_SAMPLES) % LOOKUP_SAMPLES];
        const next = roadTable[(i + 4) % LOOKUP_SAMPLES];
        const yawPrev = Math.atan2(prev.tan.x, prev.tan.z);
        const yawNext = Math.atan2(next.tan.x, next.tan.z);
        let dYaw = yawNext - yawPrev;
        while (dYaw > Math.PI) dYaw -= Math.PI * 2;
        while (dYaw < -Math.PI) dYaw += Math.PI * 2;
        const ds = (8 / LOOKUP_SAMPLES) * ROAD_TOTAL_LEN;
        roadTable[i].curvature = dYaw / ds;
      }

      function sampleRoadAtDist(s) {
        const wrapped = ((s % ROAD_TOTAL_LEN) + ROAD_TOTAL_LEN) % ROAD_TOTAL_LEN;
        const fIdx = (wrapped / ROAD_TOTAL_LEN) * LOOKUP_SAMPLES;
        const i0 = Math.floor(fIdx) % LOOKUP_SAMPLES;
        const i1 = (i0 + 1) % LOOKUP_SAMPLES;
        const alpha = fIdx - Math.floor(fIdx);
        const a = roadTable[i0];
        const b = roadTable[i1];
        const pos = new THREE.Vector3().lerpVectors(a.pos, b.pos, alpha);
        const tan = new THREE.Vector3().lerpVectors(a.tan, b.tan, alpha).normalize();
        const curvature = THREE.MathUtils.lerp(a.curvature, b.curvature, alpha);
        return { pos, tan, curvature };
      }

      function createSoftBeamMaterial(colorHex) {
        return new THREE.ShaderMaterial({
          transparent: true,
          depthWrite: false,
          blending: THREE.AdditiveBlending,
          side: THREE.DoubleSide,
          uniforms: {
            uColor: { value: new THREE.Color(colorHex) },
            uOpacity: { value: 0.0 }
          },
          vertexShader: `
            varying vec2 vUv;
            varying vec3 vNormalView;
            varying vec3 vViewPos;
            void main() {
              vUv = uv;
              vec4 mvPos = modelViewMatrix * vec4(position, 1.0);
              vViewPos = -mvPos.xyz;
              vNormalView = normalize(normalMatrix * normal);
              gl_Position = projectionMatrix * mvPos;
            }
          `,
          fragmentShader: `
            uniform vec3 uColor;
            uniform float uOpacity;
            varying vec2 vUv;
            varying vec3 vNormalView;
            varying vec3 vViewPos;
            void main() {
              vec3 N = normalize(vNormalView);
              vec3 V = normalize(vViewPos);
              float edgeSoft = pow(abs(dot(N, V)), 2.2);
              float distFade = pow(clamp(vUv.y, 0.0, 1.0), 1.6) * smoothstep(1.0, 0.9, vUv.y);
              float alpha = uOpacity * edgeSoft * distFade;
              gl_FragColor = vec4(uColor, alpha);
            }
          `
        });
      }

        new THREE.Vector3(), new THREE.Vector3(), new THREE.Vector3(), new THREE.Vector3()
      ]);
      const _nTan = new THREE.Vector3();
      const _nRight = new THREE.Vector3();
      const _nUp = new THREE.Vector3();

      function updatePelicanNeck(basePos, headPos, headForward) {
        _neckCurve.points[0].copy(basePos);
        _neckCurve.points[1].copy(basePos).add(new THREE.Vector3(0, 0.095, -0.055));
        _neckCurve.points[2].copy(headPos).add(new THREE.Vector3(-headForward.x * 0.05, -0.095, -0.055));
        _neckCurve.points[3].copy(headPos).add(new THREE.Vector3(0, -0.02, -0.02));

        let ptr = 0;
        for (let r = 0; r < NECK_RINGS; r++) {
          const t = r / (NECK_RINGS - 1);
          const center = _neckCurve.getPoint(t);
          _nTan.copy(_neckCurve.getTangent(t)).normalize();
          _nRight.set(1, 0, 0).cross(_nTan);
          if (_nRight.lengthSq() < 1e-5) _nRight.set(0, 0, 1).cross(_nTan);
          _nRight.normalize();
          _nUp.crossVectors(_nTan, _nRight).normalize();

          const radiusW = THREE.MathUtils.lerp(0.088, 0.054, Math.pow(t, 0.72));
          const radiusD = THREE.MathUtils.lerp(0.096, 0.058, Math.pow(t, 0.72));

          for (let s = 0; s < NECK_RADIAL; s++) {
            const ang = (s / NECK_RADIAL) * Math.PI * 2;
            const ca = Math.cos(ang) * radiusW;
            const sa = Math.sin(ang) * radiusD;
            neckPositions[ptr] = center.x + _nUp.x * ca + _nRight.x * sa;
            neckPositions[ptr + 1] = center.y + _nUp.y * ca + _nRight.y * sa;
            neckPositions[ptr + 2] = center.z + _nUp.z * ca + _nRight.z * sa;
            ptr += 3;
          }
        }
        neckGeo.attributes.position.needsUpdate = true;
        neckGeo.computeVertexNormals();
      }

      function buildPelicanWing(sideSign) {
        // Bone axis = +Y (start -> end); +Z = dorsal/outward away from torso!
        const upperGroup = new THREE.Group();
        const shoulderCap = new THREE.Mesh(new THREE.SphereGeometry(0.052, 12, 10), creamFeatherMat);
        shoulderCap.scale.set(0.95, 1.1, 1.2);
        upperGroup.add(shoulderCap);
        const upperMesh = new THREE.Mesh(
          new THREE.CapsuleGeometry(0.044, WING_UPPER_LEN - 0.07, 6, 12),
          creamFeatherMat
        );
        upperMesh.scale.set(0.9, 1.0, 1.25);
        upperMesh.position.y = WING_UPPER_LEN * 0.5;
        upperMesh.castShadow = true;
        upperGroup.add(upperMesh);

        // Covert feathers trailing backward (-Y) along the dorsal (+Z) surface
        const covertGeo = createFeatherGeometry(0.21, 0.054, 0.008);
        for (let i = 0; i < 4; i++) {
          const f = new THREE.Mesh(covertGeo, creamFeatherMat);
          f.position.set(0, 0.07 + i * 0.055, 0.025);
          f.rotation.set(Math.PI - 0.22 - i * 0.03, 0, sideSign * -0.14);
          f.castShadow = true;
          upperGroup.add(f);
        }

        const foreGroup = new THREE.Group();
        const foreMesh = new THREE.Mesh(
          new THREE.CapsuleGeometry(0.032, WING_FORE_LEN - 0.05, 6, 10),
          creamFeatherMat
        );
        foreMesh.scale.set(0.85, 1.0, 1.15);
        foreMesh.position.y = WING_FORE_LEN * 0.5;
        foreMesh.castShadow = true;
        foreGroup.add(foreMesh);

        // Layered secondary (cream) & primary (charcoal) flight feathers trailing backward (-Y) & outward (+Z)
        const primaryGeo = createFeatherGeometry(0.25, 0.045, 0.006);
        for (let i = 0; i < 5; i++) {
          const isDark = i >= 2;
          const pf = new THREE.Mesh(primaryGeo, isDark ? charcoalFlightMat : creamFeatherMat);
          pf.position.set(0, 0.06 + i * 0.045, 0.018);
          pf.rotation.set(Math.PI - 0.16 - i * 0.04, 0, sideSign * (-0.1 - i * 0.03));
          pf.castShadow = true;
          foreGroup.add(pf);
        }

        const tipGrip = new THREE.Mesh(new THREE.SphereGeometry(0.028, 10, 8), creamFeatherMat);
        tipGrip.scale.set(0.95, 1.25, 1.05);
        tipGrip.position.y = WING_FORE_LEN;
        foreGroup.add(tipGrip);

        pelicanRoot.add(upperGroup, foreGroup);
        return { upperGroup, foreGroup };
      }

          sim.isCoasting = false;
          document.getElementById('btn-coast').classList.remove('active');
          sim.targetSpeed = Math.max(sim.targetSpeed, 7.6);
        } else {
          sim.targetSpeed = 4.6;
        }
        speedSlider.value = sim.targetSpeed;
        speedTargetLabel.textContent = sim.targetSpeed.toFixed(1) + ' m/s';
        e.currentTarget.classList.toggle('active', sim.isSprinting);
        showToast(sim.isSprinting ? '站立摇车冲刺!' : '恢复坐姿巡航');
      });

      const telCadenceEl = document.getElementById('tel-cadence');
      const telLeanEl = document.getElementById('tel-lean');
      const telDistEl = document.getElementById('tel-dist');

      const _vHipL = new THREE.Vector3();
      const _vHipR = new THREE.Vector3();
      const _vPedalL = new THREE.Vector3();
      const _vPedalR = new THREE.Vector3();
      const _vFootOffL = new THREE.Vector3();
      const _vFootOffR = new THREE.Vector3();
      const _vAnkleL = new THREE.Vector3();
      const _vAnkleR = new THREE.Vector3();
      const _vKneeL = new THREE.Vector3();
      const _vKneeR = new THREE.Vector3();
      const _vShL = new THREE.Vector3();
      const _vShR = new THREE.Vector3();
      const _vGripL = new THREE.Vector3();
      const _vGripR = new THREE.Vector3();
      const _vElbowL = new THREE.Vector3();

      const _poleKneeR = new THREE.Vector3(-0.24, 0.35, 1.0);
      const _poleElbowL = new THREE.Vector3(0.62, -0.25, -0.74);
      const _poleElbowR = new THREE.Vector3(-0.62, -0.25, -0.74);
      const _dorsalWingL = new THREE.Vector3(0.68, 0.72, -0.15).normalize();
      const _dorsalWingR = new THREE.Vector3(-0.68, 0.72, -0.15).normalize();

      let uiUpdateAccum = 0;

      function updateBikePoseAndPhysics(dt) {
        let curvature = 0;
        let desiredSteer = 0;

        if (sim.driveMode === 'auto') {
          const effTarget = sim.isCoasting ? Math.max(sim.speed * 0.96, 1.8) : sim.targetSpeed;
          sim.speed = THREE.MathUtils.lerp(sim.speed, effTarget, 1 - Math.exp(-2.2 * dt));
          const ds = sim.speed * dt;
          sim.trackDist = (sim.trackDist + ds) % ROAD_TOTAL_LEN;
          sim.totalDistance += ds;

          const sample = sampleRoadAtDist(sim.trackDist);
          const lookAhead = sampleRoadAtDist(sim.trackDist + 1.35);
          sim.posX = sample.pos.x;
          sim.posZ = sample.pos.z;
          sim.headingYaw = Math.atan2(lookAhead.pos.x - sample.pos.x, lookAhead.pos.z - sample.pos.z);
          curvature = sample.curvature;
          desiredSteer = THREE.MathUtils.clamp(Math.atan(1.14 * curvature), -0.52, 0.52);
        } else {
          const accel = (keys['w'] || keys['arrowup']) ? 1 : ((keys['s'] || keys['arrowdown']) ? -1 : 0);
          const turnInput = (keys['a'] || keys['arrowleft']) ? 1 : ((keys['d'] || keys['arrowright']) ? -1 : 0);

          if (accel > 0) {
            sim.speed = THREE.MathUtils.lerp(sim.speed, 8.8, 1 - Math.exp(-1.8 * dt));
            sim.isCoasting = false;
          } else if (accel < 0) {
            sim.speed = THREE.MathUtils.lerp(sim.speed, 0.0, 1 - Math.exp(-4.5 * dt));
          } else {
            sim.speed = THREE.MathUtils.lerp(sim.speed, sim.targetSpeed * 0.55, 1 - Math.exp(-0.4 * dt));
          }

          const maxSteer = THREE.MathUtils.lerp(0.55, 0.24, THREE.MathUtils.clamp(sim.speed / 9.0, 0, 1));
          desiredSteer = turnInput * maxSteer;
          sim.steerAngle = THREE.MathUtils.lerp(sim.steerAngle, desiredSteer, 1 - Math.exp(-8.5 * dt));

          curvature = Math.tan(sim.steerAngle) / 1.14;
          sim.headingYaw += sim.speed * curvature * dt;

          const ds = sim.speed * dt;
          sim.totalDistance += ds;
          let nextX = sim.posX + Math.sin(sim.headingYaw) * ds;
          let nextZ = sim.posZ + Math.cos(sim.headingYaw) * ds;

          const rx = nextX / 25.5;
          const rz = nextZ / 19.5;
          const ellipR = Math.hypot(rx, rz);
          if (ellipR > 0.96) {
            const scale = 0.96 / ellipR;
            nextX *= scale;
            nextZ *= scale;
            const normAng = Math.atan2(-nextX, -nextZ);
            let diff = normAng - sim.headingYaw;
            while (diff > Math.PI) diff -= Math.PI * 2;
            while (diff < -Math.PI) diff += Math.PI * 2;
            sim.headingYaw += diff * 0.08;
          }
          sim.posX = nextX;
          sim.posZ = nextZ;
        }

        if (sim.driveMode === 'auto') {
          sim.steerAngle = THREE.MathUtils.lerp(sim.steerAngle, desiredSteer, 1 - Math.exp(-7.0 * dt));
        }

        const sinH = Math.sin(sim.headingYaw);
        const cosH = Math.cos(sim.headingYaw);
        const fY = terrainHeight(sim.posX + sinH * 0.58, sim.posZ + cosH * 0.58);
        const rY = terrainHeight(sim.posX - sinH * 0.56, sim.posZ - cosH * 0.56);
        const groundMidY = (fY + rY) * 0.5 + (sim.driveMode === 'auto' ? 0.040 : 0.02);
        const groundPitch = -Math.atan2(fY - rY, 1.14);

        bikeRoot.position.set(sim.posX, groundMidY, sim.posZ);
        bikeRoot.rotation.set(0, sim.headingYaw, 0);
        frameGroup.rotation.x = groundPitch;

        const centripetalLean = -Math.atan((sim.speed * sim.speed * curvature) / 9.81);
        const sprintRock = sim.isSprinting && !sim.isCoasting && sim.speed > 0.5
          ? Math.sin(sim.crankAngle) * 0.085
          : (sim.speed > 0.5 && !sim.isCoasting ? Math.sin(sim.crankAngle) * 0.018 : 0);
        const targetLean = THREE.MathUtils.clamp(centripetalLean, -0.36, 0.36) + sprintRock;
        sim.leanAngle = THREE.MathUtils.lerp(sim.leanAngle, targetLean, 1 - Math.exp(-6.5 * dt));
        leanPivot.rotation.z = sim.leanAngle;

        steeringPivot.rotation.y = sim.steerAngle;

        const dsFrame = sim.speed * dt;
        const dWheel = dsFrame / WHEEL_RADIUS;
        sim.wheelAngle += dWheel;
        rearWheelGroup.rotation.x = sim.wheelAngle;
        frontWheelGroup.rotation.x = sim.wheelAngle;

        if (!sim.isCoasting && sim.speed > 0.05) {
          const dCrank = dWheel / GEAR_RATIO;
          sim.crankAngle += dCrank;
          rearCogGroup.rotation.x = sim.wheelAngle;
          updateChainLinks(sim.crankAngle);
        } else if (sim.speed > 0.3) {
          const levelErr = Math.sin(sim.crankAngle - Math.PI * 0.5);
          sim.crankAngle -= levelErr * Math.min(1, dt * 3.5);
          const clickInterval = 0.32;
          const step = Math.floor(sim.wheelAngle / clickInterval);
          if (step !== audio.lastClickStep) {
            audio.lastClickStep = step;
            audio.playFreehubClick();
          }
        }

        crankPivot.rotation.x = sim.crankAngle;
        const ankleTiltL = 0.16 * Math.sin(sim.crankAngle - 0.55) + 0.08;
        const ankleTiltR = 0.16 * Math.sin(sim.crankAngle + Math.PI - 0.55) + 0.08;
        leftPedalPivot.rotation.x = -sim.crankAngle + ankleTiltL;
        rightPedalPivot.rotation.x = -sim.crankAngle + ankleTiltR;

        audio.updateWind(sim.speed);
      }

      function updatePelicanKinematics(dt, elapsed) {
        const sprintBlend = sim.isSprinting && !sim.isCoasting && sim.speed > 0.6 ? 1 : 0;
        const cadenceActive = (!sim.isCoasting && sim.speed > 0.05) ? 1 : 0;

        const bob = cadenceActive * Math.cos(sim.crankAngle * 2) * 0.011;
        const standLift = sprintBlend * 0.075;
        const hipRockZ = cadenceActive * Math.sin(sim.crankAngle) * (0.045 + sprintBlend * 0.055);
        const torsoYaw = cadenceActive * Math.sin(sim.crankAngle) * (0.032 + sprintBlend * 0.04);

        pelvisPivot.position.set(0, 0.885 + bob + standLift, -0.33 + sprintBlend * 0.04);
        pelvisPivot.rotation.set(sprintBlend * 0.22, torsoYaw, hipRockZ);

        // Exact pedal center & rotated ankle offset so webbed sole stays flush on pedal platform
        const ca = sim.crankAngle;
        const ankleTiltL = 0.16 * Math.sin(ca - 0.55) + 0.08;
        const ankleTiltR = 0.16 * Math.sin(ca + Math.PI - 0.55) + 0.08;

        _vPedalL.set(0.125, BB_POS.y + CRANK_LEN * Math.cos(ca), BB_POS.z + CRANK_LEN * Math.sin(ca));
        _vPedalR.set(-0.125, BB_POS.y - CRANK_LEN * Math.cos(ca), BB_POS.z - CRANK_LEN * Math.sin(ca));

        _vFootOffL.set(0, 0.0435, -0.042).applyAxisAngle(_axisX, ankleTiltL);
        _vFootOffR.set(0, 0.0435, -0.042).applyAxisAngle(_axisX, ankleTiltR);
        _vAnkleL.copy(_vPedalL).add(_vFootOffL);
        _vAnkleR.copy(_vPedalR).add(_vFootOffR);

        _vHipL.set(0.122, -0.03, 0.07).applyEuler(pelvisPivot.rotation).add(pelvisPivot.position);
        _vHipR.set(-0.122, -0.03, 0.07).applyEuler(pelvisPivot.rotation).add(pelvisPivot.position);

        solveTwoBoneIK(_vHipL, _vAnkleL, LEG_UPPER_LEN, LEG_LOWER_LEN, _poleKneeL, _vKneeL);
        solveTwoBoneIK(_vHipR, _vAnkleR, LEG_UPPER_LEN, LEG_LOWER_LEN, _poleKneeR, _vKneeR);

        orientBoneAlongY(leftLeg.upperGroup, _vHipL, _vKneeL, _axisX);
        orientBoneAlongY(leftLeg.lowerGroup, _vKneeL, _vAnkleL, _axisX);
        leftLeg.footGroup.position.copy(_vAnkleL);
        leftLeg.footGroup.rotation.set(ankleTiltL, 0, 0);

        orientBoneAlongY(rightLeg.upperGroup, _vHipR, _vKneeR, _axisX);
        orientBoneAlongY(rightLeg.lowerGroup, _vKneeR, _vAnkleR, _axisX);
        rightLeg.footGroup.position.copy(_vAnkleR);
        rightLeg.footGroup.rotation.set(ankleTiltR, 0, 0);

        // Grip targets transformed into frameGroup (= pelicanRoot) space
        frameGroup.updateMatrixWorld(true);
        leftGripAnchor.getWorldPosition(_vGripL);
        rightGripAnchor.getWorldPosition(_vGripR);
        frameGroup.worldToLocal(_vGripL);
        frameGroup.worldToLocal(_vGripR);

        if (sim.waveAnim > 0) {
          sim.waveAnim = Math.max(0, sim.waveAnim - dt * 0.55);
          const blend = Math.sin(sim.waveAnim * Math.PI);
          const flap = Math.sin(sim.waveAnim * Math.PI * 5.0) * 0.14;
          _waveTarget.set(-0.46, 1.48 + flap, 0.06);
          _vGripR.lerp(_waveTarget, THREE.MathUtils.smoothstep(blend, 0.05, 0.45));
        }

        _vShL.set(0.155, 0.285, 0.185).applyEuler(pelvisPivot.rotation).add(pelvisPivot.position);
        _vShR.set(-0.155, 0.285, 0.185).applyEuler(pelvisPivot.rotation).add(pelvisPivot.position);

        solveTwoBoneIK(_vShL, _vGripL, WING_UPPER_LEN, WING_FORE_LEN, _poleElbowL, _vElbowL);
        solveTwoBoneIK(_vShR, _vGripR, WING_UPPER_LEN, WING_FORE_LEN, _poleElbowR, _vElbowR);

        orientWingAlongY(leftWing.upperGroup, _vShL, _vElbowL, _dorsalWingL);
        orientWingAlongY(leftWing.foreGroup, _vElbowL, _vGripL, _dorsalWingL);

        orientWingAlongY(rightWing.upperGroup, _vShR, _vElbowR, _dorsalWingR);
        orientWingAlongY(rightWing.foreGroup, _vElbowR, _vGripR, _dorsalWingR);

        // Head Stabilization, Look-Ahead & Expressive Beak / Pouch / Blink
        _vNeckBase.set(0, 0.375, 0.195).applyEuler(pelvisPivot.rotation).add(pelvisPivot.position);

        sim.headGlanceCurrent = THREE.MathUtils.lerp(sim.headGlanceCurrent, sim.headGlanceTarget, 1 - Math.exp(-5 * dt));
        const idleBobY = Math.sin(elapsed * 2.4) * 0.008;
        const stabilizedY = 1.525 + standLift * 0.55 + bob * 0.32 + idleBobY;
        const headZ = -0.045 + sprintBlend * 0.08;
        _vHeadPos.set(hipRockZ * 0.35, stabilizedY, headZ);

        let jawOpen = 0.02 + THREE.MathUtils.clamp(sim.speed / 12, 0, 0.05);
        let headPitch = sprintBlend * 0.12 - sim.steerAngle * 0.05 - frameGroup.rotation.x * 0.55;
        if (sim.squawkAnim > 0) {
          sim.squawkAnim = Math.max(0, sim.squawkAnim - dt * 1.45);
          const sEnv = Math.sin(sim.squawkAnim * Math.PI);
          jawOpen += sEnv * 0.38;
          headPitch -= sEnv * 0.18;
        }
        if (sim.bellAnim > 0) {
          sim.bellAnim = Math.max(0, sim.bellAnim - dt * 2.4);
          bellHammerPivot.rotation.z = Math.sin(sim.bellAnim * Math.PI * 6) * 0.35 * sim.bellAnim;
        } else {
          bellHammerPivot.rotation.z = 0;
        }

        headGroup.position.copy(_vHeadPos);
        const lookYaw = sim.steerAngle * 0.72 + sim.headGlanceCurrent * 0.55 + Math.sin(elapsed * 0.65) * 0.09;
        headGroup.rotation.set(headPitch, lookYaw, -sim.leanAngle * 0.45);

        _vHeadFwd.set(0, 0, 1).applyEuler(headGroup.rotation);
        updatePelicanNeck(_vNeckBase, _vHeadPos, _vHeadFwd);

        jawPivot.rotation.x = jawOpen;
        const pouchWind = Math.sin(elapsed * (5 + sim.speed * 1.8)) * 0.004 * (0.3 + sim.speed * 0.1);
        updateGularPouch(jawOpen, pouchWind);

        sim.blinkTimer -= dt;
        if (sim.blinkTimer <= 0 && sim.blinkPhase <= 0) {
          sim.blinkPhase = 1.0;
          sim.blinkTimer = 2.4 + Math.random() * 3.2;
        }
        let lidAngle = -Math.PI * 0.55;
        if (sim.blinkPhase > 0) {
          sim.blinkPhase = Math.max(0, sim.blinkPhase - dt * 6.5);
          const b = Math.sin(sim.blinkPhase * Math.PI);
          lidAngle = THREE.MathUtils.lerp(-Math.PI * 0.55, 0.12, b);
        }
        eyelids.forEach((l) => { l.rotation.x = lidAngle; });

        const breeze = 0.025 + sim.speed * 0.008;
        tailFeathers.forEach((tf) => {
          tf.mesh.rotation.x = 0.08 * Math.abs(tf.idx) + Math.sin(elapsed * 7.5 + tf.idx * 0.6) * breeze;
        });
        crestFeathers.forEach((cf) => {
          cf.mesh.rotation.x = cf.baseRx + Math.sin(elapsed * 9.0 + cf.idx * 0.9) * breeze * 1.4;
        });

        let fishJump = Math.sin(elapsed * 6.5) * 0.004 * Math.min(1, sim.speed * 0.2);
        let fishRoll = -0.15;
        if (sim.fishFlopAnim > 0) {
          sim.fishFlopAnim = Math.max(0, sim.fishFlopAnim - dt * 1.6);
          const fEnv = Math.sin(sim.fishFlopAnim * Math.PI);
          fishJump += fEnv * 0.085;
          fishRoll += Math.sin(sim.fishFlopAnim * Math.PI * 5) * 0.55;
        }
        fishGroup.position.y = -0.02 + fishJump;
        fishGroup.rotation.z = fishRoll;
      }

      function updateEnvironmentAndParticles(dt, elapsed) {
        oceanUniforms.uTime.value = elapsed;
        skyUniforms.uTime.value = elapsed;

        for (const p of swayingPalms) {
          p.pivot.rotation.z = Math.sin(elapsed * 1.45 + p.phase) * 0.045;
          p.pivot.rotation.x = Math.cos(elapsed * 1.1 + p.phase) * 0.03;
        }

        if (lighthouseBeaconGroup) {
          lighthouseBeaconGroup.rotation.y = elapsed * 0.45;
        }

        for (const b of mooredBoats) {
          b.mesh.position.y = b.baseY + Math.sin(elapsed * 1.35 + b.phase) * 0.08;
          b.mesh.rotation.z = Math.sin(elapsed * 1.1 + b.phase) * 0.045;
          b.mesh.rotation.x = Math.cos(elapsed * 0.9 + b.phase) * 0.025;
        }

        for (const c of clouds) {
          c.orbitAng += (c.speed * dt) / c.rad;
          c.mesh.position.set(Math.cos(c.orbitAng) * c.rad, c.y, Math.sin(c.orbitAng) * c.rad);
          c.mesh.lookAt(0, c.y, 0);
        }

        for (const g of gulls) {
          const ang = elapsed * g.speed + g.phase;
          const nx = g.centerX + Math.cos(ang) * g.radius;
          const nz = g.centerZ + Math.sin(ang) * g.radius;
          const ny = g.height + Math.sin(ang * 2.0) * 0.9;
          g.group.position.set(nx, ny, nz);
          g.group.rotation.y = -ang;
          const flap = Math.sin(elapsed * 7.2 + g.phase) * 0.45;
          g.lW.rotation.z = flap;
          g.rW.rotation.z = -flap;
        }

        const sinH = Math.sin(sim.headingYaw);
        const cosH = Math.cos(sim.headingYaw);
        const rearX = bikeRoot.position.x - sinH * 0.65;
        const rearY = bikeRoot.position.y + 0.06;
        const rearZ = bikeRoot.position.z - cosH * 0.65;

        for (let i = 0; i < PARTICLE_COUNT; i++) {
          pLife[i] -= dt * (0.8 + sim.speed * 0.14);
          if (pLife[i] <= 0) {
            pLife[i] = 1.0;
            pPositions[i * 3] = rearX + (Math.random() - 0.5) * 0.16;
            pPositions[i * 3 + 1] = rearY + Math.random() * 0.08;
            pPositions[i * 3 + 2] = rearZ + (Math.random() - 0.5) * 0.16;
            pVels[i].set(
              -sinH * sim.speed * 0.18 + (Math.random() - 0.5) * 0.35,
              0.22 + Math.random() * 0.32,
              -cosH * sim.speed * 0.18 + (Math.random() - 0.5) * 0.35
            );
          } else {
            pPositions[i * 3] += pVels[i].x * dt;
            pPositions[i * 3 + 1] += pVels[i].y * dt;
            pPositions[i * 3 + 2] += pVels[i].z * dt;
          }
        }
        pGeo.attributes.position.needsUpdate = true;
        pMat.opacity = THREE.MathUtils.clamp(sim.speed * 0.065, 0.04, 0.42);
      }

      function updateCameraAndShadows(dt) {
        const deltaPos = _vSunSnap.subVectors(bikeRoot.position, prevBikePos);

        if (sim.camMode === 'pov') {
          bikeRoot.updateMatrixWorld(true);
          const eyePos = new THREE.Vector3(0, 0.095, 0.085);
          headGroup.localToWorld(eyePos);
          const lookTarget = new THREE.Vector3(0, -0.22, 4.5);
          headGroup.localToWorld(lookTarget);
          camera.position.lerp(eyePos, 1 - Math.exp(-16 * dt));
          camera.lookAt(lookTarget);
        } else if (sim.camMode === 'spotter') {
          controls.target.set(bikeRoot.position.x, bikeRoot.position.y + 0.9, bikeRoot.position.z);
          controls.update();
        } else {
          camera.position.add(deltaPos);
          controls.target.add(deltaPos);
          controls.update();
        }

        prevBikePos.copy(bikeRoot.position);

        const sunDir = skyUniforms.uSunDir.value;
        const snapStep = 0.25;
        const sx = Math.round(bikeRoot.position.x / snapStep) * snapStep;
        const sz = Math.round(bikeRoot.position.z / snapStep) * snapStep;
        sunLight.target.position.set(sx, bikeRoot.position.y, sz);
        sunLight.position.set(sx + sunDir.x * 38, bikeRoot.position.y + sunDir.y * 38, sz + sunDir.z * 38);
        sunLight.target.updateMatrixWorld();

        rimFillLight.position.set(sx - sunDir.x * 18, 12, sz - sunDir.z * 18);
      }

      updateBikePoseAndPhysics(0.016);
      updatePelicanKinematics(0.016, 0);
      setCameraMode('follow');
      updateCameraAndShadows(0.016);

      setTimeout(() => {
        loadingOverlay.classList.add('hidden');
      }, 280);

      let simElapsed = 0;

      function animate() {
        requestAnimationFrame(animate);
        const rawDt = Math.min(clock.getDelta(), 0.05);
        const dt = sim.paused ? 0 : rawDt * sim.timeScale;
        simElapsed += dt;

        if (dt > 0) {
          updateBikePoseAndPhysics(dt);
          updatePelicanKinematics(dt, simElapsed);
          updateEnvironmentAndParticles(dt, simElapsed);
        }
        updateCameraAndShadows(rawDt);

        uiUpdateAccum += rawDt;
        if (uiUpdateAccum > 0.08) {
          uiUpdateAccum = 0;
          const kmh = sim.speed * 3.6;
          const crankRPM = (!sim.isCoasting && sim.speed > 0.05)
            ? (sim.speed / (WHEEL_RADIUS * GEAR_RATIO)) * (60 / (Math.PI * 2))
            : 0;
          telSpeedEl.textContent = kmh.toFixed(1) + ' km/h';
          telCadenceEl.textContent = Math.round(crankRPM) + ' RPM';
          telLeanEl.textContent = (-THREE.MathUtils.radToDeg(sim.leanAngle)).toFixed(1) + '°';
          telDistEl.textContent = Math.round(sim.totalDistance) + ' m';
        }

        renderer.render(scene, camera);
      }
      animate();

      window.addEventListener('resize', () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
      });
    }
