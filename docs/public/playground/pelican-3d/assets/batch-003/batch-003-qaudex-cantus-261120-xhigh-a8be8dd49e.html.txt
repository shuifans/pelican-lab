// 让骨骼组从 a 指向 b：局部 +Y 沿骨骼方向，局部 +X（羽毛后缘）落在车身竖直面内
const _x = v3(), _y = v3(), _z = v3(), _m = new THREE.Matrix4();
function aim(g, a, b) {
  _y.subVectors(b, a).normalize(); _x.set(_y.y, -_y.x, 0).normalize(); _z.crossVectors(_x, _y);
  g.position.copy(a); g.quaternion.setFromRotationMatrix(_m.makeBasis(_x, _y, _z));
}
// 两段骨骼 IK：返回关节位置，pole 指定膝/肘弯曲的方向
const _dir = v3(), _perp = v3();
function solveIK(root, target, l1, l2, pole, out) {
  _dir.subVectors(target, root); const d = clamp(_dir.length(), Math.abs(l1 - l2) + 1e-3, l1 + l2 - 1e-3); _dir.normalize();
  const a = Math.acos(clamp((l1 * l1 + d * d - l2 * l2) / (2 * l1 * d), -1, 1));
  _perp.copy(pole).addScaledVector(_dir, -pole.dot(_dir)).normalize();
  return out.copy(root).addScaledVector(_dir, l1 * Math.cos(a)).addScaledVector(_perp, l1 * Math.sin(a));
}

// ---------- 骑行动力学与联动动画 ----------
const S = { u: 0, speed: 4, target: 4, dist: 0, crankA: 0, pedal: 1, lean: 0, steer: 0, honk: 0, t: 0, paused: false, night: 0, heading: 0 };
const _a = v3(), _b = v3(), _c = v3(), _hip = v3(), _knee = v3(), _ank = v3(), _sh = v3(), _el = v3(), _grip = v3(), _wave = v3(), _nb = v3(), _c1 = v3(), _c2 = v3(), _c3 = v3(), _ht = v3();
const kneePole = [v3(1, .35, .3), v3(1, .35, -.3)], elbowPole = [v3(-.5, -.6, .9), v3(-.5, -.6, -.9)];
const headPos = v3(.1, 1.6, 0);
const wrapPi = a => a - TAU * Math.round(a / TAU);

function updateRide(dt) {
  S.speed = damp(S.speed, S.target, 1.2, dt);
  const ds = S.speed * dt; S.dist += ds;
  // 沿路线前进，数值求航向与曲率
  pathAt(S.u, _a); pathAt(S.u + 1e-3, _b); S.u = (S.u + ds / (_b.distanceTo(_a) / 1e-3)) % TAU;
  pathAt(S.u, _a); pathAt(S.u + .002, _b); pathAt(S.u + .004, _c);
  const h0 = Math.atan2(_b.z - _a.z, _b.x - _a.x), h1 = Math.atan2(_c.z - _b.z, _c.x - _b.x);
  const kappa = wrapPi(h1 - h0) / _b.distanceTo(_a); // > 0 表示右转
  S.heading = h0; bike.position.copy(_a); bike.rotation.y = -h0;
  // 压弯倾角 = atan(v²κ/g)，低速时加入更明显的平衡晃动；车把转向与倾斜方向一致
  const wob = (Math.sin(S.t * 2.3) * .5 + Math.sin(S.t * 3.7 + 1) * .3) * (.015 + .06 / (1 + S.speed));
  S.lean = damp(S.lean, clamp(Math.atan(S.speed * S.speed * kappa / 9.8), -.42, .42) + wob, 5, dt);
  S.steer = damp(S.steer, -Math.atan(1.08 * kappa) * 2.5 - wob * 1.6, 8, dt);
  lean.rotation.x = S.lean; steer.rotation.y = S.steer;
  // 车轮纯滚动；曲柄经齿比联动；目标速度低于当前速度时停止蹬踏（飞轮空转）
  rearWheel.rotation.z = frontWheel.rotation.z = -S.dist / WR;
  S.pedal = damp(S.pedal, S.target < S.speed - .25 ? 0 : 1, 4, dt);
  S.crankA -= ds / WR * GEAR * S.pedal; crank.rotation.z = S.crankA;
  updateChain(-S.crankA * RC);

  const ca = S.crankA, tuck = sstep(3, 9, S.speed), eff = S.pedal;
  const p = 1.4 - S.honk, honking = S.honk > 0, open = honking ? sstep(0, .12, p) * (1 - sstep(.55, .85, p)) : 0, wave = honking ? sstep(0, .25, p) * (1 - sstep(1.05, 1.4, p)) : 0;
  S.honk = Math.max(0, S.honk - dt);
  // 躯干：随踏频左右摇摆、轻微起伏，速度越快俯身越低
  body.rotation.set(Math.cos(ca) * .05 * eff, 0, -.04 - tuck * .22 + Math.sin(ca * 2) * .012 * eff + open * .08);
  body.position.y = .85 + Math.cos(ca * 2) * .006 * eff;
  tail.rotation.set(0, Math.sin(ca) * .16, Math.sin(S.t * 3) * .05);
  body.updateMatrix(); crank.updateMatrix(); front.updateMatrix(); steer.updateMatrix(); steerPivot.updateMatrix();

  limbs.forEach((L, i) => {
    // 腿：脚掌踩在脚踏上，踝关节随行程变化（下死点附近脚尖下压），膝盖由 IK 求出
    const a = ca + (i ? Math.PI : 0), pitch = -.15 + .2 * Math.sin(a + .4), cp = Math.cos(pitch), sp = Math.sin(pitch);
    pedals[i].rotation.z = -ca + pitch;
    _ank.set(BB.x + CRANK * Math.cos(a) - .045 * cp - .048 * sp, BB.y + CRANK * Math.sin(a) - .045 * sp + .048 * cp, L.s * .118);
    L.foot.position.copy(_ank); L.foot.rotation.set(0, -L.s * .1, pitch);
    _hip.copy(L.hip.position).applyMatrix4(body.matrix);
    solveIK(_hip, _ank, TH, SH, kneePole[i], _knee); aim(L.thigh, _hip, _knee); aim(L.shin, _knee, _ank);
    // 翅膀：翼尖握住车把（随转向移动），按铃时右翼抬起挥舞
    _sh.copy(L.shoulder.position).applyMatrix4(body.matrix);
    _grip.copy(grips[i].position).applyMatrix4(front.matrix).applyMatrix4(steer.matrix).applyMatrix4(steerPivot.matrix);
    if (i === 0 && wave > 0) _grip.lerp(_wave.set(_sh.x + .1, _sh.y + .36 + Math.sin(S.t * 13) * .05, _sh.z + .3 + Math.sin(S.t * 13 + 1.5) * .12), wave);
    solveIK(_sh, _grip, UA, FW, elbowPole[i], _el); aim(L.upper, _sh, _el); aim(L.fore, _el, _grip);
  });

  // 头颈：头部滞后跟随以保持稳定，颈部沿 S 形贝塞尔曲线伸缩
  _nb.copy(neckBase.position).applyMatrix4(body.matrix);
  _ht.set(_nb.x + .08 + tuck * .1, _nb.y + .36 - tuck * .09 + open * .06, Math.sin(ca) * .008);
  headPos.lerp(_ht, 1 - Math.exp(-10 * dt)); head.position.copy(headPos);
  head.rotation.set(-S.lean * .7, S.steer * 1.2 + Math.sin(S.t * .45) * .14 + wave * .5, -.16 - tuck * .04 + open * .5 + Math.sin(ca * 2) * .01 * eff);
  _c1.copy(_nb).add(_a.set(.17, .1, 0)); _c2.copy(headPos).add(_a.set(-.17, -.15, 0)); _c3.copy(headPos).add(_a.set(-.03, -.03, 0));
  neck.forEach((n, j) => { const t = j / (NECK - 1), m = 1 - t; n.position.set(0, 0, 0).addScaledVector(_nb, m * m * m).addScaledVector(_c1, 3 * m * m * t).addScaledVector(_c2, 3 * m * t * t).addScaledVector(_c3, t * t * t); });
  jaw.rotation.z = -(.02 + open * .55 + Math.sin(S.t * 2.2) * .008);
  pouch.scale.y = .06 * (1 + Math.sin(S.t * 9) * .05 * (.4 + tuck) + open * .6); pouch.rotation.x = Math.sin(S.t * 5) * .06 + S.lean * .5;
  fish.rotation.set(0, Math.sin(S.t * 4.3) * .2, 1.1 + Math.sin(S.t * 6) * .08);
  bell.scale.setScalar(1 + (honking ? Math.max(0, .35 - p) * Math.abs(Math.sin(p * 60)) : 0));
}

function applyTOD(t) {

  mix(A.top, B.top, k, skyU.top.value); mix(A.hor, B.hor, k, skyU.hor.value); mix(A.sun, B.sun, k, skyU.sunCol.value);
  mix(A.hemiS, B.hemiS, k, hemi.color); mix(A.hemiG, B.hemiG, k, hemi.groundColor); mix(A.sea, B.sea, k, seaMat.color);
  scene.fog.color.copy(skyU.hor.value); sun.color.copy(skyU.sunCol.value); hemi.intensity = lerp(A.hemi, B.hemi, k); renderer.toneMappingExposure = lerp(A.exp, B.exp, k);
  // 白天→黄昏太阳西沉；入夜后太阳落下、强度归零，再由月光从另一方向亮起
  let el, az, I;
  if (i === 0) { el = lerp(55, 8, k); az = lerp(35, -65, k); I = lerp(3.2, 2.3, k); }
  else if (k < .5) { el = lerp(8, -5, k * 2); az = -65 - k * 30; I = 2.3 * (1 - k * 2); }
  else { el = 42; az = 150; I = .7 * (k * 2 - 1); }
  el *= Math.PI / 180; az *= Math.PI / 180;
  skyU.sunDir.value.set(Math.cos(el) * Math.cos(az), Math.sin(el), Math.cos(el) * Math.sin(az));

  clouds.material.emissive.copy(skyU.hor.value); clouds.material.emissiveIntensity = .5 * hemi.intensity;
