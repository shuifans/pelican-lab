function hash2(i, j, s) {
  i = ((i % s) + s) % s; j = ((j % s) + s) % s;
  const n = Math.sin(i * 127.1 + j * 311.7) * 43758.5453;
  return n - Math.floor(n);
}
function noise(u, v, s) {
  const x = u * s, y = v * s;
  const xi = Math.floor(x), yi = Math.floor(y);
  let fx = x - xi, fy = y - yi;
  fx = fx * fx * (3 - 2 * fx); fy = fy * fy * (3 - 2 * fy);
  const a = hash2(xi, yi, s), b = hash2(xi + 1, yi, s);
  const c = hash2(xi, yi + 1, s), d = hash2(xi + 1, yi + 1, s);
  return a * (1 - fx) * (1 - fy) + b * fx * (1 - fy) + c * (1 - fx) * fy + d * fx * fy;
}
function makeTexture(kind) {
  const S = 256, data = new Uint8Array(S * S * 4);
  for (let y = 0; y < S; y++) {
    for (let x = 0; x < S; x++) {
      const u = x / S, v = y / S;
      const n1 = noise(u, v, 8), n2 = noise(u, v, 32), n3 = noise(u, v, 128);
      let r, g, b;
      if (kind === 'grass') {
        const k = (n1 - 0.5) * 0.24 + (n2 - 0.5) * 0.11 + (n3 - 0.5) * 0.05;
        r = 118 + k * 180 + (n3 > 0.83 ? 20 : 0);
        g = 152 + k * 145 + (n3 > 0.83 ? 13 : 0);
        b = 86 + k * 120;
        if (n2 > 0.78 && n3 > 0.60) { r += 17; g += 14; b -= 4; }
      } else {
        const k = (n2 - 0.5) * 0.16 + (n3 - 0.5) * 0.14;
        const g0 = 63 + k * 210;
        r = g0 * 0.98; g = g0 * 1.0; b = g0 * 1.06;
        if (n3 > 0.88) { r += 24; g += 24; b += 24; }
      }
      const i = (y * S + x) * 4;
      data[i] = clamp(r, 0, 255); data[i + 1] = clamp(g, 0, 255);
      data[i + 2] = clamp(b, 0, 255); data[i + 3] = 255;
    }
  }
  const t = new THREE.DataTexture(data, S, S, THREE.RGBAFormat);
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  t.colorSpace = THREE.SRGBColorSpace;
  t.magFilter = THREE.LinearFilter;
  t.minFilter = THREE.LinearMipmapLinearFilter;
  t.generateMipmaps = true;
  t.anisotropy = Math.min(8, renderer.capabilities.getMaxAnisotropy());
  t.needsUpdate = true;
  return t;
}

const UNIT_SPHERE = new THREE.SphereGeometry(1, 20, 14);
function mesh(parent, geo, mat) { const m = new THREE.Mesh(geo, mat); parent.add(m); return m; }
function ellipsoid(parent, mat, rx, ry, rz, pos, geo) {
  const m = mesh(parent, geo || UNIT_SPHERE, mat);
  m.scale.set(rx, ry, rz);
  if (pos) m.position.copy(pos);

}
function cyl(parent, rt, rb, mat, seg) {

function setSegment(m, a, b) {
  const d = new THREE.Vector3().subVectors(b, a);
  const len = Math.max(d.length(), 1e-5);
  m.position.copy(a).add(b).multiplyScalar(0.5);
  m.quaternion.setFromUnitVectors(UP, d.clone().normalize());
  m.scale.set(1, len, 1);
}
function tube(parent, pts, r, mat, tubular, closed) {

  return mesh(parent, new THREE.TubeGeometry(c, tubular || 32, r, 8, !!closed), mat);
}
/* 变截面管：用于颈部、羽臂等有机形体 */
function variableTube(parent, pts, rFn, mat, tubular, radial) {
  tubular = tubular || 44; radial = radial || 12;
  const curve = new THREE.CatmullRomCurve3(pts.map(p => V(p[0], p[1], p[2])), false, 'catmullrom', 0.5);
  const frames = curve.computeFrenetFrames(tubular, false);
  const pos = [], nor = [], uvs = [], idx = [];
  for (let i = 0; i <= tubular; i++) {
    const t = i / tubular, p = curve.getPointAt(t);
    const rr = rFn(t);
    const rx = Array.isArray(rr) ? rr[0] : rr, ry = Array.isArray(rr) ? rr[1] : rr;
    const N = frames.normals[i], B = frames.binormals[i];
    for (let j = 0; j <= radial; j++) {
      const a = j / radial * Math.PI * 2, ca = Math.cos(a), sa = Math.sin(a);
      const nx = N.x * ca + B.x * sa, ny = N.y * ca + B.y * sa, nz = N.z * ca + B.z * sa;
      const nl = Math.hypot(nx, ny, nz) || 1;
      pos.push(p.x + N.x * ca * rx + B.x * sa * ry,
               p.y + N.y * ca * rx + B.y * sa * ry,
               p.z + N.z * ca * rx + B.z * sa * ry);
      nor.push(nx / nl, ny / nl, nz / nl);
      uvs.push(j / radial, t);
    }
  }
  for (let i = 0; i < tubular; i++) for (let j = 0; j < radial; j++) {
    const a = i * (radial + 1) + j, b = a + radial + 1;
    idx.push(a, b, a + 1, b, b + 1, a + 1);
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('normal', new THREE.Float32BufferAttribute(nor, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  g.setIndex(idx);
  return mesh(parent, g, mat);
}

function loft(sections, sides) {
  sides = sides || 12;
  const pos = [], uvs = [], idx = [];
  for (let i = 0; i < sections.length; i++) {
    const s = sections[i];
    for (let j = 0; j <= sides; j++) {
      const a = j / sides * Math.PI * 2;
      pos.push(Math.cos(a) * s.rx, s.cy + Math.sin(a) * s.ry, s.z);
      uvs.push(j / sides, i / (sections.length - 1));
    }
  }
  for (let i = 0; i < sections.length - 1; i++) for (let j = 0; j < sides; j++) {
    const a = i * (sides + 1) + j, b = a + sides + 1;
    idx.push(a, b, a + 1, b, b + 1, a + 1);
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  g.setIndex(idx);
  g.computeVertexNormals();
  return g;
}

const L_RING = Math.PI * RING_R, L_COG = Math.PI * COG_R;
const L_CHAIN = L_TOP + L_RING + L_BOT + L_COG;
function chainPoint(s) {
  s = ((s % L_CHAIN) + L_CHAIN) % L_CHAIN;
  if (s < L_TOP) { const t = s / L_TOP; return { y: lerp(P2T.y, P1T.y, t), z: lerp(P2T.z, P1T.z, t) }; }
  s -= L_TOP;
  if (s < L_RING) { const a = Math.PI / 2 + s / RING_R; return { y: C1.y + RING_R * Math.sin(a), z: C1.z + RING_R * Math.cos(a) }; }
  s -= L_RING;
  if (s < L_BOT) { const t = s / L_BOT; return { y: lerp(P1B.y, P2B.y, t), z: lerp(P1B.z, P2B.z, t) }; }
  s -= L_BOT;
  const a = 3 * Math.PI / 2 + s / COG_R;
  return { y: C2.y + COG_R * Math.sin(a), z: C2.z + COG_R * Math.cos(a) };
}

/* ---------------------------------------------------------------- 动画 */
const FWD_UP = V(0, 0.6, -0.8).normalize();
const tmpA = new THREE.Vector3(), tmpB = new THREE.Vector3(), tmpC2 = new THREE.Vector3();
function updateRide(dt) {
  const omega = rpm * Math.PI * 2 / 60;
  theta += omega * dt;
  const speed = omega * GEAR * R;
  distance += speed * dt;
  const wh = theta * GEAR;
  const bob = 0.022 * Math.sin(2 * theta) + 0.011 * Math.cos(theta);

  /* 车轮 / 传动 */
  frontSpin.rotation.x = -wh;
  rearSpin.rotation.x = -wh;
  ringGrp.rotation.x = -theta;
  cogGrp.rotation.x = -wh;
  chainOffset += omega * RING_R * dt;

  for (let i = 0; i < N_LINKS; i++) {
    const s = chainOffset + i * L_CHAIN / N_LINKS;
    const p = chainPoint(s), a = chainPoint(s - 0.003), b = chainPoint(s + 0.003);
    const ty = b.y - a.y, tz = b.z - a.z;
    const tl = Math.hypot(ty, tz) || 1;
    const m = links[i];
    m.position.set(CHAIN_X, p.y, p.z);
    m.quaternion.setFromUnitVectors(UP, tmpA.set(0, ty / tl, tz / tl));
  }

  /* 曲柄、脚踏、蹼足 */
  for (let i = 0; i < 2; i++) {
    const side = pedals[i].side;
    const a = -theta + (side > 0 ? Math.PI : 0);
    const pedal = tmpB.set(side * PEDAL_X,
      BB.y - CRANK_R * Math.cos(a), BB.z - CRANK_R * Math.sin(a));
    setSegment(cranks[i].arm, V(side * 0.145, BB.y, BB.z), pedal);
    pedals[i].pg.position.copy(pedal);
    pedals[i].pg.rotation.set(0, 0, 0);

    /* 两段 IK：髋 → 膝 → 踝 */
    const lg = legs[i];
    const hip = V(side * PEDAL_X, 1.20 + bob * 0.35, 0.30);
    const ankle = V(pedal.x, pedal.y + 0.078, pedal.z + 0.02);
    const dir = tmpC2.subVectors(ankle, hip);
    const d = Math.min(dir.length(), L1 + L2 - 0.002);
    dir.normalize();
    const segA = (L1 * L1 - L2 * L2 + d * d) / (2 * d);
    const segH = Math.sqrt(Math.max(0, L1 * L1 - segA * segA));
    const perp = FWD_UP.clone().addScaledVector(dir, -FWD_UP.dot(dir)).normalize();
    const knee = hip.clone().addScaledVector(dir, segA).addScaledVector(perp, segH);
    setSegment(lg.thigh, hip, knee);
    setSegment(lg.shin, knee, ankle);
    lg.knee.position.copy(knee);
    lg.ankle.position.copy(ankle);
  }

  /* 身体起伏、平衡与翅膀 */
  body.position.y = bob;
  body.rotation.z = 0.028 * Math.sin(theta);
  body.rotation.x = 0.017 * Math.sin(2 * theta + 0.6);
  body.rotation.y = 0.018 * Math.sin(theta * 0.5);
  rig.rotation.z = 0.009 * Math.sin(theta);
  rig.rotation.y = 0.011 * Math.sin(theta * 0.5);
  rig.position.y = 0.007 * Math.sin(2 * theta + 1.1);

  head.rotation.y = mouseX * 0.11 + 0.018 * Math.sin(theta);
  head.rotation.x = -0.028 * Math.sin(2 * theta) + flap * 0.14 + mouseY * 0.045;
  head.rotation.z = 0.012 * Math.sin(theta + 1);
  crown.rotation.z = 0.09 * Math.sin(theta * 1.5) + flap * 0.2;
  crown.rotation.x = 0.06 * Math.sin(2 * theta);
  tail.rotation.y = 0.06 * Math.sin(theta + 0.8);
  tail.rotation.x = 0.03 * Math.sin(2 * theta);
  pouch.scale.set(1 + flap * 0.08, 1 + 0.045 * Math.sin(2 * theta + 1) + flap * 0.16, 1 + flap * 0.05);
  for (const w of wings) {
    const amp = 0.055 * Math.sin(theta + 0.4) + flap * 0.55;
    w.w.rotation.z = w.side * amp;
    w.w.rotation.x = 0.045 * Math.sin(2 * theta) + flap * 0.18;
    w.w.rotation.y = w.side * (0.02 + flap * 0.13);
  }
  /* 眨眼 */
  const blink = Math.pow(Math.max(0, Math.sin(performance.now() * 0.0011)), 34);
  for (const e of eyes) e.scale.y = 0.042 * (1 - blink * 0.88);

  /* 羽臂：肩随身体起伏，末端锁定握把 */
  for (const arm of arms) {
    const s = arm.side;
    const shoulder = V(s * 0.30, 1.80 + bob, -0.06);
    const grip = V(s * 0.322, 1.225, -0.99);
    const elbow = shoulder.clone().lerp(grip, 0.52).add(V(s * 0.075, -0.10, 0.02));
    setSegment(arm.upper, shoulder, elbow);
    setSegment(arm.fore, elbow, grip);
    arm.elbow.position.copy(elbow);
    arm.tip.position.copy(grip).add(V(0, 0.015, -0.02));
    arm.tip.rotation.x = 0.2;
  }

  /* 环境反向滚动 + 纹理流动 */
  const dz = speed * dt;
  sceneryA.position.z = (sceneryA.position.z + dz) % PER_A;
  sceneryB.position.z = (sceneryB.position.z + dz) % PER_B;
  grassTex.offset.y -= dz / 5;
  roadTex.offset.y -= dz / 4;
  const arr = pGeo.attributes.position.array;
  for (let i = 0; i < P_N; i++) {
    arr[i * 3 + 2] += dz * 0.8;
    arr[i * 3] += Math.sin(theta * 0.35 + i) * dt * 0.06;
    if (arr[i * 3 + 2] > 24) { arr[i * 3 + 2] = -100; arr[i * 3] = (Math.random() - 0.5) * 26; }
  }
  pGeo.attributes.position.needsUpdate = true;
  return paused ? 0 : speed;
}

function updateWorldAmbient(dt) {
  for (const c of clouds) {
    c.position.x += dt * 0.42;
    if (c.position.x > 68) c.position.x = -68;
  }
}
function updateCamera(dt) {
  const k = Math.min(1, dt * 4.2);
  cam.az += (cam.tAz - cam.az) * k;
  cam.pol += (cam.tPol - cam.pol) * k;
  cam.dist += (cam.tDist - cam.dist) * k;
  const sp = Math.sin(cam.pol), cp = Math.cos(cam.pol);
  camera.position.set(
    cam.target.x + cam.dist * sp * Math.sin(cam.az),
    cam.target.y + cam.dist * cp,
    cam.target.z + cam.dist * sp * Math.cos(cam.az)
  );
  camera.lookAt(cam.target);
  sky.position.copy(camera.position);
}

let last = performance.now(), hudT = 0, currentSpeed = 0;
function frame(now) {
  const dt = Math.min((now - last) / 1000, 0.05);
  last = now;
  const pdt = paused ? 0 : dt;
  flap *= Math.exp(-dt * 2.3);
  currentSpeed = updateRide(pdt);
  updateWorldAmbient(dt);
  applyTheme(dt);
  updateCamera(dt);
  hudT += dt;
  if (hudT > 0.12) {
    hudT = 0;
    document.getElementById('s-speed').innerHTML = (currentSpeed * 3.6).toFixed(1) + '<small>km/h</small>';
    const d = distance;
    document.getElementById('s-dist').innerHTML = d < 1000
      ? Math.floor(d) + '<small>m</small>'
      : (d / 1000).toFixed(2) + '<small>km</small>';
  }
  renderer.render(scene, camera);
  requestAnimationFrame(frame);
}
