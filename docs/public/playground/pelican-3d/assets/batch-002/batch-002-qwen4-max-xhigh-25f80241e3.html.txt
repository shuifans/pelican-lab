function radialTex(size, fn) {
  const d = new Uint8Array(size * size * 4);
  for (let y = 0; y < size; y++) for (let x = 0; x < size; x++) {
    const u = (x + .5) / size * 2 - 1, v = (y + .5) / size * 2 - 1;
    const c = fn(Math.min(1, Math.hypot(u, v)), u, v);
    const i = (y * size + x) * 4;
    d[i] = c[0]; d[i + 1] = c[1]; d[i + 2] = c[2]; d[i + 3] = c[3];
  }
  const t = new THREE.DataTexture(d, size, size, THREE.RGBAFormat);
  t.needsUpdate = true; t.colorSpace = THREE.SRGBColorSpace;
  t.minFilter = THREE.LinearMipmapLinearFilter; t.magFilter = THREE.LinearFilter;
  t.generateMipmaps = true; t.anisotropy = MAX_ANISO;
  return t;
}

/** Overlapping plumage scallops → normal + roughness. Typed arrays only. */
function makePlumage() {
  const W = 512, H = 512, ROWS = 9, COLS = 7;
  const hgt = new Float32Array(W * H);
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
    const v = y / H, u = x / W;
    const rv = v * ROWS, row = Math.floor(rv), fv = rv - row;
    const cu = u * COLS + (row % 2) * 0.5, fu = cu - Math.floor(cu);
    const dx = (fu - 0.5) * 2;
    const w = 1 - 0.30 * fv * fv;
    const inside = 1 - smoothstep(clamp((Math.abs(dx) / w - 0.55) / 0.45, 0, 1));
    const dome = Math.sin(Math.PI * Math.pow(fv, 0.85));
    let h = inside * (0.30 + 0.70 * dome) * (1 - 0.18 * fv);
    h += Math.exp(-Math.pow(dx * 5.5, 2)) * 0.10 * inside;
    h -= smoothstep(clamp((fv - 0.90) / 0.10, 0, 1)) * 0.13 * inside;
    hgt[y * W + x] = h;
  }
  const nrm = new Uint8Array(W * H * 4), rgh = new Uint8Array(W * H * 4);
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
    const l = hgt[y * W + ((x - 1 + W) % W)], r = hgt[y * W + ((x + 1) % W)];
    const d = hgt[((y - 1 + H) % H) * W + x], u2 = hgt[((y + 1) % H) * W + x];
    let nx = (l - r) * 4.0, ny = (d - u2) * 4.0, nz = 1;
    const inv = 1 / Math.hypot(nx, ny, nz);
    const i = (y * W + x) * 4;
    nrm[i] = (nx * inv * 0.5 + 0.5) * 255; nrm[i + 1] = (ny * inv * 0.5 + 0.5) * 255;
    nrm[i + 2] = (nz * inv * 0.5 + 0.5) * 255; nrm[i + 3] = 255;
    const g = clamp(0.70 + (1 - hgt[y * W + x]) * 0.26, 0, 1) * 255;
    rgh[i] = g; rgh[i + 1] = g; rgh[i + 2] = g; rgh[i + 3] = 255;
  }
  const mk = (data, rx, ry) => {
    const t = new THREE.DataTexture(data, W, H, THREE.RGBAFormat);
    t.wrapS = t.wrapT = THREE.RepeatWrapping; t.repeat.set(rx, ry);
    t.minFilter = THREE.LinearMipmapLinearFilter; t.magFilter = THREE.LinearFilter;
    t.generateMipmaps = true; t.anisotropy = MAX_ANISO; t.needsUpdate = true;
    return t;
  };
  return { nrm, rgh, mk };
}

/** Fine diagonal barbs for individual flight feathers. */
function makeBarbs() {
  const W = 256, H = 256, d = new Uint8Array(W * H * 4);
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
    const u = x / W, v = y / H;
    const nx = Math.cos((u * 46 + Math.abs(v - 0.5) * 13) * TAU) * 0.30;
    const i = (y * W + x) * 4;
    d[i] = (nx * 0.5 + 0.5) * 255; d[i + 1] = 128; d[i + 2] = (Math.sqrt(1 - nx * nx) * 0.5 + 0.5) * 255; d[i + 3] = 255;
  }
  const t = new THREE.DataTexture(d, W, H, THREE.RGBAFormat);
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  t.minFilter = THREE.LinearMipmapLinearFilter; t.magFilter = THREE.LinearFilter;
  t.generateMipmaps = true; t.anisotropy = MAX_ANISO; t.needsUpdate = true;
  return t;
}

/** Collapse a static subtree into a single vertex-coloured mesh (fewer draws). */
function bakeMeshes(root, mat) {
  root.updateMatrixWorld(true);
  const inv = new THREE.Matrix4().copy(root.matrixWorld).invert();
  const pos = [], nor = [], uv = [], col = [], idx = [];
  const m = new THREE.Matrix4(), nm = new THREE.Matrix3(), v = new THREE.Vector3();
  let off = 0;
  root.traverse((o) => {
    if (!o.isMesh || o.isInstancedMesh) return;
    const g = o.geometry;
    m.multiplyMatrices(inv, o.matrixWorld); nm.getNormalMatrix(m);
    const p = g.attributes.position, n = g.attributes.normal, u = g.attributes.uv, c = g.attributes.color;
    const base = o.material.color || COL('#ffffff');
    for (let i = 0; i < p.count; i++) {
      v.fromBufferAttribute(p, i).applyMatrix4(m); pos.push(v.x, v.y, v.z);
      if (n) { v.fromBufferAttribute(n, i).applyMatrix3(nm).normalize(); nor.push(v.x, v.y, v.z); } else nor.push(0, 1, 0);
      if (u) uv.push(u.getX(i), u.getY(i)); else uv.push(0, 0);
      if (c) col.push(c.getX(i) * base.r, c.getY(i) * base.g, c.getZ(i) * base.b);
      else col.push(base.r, base.g, base.b);
    }
    if (g.index) { const ia = g.index.array; for (let i = 0; i < ia.length; i++) idx.push(ia[i] + off); }
    else for (let i = 0; i < p.count; i++) idx.push(i + off);
    off += p.count;
  });
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('normal', new THREE.Float32BufferAttribute(nor, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
  g.setIndex(idx);
  const mesh = new THREE.Mesh(g, mat);
  mesh.castShadow = true; mesh.receiveShadow = true;
  return mesh;
}

/** Bake a subset of a parent's children into one mesh, preserving local transforms. */
function bakeChildren(parent, kids, mat) {
  const tmp = new THREE.Group();
  parent.add(tmp);
  for (const k of kids) tmp.add(k);
  const baked = bakeMeshes(tmp, mat);
  parent.remove(tmp);
  parent.add(baked);
  return baked;
}

/* chain path: external tangents between chainring and cog */
function buildChainPath(cogR) {
  const c1 = V3(B.bbX, B.bbY, CHAIN_Z), c2 = V3(B.rearX, B.hubY, CHAIN_Z);
  const d = c2.clone().sub(c1), dist = d.length(), beta = Math.atan2(d.y, d.x);
  const phi = Math.acos(clamp((RING_R - cogR) / dist, -1, 1));
  const pts = [];
  const A = beta - phi, Bb = beta + phi;
  for (let i = 0; i <= 30; i++) { const a = A - (A - (Bb - TAU)) * (i / 30); pts.push(V3(c1.x + Math.cos(a) * RING_R, c1.y + Math.sin(a) * RING_R, CHAIN_Z)); }
  for (let i = 1; i <= 4; i++) { const t = i / 5; pts.push(V3(lerp(c1.x + Math.cos(Bb) * RING_R, c2.x + Math.cos(Bb) * cogR, t), lerp(c1.y + Math.sin(Bb) * RING_R, c2.y + Math.sin(Bb) * cogR, t), CHAIN_Z)); }
  for (let i = 0; i <= 20; i++) { const a = Bb - (Bb - A) * (i / 20); pts.push(V3(c2.x + Math.cos(a) * cogR, c2.y + Math.sin(a) * cogR, CHAIN_Z)); }
  for (let i = 1; i <= 4; i++) { const t = i / 5; pts.push(V3(lerp(c2.x + Math.cos(A) * cogR, c1.x + Math.cos(A) * RING_R, t), lerp(c2.y + Math.sin(A) * cogR, c1.y + Math.sin(A) * RING_R, t), CHAIN_Z)); }
  const cum = [0]; let total = 0;
  for (let i = 1; i <= pts.length; i++) { total += pts[i % pts.length].distanceTo(pts[i - 1]); cum.push(total); }
  return { pts, cum, total };
}
let chainPath = buildChainPath(COG_R);

const _cp = V3(), _ct = V3(), _cq = new THREE.Quaternion(), _cm = new THREE.Matrix4(), _cs = V3();
function chainAt(s, outP, outT) {
  s = ((s % chainPath.total) + chainPath.total) % chainPath.total;
  const cum = chainPath.cum, pts = chainPath.pts;
  let lo = 0, hi = cum.length - 1;
  while (lo < hi - 1) { const mid = (lo + hi) >> 1; if (cum[mid] <= s) lo = mid; else hi = mid; }
  const a = pts[lo % pts.length], b = pts[(lo + 1) % pts.length];
  const seg = cum[lo + 1] - cum[lo] || 1e-6; const t = (s - cum[lo]) / seg;
  outP.copy(a).lerp(b, t); outT.copy(b).sub(a).normalize();
}
function updateChain(phase) {
  for (let i = 0; i < CHAIN_N; i++) {
    chainAt(i * CHAIN_PITCH + phase, _cp, _ct);
    _cq.setFromUnitVectors(V3(1, 0, 0), _ct);
    _cs.set(CHAIN_PITCH * 1.02, 0.0105, i % 2 ? 0.0092 : 0.0062);
    _cm.compose(_cp, _cq, _cs); chainMesh.setMatrixAt(i, _cm);
  }
  chainMesh.instanceMatrix.needsUpdate = true;
}

/** A single feather: cambered vane, pointed calamus, rounded tip, graded colour. */
function featherGeo(len, wid, o = {}) {
  const NL = 10, NW = 6;
  const droop = o.droop ?? 0.10, camber = o.camber ?? -0.34, curl = o.curl ?? 0.05;
  const cBase = COL(o.base || '#F7F0E4'), cTip = COL(o.tip || '#F7F0E4'), cMid = COL(o.mid || o.base || '#F7F0E4');
  const tipAt = o.tipAt ?? 0.72;
  const pos = [], col = [], uv = [], idx = [];
  for (let i = 0; i <= NL; i++) {
    const u = i / NL;
    let w = wid * Math.pow(Math.max(0, Math.sin(Math.PI * Math.pow(u, 1.50))), 0.80);
    if (u > 0.86) w *= Math.sqrt(Math.max(0, 1 - Math.pow((u - 0.86) / 0.14, 2)));
    const x = u * len, dy = -droop * u * u * len + curl * u * u * u * len;
    let c;
    if (u < tipAt) c = cBase.clone().lerp(cMid, smoothstep(u / tipAt));
    else c = cMid.clone().lerp(cTip, smoothstep((u - tipAt) / (1 - tipAt)));
    for (let j = 0; j <= NW; j++) {
      const v = (j / NW) * 2 - 1;
      const wv = v > 0 ? w * 0.72 : w;
      pos.push(x, dy + camber * wv * v * v + (1 - Math.abs(v)) * wid * 0.09, v * wv);
      uv.push(u, (v + 1) / 2);
      const shade = 0.90 + 0.10 * (1 - Math.abs(v));
      col.push(c.r * shade, c.g * shade, c.b * shade);
    }
  }
  for (let i = 0; i < NL; i++) for (let j = 0; j < NW; j++) {
    const a = i * (NW + 1) + j, b = a + NW + 1;
    idx.push(a, b, a + 1, a + 1, b, b + 1);
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  g.setIndex(idx); g.computeVertexNormals();
  return g;
}
function feather(len, wid, o) {

  m.castShadow = true; m.receiveShadow = true; return m;
}

/* ---- skinned neck ---- */
function skinnedTube(len, boneCount, rings, radial, radiusFn, mat) {
  const seg = len / boneCount;
  const pos = [], nor = [], uv = [], idx = [], si = [], sw = [];
  for (let i = 0; i <= rings; i++) {
    const t = i / rings, y = t * len, r = radiusFn(t);
    for (let j = 0; j <= radial; j++) {
      const a = (j / radial) * TAU, nx = Math.cos(a), nz = Math.sin(a);
      pos.push(nx * r, y, nz * r); nor.push(nx, 0, nz); uv.push(j / radial, t);
      const f = y / seg; let i0 = Math.min(Math.floor(f), boneCount - 1);
      const i1 = Math.min(i0 + 1, boneCount - 1), w = clamp(f - i0, 0, 1);
      si.push(i0, i1, 0, 0); sw.push(1 - w, w, 0, 0);
    }
  }
  for (let i = 0; i < rings; i++) for (let j = 0; j < radial; j++) {
    const a = i * (radial + 1) + j, b = a + radial + 1;
    idx.push(a, b, a + 1, a + 1, b, b + 1);
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('normal', new THREE.Float32BufferAttribute(nor, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  g.setAttribute('skinIndex', new THREE.Uint16BufferAttribute(si, 4));
  g.setAttribute('skinWeight', new THREE.Float32BufferAttribute(sw, 4));
  g.setIndex(idx);
  const bones = [];
  for (let i = 0; i < boneCount; i++) {
    const b = new THREE.Bone(); b.position.y = i === 0 ? 0 : seg;
    if (i) bones[i - 1].add(b); bones.push(b);
  }
  const mesh = new THREE.SkinnedMesh(g, mat);
  mesh.add(bones[0]); mesh.bind(new THREE.Skeleton(bones));
  mesh.castShadow = true; mesh.receiveShadow = true; mesh.frustumCulled = false;
  return { mesh, bones, seg, len };
}

const GLSL_NOISE = `
float hash21(vec2 p){ p=fract(p*vec2(123.34,456.21)); p+=dot(p,p+45.32); return fract(p.x*p.y); }
float vnoise(vec2 p){
  vec2 i=floor(p), f=fract(p); f=f*f*(3.0-2.0*f);
  float a=hash21(i), b=hash21(i+vec2(1,0)), c=hash21(i+vec2(0,1)), d=hash21(i+vec2(1,1));
  return mix(mix(a,b,f.x),mix(c,d,f.x),f.y);
}
float fbm(vec2 p){ float s=0.0,a=0.5; for(int i=0;i<4;i++){ s+=a*vnoise(p); p*=2.03; a*=0.5; } return s*1.067; }
`;
const GLSL_FOG = `
uniform vec3 uFogColor; uniform float uFogDen;
vec3 applyFog(vec3 c, float dist){
  float f = 1.0 - exp(-uFogDen*uFogDen*dist*dist);
  return mix(c, uFogColor, clamp(f,0.0,1.0));
}
`;
function shaderMat(o) { return new THREE.ShaderMaterial(o); }

function groundY(x, z) {
  const shore = smoothstep(clamp((z + 15) / 12.5, 0, 1));
  let y = lerp(-1.30, -0.145, shore);
  const dune = smoothstep(clamp((z - 9) / 35, 0, 1));
  y += dune * (1.05 + 0.85 * Math.sin(x * 0.055) + 0.55 * Math.sin(x * 0.13 + 1.7));

}

/* ---------------- boardwalk ----------------
   Typed-array wood textures on a standard material so the deck genuinely
   receives the rider's shadow and picks up the baked sky reflection.      */
const DECK_PLANKS = 8, DECK_PW = 0.165, DECK_TILE = DECK_PLANKS * DECK_PW;
function makeDeckTextures() {
  const W = 1024, H = 192;
  const col = new Uint8Array(W * H * 4), nrm = new Uint8Array(W * H * 4);
  const hgt = new Float32Array(W * H);
  const n2 = (x, y) => {
    const ix = Math.floor(x), iy = Math.floor(y), fx = x - ix, fy = y - iy;
    const sx = fx * fx * (3 - 2 * fx), sy = fy * fy * (3 - 2 * fy);
    const h = (a, b) => { const s = Math.sin(a * 127.1 + b * 311.7) * 43758.5453; return s - Math.floor(s); };
    return lerp(lerp(h(ix, iy), h(ix + 1, iy), sx), lerp(h(ix, iy + 1), h(ix + 1, iy + 1), sx), sy);
  };
  const fb = (x, y) => { let s = 0, a = 0.5, fx = x, fy = y; for (let i = 0; i < 4; i++) { s += a * n2(fx, fy); fx *= 2.03; fy *= 2.03; a *= 0.5; } return s; };
  const base = COL('#93876F'), dark = COL('#4E4638');
  for (let y = 0; y < H; y++) {
    const v = y / (H - 1), z = (0.5 - v) * 3.4;
    for (let x = 0; x < W; x++) {
      const u = x / W, px = u * DECK_PLANKS, id = Math.floor(px), f = px - id;
      const tone = hash1(id * 3.77 + 1.3);
      const gap = smoothstep(clamp(f / 0.055, 0, 1)) * (1 - smoothstep(clamp((f - 0.945) / 0.055, 0, 1)));
      const grain = fb(px * 5.0, v * 34 + tone * 40);
      const rings = fb(px * 62 + tone * 30, v * 5.5);
      let t = 0.40 + tone * 0.44;
      let shade = 0.78 + grain * 0.36 + rings * 0.16;
      const knot = hash1(id * 9.13 + 4.2) > 0.80 ? 1 : 0;
      const kc = Math.hypot((f - 0.5) * 1.0, (v - (0.2 + tone * 0.6)) * 2.4);
      const knotM = knot * (1 - smoothstep(clamp(kc / 0.13, 0, 1)));
      const edge = smoothstep(clamp((Math.abs(z) - 1.50) / 0.10, 0, 1));
      let c = dark.clone().lerp(base, t).multiplyScalar(shade);
      c.lerp(dark.clone().multiplyScalar(0.62), knotM * 0.85);
      c.multiplyScalar(lerp(0.20, 1.0, gap));
      c.lerp(dark.clone().multiplyScalar(1.06), edge * 0.8);
      const nailD = Math.hypot((f - 0.5) * 1.0, (Math.abs(z) - 1.32) * 1.9);
      const nail = (1 - smoothstep(clamp(nailD / 0.045, 0, 1))) * (Math.abs(z) > 1.15 ? 1 : 0);
      c.lerp(COL('#6E7478'), nail * 0.75);
      const i = (y * W + x) * 4;
      col[i] = clamp(c.r, 0, 1) * 255; col[i + 1] = clamp(c.g, 0, 1) * 255; col[i + 2] = clamp(c.b, 0, 1) * 255; col[i + 3] = 255;
      hgt[y * W + x] = gap * 0.85 + grain * 0.12 - knotM * 0.10 - edge * 0.25 + nail * 0.10;
    }
  }
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
    const l = hgt[y * W + ((x - 1 + W) % W)], r = hgt[y * W + ((x + 1) % W)];
    const d = hgt[Math.max(0, y - 1) * W + x], u2 = hgt[Math.min(H - 1, y + 1) * W + x];
    let nx = (l - r) * 5.0, ny = (d - u2) * 5.0, nz = 1;
    const inv = 1 / Math.hypot(nx, ny, nz); nx *= inv; ny *= inv; nz *= inv;
    const i = (y * W + x) * 4;
    nrm[i] = (nx * 0.5 + 0.5) * 255; nrm[i + 1] = (ny * 0.5 + 0.5) * 255; nrm[i + 2] = (nz * 0.5 + 0.5) * 255; nrm[i + 3] = 255;
  }
  const mk = (data, srgb) => {
    const t = new THREE.DataTexture(data, W, H, THREE.RGBAFormat);
    t.wrapS = THREE.RepeatWrapping; t.wrapT = THREE.ClampToEdgeWrapping;
    t.colorSpace = srgb ? THREE.SRGBColorSpace : THREE.NoColorSpace;
    t.minFilter = THREE.LinearMipmapLinearFilter; t.magFilter = THREE.LinearFilter;
    t.generateMipmaps = true; t.anisotropy = MAX_ANISO; t.needsUpdate = true;
    return t;
  };
  return { map: mk(col, true), normalMap: mk(nrm, false) };
}
const deckTex = makeDeckTextures();
const DECK_REPEAT = 420 / DECK_TILE;

function frondGeo(len, wid) {
  const NL = 16, pos = [], col = [], idx = [];
  const cA = COL('#3F7A34'), cB = COL('#7FBF4A'), cC = COL('#C8D96A');
  let vi = 0;
  const push = (x, y, z, c) => { pos.push(x, y, z); col.push(c.r, c.g, c.b); return vi++; };
  // rachis
  const NS = 14;
  const spine = [];
  for (let i = 0; i <= NS; i++) {
    const t = i / NS;
    spine.push([t * len, -Math.pow(t, 1.9) * len * 0.42 + Math.sin(t * Math.PI) * len * 0.10, 0]);
  }
  for (let i = 0; i < NS; i++) {
    const a = spine[i], b = spine[i + 1];
    const w = 0.012 * (1 - i / NS) + 0.003;
    const c = cA.clone().lerp(cB, i / NS);
    const i0 = push(a[0], a[1], -w, c), i1 = push(a[0], a[1], w, c);
    const i2 = push(b[0], b[1], -w, c), i3 = push(b[0], b[1], w, c);
    idx.push(i0, i2, i1, i1, i2, i3);
  }
  // leaflets
  for (let i = 1; i < NS; i++) {
    const t = i / NS, a = spine[i], b = spine[i + 1];
    const lw = wid * Math.sin(Math.PI * Math.pow(t, 0.72)) * (1 - 0.25 * t);
    for (const s of [-1, 1]) {
      const c0 = cA.clone().lerp(cB, t), c1 = cB.clone().lerp(cC, t * 0.8);
      const droop = -lw * 0.34;
      const i0 = push(a[0], a[1], 0, c0);
      const i1 = push(b[0], b[1], 0, c0);
      const i2 = push(a[0] + lw * 0.30, a[1] + droop, s * lw, c1);
      const i3 = push(b[0] + lw * 0.16, b[1] + droop * 1.05, s * lw * 0.82, c1);
      if (s > 0) idx.push(i0, i2, i1, i1, i2, i3); else idx.push(i0, i1, i2, i1, i3, i2);
    }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
  g.setIndex(idx); g.computeVertexNormals();
  return g;
}

/* wind sway injected into the standard material */
const windU = { value: 0.7 }, timeU = { value: 0 };
function addWind(mat, strength) {
  mat.onBeforeCompile = (sh) => {
    sh.uniforms.uWind = windU; sh.uniforms.uT = timeU;
    sh.vertexShader = 'uniform float uWind; uniform float uT;\n' + sh.vertexShader.replace(
      '#include <begin_vertex>',
      `#include <begin_vertex>
       #ifdef USE_INSTANCING
         vec3 ip = vec3(instanceMatrix[3][0],instanceMatrix[3][1],instanceMatrix[3][2]);
       #else
         vec3 ip = vec3(0.0);
       #endif
       float ph = ip.x*0.7 + ip.z*0.31;
       float sway = sin(uT*1.7 + ip.x*0.35 + ph) * 0.5 + sin(uT*3.1 + ip.z*0.5)*0.25;
       float infl = max(transformed.y,0.0);
       transformed.x += sway*uWind*${strength.toFixed(3)}*infl*infl*0.35;
       transformed.z += sway*uWind*${(strength * 0.55).toFixed(3)}*infl*infl*0.22;`
    );
  };
  mat.customProgramCacheKey = () => 'wind' + strength;
}

  }`, { tSrc: { value: null }, uTexel: { value: new THREE.Vector2() }, uThresh: { value: 1.20 }, uSoft: { value: 0.80 } });

const compositeMat = passMat(`

});

const fxaaMat = passMat(`

  void main(){

function disposeTargets() {
  if (sceneRT) { sceneRT.depthTexture && sceneRT.depthTexture.dispose(); sceneRT.dispose(); }
  if (ldrRT) ldrRT.dispose();
  bloomA.forEach(t => t.dispose()); bloomB.forEach(t => t.dispose());
  if (dofA) dofA.dispose(); if (dofB) dofB.dispose();
  bloomA = []; bloomB = [];
}
function buildTargets() {
  disposeTargets();
  const dpr = Math.min(devicePixelRatio || 1, 2);
  let pr = dpr * Q.scale;
  const px = innerWidth * innerHeight * pr * pr;
  if (px > Q.budget) pr *= Math.sqrt(Q.budget / px);
  pr = Math.max(0.5, pr);
  renderer.setPixelRatio(pr);
  renderer.setSize(innerWidth, innerHeight, false);
  RW = Math.max(2, renderer.domElement.width);
  RH = Math.max(2, renderer.domElement.height);
  canvas.style.width = innerWidth + 'px'; canvas.style.height = innerHeight + 'px';

  const dt = new THREE.DepthTexture(RW, RH);
  dt.type = THREE.UnsignedIntType; dt.format = THREE.DepthFormat;
  dt.minFilter = THREE.NearestFilter; dt.magFilter = THREE.NearestFilter;
  sceneRT = new THREE.WebGLRenderTarget(RW, RH, { ...rtOpts, depthBuffer: true, depthTexture: dt });
  ldrRT = new THREE.WebGLRenderTarget(RW, RH, { ...rtOpts, type: THREE.UnsignedByteType, depthBuffer: false });
  let w = RW >> 1, h = RH >> 1;
  for (let i = 0; i < BLOOM_LEVELS; i++) {
    w = Math.max(2, w); h = Math.max(2, h);
    bloomA.push(new THREE.WebGLRenderTarget(w, h, rtOpts));
    bloomB.push(new THREE.WebGLRenderTarget(w, h, rtOpts));
    w >>= 1; h >>= 1;
  }
  dofA = new THREE.WebGLRenderTarget(Math.max(2, RW >> 1), Math.max(2, RH >> 1), rtOpts);
  dofB = new THREE.WebGLRenderTarget(Math.max(2, RW >> 1), Math.max(2, RH >> 1), rtOpts);
  compositeMat.uniforms.uRes.value.set(RW, RH);
  compositeMat.uniforms.uLevels.value = Q.levels;
  fxaaMat.uniforms.uTexel.value.set(1 / RW, 1 / RH);
  sprayU.uPx.value = RH / Math.max(1, innerHeight);
  sprayGeo.setDrawRange(0, Q.spray);
  if (keyLight.shadow.mapSize.x !== Q.shadow) {
    keyLight.shadow.mapSize.set(Q.shadow, Q.shadow);
    if (keyLight.shadow.map) { keyLight.shadow.map.dispose(); keyLight.shadow.map = null; }
  }

}
/** Lens shift keeps the rider centred in the space the console does not cover. */
function updateLensShift() {
  const w = innerWidth, h = innerHeight;
  const panel = $('#panel');
  let ox = 0, oy = 0;
  if (!panel.classList.contains('hidden')) {
    const r = panel.getBoundingClientRect();
    if (w > 860) ox = (r.width + 16) * 0.5;
    else oy = (r.height + 8) * 0.46;
  }
  if (ox > 0.5 || oy > 0.5) camera.setViewOffset(w, h, ox, oy, w, h);
  else { camera.clearViewOffset(); camera.aspect = w / h; }
  camera.updateProjectionMatrix();
}

function renderFrame() {
  renderer.setRenderTarget(sceneRT);
  renderer.clear();
  renderer.render(scene, camera);

  // DOF source blur
  if (Q.dof && S.dof > 0.001) {
    copyMat.uniforms.tSrc.value = sceneRT.texture; blit(copyMat, dofA);
    const tx = 1 / dofA.width, ty = 1 / dofA.height;
    for (const r of [1.6, 3.4]) {
      blurMat.uniforms.tSrc.value = dofA.texture; blurMat.uniforms.uDir.value.set(tx * r, 0); blit(blurMat, dofB);
      blurMat.uniforms.tSrc.value = dofB.texture; blurMat.uniforms.uDir.value.set(0, ty * r); blit(blurMat, dofA);
    }
  }
  // bloom chain
  brightMat.uniforms.tSrc.value = sceneRT.texture;
  brightMat.uniforms.uTexel.value.set(1 / RW, 1 / RH);
  blit(brightMat, bloomA[0]);
  for (let i = 0; i < Q.levels; i++) {
    const a = bloomA[i], b = bloomB[i];
    blurMat.uniforms.tSrc.value = a.texture; blurMat.uniforms.uDir.value.set(1 / a.width, 0); blit(blurMat, b);
    blurMat.uniforms.tSrc.value = b.texture; blurMat.uniforms.uDir.value.set(0, 1 / a.height); blit(blurMat, a);
    if (i < Q.levels - 1) { copyMat.uniforms.tSrc.value = a.texture; blit(copyMat, bloomA[i + 1]); }
  }
  const cu = compositeMat.uniforms;
  cu.tScene.value = sceneRT.texture; cu.tDepth.value = sceneRT.depthTexture; cu.tDof.value = dofA.texture;
  cu.tB0.value = bloomA[0].texture; cu.tB1.value = bloomA[1].texture; cu.tB2.value = bloomA[2].texture;
  cu.tB3.value = bloomA[Math.min(3, Q.levels - 1)].texture; cu.tB4.value = bloomA[Math.min(4, Q.levels - 1)].texture;
  cu.uDof.value = Q.dof ? S.dof : 0;
  blit(compositeMat, ldrRT);
  fxaaMat.uniforms.tSrc.value = ldrRT.texture;
  blit(fxaaMat, null);
  renderer.setRenderTarget(null);
}

function updateCamera(dt) {
  if (!CAM.dragging) {
    CAM.dTheta += CAM.flickT * dt; CAM.dPhi += CAM.flickP * dt;
    CAM.flickT *= Math.exp(-4.2 * dt); CAM.flickP *= Math.exp(-4.2 * dt);
  }
  if (!CAM.dragging && S.orbit > 0.001) CAM.dTheta += S.orbit * 0.22 * dt;
  CAM.dPhi = clamp(CAM.dPhi, 0.16, 1.595);
  CAM.dRadius = clamp(CAM.dRadius, 1.05, 16);
  const k = 1 - Math.exp(-9.5 * dt);
  CAM.theta = lerp(CAM.theta, CAM.dTheta, k);
  CAM.phi = lerp(CAM.phi, CAM.dPhi, k);
  CAM.radius = lerp(CAM.radius, CAM.dRadius, k);
  CAM.target.lerp(CAM.dTarget, k);
  const sp = Math.sin(CAM.phi);
  const halfW = CAM.radius * Math.tan(camera.fov * Math.PI / 360) * camera.aspect;
  const fit = clamp(1.12 / Math.max(halfW, 0.05), 1, 2.6);
  const rad = CAM.radius * fit;

const A = { ctx: null, master: null, on: true, ready: false, chain: null, wind: null, surf: null, tire: null };
function noiseBuffer(ctx, sec) {
  const b = ctx.createBuffer(1, Math.floor(ctx.sampleRate * sec), ctx.sampleRate);
  const d = b.getChannelData(0);
  let last = 0;
  for (let i = 0; i < d.length; i++) { const w = Math.random() * 2 - 1; last = (last + 0.02 * w) / 1.02; d[i] = w * 0.6 + last * 3.2; }
  return b;
}
function chainBuffer(ctx) {
  const N = 40, sec = 1, len = Math.floor(ctx.sampleRate * sec);
  const b = ctx.createBuffer(1, len, ctx.sampleRate), d = b.getChannelData(0);
  for (let k = 0; k < N; k++) {
    const start = Math.floor((k / N) * len), dur = Math.floor(ctx.sampleRate * 0.007);
    const amp = (k % 2 ? 0.55 : 1.0) * 0.5;
    for (let i = 0; i < dur && start + i < len; i++) {
      const t = i / dur;
      d[start + i] += (Math.random() * 2 - 1) * Math.exp(-t * 9) * amp * (0.6 + 0.4 * Math.sin(t * 90));
    }
  }
  return b;
}
function audioInit() {
  if (A.ctx) return;
  const AC = window.AudioContext || window.webkitAudioContext;
  if (!AC) { A.on = false; return; }
  const ctx = new AC(); A.ctx = ctx;
  const comp = ctx.createDynamicsCompressor();
  comp.threshold.value = -16; comp.ratio.value = 5; comp.attack.value = 0.004; comp.release.value = 0.22;
  const master = ctx.createGain(); master.gain.value = A.on ? 0.85 : 0;
  master.connect(comp); comp.connect(ctx.destination); A.master = master;

  const nb = noiseBuffer(ctx, 2.5);
  const mkLoop = (buf, type, freq, q, gain) => {
    const src = ctx.createBufferSource(); src.buffer = buf; src.loop = true;
    const f = ctx.createBiquadFilter(); f.type = type; f.frequency.value = freq; f.Q.value = q;
    const g = ctx.createGain(); g.gain.value = gain;
    src.connect(f); f.connect(g); g.connect(master); src.start();
    return { src, f, g };
  };
  A.wind = mkLoop(nb, 'bandpass', 520, 0.6, 0.0);
  A.surf = mkLoop(nb, 'lowpass', 620, 0.9, 0.05);
  A.tire = mkLoop(nb, 'bandpass', 1500, 1.4, 0.0);
  const cs = ctx.createBufferSource(); cs.buffer = chainBuffer(ctx); cs.loop = true;
  const cf = ctx.createBiquadFilter(); cf.type = 'bandpass'; cf.frequency.value = 2600; cf.Q.value = 1.1;
  const cg = ctx.createGain(); cg.gain.value = 0;
  cs.connect(cf); cf.connect(cg); cg.connect(master); cs.start();
  A.chain = { src: cs, f: cf, g: cg };
  // slow surf swell
  const lfo = ctx.createOscillator(); lfo.frequency.value = 0.11;
  const lg = ctx.createGain(); lg.gain.value = 0.030;
  lfo.connect(lg); lg.connect(A.surf.g.gain); lfo.start();
  A.ready = true;
}
function audioUnlock() {
  if (navigator.userActivation && !navigator.userActivation.hasBeenActive) return;

  if (A.ctx && A.ctx.state === 'suspended') A.ctx.resume();
}

function envGain(node, t0, peak, atk, dec) {
  node.gain.cancelScheduledValues(t0);
  node.gain.setValueAtTime(0.0001, t0);
  node.gain.exponentialRampToValueAtTime(Math.max(peak, 0.0002), t0 + atk);
  node.gain.exponentialRampToValueAtTime(0.0001, t0 + atk + dec);
}
function sfxBell() {
  if (!A.ready || !A.on) return;
  const ctx = A.ctx, t = ctx.currentTime;
  const partials = [[1, 0.30, 1.9], [2.76, 0.16, 1.3], [5.40, 0.09, 0.8], [8.93, 0.05, 0.5]];
  for (const [m, g, d] of partials) {
    const o = ctx.createOscillator(); o.type = 'sine'; o.frequency.value = 1720 * m;
    const gn = ctx.createGain(); o.connect(gn); gn.connect(A.master);
    envGain(gn, t, g, 0.004, d); o.start(t); o.stop(t + d + 0.1);
  }
  const n = ctx.createBufferSource(); n.buffer = noiseBuffer(ctx, 0.05);

  const gn = ctx.createGain(); n.connect(f); f.connect(gn); gn.connect(A.master);
  envGain(gn, t, 0.14, 0.002, 0.05); n.start(t); n.stop(t + 0.09);
}
function sfxSquawk() {
  if (!A.ready || !A.on) return;
  const ctx = A.ctx, t = ctx.currentTime;
  for (let k = 0; k < 2; k++) {
    const t0 = t + k * 0.20;
    const o = ctx.createOscillator(); o.type = 'sawtooth';
    o.frequency.setValueAtTime(330, t0);

const _ikd = V3(), _ikp = V3();
function ik2(H, T, L1, L2, pole, outK, outEnd) {
  _ikd.copy(T).sub(H);
  let dist = _ikd.length();
  const maxD = (L1 + L2) * 0.9985, minD = Math.abs(L1 - L2) + 0.004;
  dist = clamp(dist, minD, maxD);
  _ikd.normalize();
  const a = (L1 * L1 - L2 * L2 + dist * dist) / (2 * dist);
  const h = Math.sqrt(Math.max(0, L1 * L1 - a * a));
  _ikp.copy(pole).addScaledVector(_ikd, -pole.dot(_ikd));
  if (_ikp.lengthSq() < 1e-9) { _ikp.set(0, 0, 1).addScaledVector(_ikd, -_ikd.z); }
  _ikp.normalize();
  outK.copy(H).addScaledVector(_ikd, a).addScaledVector(_ikp, h);
  outEnd.copy(H).addScaledVector(_ikd, dist);
}

  R.speed = damp(R.speed, speed, 6, dt);
  R.crank -= cad * dt;
  R.wheel -= cad * S.gear * dt;

  /* hop physics */
  if (R.hopVel !== 0 || R.hop > 0) {
    R.hopVel -= 9.4 * dt; R.hop += R.hopVel * dt;
    if (R.hop <= 0) { R.hop = 0; if (R.hopVel < -0.4) sfxThud(clamp(-R.hopVel / 2.4, 0.2, 1)); R.hopVel = 0; }
  }
  /* flap spring */
  if (R.flapVel > 0 || R.flap > 0) {
    R.flap += R.flapVel * dt; R.flapVel -= 13 * dt;
    if (R.flap <= 0) { R.flap = 0; R.flapVel = 0; }
    R.flap = clamp(R.flap, 0, 1);
  }
  R.bell = Math.max(0, R.bell - dt * 1.5);
  R.squawk = Math.max(0, R.squawk - dt * 1.35);

  const cr = R.crank;
  const swayAmt = S.sway * (0.35 + 0.65 * clamp(S.cadence / 80, 0, 1.3));
  const bobAmt = S.bob * (0.35 + 0.65 * clamp(S.cadence / 80, 0, 1.3));
  const leanTarget = Math.sin(cr) * 0.030 * swayAmt + Math.sin(R.t * 0.31) * 0.012 * S.wind;
  R.lean = damp(R.lean, leanTarget, 8, dt);
  R.steer = damp(R.steer, Math.sin(cr + 0.9) * 0.030 * swayAmt + Math.sin(R.t * 0.23) * 0.020, 7, dt);

  rig.rotation.x = R.lean;
  rig.rotation.z = Math.sin(cr * 2 + 0.5) * 0.006 * bobAmt;
  rig.position.y = R.hop;
  contact.position.y = 0.012 - R.hop;
  contact.material.opacity = 0.40 * clamp(1 - R.hop * 2.2, 0.08, 1);
  contact.scale.setScalar(1 + R.hop * 0.5);

  /* steering assembly rotates about the head axis */
  steerQ.setFromAxisAngle(steerAxis, R.steer);
  frontEnd.quaternion.copy(steerQ);
  steerOff.copy(headBot).applyQuaternion(steerQ);
  frontEnd.position.copy(headBot).sub(steerOff);

  /* torso */
  pelvis.position.set(
    BODY_POS.x + Math.sin(cr) * 0.007 * bobAmt,
    BODY_POS.y + Math.sin(cr * 2 + 0.35) * 0.016 * bobAmt - Math.abs(Math.sin(cr)) * 0.004,
    BODY_POS.z + Math.sin(cr) * 0.010 * swayAmt);
  pelvis.rotation.set(
    Math.sin(cr) * 0.062 * swayAmt,
    Math.sin(cr * 2 + 1.1) * 0.018 * swayAmt - R.steer * 0.5,
    0.020 + Math.sin(cr * 2 + 0.6) * 0.024 * bobAmt - R.hop * 0.12);
  pelvis.updateMatrix();
  tailFan.rotation.set(Math.sin(cr) * 0.10 * swayAmt, Math.sin(cr * 2) * 0.07, 0.30 + Math.sin(R.t * 2.4) * 0.06 * (0.3 + S.wind));

  /* neck */
  let cum = 0;
  for (let i = 0; i < 6; i++) {
    const wave = Math.sin(R.t * 1.5 + i * 0.8) * 0.018 * (0.4 + S.wind * 0.6)
               + Math.sin(cr * 2 + i * 0.5) * 0.016 * bobAmt;
    const a = NECK_BASE_ANG[i] * S.neck + wave;
    neck.bones[i].rotation.z = a;
    neck.bones[i].rotation.x = Math.sin(cr + i * 0.6) * 0.020 * swayAmt;
    cum += a;
  }
  /* head: level, stabilised, occasionally glances at the viewer */
  glanceTimer -= dt;
  if (glanceTimer < 0) { glanceTimer = rnd(5.5, 11); glance = 1; }
  glance = Math.max(0, glance - dt * 0.55);
  const gEnv = Math.sin(clamp(glance, 0, 1) * Math.PI);
  tmpV.copy(camera.position); rig.worldToLocal(tmpV); tmpV.sub(headWorld);
  const yaw = clamp(Math.atan2(-tmpV.z, Math.max(tmpV.x, 0.05)), -0.62, 0.62);
  headPivot.rotation.z = -cum - 0.05 - pelvis.rotation.z * 0.55 + R.squawk * 0.16 - R.flap * 0.10;
  headPivot.rotation.y = yaw * gEnv * 0.85 + Math.sin(R.t * 0.6) * 0.035;
  headPivot.rotation.x = -pelvis.rotation.x * 0.62;
  head.rotation.z = Math.sin(cr * 2 + 0.9) * 0.014 * bobAmt;

  /* blink */
  R.nextBlink -= dt;
  if (R.nextBlink < 0) { R.nextBlink = rnd(2.2, 6.5); R.blink = 1; }
  R.blink = Math.max(0, R.blink - dt * 7.5);
  const bl = Math.sin(clamp(R.blink, 0, 1) * Math.PI);
  for (const e of eyes) e.grp.scale.y = 1 - 0.88 * bl;

  /* bill + pouch */
  lowerBill.rotation.z = -0.26 * R.squawk - 0.02;
  const pTarget = 0.086 * S.pouch * (1 + 0.09 * Math.sin(cr * 2 + 1.1) + 0.42 * R.squawk + 0.16 * R.flap);
  pouchV += (pTarget - pouchCur) * 120 * dt; pouchV *= Math.exp(-8.5 * dt); pouchCur += pouchV * dt;
  setPouchDepth(clamp(pouchCur, 0.004, 0.42));
  pouchGlow.intensity = 0.25 + R.squawk * 0.8;

  /* bell + fish react */
  bellGrp.rotation.z = Math.sin(R.t * 46) * 0.16 * R.bell;
  bellGrp.children[2].rotation.z = -0.5 * R.bell;
  basket.userData.fish.rotation.z = 0.62 + Math.sin(R.t * 3.1 + 1) * 0.06 + Math.sin(cr * 2) * 0.05 * bobAmt;
  basket.rotation.z = Math.sin(cr * 2 + 0.4) * 0.012 * bobAmt;

  /* ---- legs ---- */
  let ji = 0;
  for (const leg of [legL, legR]) {
    H.copy(leg.side > 0 ? HIP_L : HIP_R);
    H.y -= (S.plump - 1) * 0.10; H.z *= 1 + (S.plump - 1) * 0.35;
    H.applyMatrix4(pelvis.matrix);
    const th = cr + (leg.side > 0 ? 0 : Math.PI);
    P.set(B.bbX - B.crank * Math.sin(th), B.bbY + B.crank * Math.cos(th), leg.side * B.pedalZ);
    T.copy(P); T.y += 0.024; T.x += 0.004;
    POLE.set(-1, 0.20, leg.side * 0.12);
    ik2(H, T, LEG_L1, LEG_L2, POLE, K, E);
    orient(leg.thigh, H, K); orient(leg.shank, K, E);
    leg.knee.position.copy(K); leg.cuff.position.copy(K); leg.ankle.position.copy(E);
    leg.foot.position.copy(E);
    leg.foot.rotation.set(leg.side * 0.04, 0, -0.06 + Math.sin(th) * 0.13);
    jointPts[ji++].copy(H); jointPts[ji++].copy(K); jointPts[ji++].copy(E);
  }
  /* ---- wings ---- */
  const spread = clamp(S.wing * 0.85 + R.flap * 0.9, 0, 1);
  const flapWave = Math.sin(R.t * 15) * 0.5 + 0.5;
  for (const w of [wingL, wingR]) {
    SH.copy(w.side > 0 ? SHO_L : SHO_R).applyMatrix4(pelvis.matrix);
    GR.set(B.gripX, B.gripY - 0.012, w.side * B.gripZ).applyQuaternion(steerQ).add(frontEnd.position);
    tmpV.set(SH.x - 0.03, SH.y + 0.24 + flapWave * 0.20, SH.z + w.side * (0.34 + flapWave * 0.12));
    GR.lerp(tmpV, R.flap);
    GR.y += S.wing * 0.035; GR.z += w.side * S.wing * 0.030;
    POLE.set(-0.55, 1.0, w.side * 0.30);
    ik2(SH, GR, WING_L1, WING_L2, POLE, K, E);
    orient(w.humerus, SH, K); orient(w.forearm, K, E);
    w.elbow.position.copy(K); w.wrist.position.copy(E);
    w.hand.position.copy(E);
    w.hand.quaternion.setFromUnitVectors(XV, tmpV.copy(E).sub(K).normalize());
    /* Feather fan: X trails backward, Y points outboard, Z points up — so the
       vanes stand vertical against the flank instead of lying flat. */
    fx.set(-0.955, -0.055 - R.flap * 0.34, w.side * (0.13 + spread * 0.52)).normalize();
    fy.set(0, 0, w.side).addScaledVector(fx, -fx.z * w.side).normalize();
    fz.crossVectors(fx, fy).normalize();
    if (fz.y < 0) { fy.negate(); fz.negate(); }
    fanM.makeBasis(fx, fy, fz);
    w.fan.quaternion.setFromRotationMatrix(fanM);
    w.fan.position.copy(SH).lerp(K, 0.34).addScaledVector(fy, 0.052).addScaledVector(fz, 0.014);
    for (let i = 0; i < w.primaries.length; i++) {
      const { m, t } = w.primaries[i];
      m.rotation.set(t * 0.07, lerp(-0.02 - t * 0.07, -0.32 + t * 0.78, spread), lerp(0.01 + t * 0.03, -0.12 + t * 0.34, spread));
      m.position.set(0.098 + t * 0.058, 0.004 + t * 0.015, -0.004 - t * 0.020);
    }
    for (let i = 0; i < w.secondaries.length; i++) {
      const { m, t } = w.secondaries[i];
      m.rotation.set(t * 0.05, lerp(0.03 + t * 0.05, -0.18 + t * 0.52, spread), lerp(0.02, 0.18, spread));
      m.position.set(0.000 + t * 0.080, 0.010 + t * 0.011, 0.016 - t * 0.012);
    }
    for (let i = 0; i < w.coverts.length; i++) {
      const { m, t } = w.coverts[i];
      m.rotation.set(0.10, 0.12 + t * 0.07, 0.03);
      m.position.set(-0.020 + t * 0.090, 0.014 + t * 0.010, 0.040 - t * 0.012);
    }
    /* coverts sheathing the arm, walked along shoulder → elbow → wrist */
    for (let i = 0; i < w.armFeathers.length; i++) {
      const { m, t } = w.armFeathers[i];
      if (t < 0.46) tmpV.copy(SH).lerp(K, t / 0.46);
      else tmpV.copy(K).lerp(E, (t - 0.46) / 0.54);
      m.position.copy(tmpV).addScaledVector(fy, 0.040 + t * 0.006).addScaledVector(fz, 0.034 - t * 0.010);
      m.quaternion.copy(w.fan.quaternion);
      m.rotateY(0.20 + t * 0.10);
      m.rotateX(0.18);
    }
    jointPts[ji++].copy(SH); jointPts[ji++].copy(K); jointPts[ji++].copy(E);
  }
  head.getWorldPosition(headWorld); rig.worldToLocal(headWorld);
  jointPts[12].copy(NECK_BASE).applyMatrix4(pelvis.matrix);
  jointPts[13].copy(headWorld);

  if (guides.visible) {
    for (let i = 0; i < 14; i++) { guideM.makeTranslation(jointPts[i].x, jointPts[i].y, jointPts[i].z); jointIM.setMatrixAt(i, guideM); }
    jointIM.instanceMatrix.needsUpdate = true;
    const pairs = [[0, 1], [1, 2], [3, 4], [4, 5], [6, 7], [7, 8], [9, 10], [10, 11], [0, 3], [6, 9], [12, 13], [0, 12], [2, 5], [8, 11]];
    for (let i = 0; i < pairs.length; i++) {
      const [a, b] = pairs[i];
      boneArr[i * 6] = jointPts[a].x; boneArr[i * 6 + 1] = jointPts[a].y; boneArr[i * 6 + 2] = jointPts[a].z;
      boneArr[i * 6 + 3] = jointPts[b].x; boneArr[i * 6 + 4] = jointPts[b].y; boneArr[i * 6 + 5] = jointPts[b].z;
    }
    boneGeo.attributes.position.needsUpdate = true;
  }
}

const _m4 = new THREE.Matrix4(), _p3 = V3(), _s3 = V3(), _q4 = new THREE.Quaternion();
const wrap = (x, w) => ((x % w) + w) % w;
function solveWorld(dt) {
  const t = R.t;
  skyU.uTime.value = t; oceanU.uTime.value = t; beachU.uTime.value = t;
  sprayU.uTime.value = t; timeU.value = t;
  deckTex.map.offset.x = R.scroll / DECK_TILE;
  deckTex.normalMap.offset.x = R.scroll / DECK_TILE;
  piles.position.x = -wrap(R.scroll, PILE_SP);
  rail.position.x = -wrap(R.scroll, PILE_SP);
  sky.position.copy(camera.position);
  sunCard.quaternion.copy(camera.quaternion);

  for (const p of palms) {
    p.position.x = wrap(p.userData.baseX - R.scroll + 70, 140) - 70;
    p.position.y = groundY(p.position.x, p.userData.z) - 0.06;
    const s = Math.sin(t * 1.25 + p.userData.phase) * 0.045 * S.wind + Math.sin(t * 2.7 + p.userData.phase) * 0.018 * S.wind;
    p.userData.crown.rotation.z = s;
    p.userData.crown.rotation.x = s * 0.6;
    for (let i = 0; i < p.userData.fronds.length; i++) {
      const f = p.userData.fronds[i];
      f.rotation.z = 0.24 + Math.sin(t * 2.1 + i * 1.3 + p.userData.phase) * 0.10 * (0.25 + S.wind);
    }
  }
  for (let i = 0; i < GRASS_N; i++) {
    const d = grassData[i];
    const x = wrap(d.x - R.scroll, GRASS_WIN) - GRASS_WIN * 0.35;
    _p3.set(x, groundY(x, d.z) - 0.03, d.z); _q4.setFromAxisAngle(UPV, d.r); _s3.setScalar(d.s);
    _m4.compose(_p3, _q4, _s3); grass.setMatrixAt(i, _m4);
  }
  grass.instanceMatrix.needsUpdate = true;
  for (let i = 0; i < ROCK_N; i++) {
    const d = rockData[i];
    const x = wrap(d.x - R.scroll, GRASS_WIN) - GRASS_WIN * 0.35;
    _p3.set(x, groundY(x, d.z) + 0.03, d.z); _q4.setFromAxisAngle(UPV, d.r); _s3.setScalar(d.s);
    _m4.compose(_p3, _q4, _s3); rocks.setMatrixAt(i, _m4);
  }
  rocks.instanceMatrix.needsUpdate = true;

  for (const p of farPalms) {
    p.position.x = wrap(p.userData.baseX - R.scroll * 0.22 + 230, 460) - 230;
    p.rotation.z = Math.sin(t * 1.05 + p.userData.phase) * 0.012 * S.wind;
  }
  for (const b of boats) {
    b.position.x += b.userData.sp * dt;
    if (b.position.x > 130) b.position.x = -130; if (b.position.x < -130) b.position.x = 130;
    b.position.y = -0.30 + Math.sin(t * 0.8 + b.userData.ph) * 0.09 * S.wave;
    b.rotation.z = Math.sin(t * 0.7 + b.userData.ph) * 0.05 * S.wave;
  }
  if (S.gulls) for (const g of gulls) {
    const u = g.userData;
    const a = t * u.sp + u.ph;
    g.position.set(u.cx + Math.cos(a) * u.rad, u.h + Math.sin(a * 2.1) * 0.7, u.cz + Math.sin(a) * u.rad * 0.55);
    g.rotation.y = -a + Math.PI / 2;
    g.rotation.z = Math.sin(a) * 0.28;
    const fl = Math.sin(t * 5.4 + u.ph) * 0.55 + 0.12;
    u.wings[0].rotation.x = -fl; u.wings[1].rotation.x = fl;
  }
  if (S.spray) {
    const arr = sprayGeo.attributes.position.array;
    const drift = (R.speed * 0.35 + S.wind * 1.2);
    for (let i = 0; i < Q.spray; i++) {
      const d = sprayData[i];
      d.x -= drift * dt * (0.4 + d.s * 0.3);
      d.y += Math.sin(t * 0.9 + d.ph) * 0.10 * dt + 0.05 * dt * S.wind;
      if (d.x < -24) { d.x = 24; d.y = rnd(0.02, 6.5); d.z = rnd(-26, 12); }
      if (d.y > 7) d.y = 0.05;
      arr[i * 3] = d.x; arr[i * 3 + 1] = d.y; arr[i * 3 + 2] = d.z + Math.sin(t * 0.6 + d.ph) * 0.3;
    }
    sprayGeo.attributes.position.needsUpdate = true;
  }
}

let raf = 0, last = performance.now(), fpsAcc = 0, fpsN = 0, fpsT = 0, hudT = 0;
function tick(now) {
  raf = requestAnimationFrame(tick);
  const real = Math.max(0.0005, (now - last) / 1000); last = now;
  const dt = Math.min(0.08, real);
  if (!R.paused) R.t += dt;
  const sdt = R.paused ? 0 : dt;
  solveRig(sdt);
  solveWorld(sdt);
  updateCamera(dt);
  audioUpdate(dt);
  if (envDirty && now - envLast > 220) bakeEnv();
  compositeMat.uniforms.uTime.value = now / 1000;
  renderFrame();

  fpsAcc += real; fpsN++;
  if (now - fpsT > 500) {
    $('#fpsLbl').textContent = Math.round(fpsN / Math.max(fpsAcc, 1e-6)) + ' fps';
    fpsAcc = 0; fpsN = 0; fpsT = now;
  }
  if (now - hudT > 110) {
    hudT = now;
    const kmh = R.speed * 3.6;
    $('#gSpeed').textContent = kmh.toFixed(1);
    $('#gCad').textContent = Math.round(R.paused ? 0 : S.cadence);
    $('#gDist').textContent = R.distance < 1000 ? Math.round(R.distance) : (R.distance / 1000).toFixed(2) + 'k';
    $('#speedBarFill').style.width = clamp(kmh / 45 * 100, 0, 100) + '%';
  }
}
