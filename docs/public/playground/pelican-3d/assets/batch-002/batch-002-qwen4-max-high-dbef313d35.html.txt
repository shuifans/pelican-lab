import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';
import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js';

/* ============================================================
   状态
============================================================ */
const DEFAULTS = { speed:14, time:0.7, wave:0.6, wind:0.7, glow:0.55, dust:0.6,
                   helmet:false, scarf:true, cam:'orbit' };
const S = { ...DEFAULTS, sound:true };
const FRAME_COLORS = [0xe4572e, 0x1f8a8c, 0xd9a441, 0xc2417f, 0x4f7d3a, 0x3a6ea5];

const $ = id => document.getElementById(id);
const clamp = THREE.MathUtils.clamp, lerp = THREE.MathUtils.lerp;
const smooth = (a,b,x)=>{ const t = clamp((x-a)/(b-a),0,1); return t*t*(3-2*t); };

/* ============================================================
   渲染器 / 场景 / 相机
============================================================ */
const renderer = new THREE.WebGLRenderer({ antialias:true, powerPreference:'high-performance' });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.05;
$('stage').appendChild(renderer.domElement);

const scene = new THREE.Scene();
scene.fog = new THREE.Fog(0xf2a869, 60, 260);

const camera = new THREE.PerspectiveCamera(46, innerWidth/innerHeight, 0.1, 600);
camera.position.set(3.6, 1.85, 4.1);

const controls = new OrbitControls(camera, renderer.domElement);
controls.target.set(0, 0.95, 0);
controls.enableDamping = true;
controls.dampingFactor = 0.06;
controls.minDistance = 1.4;
controls.maxDistance = 16;
controls.maxPolarAngle = 1.53;
controls.autoRotate = true;
controls.autoRotateSpeed = 0.55;

const composer = new EffectComposer(renderer);
composer.addPass(new RenderPass(scene, camera));
const bloom = new UnrealBloomPass(new THREE.Vector2(innerWidth, innerHeight), S.glow, 0.5, 0.85);
composer.addPass(bloom);
composer.addPass(new OutputPass());

/* ============================================================
   时辰调色板（清晨 → 正午 → 黄金时刻 → 暮色）
============================================================ */
const KEYS = [
  { t:0.00, top:0x4a4e8f, hor:0xffb98a, sun:0xffd9a0, li:1.7, fog:0xf0bd97, sea:0x2e6f7e, amb:0.55, elev:0.16, az:-0.9 },
  { t:0.35, top:0x3d7ecc, hor:0xc4e5f2, sun:0xfff3d6, li:2.7, fog:0xcfe9f4, sea:0x177f96, amb:0.85, elev:1.05, az:-0.35 },
  { t:0.70, top:0x35548f, hor:0xff9e5e, sun:0xffb347, li:2.2, fog:0xf2a869, sea:0x1f6a80, amb:0.52, elev:0.20, az:-0.75 },
  { t:1.00, top:0x141b3d, hor:0x6a4a7e, sun:0xff7e5f, li:0.65, fog:0x4a3f66, sea:0x10263c, amb:0.30, elev:0.045, az:-0.55 },
];
const _cA = new THREE.Color(), _cB = new THREE.Color();
function palette(t){
  let i = 0;
  while (i < KEYS.length-2 && t > KEYS[i+1].t) i++;
  const a = KEYS[i], b = KEYS[i+1];
  const f = smooth(a.t, b.t, t);
  const mix = (ka,kb)=> _cA.setHex(ka).clone().lerp(_cB.setHex(kb), f);
  return {
    top: mix(a.top,b.top), hor: mix(a.hor,b.hor), sun: mix(a.sun,b.sun),
    fog: mix(a.fog,b.fog), sea: mix(a.sea,b.sea),
    li: lerp(a.li,b.li,f), amb: lerp(a.amb,b.amb,f),
    elev: lerp(a.elev,b.elev,f), az: lerp(a.az,b.az,f)
  };
}

/* ============================================================
   天空穹顶 / 太阳 / 星辰
============================================================ */
const skyUni = {
  topColor:{value:new THREE.Color()}, horColor:{value:new THREE.Color()},
  sunDir:{value:new THREE.Vector3(0,0.2,-1)}, sunColor:{value:new THREE.Color()}
};
const sky = new THREE.Mesh(
  new THREE.SphereGeometry(420, 32, 20),
  new THREE.ShaderMaterial({
    side: THREE.BackSide, depthWrite:false, fog:false,
    uniforms: skyUni,
    vertexShader:`varying vec3 vDir; void main(){ vDir = normalize(position);
      gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    fragmentShader:`
      uniform vec3 topColor, horColor, sunColor; uniform vec3 sunDir;
      varying vec3 vDir;
      void main(){
        float h = clamp(vDir.y, 0.0, 1.0);
        vec3 col = mix(horColor, topColor, pow(h, 0.62));
        float s = max(dot(normalize(vDir), normalize(sunDir)), 0.0);
        col += sunColor * (pow(s, 260.0)*1.5 + pow(s, 24.0)*0.22 + pow(s,3.0)*0.08);
        gl_FragColor = vec4(col, 1.0);
      }`
  })
);
scene.add(sky);

const sunMat = new THREE.MeshBasicMaterial({ color:0xffc978, fog:false });
const sunBall = new THREE.Mesh(new THREE.SphereGeometry(16, 24, 16), sunMat);
scene.add(sunBall);
const haloMat = new THREE.MeshBasicMaterial({ color:0xffb347, transparent:true, opacity:0.10,
  blending:THREE.AdditiveBlending, depthWrite:false, fog:false });
const halo = new THREE.Mesh(new THREE.SphereGeometry(27, 24, 16), haloMat);
scene.add(halo);

const starGeo = new THREE.BufferGeometry();
{
  const n = 520, p = new Float32Array(n*3);
  for (let i=0;i<n;i++){
    const th = Math.random()*Math.PI*2, ph = Math.acos(Math.random()*0.85);
    const r = 390;
    p[i*3]   = r*Math.sin(ph)*Math.cos(th);
    p[i*3+1] = r*Math.cos(ph)+8;
    p[i*3+2] = r*Math.sin(ph)*Math.sin(th);
  }
  starGeo.setAttribute('position', new THREE.BufferAttribute(p,3));
}
const starMat = new THREE.PointsMaterial({ color:0xfff2d8, size:1.6, sizeAttenuation:false,
  transparent:true, opacity:0, depthWrite:false, fog:false });
scene.add(new THREE.Points(starGeo, starMat));

/* ============================================================
   程序化环境反射（时辰渐变 + 太阳高光，烘焙为 PMREM）
============================================================ */
const pmrem = new THREE.PMREMGenerator(renderer);
pmrem.compileEquirectangularShader();
const ENV_W = 64, ENV_H = 32;
const envData = new Uint8Array(ENV_W*ENV_H*4);
const envTex = new THREE.DataTexture(envData, ENV_W, ENV_H);
envTex.mapping = THREE.EquirectangularReflectionMapping;
envTex.colorSpace = THREE.SRGBColorSpace;
let envRT = null, lastEnvTime = -1;
const _envC = new THREE.Color(), _envSun = new THREE.Color();
function bakeEnvironment(p, sunDir){
  if (Math.abs(S.time - lastEnvTime) < 0.02 && envRT) return;
  lastEnvTime = S.time;
  const su = Math.atan2(sunDir.z, sunDir.x)/(Math.PI*2) + 0.5;
  const sv = Math.asin(clamp(sunDir.y, -1, 1))/Math.PI + 0.5;
  _envSun.copy(p.sun).convertLinearToSRGB();
  for (let y=0;y<ENV_H;y++){
    const t = y/(ENV_H-1);
    if (t < 0.5) _envC.copy(p.sea).lerp(p.hor, smooth(0.0, 0.5, t));
    else _envC.copy(p.hor).lerp(p.top, Math.pow((t-0.5)/0.5, 0.6));
    _envC.convertLinearToSRGB();
    for (let x=0;x<ENV_W;x++){
      const u = x/(ENV_W-1);
      let du = Math.abs(u-su); du = Math.min(du, 1-du);
      const dv = t-sv;
      const g = Math.exp(-(du*du*14 + dv*dv*22)/0.02);
      const i = (y*ENV_W+x)*4;
      envData[i]   = clamp(_envC.r + _envSun.r*g*1.6, 0, 1)*255;
      envData[i+1] = clamp(_envC.g + _envSun.g*g*1.6, 0, 1)*255;
      envData[i+2] = clamp(_envC.b + _envSun.b*g*1.6, 0, 1)*255;
      envData[i+3] = 255;
    }
  }
  envTex.needsUpdate = true;
  const rt = pmrem.fromEquirectangular(envTex);
  if (envRT) envRT.dispose();
  envRT = rt;
  scene.environment = rt.texture;
}

const unitCyl = new THREE.CylinderGeometry(1, 1, 1, 10);
const YAXIS = new THREE.Vector3(0,1,0);
function tubeBetween(a, b, r, mat, parent, rTop){
  const g = rTop === undefined ? unitCyl : new THREE.CylinderGeometry(rTop, r, 1, 10);
  const m = new THREE.Mesh(g, mat);
  const dir = new THREE.Vector3().subVectors(b, a);
  const len = dir.length();
  m.position.copy(a).addScaledVector(dir, 0.5);
  m.quaternion.setFromUnitVectors(YAXIS, dir.normalize());
  if (rTop === undefined) m.scale.set(r, len, r); else m.scale.set(1, len, 1);
  m.castShadow = true;
  parent.add(m);
  return m;
}
const sphereGeo = new THREE.SphereGeometry(1, 22, 16);
function blob(mat, sx,sy,sz, x,y,z, parent){
  const m = new THREE.Mesh(sphereGeo, mat);
  m.scale.set(sx,sy,sz); m.position.set(x,y,z);
  m.castShadow = true; parent.add(m); return m;
}

function frondGeometry(){
  const seg = 9, pos = [], col = [], idx = [];
  const cBase = new THREE.Color(0x4a8a3e), cTip = new THREE.Color(0x8a9a3a);
  for (let i=0;i<=seg;i++){
    const s = i/seg;
    const x = s*1.7, y = Math.sin(s*Math.PI*0.5)*0.55 - s*s*1.05;
    const w = 0.24*Math.sin(Math.PI*Math.pow(s,0.72)) + 0.015;
    pos.push(x, y, -w,  x, y, w);
    const c = cBase.clone().lerp(cTip, s);
    col.push(c.r,c.g,c.b, c.r,c.g,c.b);
    if (i<seg){ const a=i*2; idx.push(a,a+1,a+2, a+1,a+3,a+2); }
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('color', new THREE.Float32BufferAttribute(col,3));
  g.setIndex(idx); g.computeVertexNormals();
  return g;
}

const _hip = new THREE.Vector3(), _tgt = new THREE.Vector3(), _u = new THREE.Vector3(),
      _knee = new THREE.Vector3(), _perp = new THREE.Vector3(), _dir = new THREE.Vector3(),
      _mid = new THREE.Vector3(), ZAXIS = new THREE.Vector3(0,0,1);
function solveLeg(leg, theta, bob){
  const ph = theta + (leg.side>0 ? 0 : Math.PI);
  _hip.set(HIP_X, HIP_Y + bob, leg.side*HIP_Z);
  _tgt.set(BB.x + Math.cos(ph)*CRANK_L, BB.y + Math.sin(ph)*CRANK_L, leg.side*0.115);
  _u.subVectors(_tgt, _hip);
  let d = _u.length();
  d = clamp(d, Math.abs(L1-L2)+0.005, L1+L2-0.005);
  _u.normalize();
  const a = (L1*L1 - L2*L2 + d*d)/(2*d);
  const h = Math.sqrt(Math.max(L1*L1 - a*a, 0));
  _perp.crossVectors(ZAXIS, _u);           // 膝盖朝前
  _knee.copy(_hip).addScaledVector(_u, a).addScaledVector(_perp, h);
  const place = (mesh, from, to, r)=>{
    _dir.subVectors(to, from);
    const len = _dir.length();
    _mid.copy(from).addScaledVector(_dir, 0.5);
    mesh.position.copy(_mid);
    mesh.quaternion.setFromUnitVectors(YAXIS, _dir.normalize());
    mesh.scale.set(r, len, r);
  };
  place(leg.thigh, _hip, _knee, 0.036);
  place(leg.shin, _knee, _tgt, 0.026);
  leg.knee.position.copy(_knee);
  leg.foot.position.set(_tgt.x+0.01, _tgt.y+0.026, _tgt.z);
  leg.foot.rotation.z = Math.sin(ph)*0.12;
}

let AC = null, master = null, seaGain = null, windGain = null;
function initAudio(){
  if (AC) return;
  AC = new (window.AudioContext || window.webkitAudioContext)();
  master = AC.createGain(); master.gain.value = S.sound ? 0.8 : 0;
  master.connect(AC.destination);
  const noiseBuf = AC.createBuffer(1, AC.sampleRate*2, AC.sampleRate);
  const data = noiseBuf.getChannelData(0);
  for (let i=0;i<data.length;i++) data[i] = Math.random()*2-1;
  const mkNoise = (filterType, freq, q, gain)=>{
    const src = AC.createBufferSource(); src.buffer = noiseBuf; src.loop = true;
    const f = AC.createBiquadFilter(); f.type = filterType; f.frequency.value = freq; f.Q.value = q;
    const g = AC.createGain(); g.gain.value = gain;
    src.connect(f); f.connect(g); g.connect(master); src.start();
    return { src, f, g };
  };
  const seaN = mkNoise('lowpass', 380, 0.6, 0.10);
  seaGain = seaN.g;
  const lfo = AC.createOscillator(); lfo.frequency.value = 0.09;
  const lfoG = AC.createGain(); lfoG.gain.value = 0.05;
  lfo.connect(lfoG); lfoG.connect(seaN.g.gain); lfo.start();
  const windN = mkNoise('bandpass', 700, 0.8, 0.0);
  windGain = windN.g;
}
function blip(freq, dur, type, gain, sweepTo){
  if (!AC || !S.sound) return;
  const t = AC.currentTime;
  const o = AC.createOscillator(); o.type = type; o.frequency.setValueAtTime(freq, t);
  if (sweepTo) o.frequency.exponentialRampToValueAtTime(sweepTo, t+dur);
  const g = AC.createGain();
  g.gain.setValueAtTime(gain, t);
  g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
  o.connect(g); g.connect(master); o.start(t); o.stop(t+dur+0.02);
}

function updateEnvironment(dt, v){
  /* 木板 */
  for (let i=0;i<PLANK_N;i++){
    plankX[i] -= v*dt;
    if (plankX[i] < -32) plankX[i] += PLANK_SPAN;
    _p.set(plankX[i], -0.025, 0);
    _m4.compose(_p, _q, _s);
    planks.setMatrixAt(i, _m4);
  }
  planks.instanceMatrix.needsUpdate = true;
  /* 灯柱 / 棕榈 */
  for (const g of posts){ g.position.x -= v*dt; if (g.position.x < -33) g.position.x += POST_SPAN; }
  for (const p of palms){
    p.position.x -= v*dt;
    if (p.position.x < -32) p.position.x += PALM_SPAN;
    const c = p.userData.crown;
    c.rotation.z = Math.sin(elapsed*1.25 + p.userData.phase)*0.045*(0.3+S.wind);
    c.rotation.x = Math.cos(elapsed*0.9 + p.userData.phase)*0.03*(0.3+S.wind);
  }
  /* 海浪 */
  const pos = seaGeo.attributes.position;
  const amp = S.wave*0.22;
  for (let i=0;i<pos.count;i++){
    const x = seaBase[i*3], y = seaBase[i*3+1];
    pos.array[i*3+2] = amp*( Math.sin(x*0.22 + elapsed*1.15)
      + Math.sin(y*0.34 + elapsed*0.85)*0.6
      + Math.sin((x+y)*0.11 + elapsed*0.5)*0.8 );
  }
  pos.needsUpdate = true;
  seaGeo.computeVertexNormals();
  foam.material.opacity = 0.28 + Math.sin(elapsed*0.8)*0.14*S.wave;
  foam.position.z = -8.2 + Math.sin(elapsed*0.55)*0.5*S.wave;
  /* 云 / 海鸥 */
  for (const c of clouds){ c.position.x -= dt*(0.25 + S.wind*0.35); if (c.position.x < -130) c.position.x = 130; }
  for (const g of gulls){
    const u = g.userData, a = elapsed*u.sp + u.ph;
    g.position.set(-6 + Math.cos(a)*u.r, u.h + Math.sin(elapsed*0.7+u.ph)*0.5, -15 + Math.sin(a)*u.r*0.6);
    g.rotation.y = -a + Math.PI/2;
    const flap = Math.sin(elapsed*7 + u.ph)*0.55;
    u.pl.rotation.x = flap; u.pr.rotation.x = -flap;
  }
  /* 浮尘 */
  const dp = dustGeo.attributes.position;
  const n = Math.floor(DUST_N * S.dust);
  dustGeo.setDrawRange(0, n);
  for (let i=0;i<n;i++){
    dp.array[i*3] -= dt*(v*0.25 + 0.15);
    dp.array[i*3+1] += Math.sin(elapsed*0.8 + dustSeed[i])*dt*0.12;
    if (dp.array[i*3] < -20) dp.array[i*3] += 45;
  }
  dp.needsUpdate = true;
}

function updateRider(dt){
  const v = S.speed/3.6;
  const omegaC = v/(GEAR*WHEEL_R);
  theta += omegaC*dt;
  const wheelOmega = v/WHEEL_R;
  wheelRear.rotation.z -= wheelOmega*dt;
  wheelFront.rotation.z -= wheelOmega*dt;
  crank.rotation.z = theta;
  pedals[0].rotation.z = -theta;
  pedals[1].rotation.z = -theta;

  /* 链条咔哒声（每半圈） */
  const step = Math.floor(theta/Math.PI);
  if (step !== lastTickStep){ lastTickStep = step; sndTick(); }

  /* 身体起伏与摇摆 */
  bob = Math.sin(theta*2)*0.018;
  pelican.position.y = 0.90 + bob;
  pelican.rotation.x = Math.sin(theta)*0.035;
  pelican.rotation.z = Math.sin(theta*2)*0.012;
  headG.rotation.z = -Math.sin(theta*2)*0.045 + Math.sin(elapsed*0.6)*0.03;
  headG.rotation.y = Math.sin(elapsed*0.35)*0.12;
  for (let i=0;i<wings.length;i++){
    wings[i].rotation.x = Math.sin(theta*2 + i*Math.PI)*0.03;
  }
  /* 鱼尾摆动 */
  fishTail.rotation.z = Math.sin(elapsed*5.5)*0.3;
  pouch.scale.y = 0.075*(1 + Math.sin(elapsed*2.2)*0.06);

  /* 围巾飘动 */
  if (S.scarf){
    const amp = 0.10 + S.speed*0.008 + S.wind*0.10;
    for (let i=0;i<scarfSegs.length;i++){
      scarfSegs[i].rotation.z = Math.sin(elapsed*7 - i*0.9)*amp*(i?1:0.4) + (i?0.06:0);
      scarfSegs[i].rotation.y = Math.sin(elapsed*5.2 - i*0.7)*amp*0.5;
    }
  }

  /* 鸣叫动画 */
  if (squawkT >= 0){
    squawkT += dt/0.65;
    const k = squawkT >= 1 ? 0 : Math.sin(Math.PI*squawkT);
    headG.rotation.z += k*0.38;
    billG.rotation.z = k*0.30;
    pouch.scale.set(0.20*(1+k*0.25), 0.075*(1+k*0.85), 0.06*(1+k*0.3));
    fishTail.position.y = -0.03 + k*0.07;
    if (squawkT >= 1){ squawkT = -1; billG.rotation.z = 0; fishTail.position.y = -0.03; }
  }

  /* 车铃动画 + 车把 */
  if (bellT >= 0){
    bellT += dt/0.5;
    const b = steerGroup.userData.bell;
    const k = bellT >= 1 ? 0 : Math.sin(Math.PI*bellT);
    b.scale.setScalar(1 + k*0.35);
    if (bellT >= 1) bellT = -1;
  }
  const wobble = Math.sin(elapsed*2.1)*0.02 + (bellT>=0 ? Math.sin(bellT*28)*0.05*(1-bellT) : 0);
  steerGroup.rotation.y = steer*0.35 + wobble;
  steerGroup.userData.basket.rotation.z = Math.sin(theta*2)*0.02;

  /* 腿部 IK */
  for (const leg of legs) solveLeg(leg, theta, bob);
  return v;
}

function updateVehicle(dt){
  steerTarget = keys['ArrowLeft']||keys['KeyA'] ? 1 : keys['ArrowRight']||keys['KeyD'] ? -1 : steerTarget;
  if (!keys['ArrowLeft']&&!keys['KeyA']&&!keys['ArrowRight']&&!keys['KeyD'] &&
      !document.querySelector('#steer button.hold')) steerTarget = 0;
  steer = lerp(steer, steerTarget, 1-Math.pow(0.002, dt));
  vehicle.position.z = clamp(vehicle.position.z + steer*1.5*dt, -1.0, 1.0);
  vehicle.rotation.y = -steer*0.14;
  vehicle.rotation.x = steer*0.13;
}

function updateCamera(dt){
  if (camTween){
    camTween.t += dt/camTween.dur;
    const f = camTween.t >= 1 ? 1 : 1-Math.pow(1-camTween.t, 3);
    camera.position.lerpVectors(camTween.fromP, camTween.toP, f);
    controls.target.lerpVectors(camTween.fromT, camTween.toT, f);
    if (camTween.t >= 1) camTween = null;
  } else if (S.cam === 'follow'){
    camera.position.z = lerp(camera.position.z, 2.7 + vehicle.position.z*0.75, 1-Math.pow(0.01, dt));
    controls.target.z = lerp(controls.target.z, vehicle.position.z*0.75, 1-Math.pow(0.01, dt));
  }
  controls.update();
}

let statTimer = 0;
function updateStats(dt){
  statTimer += dt;
  if (statTimer < 0.25) return;
  statTimer = 0;
  const v = S.speed/3.6;
  $('stSpeed').textContent = S.speed.toFixed(0);
  $('stCad').textContent = Math.round(v/(GEAR*WHEEL_R)*60/(Math.PI*2));
  $('stWheel').textContent = Math.round(v/WHEEL_R*60/(Math.PI*2));
}

function capture(){
  try{
    const url = renderer.domElement.toDataURL('image/png');
    const a = document.createElement('a');
    a.href = url; a.download = 'pelican-velo-club.png';
    a.click();
    toast('截图已保存');
  }catch(err){ toast('截图失败：' + err.message); }
}

function animate(){
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  elapsed += dt;
  applyPalette();
  updateVehicle(dt);
  const v = updateRider(dt);
  updateEnvironment(dt, v);
  updateCamera(dt);
  updateStats(dt);
  if (windGain) windGain.gain.value = S.sound ? S.wind*0.035 : 0;
  composer.render();
  if (captureNext){ captureNext = false; capture(); }
  if (firstFrame){
    firstFrame = false;
    setTimeout(()=>$('loader').classList.add('gone'), 350);
  }
}
