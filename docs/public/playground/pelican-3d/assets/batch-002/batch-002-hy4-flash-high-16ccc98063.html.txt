(function buildSky(){
  const g = new THREE.SphereGeometry(90, 32, 20);
  const pos = g.attributes.position;
  const colors = [];
  const top = new THREE.Color(0x7db8dd);
  const mid = new THREE.Color(0xc7e3f1);

      c.copy(mid).lerp(top, t);

      c.copy(bot).lerp(mid, t);

  const sky = new THREE.Mesh(g,m);

})();

const UP = new THREE.Vector3(0,1,0);
function mat(color, o={}){
  return new THREE.MeshStandardMaterial({
    color, roughness:o.roughness??0.78, metalness:o.metalness??0.0,
    flatShading:o.flat||false, emissive:o.emissive??0x000000,
    emissiveIntensity:o.ei??0, transparent:o.transparent||false, opacity:o.opacity??1,
    side:o.side??THREE.FrontSide
  });
}
function add(parent, geo, material, o={}){
  const m = new THREE.Mesh(geo, material);
  m.castShadow = o.cast!==false;
  m.receiveShadow = !!o.receive;

  return m;
}
function rod(parent, r, material){ // 单位高度圆柱，沿 Y，便于缩放定向

function setBone(mesh, a, b){
  const dx=b.x-a.x, dy=b.y-a.y, dz=b.z-a.z;
  const len=Math.hypot(dx,dy,dz)||1e-5;
  mesh.position.set((a.x+b.x)/2,(a.y+b.y)/2,(a.z+b.z)/2);
  mesh.quaternion.setFromUnitVectors(UP, new THREE.Vector3(dx/len,dy/len,dz/len));
  mesh.scale.set(1,len,1);
}

const H=new THREE.Vector3(), F=new THREE.Vector3(), K=new THREE.Vector3();
function solveIK(hx,hy,hz, fx,fy,fz, out){
  let dx=fx-hx, dy=fy-hy;
  const raw=Math.hypot(dx,dy)||1e-5;
  let d=THREE.MathUtils.clamp(raw, Math.abs(L1-L2)+1e-4, L1+L2-1e-4);
  const a=(L1*L1-L2*L2+d*d)/(2*d);
  const h=Math.sqrt(Math.max(0,L1*L1-a*a));
  const ux=dx/raw, uy=dy/raw;
  const px=hx+a*ux, py=hy+a*uy;
  out.set(px + (-uy)*h, py + (ux)*h, fz);
}

const clock=new THREE.Clock();
let phase=0;

function animate(){
  requestAnimationFrame(animate);
  const dt=Math.min(clock.getDelta(),0.05);

  if(playing) phase += dt*speed*2.0;
  const theta=-phase;
  const scrolling=phase*WHEEL_R;

  // 车轮、曲柄
  wheels.forEach(w=>w.rotation.z=theta);
  crank.rotation.z=theta;

  // 环境滚动
  for(const s of scrollers){
    let x=s.baseX-(scrolling%s.loop);
    if(x < -s.loop/2) x+=s.loop;
    if(x >  s.loop/2) x-=s.loop;
    s.obj.position.x=x;
  }
  for(const c of clouds){
    c.position.x += c.userData.speed*dt;
    if(c.position.x>24) c.position.x=-24;
  }

  // 身体起伏 / 头部点动 / 翅膀微扇
  const bob=Math.sin(theta*2)*0.016*Math.min(speed,1.4);
  bird.position.y=1.06+bob;
  head.rotation.z=Math.sin(theta*2+0.6)*0.045;
  head.rotation.y=Math.sin(theta*0.7)*0.06;
  wings.forEach(({pivot,s})=>{
    pivot.rotation.x = s*(0.05 + Math.sin(theta*4+s)*0.05);
    pivot.rotation.z = Math.sin(theta*2)*0.03;
  });
  scarfTail.rotation.z=0.5+Math.sin(theta*3)*0.18;

  // 鸣笛反应
  if(honkTimer>0){
    honkTimer=Math.max(0,honkTimer-dt);
    const k=Math.sin((1.15-honkTimer)/1.15*Math.PI);
    lower.rotation.z = -0.42*k;
    wings.forEach(({pivot,s})=> pivot.rotation.x = s*(0.05+0.75*k));
    head.rotation.z += 0.12*k;
  }else{
    lower.rotation.z += (0-lower.rotation.z)*Math.min(1,dt*8);
  }

  // 双腿 IK 跟随脚踏
  const hipX=-0.30, hipY=1.06+bob;
  legs.forEach((leg,i)=>{
    const ang=theta + (i===0?Math.PI:0);
    const px=BB.x+CRANK_R*Math.cos(ang);
    const py=BB.y+CRANK_R*Math.sin(ang);
    H.set(hipX, hipY, leg.z);
    F.set(px, py, leg.z);
    solveIK(H.x,H.y,H.z,F.x,F.y,F.z,K);
    setBone(leg.thigh,H,K);
    setBone(leg.shin,K,F);
    leg.knee.position.copy(K);
    leg.footG.position.set(px,py,leg.z);
    leg.footG.rotation.y=ang*0.15;
  });

  // 相机缓动
  if(camTween){
    camTween.k=Math.min(1,camTween.k+dt*1.6);
    const e=camTween.k<0.5 ? 2*camTween.k*camTween.k : 1-Math.pow(-2*camTween.k+2,2)/2;
    camera.position.lerpVectors(camTween.fp,camTween.tp,e);
    controls.target.lerpVectors(camTween.ft,camTween.tt,e);
    if(camTween.k>=1) camTween=null;
  }
  controls.update();
  renderer.render(scene,camera);
}
