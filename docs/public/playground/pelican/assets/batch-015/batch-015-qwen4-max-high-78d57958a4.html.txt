  /* ---------- leg IK ---------- */
  const L1 = 112, L2 = 112;
  function solveLeg(hx, hy, fx, fy) {
    let dx = fx - hx, dy = fy - hy;
    let d = Math.hypot(dx, dy);
    d = Math.max(4, Math.min(d, L1 + L2 - 2));
    const ux = dx / (Math.hypot(dx, dy) || 1), uy = dy / (Math.hypot(dx, dy) || 1);
    const a = d / 2;
    const h = Math.sqrt(Math.max(0, L1 * L1 - a * a));
    const kx = hx + ux * a + uy * h;
    const ky = hy + uy * a - ux * h;
    return [kx, ky];
  }
  function legPath(hx, hy, fx, fy) {
    const [kx, ky] = solveLeg(hx, hy, fx, fy);
    return "M" + hx.toFixed(1) + " " + hy.toFixed(1) +
           " L" + kx.toFixed(1) + " " + ky.toFixed(1) +
           " L" + fx.toFixed(1) + " " + fy.toFixed(1);
  }
  function footD(fx, fy) {
    return "M" + (fx - 15) + " " + (fy - 4) + " L" + (fx + 19) + " " + (fy - 4) +
           " L" + (fx + 27) + " " + (fy + 3) + " L" + (fx - 11) + " " + (fy + 4) + " Z";
  }

  function hslToHex(h, s, l) {
    s /= 100; l /= 100;
    const k = n => (n + h / 30) % 12;
    const a = s * Math.min(l, 1 - l);
    const f = n => l - a * Math.max(-1, Math.min(k(n) - 3, Math.min(9 - k(n), 1)));
    const to = x => Math.round(x * 255).toString(16).padStart(2, "0");
    return "#" + to(f(0)) + to(f(8)) + to(f(4));
  }
