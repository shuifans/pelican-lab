function computeGeom(R){
  const gy=612, cx=600, wb=3.35*R;
  const rear={x:cx-wb/2,y:gy-R}, front={x:cx+wb/2,y:gy-R};
  const bb={x:rear.x+1.22*R,y:rear.y+.2*R};
  const seatTop={x:bb.x-.5*R,y:bb.y-1.55*R};
  const seatCluster={x:bb.x-.42*R,y:bb.y-1.32*R};
  const headTop={x:front.x-.34*R,y:front.y-1.5*R};
  const headBot={x:front.x-.16*R,y:front.y-.95*R};
  const bar={x:headTop.x-.06*R,y:headTop.y-.3*R};
  const grip={x:bar.x-.42*R,y:bar.y+.1*R};
  const saddle={x:seatTop.x-.02*R,y:seatTop.y-.16*R};
  const bodyC={x:saddle.x+.04*R,y:saddle.y-.95*R};
  const hip={x:seatTop.x+.06*R,y:seatTop.y-.3*R};
  const head0={x:bodyC.x+1.02*R,y:bodyC.y-1.72*R};
  return {R,gy,rear,front,bb,seatTop,seatCluster,headTop,headBot,bar,grip,saddle,bodyC,hip,head0,
    crank:.44*R,ring:.3*R,cog:.13*R,L1:1.3*R,L2:1.3*R,wb};
}
function ik(hip,t,L1,L2,sign){
  let dx=t.x-hip.x, dy=t.y-hip.y, d=Math.hypot(dx,dy);
  d=clamp(d,Math.abs(L1-L2)+.01,L1+L2-.01);
  const a=Math.atan2(dy,dx);
  const A=Math.acos(clamp((L1*L1+d*d-L2*L2)/(2*L1*d),-1,1));
  const ang=a+sign*A;
  return {x:hip.x+Math.cos(ang)*L1,y:hip.y+Math.sin(ang)*L1};
}
function chainPath(c1,r1,c2,r2){
  const dx=c2.x-c1.x, dy=c2.y-c1.y, d=Math.hypot(dx,dy), b=Math.atan2(dy,dx);
  const g=Math.acos(clamp((r1-r2)/d,-1,1));
  const n1=b+g, n2=b-g;
  const p1a={x:c1.x+r1*Math.cos(n1),y:c1.y+r1*Math.sin(n1)};
  const p2a={x:c2.x+r2*Math.cos(n1),y:c2.y+r2*Math.sin(n1)};
  const p2b={x:c2.x+r2*Math.cos(n2),y:c2.y+r2*Math.sin(n2)};
  const p1b={x:c1.x+r1*Math.cos(n2),y:c1.y+r1*Math.sin(n2)};
  return `M${P(p1a.x,p1a.y)}L${P(p2a.x,p2a.y)}A${fmt(r2,2)},${fmt(r2,2)} 0 0 0 ${P(p2b.x,p2b.y)}L${P(p1b.x,p1b.y)}A${fmt(r1,2)},${fmt(r1,2)} 0 1 0 ${P(p1a.x,p1a.y)}Z`;
}

function restyle(){
  if(wireApplied)restoreWire();
  const rig=$("rig");
  rig.querySelectorAll("*").forEach(n=>{ if(n.__sw!==undefined)n.setAttribute("stroke-width",fmt(n.__sw*S.stroke,3)); });
  if(S.wire)applyWire();
  $("guides").setAttribute("opacity",S.guides?1:0);
  $("grainRect").setAttribute("opacity",S.grain&&S.quality!=="low"?.085:0);
  $("glowBlur").setAttribute("stdDeviation",fmt(1.2+S.glow*2.6,2));
  $("dofBlur").setAttribute("stdDeviation",fmt(S.quality==="low"?0:S.dof,2));
  $("farBlur").setAttribute("filter",S.dof>.05&&S.quality!=="low"?"url(#fDof)":"none");
  $("sunHalo").setAttribute("opacity",fmt(clamp(.45+S.glow*.5,0,1.4),2));
  if(refs.lampG)refs.lampG.setAttribute("opacity",S.lamp?1:0);
  if(refs.basketG)refs.basketG.setAttribute("opacity",S.basket?1:0);
  if(refs.scarf)refs.scarf.setAttribute("opacity",S.scarf?1:0);
  if(refs.helmet)refs.helmet.setAttribute("opacity",S.helmet?1:0);
  if(refs.crest)refs.crest.setAttribute("opacity",S.helmet?0:1);
  $("gulls").setAttribute("opacity",S.gulls?1:0);
  applyCamera();
}
function applyWire(){

    wfMap.set(n,{f:n.getAttribute("fill"),s:n.getAttribute("stroke"),w:n.getAttribute("stroke-width"),o:n.getAttribute("opacity")});
    if(n.tagName==="text"){n.setAttribute("fill","#B6F24A");n.setAttribute("opacity",.9);return;}
    const keep=n.getAttribute("stroke")==="transparent";
    n.setAttribute("fill","none");

  });
  /* particles */

  const want=Math.min(S.particles,cap);
  let live=0;
  for(let i=0;i<particles.length;i++){

    if(fpsAcc>.5){fps=fpsN/fpsAcc;fpsAcc=0;fpsN=0;D.tagFps.textContent=Math.round(fps);}
    D.footState.textContent=`${th.name} · ${S.playing?"骑行中":"已暂停"} · 轮径 ${R}px · ${fmt(dist*mPerPx,0)} m`;
  }
}

function makeSound(){
  const AC=window.AudioContext||window.webkitAudioContext;
  if(!AC)return null;
  let ctx=null,master=null,seaG=null,whirG=null,whirF=null,started=false,lastTick=0;
  function noiseBuf(){
    const b=ctx.createBuffer(1,ctx.sampleRate*2,ctx.sampleRate), d=b.getChannelData(0);
    for(let i=0;i<d.length;i++)d[i]=Math.random()*2-1;
    return b;
  }
  function start(){
    if(started)return;
    ctx=new AC(); started=true;
    master=ctx.createGain(); master.gain.value=0; master.connect(ctx.destination);
    const nb=noiseBuf();
    const s1=ctx.createBufferSource(); s1.buffer=nb; s1.loop=true;
    const lp=ctx.createBiquadFilter(); lp.type="lowpass"; lp.frequency.value=430;
    seaG=ctx.createGain(); seaG.gain.value=.05;
    s1.connect(lp); lp.connect(seaG); seaG.connect(master); s1.start();
    const lfo=ctx.createOscillator(); lfo.frequency.value=.09;
    const lg=ctx.createGain(); lg.gain.value=.028;
    lfo.connect(lg); lg.connect(seaG.gain); lfo.start();
    const s2=ctx.createBufferSource(); s2.buffer=nb; s2.loop=true;
    whirF=ctx.createBiquadFilter(); whirF.type="bandpass"; whirF.frequency.value=500; whirF.Q.value=7;
    whirG=ctx.createGain(); whirG.gain.value=0;
    s2.connect(whirF); whirF.connect(whirG); whirG.connect(master); s2.start();
  }
  return {
    get on(){return started&&ctx&&ctx.state==="running";},
    resume(){ start(); if(ctx.state!=="running")ctx.resume(); },
    suspend(){ if(ctx&&ctx.state==="running")ctx.suspend(); },
    update(dt,S,ca){
      if(!started||!ctx||ctx.state!=="running")return;
      const tgt=S.sound?.5:0;
      master.gain.setTargetAtTime(tgt,ctx.currentTime,.15);
      const sp=S.playing?S.speed:0;
      whirG.gain.setTargetAtTime(sp*.035,ctx.currentTime,.12);
      whirF.frequency.setTargetAtTime(320+sp*1100,ctx.currentTime,.15);
      seaG.gain.setTargetAtTime(S.playing?.045:.012,ctx.currentTime,.4);
      const half=Math.floor(ca/Math.PI);
      if(sp>.06&&half!==lastTick){lastTick=half;this.tick(sp);}
    },
    tick(sp){
      if(!ctx||ctx.state!=="running")return;
      const t=ctx.currentTime, o=ctx.createOscillator(), g=ctx.createGain();
      o.type="triangle"; o.frequency.setValueAtTime(190+sp*60,t);
      g.gain.setValueAtTime(0,t); g.gain.linearRampToValueAtTime(.035*Math.min(1,sp),t+.004);
      g.gain.exponentialRampToValueAtTime(.0001,t+.07);
      o.connect(g); g.connect(master); o.start(t); o.stop(t+.09);
    },
    bell(){
      this.resume(); if(!ctx)return;
      const t=ctx.currentTime;
      [[1046,.16],[1568,.1],[2093,.06],[2637,.03]].forEach(([f,a],i)=>{
        const o=ctx.createOscillator(), g=ctx.createGain();
        o.type="sine"; o.frequency.value=f*(1+i*.001);
        g.gain.setValueAtTime(0,t); g.gain.linearRampToValueAtTime(a,t+.006);
        g.gain.exponentialRampToValueAtTime(.0001,t+1.5-i*.2);
        o.connect(g); g.connect(master); o.start(t); o.stop(t+1.6);
      });
    },
    squawk(){
      this.resume(); if(!ctx)return;
      const t=ctx.currentTime, o=ctx.createOscillator(), g=ctx.createGain(), f=ctx.createBiquadFilter();
      o.type="sawtooth"; o.frequency.setValueAtTime(620,t);
      o.frequency.exponentialRampToValueAtTime(980,t+.09);
      o.frequency.exponentialRampToValueAtTime(430,t+.3);
      f.type="bandpass"; f.frequency.value=1200; f.Q.value=3;
      g.gain.setValueAtTime(0,t); g.gain.linearRampToValueAtTime(.09,t+.03);
      g.gain.exponentialRampToValueAtTime(.0001,t+.34);
      o.connect(f); f.connect(g); g.connect(master); o.start(t); o.stop(t+.36);
    }
  };
}

  }
}

/* ---------- actions ---------- */
function setPlaying(v){
  S.playing=v; syncUI();
  if(v&&S.sound&&sound)sound.resume();
}
function randomize(){
  const th=["dawn","noon","dusk","night"];
  S.theme=th[Math.floor(Math.random()*4)];
  S.speed=+rnd(.5,2.1).toFixed(2);
  S.cadence=+rnd(.7,1.6).toFixed(2);
  S.R=Math.round(rnd(70,110));
  S.bob=+rnd(.2,1.4).toFixed(2);
  S.wind=+rnd(-.2,1.8).toFixed(2);
  S.zoom=+rnd(.85,1.3).toFixed(2);
  S.stroke=+rnd(.8,1.7).toFixed(2);
  S.glow=+rnd(.3,1.8).toFixed(2);
  S.dof=+rnd(0,2.2).toFixed(2);
  S.particles=Math.round(rnd(16,80));
  const p=PRESETS[Math.floor(Math.random()*PRESETS.length)];
  S.body=p[0]; S.bill=p[1];
  S.scarf=Math.random()<.8; S.helmet=Math.random()<.45;
  S.basket=Math.random()<.8; S.gulls=Math.random()<.85;
  S.lamp=S.theme==="night"?true:Math.random()<.25;
  lampAuto=false;
  S.wire=false; S.guides=false;
  PAN.x=0;PAN.y=0;
  buildRig(); applyTheme(); applyParticleCount(); syncUI();
  toast("随机生成一套新的骑行配置");
}
function reset(){
  Object.assign(S,DEF);
  PAN.x=0;PAN.y=0; dist=0; simT=0; lampAuto=false;
  buildRig(); applyTheme(); applyParticleCount(); syncUI();
  toast("已恢复默认配置");
}

/* ---------- interaction ---------- */
function bindStage(){
  const stage=$("stage"), svg=$("scene");
  let dragging=false,sx=0,sy=0,px=0,py=0,moved=0;
  stage.addEventListener("pointerdown",e=>{
    if(e.target.closest("[style*=cursor]"))return;
    dragging=true;moved=0;sx=e.clientX;sy=e.clientY;px=PAN.x;py=PAN.y;
    stage.classList.add("dragging"); stage.setPointerCapture(e.pointerId);
  });
  stage.addEventListener("pointermove",e=>{
    if(!dragging)return;
    const r=stage.getBoundingClientRect(), k=1200/r.width;
    PAN.x=clamp(px+(e.clientX-sx)*k,-900,900);
    PAN.y=clamp(py+(e.clientY-sy)*k,-500,500);
    moved+=Math.abs(e.clientX-sx)+Math.abs(e.clientY-sy);
    applyCamera();
  });
  const end=e=>{ if(!dragging)return; dragging=false; stage.classList.remove("dragging"); };
  stage.addEventListener("pointerup",end); stage.addEventListener("pointercancel",end);
  stage.addEventListener("wheel",e=>{
    e.preventDefault();
    S.zoom=clamp(S.zoom*(e.deltaY>0?.92:1.08),.65,1.8);
    syncUI(); applyCamera();
  },{passive:false});
  stage.addEventListener("dblclick",()=>{PAN.x=0;PAN.y=0;S.zoom=1;syncUI();applyCamera();toast("镜头已复位");});
  svg.addEventListener("click",e=>{
    if(moved>8)return;
    if(e.target.closest("[style*=cursor]")){
      const hit=e.target.closest("[style*=cursor]");
      if(hit===refs.bell)bellPulse(); else squawk();
    }
  });
}
function bindKeys(){
  window.addEventListener("keydown",e=>{
    if(e.target.tagName==="INPUT"&&e.target.type!=="range")return;
    const k=e.key.toLowerCase();
    if(e.code==="Space"){e.preventDefault();setPlaying(!S.playing);}
    else if(k==="r"){randomize();}
    else if(k==="s"){toggleSound();}
    else if(k==="b"){bellPulse();}
    else if(k==="g"){S.guides=!S.guides;syncUI();restyle();}
    else if(e.key==="ArrowRight"){S.speed=clamp(S.speed+.1,0,2.4);syncUI();}
    else if(e.key==="ArrowLeft"){S.speed=clamp(S.speed-.1,0,2.4);syncUI();}
  });
}
function toggleSound(){
  S.sound=!S.sound;
  if(S.sound){ if(!sound)sound=makeSound(); if(sound)sound.resume(); toast("声音已开启：海风、链条与车铃"); }
  else { if(sound)sound.suspend(); toast("声音已静音"); }
  syncUI();
}
function firstGesture(){
  if(S.sound){ if(!sound)sound=makeSound(); if(sound)sound.resume(); }
  window.removeEventListener("pointerdown",firstGesture);
  window.removeEventListener("keydown",firstGesture);
}

})();
