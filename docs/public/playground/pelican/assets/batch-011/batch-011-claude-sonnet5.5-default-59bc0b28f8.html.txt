<script>
const $ = id => document.getElementById(id);
const P = 800;                      // 背景图层的循环周期

// ---------- 背景 ----------
function hillPath(base, comps) {
  const pts = [];
  for (let x = 0; x <= 1600; x += 16) {
    let y = base;
    for (const [k, a, ph] of comps) y -= a * Math.sin(2 * Math.PI * k * x / P + ph);
    pts.push(x + ',' + y.toFixed(1));
  }
  return 'M0,440 L' + pts.join(' L') + ' L1600,440 Z';
}
$('hillsFar').innerHTML = '<path d="' + hillPath(375, [[1,28,0],[2,14,1.3],[3,8,2.1]]) + '" fill="#a5d6a7"/>';
$('hillsNear').innerHTML = '<path d="' + hillPath(405, [[2,16,.5],[3,9,2.4],[5,4,0]]) + '" fill="#81c784"/>';

let c = '';
for (const [x, y, s] of [[110,70,1],[430,115,.8],[910,70,1],[1230,115,.8]])
  c += `<use href="#cloud" transform="translate(${x} ${y}) scale(${s})"/>`;
$('clouds').innerHTML = c;

let t = '';
for (const x of [90, 330, 560, 700]) for (const o of [0, P]) {
  t += `<g transform="translate(${x + o} 432)"><rect x="-4" y="-34" width="8" height="34" fill="#6d4c41"/>` +
       `<circle cy="-48" r="22" fill="#388e3c"/><circle cx="-12" cy="-38" r="14" fill="#43a047"/>` +
       `<circle cx="12" cy="-40" r="15" fill="#43a047"/></g>`;
}
$('trees').innerHTML = t;

let d = '';
for (let i = 0; i < 13; i++) d += `<rect x="${i * 80}" y="474" width="40" height="5" rx="2"/>`;
$('dashes').innerHTML = d;

// ---------- 车轮 ----------
function wheelHTML() {
  let s = '<circle r="68" fill="none" stroke="#263238" stroke-width="6"/>' +
          '<circle r="62" fill="none" stroke="#b0bec5" stroke-width="2"/>';
  for (let i = 0; i < 16; i++) {
    const a = i * Math.PI / 8;
    s += `<line x2="${(62 * Math.cos(a)).toFixed(1)}" y2="${(62 * Math.sin(a)).toFixed(1)}" stroke="#cfd8dc" stroke-width="1.2"/>`;
  }
  return s + '<circle cy="-54" r="3.5" fill="#ffca28"/><circle r="6" fill="#455a64"/>';
}
$('rw').innerHTML = wheelHTML();
$('fw').innerHTML = wheelHTML();

// ---------- 骑行运动学 ----------
const BB = [372, 384], CR = 24;      // 中轴位置、曲柄长度
const HIP0 = [348, 291], THIGH = 60, SHIN = 64;
const WHEEL_R = 68;

// 两段式腿的反向运动学：膝盖朝前
function kneeFor(hip, foot) {
  const dx = foot[0] - hip[0], dy = foot[1] - hip[1];
  const dist = Math.min(Math.hypot(dx, dy), THIGH + SHIN - 0.5);
  const base = Math.atan2(dy, dx);
  const cosA = (THIGH * THIGH + dist * dist - SHIN * SHIN) / (2 * THIGH * dist);
  const ang = Math.acos(Math.max(-1, Math.min(1, cosA)));
  return [hip[0] + THIGH * Math.cos(base - ang), hip[1] + THIGH * Math.sin(base - ang)];
}

function setLeg(legId, pedId, armId, hip, a) {
  const foot = [BB[0] + CR * Math.cos(a), BB[1] + CR * Math.sin(a)];
  const knee = kneeFor(hip, foot);
  $(legId).setAttribute('d', `M${hip[0]},${hip[1]} L${knee[0].toFixed(1)},${knee[1].toFixed(1)} L${foot[0].toFixed(1)},${foot[1].toFixed(1)}`);
  $(pedId).setAttribute('transform', `translate(${foot[0].toFixed(1)} ${(foot[1] + 3).toFixed(1)})`);
  $(armId).setAttribute('x2', foot[0].toFixed(1));
  $(armId).setAttribute('y2', foot[1].toFixed(1));
}

function layer(id, off) { $(id).setAttribute('transform', `translate(${-off.toFixed(2)} 0)`); }

// ---------- 动画循环 ----------
let dist = 0, last = performance.now(), speed = 1, paused = false;

function render(now) {
  const wheelDeg = dist / WHEEL_R * 180 / Math.PI;
  $('rw').setAttribute('transform', `translate(250 380) rotate(${wheelDeg.toFixed(2)})`);
  $('fw').setAttribute('transform', `translate(520 380) rotate(${wheelDeg.toFixed(2)})`);

  const crank = dist / WHEEL_R * 0.8;
  const bob = Math.sin(crank * 2) * 1.6;
  $('pel').setAttribute('transform', `translate(0 ${bob.toFixed(2)})`);
  $('head').setAttribute('transform', `rotate(${(Math.sin(crank) * 1.8).toFixed(2)} 385 235)`);

  const hip = [HIP0[0], HIP0[1] + bob];
  setLeg('legN', 'pedN', 'crN', hip, crank);
  setLeg('legF', 'pedF', 'crF', hip, crank + Math.PI);

  const ry = (now % 3600 < 130) ? 0.4 : 3.2;
  $('eye').setAttribute('ry', ry);

  layer('clouds', (dist * 0.05) % P);
  layer('hillsFar', (dist * 0.12) % P);
  layer('hillsNear', (dist * 0.3) % P);
  layer('trees', (dist * 0.9) % P);
  layer('dashes', dist % 80);
}

function frame(now) {
  const dt = Math.min(0.05, (now - last) / 1000);
  last = now;
  if (!paused) dist += 220 * speed * dt;
  render(now);
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

$('toggle').addEventListener('click', () => {
  paused = !paused;
  $('toggle').textContent = paused ? '继续' : '暂停';
});
$('speed').addEventListener('input', e => { speed = parseFloat(e.target.value); });
