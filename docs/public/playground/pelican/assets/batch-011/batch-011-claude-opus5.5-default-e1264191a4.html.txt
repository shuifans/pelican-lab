<script>
const NS = "http://www.w3.org/2000/svg";
const $ = id => document.getElementById(id);
const el = (tag, attrs, parent) => {
  const e = document.createElementNS(NS, tag);
  for (const k in attrs) e.setAttribute(k, attrs[k]);
  parent && parent.appendChild(e);
  return e;
};

// 辐条
["spokesR", "spokesF"].forEach(id => {
  for (let i = 0; i < 12; i++) {
    const a = i * Math.PI / 6;
    el("line", {x1: 0, y1: 0, x2: 54 * Math.cos(a), y2: 54 * Math.sin(a)}, $(id));
  }
});

// 周期为 800 的山丘轮廓，绘制 0..1600 以便无缝循环
function hillPath(base, amps) {
  let d = `M0,400 L0,${base}`;
  for (let x = 0; x <= 1600; x += 10) {
    let y = base;
    amps.forEach(([k, a, p]) => y -= a * Math.sin(2 * Math.PI * k * x / 800 + p));
    d += ` L${x},${y.toFixed(1)}`;
  }
  return d + " L1600,400 Z";
}
el("path", {d: hillPath(300, [[1, 30, 0], [3, 14, 1]]), fill: "#a9cfa0"}, $("farHills"));
el("path", {d: hillPath(340, [[2, 22, 2], [5, 8, 0.5]]), fill: "#8cc274"}, $("nearHills"));

// 云、树、路面虚线（每层内容复制两份，平移取模实现循环）
[[80, 90, 1], [330, 60, 0.8], [560, 140, 1.2]].forEach(([x, y, s]) => {
  [0, 800].forEach(o => {
    const g = el("g", {transform: `translate(${x + o},${y}) scale(${s})`, fill: "#fff", opacity: .9}, $("clouds"));
    el("ellipse", {cx: 0, cy: 0, rx: 40, ry: 18}, g);
    el("ellipse", {cx: 25, cy: -12, rx: 26, ry: 20}, g);
    el("ellipse", {cx: -22, cy: -6, rx: 20, ry: 14}, g);
  });
});
[40, 190, 300, 470, 620, 730].forEach((x, i) => {
  [0, 800].forEach(o => {
    const g = el("g", {transform: `translate(${x + o},372)`}, $("trees"));
    const h = 30 + (i % 3) * 10;
    el("rect", {x: -4, y: -h * 0.4, width: 8, height: h * 0.4 + 6, fill: "#7a5432"}, g);
    el("circle", {cx: 0, cy: -h * 0.6, r: h * 0.45, fill: i % 2 ? "#3f8f4a" : "#4fa35a"}, g);
  });
});
for (let x = 0; x <= 880; x += 80) el("rect", {x, y: 446, width: 44, height: 6, rx: 3, fill: "#f5d76e"}, $("dashes"));

// 两段腿的反向运动学：已知髋、脚位置，求膝盖
const L1 = 70, L2 = 72;
function knee(hx, hy, px, py) {
  const dx = px - hx, dy = py - hy;
  const d = Math.min(Math.hypot(dx, dy), L1 + L2 - 0.01);
  const a = Math.atan2(dy, dx);
  const b = Math.acos((L1 * L1 + d * d - L2 * L2) / (2 * L1 * d));
  return [hx + L1 * Math.cos(a - b), hy + L1 * Math.sin(a - b)]; // 膝盖朝前
}

const BB = [340, 345], CRANK = 28, R = 60, SPEED = 220; // 车速 px/s
const svg = document.querySelector("svg");
let paused = false, last = null, dist = 0, t = 0;
svg.addEventListener("click", () => { paused = !paused; svg.classList.toggle("paused", paused); });

function setLeg(side, ang, bob) {
  const px = BB[0] + CRANK * Math.cos(ang), py = BB[1] + CRANK * Math.sin(ang);
  $("crank" + side).setAttribute("x1", BB[0]); $("crank" + side).setAttribute("y1", BB[1]);
  $("crank" + side).setAttribute("x2", px);    $("crank" + side).setAttribute("y2", py);
  $("pedal" + side).setAttribute("x", px - 8); $("pedal" + side).setAttribute("y", py - 2);
  const hx = side === "N" ? 322 : 316, hy = 232 + bob;
  const [kx, ky] = knee(hx, hy, px, py - 4);
  $("leg" + side).setAttribute("points", `${hx},${hy} ${kx.toFixed(1)},${ky.toFixed(1)} ${px.toFixed(1)},${(py - 4).toFixed(1)}`);
  $("foot" + side).setAttribute("transform", `translate(${px.toFixed(1)},${(py - 4).toFixed(1)}) rotate(${10 * Math.sin(ang)})`);
}

function frame(now) {
  if (last === null) last = now;
  const dt = Math.min((now - last) / 1000, 0.05);
  last = now;
  if (!paused) { dist += SPEED * dt; t += dt; }

  const wheelDeg = (dist / R) * 180 / Math.PI;
  $("spokesR").setAttribute("transform", `rotate(${wheelDeg})`);
  $("spokesF").setAttribute("transform", `rotate(${wheelDeg})`);
  $("chain").setAttribute("stroke-dashoffset", -dist * 0.3);

  const crank = dist / R / 2.2; // 齿比 2.2
  const bob = 2.5 * Math.sin(crank * 2);
  setLeg("F", crank + Math.PI, bob);
  setLeg("N", crank, bob);

  $("upper").setAttribute("transform", `translate(0,${bob})`);
  $("head").setAttribute("transform", `rotate(${3 * Math.sin(crank * 2 + 0.6)} 360 190)`);
  $("pouch").setAttribute("transform", `translate(0,${126 * (1 - (1 + 0.08 * Math.sin(t * 9)))}) scale(1,${1 + 0.08 * Math.sin(t * 9)})`);
  const blink = (t % 3.2) < 0.12;
  $("lid").setAttribute("d", blink ? "M406,110 Q412,112 418,110" : "M405,104 Q412,101 419,105");
  $("rider").setAttribute("transform", `translate(0,${Math.abs(Math.sin(t * 7)) * -1.5})`);

  const shift = (k, w) => `translate(${-((dist * k) % w)},0)`;
  $("clouds").setAttribute("transform", shift(0.08, 800));
  $("farHills").setAttribute("transform", shift(0.2, 800));
  $("nearHills").setAttribute("transform", shift(0.45, 800));
  $("trees").setAttribute("transform", shift(0.7, 800));
  $("dashes").setAttribute("transform", shift(1, 80));

  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
