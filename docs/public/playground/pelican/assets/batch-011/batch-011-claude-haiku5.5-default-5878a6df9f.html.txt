<script>
const NS = 'http://www.w3.org/2000/svg';
const $ = id => document.getElementById(id);
const BB = { x: 290, y: 300 }, HIP = { x: 265, y: 205 }, CR = 24, L = 60;

// spokes
for (const id of ['rs', 'fs']) {
  for (let i = 0; i < 6; i++) {
    const a = i * Math.PI / 6, dx = 55 * Math.cos(a), dy = 55 * Math.sin(a);
    const cx = id === 'rs' ? 170 : 430;
    const l = document.createElementNS(NS, 'line');
    l.setAttribute('x1', cx - dx); l.setAttribute('y1', 300 - dy);
    l.setAttribute('x2', cx + dx); l.setAttribute('y2', 300 + dy);
    $(id).appendChild(l);
  }
}
// road dashes
for (let x = -80; x < 700; x += 80) {
  const r = document.createElementNS(NS, 'rect');
  r.setAttribute('x', x); r.setAttribute('y', 380); r.setAttribute('width', 40); r.setAttribute('height', 5);
  $('lines').appendChild(r);
}

function setLine(el, x1, y1, x2, y2) {
  el.setAttribute('x1', x1); el.setAttribute('y1', y1);
  el.setAttribute('x2', x2); el.setAttribute('y2', y2);
}

// two-bone IK: returns foot position, sets leg path with knee bent forward
function leg(path, a) {
  const fx = BB.x + CR * Math.cos(a), fy = BB.y + CR * Math.sin(a);
  const dx = fx - HIP.x, dy = fy - HIP.y, dist = Math.hypot(dx, dy);
  const d = Math.min(dist, 2 * L - 0.5), h = Math.sqrt(L * L - (d / 2) ** 2);
  let px = -dy / dist, py = dx / dist;
  if (px < 0) { px = -px; py = -py; }
  const k = d / dist;
  const kx = HIP.x + dx * k / 2 + px * h, ky = HIP.y + dy * k / 2 + py * h;
  path.setAttribute('d', `M${HIP.x},${HIP.y} L${kx.toFixed(1)},${ky.toFixed(1)} L${fx.toFixed(1)},${fy.toFixed(1)}`);
  return [fx, fy];
}

let th = 0, wh = 0, road = 0, c1 = 0, c2 = 350, paused = false, last = performance.now();
const deg = a => a * 180 / Math.PI;

function frame(now) {
  const dt = Math.min((now - last) / 1000, 0.05);
  last = now;
  if (!paused) {
    th += 2 * dt;
    wh += 4 * dt;
    road = (road + 248 * dt) % 80;
    c1 = (c1 + 14 * dt) % 700;
    c2 = (c2 + 9 * dt) % 700;
  }

  $('rs').setAttribute('transform', `rotate(${deg(wh)} 170 300)`);
  $('fs').setAttribute('transform', `rotate(${deg(wh)} 430 300)`);
  $('lines').setAttribute('transform', `translate(${-road} 0)`);
  $('c1').setAttribute('transform', `translate(${c1 - 100} 70)`);
  $('c2').setAttribute('transform', `translate(${c2 - 100} 120)`);
  $('wing').setAttribute('transform', `rotate(${Math.sin(now / 200) * 10} 280 170)`);

  const f1 = leg($('leg1'), th);
  const f2 = leg($('leg2'), th + Math.PI);
  setLine($('ca1'), BB.x, BB.y, f1[0], f1[1]);
  setLine($('ca2'), BB.x, BB.y, f2[0], f2[1]);
  $('ft1').setAttribute('cx', f1[0] + 5); $('ft1').setAttribute('cy', f1[1] + 3);
  $('ft2').setAttribute('cx', f2[0] + 5); $('ft2').setAttribute('cy', f2[1] + 3);

  requestAnimationFrame(frame);
}

$('scene').addEventListener('click', () => { paused = !paused; });
requestAnimationFrame(frame);
</script>
