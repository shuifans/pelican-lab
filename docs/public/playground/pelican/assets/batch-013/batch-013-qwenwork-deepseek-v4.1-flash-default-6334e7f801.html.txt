<script>
(function () {
  'use strict';

  const NS = 'http://www.w3.org/2000/svg';
  const $ = id => document.getElementById(id);
  const W = 1200;

  /* -------- Catmull-Rom → 三次贝塞尔 -------- */
  function catmull(pts) {
    if (pts.length < 2) return '';
    let d = `M${pts[0][0].toFixed(1)},${pts[0][1].toFixed(1)}`;
    for (let i = 0; i < pts.length - 1; i++) {
      const p0 = pts[i - 1] || pts[i];
      const p1 = pts[i];
      const p2 = pts[i + 1];
      const p3 = pts[i + 2] || pts[i + 1];
      const c1x = p1[0] + (p2[0] - p0[0]) / 6, c1y = p1[1] + (p2[1] - p0[1]) / 6;
      const c2x = p2[0] - (p3[0] - p1[0]) / 6, c2y = p2[1] - (p3[1] - p1[1]) / 6;
      d += ` C${c1x.toFixed(1)},${c1y.toFixed(1)} ${c2x.toFixed(1)},${c2y.toFixed(1)}`
         + ` ${p2[0].toFixed(1)},${p2[1].toFixed(1)}`;
    }
    return d;
  }

  function ridge(base, a1, k1, p1, a2, k2, p2) {
    const pts = [];
    for (let x = -80; x <= W + 80; x += 20) {
      const y = base
        - a1 * Math.sin(2 * Math.PI * k1 * x / W + p1)
        - a2 * Math.sin(2 * Math.PI * k2 * x / W + p2);
      pts.push([x, y]);
    }
    return catmull(pts) + ` L${W + 80},720 L-80,720 Z`;
  }

  function tile(layer, d, fill) {
    for (let i = -2; i <= 2; i++) {
      const p = document.createElementNS(NS, 'path');
      p.setAttribute('d', d);
      p.setAttribute('fill', fill);
      p.setAttribute('transform', `translate(${i * W} 0)`);
      layer.appendChild(p);
    }
  }

  /* -------- 地形 -------- */
  tile($('farHills'), ridge(468, 42, 2, 0.6, 16, 5, 1.2), '#aecdb5');
  tile($('nearHills'), ridge(510, 44, 3, 2.1, 15, 7, 0.4), '#8ebc97');

  /* -------- 云 -------- */
  function cloud(x, y, s, o) {
    const g = document.createElementNS(NS, 'g');
    g.setAttribute('fill', '#ffffff');
    g.setAttribute('opacity', o);
    [[0, 0, 46, 26], [44, -13, 38, 24], [84, 1, 32, 22], [-38, 6, 30, 18]]
      .forEach(([dx, dy, rx, ry]) => {
        const e = document.createElementNS(NS, 'ellipse');
        e.setAttribute('cx', x + dx * s);
        e.setAttribute('cy', y + dy * s);
        e.setAttribute('rx', rx * s);
        e.setAttribute('ry', ry * s);
        g.appendChild(e);
      });
    return g;
  }
  const CLOUD_PERIOD = 1800;
  const cloudSpec = [[300, 112, 1.15, 0.85], [700, 76, 0.72, 0.55], [1250, 226, 0.88, 0.55]];
  for (const dx of [-CLOUD_PERIOD, 0, CLOUD_PERIOD]) {
    cloudSpec.forEach(([x, y, s, o]) => $('clouds').appendChild(cloud(x + dx, y, s, o)));
  }

  /* -------- 路肩植被 -------- */
  let seed = 20261105;
  const rnd = () => (seed = (seed * 1103515245 + 12345) % 2147483648) / 2147483648;

  const greens = ['#6c9a72', '#7bab81', '#5f8f69'];
  const bushSpec = [];
  for (let i = 0; i < 7; i++) {
    bushSpec.push({
      x: i * (W / 7) + rnd() * (W / 7) * 0.5,
      s: 0.72 + rnd() * 0.55,
      c: i % 3
    });
  }
  const grassSpec = [];
  for (let i = 0; i < 13; i++) {
    grassSpec.push(i * (W / 13) + rnd() * (W / 13) * 0.5);
  }

  for (const shift of [-W, 0, W]) {
    for (const b of bushSpec) {
      const g = document.createElementNS(NS, 'g');
      g.setAttribute('transform', `translate(${(b.x + shift).toFixed(1)} 552) scale(${b.s.toFixed(2)})`);
      g.setAttribute('fill', greens[b.c]);
      [[-16, -9, 17, 13], [4, -15, 20, 15], [22, -7, 13, 11]]
        .forEach(([cx, cy, rx, ry]) => {
          const e = document.createElementNS(NS, 'ellipse');
          e.setAttribute('cx', cx); e.setAttribute('cy', cy);
          e.setAttribute('rx', rx); e.setAttribute('ry', ry);
          g.appendChild(e);
        });
      $('roadside').appendChild(g);
    }
  }
  for (const shift of [-W, 0, W]) {
    for (const gx of grassSpec) {
      const x = (gx + shift).toFixed(1);
      const p = document.createElementNS(NS, 'path');
      p.setAttribute('d', `M${x},552 q -4,-13 -9,-18 M${x},552 q 1,-15 6,-20`
                        + ` M${x},552 q 7,-11 13,-13`);
      p.setAttribute('fill', 'none');
      p.setAttribute('stroke', '#79a882');
      p.setAttribute('stroke-width', '2.4');
      p.setAttribute('stroke-linecap', 'round');
      $('roadside').appendChild(p);
    }
  }

  /* -------- 路面标线 + 碎石 -------- */
  const DASH_PERIOD = 200;
  for (let i = -6; i < 14; i++) {
    const r = document.createElementNS(NS, 'rect');
    r.setAttribute('x', i * DASH_PERIOD);
    r.setAttribute('y', 614);
    r.setAttribute('width', 96);
    r.setAttribute('height', 9);
    r.setAttribute('rx', 4.5);
    r.setAttribute('fill', '#fbf2df');
    r.setAttribute('opacity', '0.82');
    $('dashes').appendChild(r);
  }
  for (let i = 0; i < 70; i++) {
    const c = document.createElementNS(NS, 'ellipse');
    c.setAttribute('cx', (rnd() * 2600 - 700).toFixed(1));
    c.setAttribute('cy', (574 + rnd() * 126).toFixed(1));
    c.setAttribute('rx', (1 + rnd() * 2.4).toFixed(1));
    c.setAttribute('ry', (1 + rnd() * 1.4).toFixed(1));
    c.setAttribute('fill', '#a58c6b');
    c.setAttribute('opacity', (0.3 + rnd() * 0.3).toFixed(2));
    $('dashes').appendChild(c);
  }

  /* ================= 骑行动力学 ================= */
  const BB = { x: 520, y: 512 };
  const CRANK_R = 30;
  const HIP = { x: 500, y: 352 };
  const FAR_HIP = { x: 468, y: 346 };
  const L1 = 100, L2 = 96;
  const DEG = Math.PI / 180;

  const nearFoot = $('nearFoot'), farFoot = $('farFoot');
  const nearCrank = $('nearCrank'), farCrank = $('farCrank');
  const nearPedal = $('nearPedal'), farPedal = $('farPedal');
  const limbs = {
    near: { thigh: $('nearThigh'), shin: $('nearShin'), knee: $('nearKnee'), hip: HIP },
    far:  { thigh: $('farThigh'),  shin: $('farShin'),  knee: null, hip: FAR_HIP }
  };

  function solveKnee(hip, foot) {
    const dx = foot.x - hip.x, dy = foot.y - hip.y;
    const raw = Math.hypot(dx, dy) || 1;
    const d = Math.min(Math.max(raw, Math.abs(L1 - L2) + 1), L1 + L2 - 1);
    const ux = dx / raw, uy = dy / raw;
    const a = (L1 * L1 - L2 * L2 + d * d) / (2 * d);
    const h = Math.sqrt(Math.max(0, L1 * L1 - a * a));
    let px = -uy, py = ux;
    if (px < 0) { px = -px; py = -py; }
    return { x: hip.x + ux * a + px * h, y: hip.y + uy * a + py * h };
  }

  function poseLeg(set, foot) {
    const k = solveKnee(set.hip, foot);
    set.thigh.setAttribute('d',
      `M${set.hip.x},${set.hip.y} L${k.x.toFixed(1)},${k.y.toFixed(1)}`);
    set.shin.setAttribute('d',
      `M${k.x.toFixed(1)},${k.y.toFixed(1)} L${foot.x.toFixed(1)},${(foot.y - 3).toFixed(1)}`);
    if (set.knee) {
      set.knee.setAttribute('cx', k.x.toFixed(1));
      set.knee.setAttribute('cy', k.y.toFixed(1));
    }
  }

  const rearWheel = $('rearWheel'), frontWheel = $('frontWheel');
  const chain = $('chain'), upper = $('upper'), head = $('head'), wing = $('wing');
  const scarfTail = $('scarfTail'), scarfTail2 = $('scarfTail2'), birds = $('birds');
  const bird1 = $('bird1'), bird2 = $('bird2'), bird3 = $('bird3');
  const eyelid = $('eyelid');
  const layers = {
    clouds: $('clouds'), farHills: $('farHills'),
    nearHills: $('nearHills'), roadside: $('roadside'), dashes: $('dashes')
  };

  const SPEED = { clouds: 14, farHills: 38, nearHills: 104, roadside: 300, dashes: 470 };
  const PERIOD = { clouds: CLOUD_PERIOD, farHills: W, nearHills: W, roadside: W, dashes: 200 };
  const off = { clouds: 0, farHills: 0, nearHills: 0, roadside: 0, dashes: 0 };

  let last = 0, crankDeg = 30, wheelDeg = 0;
  const reduceMotion = !!(window.matchMedia
    && window.matchMedia('(prefers-reduced-motion: reduce)').matches);
  const MOTION = reduceMotion ? 0.22 : 1;

  function frame(now) {
    if (!last) last = now;
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    const t = now / 1000;

    crankDeg += dt * 320 * MOTION;
    wheelDeg += dt * 540 * MOTION;
    const th = crankDeg * DEG;

    const nearP = { x: BB.x + CRANK_R * Math.cos(th), y: BB.y + CRANK_R * Math.sin(th) };
    const farP = { x: BB.x - CRANK_R * Math.cos(th), y: BB.y - CRANK_R * Math.sin(th) };

    poseLeg(limbs.near, nearP);
    poseLeg(limbs.far, farP);

    nearFoot.setAttribute('transform', `translate(${nearP.x.toFixed(1)} ${(nearP.y - 2).toFixed(1)})`);
    farFoot.setAttribute('transform', `translate(${farP.x.toFixed(1)} ${(farP.y - 2).toFixed(1)})`);
    nearPedal.setAttribute('x', (nearP.x - 15).toFixed(1));
    nearPedal.setAttribute('y', (nearP.y - 4).toFixed(1));
    farPedal.setAttribute('x', (farP.x - 15).toFixed(1));
    farPedal.setAttribute('y', (farP.y - 4).toFixed(1));
    nearCrank.setAttribute('d', `M520,512 L${nearP.x.toFixed(1)},${nearP.y.toFixed(1)}`);
    farCrank.setAttribute('d', `M520,512 L${farP.x.toFixed(1)},${farP.y.toFixed(1)}`);

    const wDeg = (wheelDeg % 360).toFixed(1);
    rearWheel.setAttribute('transform', `rotate(${wDeg} 390 480)`);
    frontWheel.setAttribute('transform', `rotate(${wDeg} 700 480)`);
    chain.setAttribute('stroke-dashoffset', (-crankDeg * 0.32).toFixed(1));

    // 身体律动
    const bob = Math.sin(t * 2.4);
    upper.setAttribute('transform',
      `rotate(${(bob * 0.9).toFixed(2)} 516 374) translate(0 ${(bob * 1.4).toFixed(2)})`);
    head.setAttribute('transform', `rotate(${(Math.sin(t * 2.4 + 0.5) * 1.6).toFixed(2)} 618 200)`);
    wing.setAttribute('transform', `rotate(${(Math.sin(t * 2.4 + 1.5) * 1.2).toFixed(2)} 500 286)`);

    // 眨眼：每 ~3.6 秒眨一次
    const blinkPhase = (t % 3.6) / 0.16;
    const blink = blinkPhase < 1 ? Math.sin(blinkPhase * Math.PI) * 9.5 : 0;
    eyelid.setAttribute('ry', blink.toFixed(2));

    // 围巾飘动
    const p1 = [[510, 242]], p2 = [[513, 250]];
    for (let i = 1; i <= 4; i++) {
      const w1 = Math.sin(t * 6.6 - i * 1.0) * (1.5 + i * 2.8);
      const w2 = Math.sin(t * 6.6 - i * 1.0 + 1.3) * (1.2 + i * 2.0);
      p1.push([510 - i * 18, 242 - i * 2 + w1]);
      p2.push([513 - i * 17, 250 + i * 1 + w2]);
    }
    scarfTail.setAttribute('d', catmull(p1));
    scarfTail2.setAttribute('d', catmull(p2));

    // 视差滚动
    for (const k in SPEED) {
      off[k] = (off[k] + SPEED[k] * dt * MOTION) % PERIOD[k];
      layers[k].setAttribute('transform', `translate(${(-off[k]).toFixed(1)} 0)`);
    }

    // 飞鸟：在左侧天空中缓慢往返，翅膀轻微扑动
    const birdX = Math.sin(t * 0.16) * 34 - 40;
    const birdY = Math.sin(t * 0.42) * 7;
    birds.setAttribute('transform',
      `translate(${birdX.toFixed(1)} ${birdY.toFixed(1)})`);
    const flap = Math.sin(t * 5.4) * 0.22 + 1;
    bird1.setAttribute('transform', `translate(212 236) scale(1 ${flap.toFixed(3)}) translate(-212 -236)`);
    bird2.setAttribute('transform', `translate(296 192) scale(1 ${(2 - flap).toFixed(3)}) translate(-296 -192)`);
    bird3.setAttribute('transform', `translate(250 284) scale(1 ${flap.toFixed(3)}) translate(-250 -284)`);

    requestAnimationFrame(frame);
  }

  requestAnimationFrame(frame);

  /* -------- 自适应画幅：viewBox 比例始终等于视口，永不裁切/留白 -------- */
  const svgEl = $('scene');
  function fitViewBox() {
    const w = window.innerWidth || 1200;
    const h = window.innerHeight || 700;
    const aspect = w / h;
    let worldW = 700 * aspect;
    let worldH = 700;
    if (worldW < 760) { worldW = 760; worldH = 760 / aspect; }
    if (worldW > 2800) { worldW = 2800; worldH = 2800 / aspect; }
    const cx = 570, cy = 380;
    svgEl.setAttribute('viewBox',
      `${(cx - worldW / 2).toFixed(1)} ${(cy - worldH / 2).toFixed(1)}`
      + ` ${worldW.toFixed(1)} ${worldH.toFixed(1)}`);
  }
  fitViewBox();
  window.addEventListener('resize', fitViewBox);
})();
