        farLeg: byId('far-leg'), nearFoot: byId('near-foot'), farFoot: byId('far-foot'),
        scarf: byId('scarf-tail'), scarfSmall: byId('scarf-tail-small'), eye: byId('eye'),
        gull1: byId('gull-one'), gull2: byId('gull-two'), speedLines: byId('speed-lines')
      };
      const drifting = [...document.querySelectorAll('[data-drift]')].map(el => ({
        el, kind: el.dataset.drift, x: Number(el.dataset.x), y: Number(el.dataset.y),
        scale: Number(el.dataset.scale || 1)
      }));
      const driftRatio = { cloud: .055, wave: .18, middle: .4, road: 1, plant: 1, pebble: 1 };
      const wrap = (n, width) => ((n % width) + width) % width;
      const fmt = n => n.toFixed(2);
      function solveLeg(hip, foot) {
        const upper = 86;
        const lower = 94;
        const dx = foot.x - hip.x;
        const dy = foot.y - hip.y;
        const distance = Math.max(.001, Math.min(Math.hypot(dx, dy), upper + lower - .001));
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const outward = Math.sqrt(Math.max(0, upper * upper - along * along));
        // Both knees bend forward, rather than merely spinning the entire legs.
        const knee = {
          x: hip.x + along * dx / distance + outward * dy / distance,
          y: hip.y + along * dy / distance - outward * dx / distance
        };
        return `M${fmt(hip.x)} ${fmt(hip.y)}L${fmt(knee.x)} ${fmt(knee.y)}L${fmt(foot.x)} ${fmt(foot.y)}`;
      }
      function render() {
        const angle = elapsed * crankRate;
        const wheelAngle = angle * gearRatio;
        const wheelDegrees = wheelAngle * 180 / Math.PI;
        const roadDistance = wheelAngle * wheelRadius;
        const bob = Math.sin(angle * 2) * 1.8;
        parts.rear.setAttribute('transform', `rotate(${fmt(wheelDegrees)})`);
        parts.front.setAttribute('transform', `rotate(${fmt(wheelDegrees)})`);
        parts.crank.setAttribute('transform', `rotate(${fmt(angle * 180 / Math.PI)} 610 480)`);
        parts.pelican.setAttribute('transform', `translate(0 ${fmt(bob)})`);
        const near = { x: 610 + Math.cos(angle) * crankRadius, y: 480 + Math.sin(angle) * crankRadius };
        const far = { x: 610 - Math.cos(angle) * crankRadius, y: 480 - Math.sin(angle) * crankRadius };
        const nearPath = solveLeg({ x: 571, y: 348 + bob }, near);
        parts.nearLeg.setAttribute('d', nearPath);
        parts.nearHighlight.setAttribute('d', nearPath);
        parts.farLeg.setAttribute('d', solveLeg({ x: 550, y: 348 + bob }, far));
        parts.nearFoot.setAttribute('transform', `translate(${fmt(near.x)} ${fmt(near.y)})`);
        parts.farFoot.setAttribute('transform', `translate(${fmt(far.x)} ${fmt(far.y)})`);
        drifting.forEach(item => {
          // Road dashes use a 1500-unit loop, divisible by their 300-unit spacing.
          const width = item.kind === 'road' ? 1500 : 1460;
          const x = wrap(item.x + 180 - roadDistance * driftRatio[item.kind], width) - 180;
          item.el.setAttribute('transform', `translate(${fmt(x)} ${item.y}) scale(${item.scale})`);
        });
        const flutter = Math.sin(elapsed * 6.5) * 5;
        parts.scarf.setAttribute('d', `M624 247C597 247 580 ${238 + flutter} 558 ${242 + flutter}S512 ${256 - flutter} 492 ${243 + flutter}L503 ${261 + flutter}C524 ${274 - flutter} 546 ${259 + flutter} 567 ${260 + flutter}S601 270 627 264Z`);
        parts.scarfSmall.setAttribute('d', `M619 257Q581 ${272 - flutter} 548 ${261 - flutter}L556 ${276 - flutter}Q585 ${287 + flutter} 626 269Z`);
        const blinkPhase = wrap(elapsed, 5.8);
        const eyeScale = blinkPhase > 5.6 ? Math.max(.08, Math.abs(blinkPhase - 5.7) * 10) : 1;
        parts.eye.setAttribute('transform', `translate(0 178) scale(1 ${fmt(eyeScale)}) translate(0 -178)`);
        parts.gull1.setAttribute('transform', `translate(${319 - Math.sin(elapsed * .2) * 28} ${230 + Math.sin(elapsed * 1.6) * 3}) scale(.8)`);
        parts.gull2.setAttribute('transform', `translate(${350 - Math.sin(elapsed * .2) * 28} ${209 + Math.sin(elapsed * 1.6 + 1) * 3}) scale(.55)`);
        parts.speedLines.setAttribute('opacity', fmt(.4 + Math.sin(elapsed * 4) * .15));
      }
      function tick(now) {
        frameId = null;
        if (!playing || document.hidden) return;
        if (lastTime !== null) elapsed += Math.min((now - lastTime) / 1000, .05) * speed;
        lastTime = now;
        render();
        frameId = requestAnimationFrame(tick);
      }
      function syncPlayback() {
        ui.label.textContent = playing ? '暂停一下' : '继续骑行';
        ui.button.setAttribute('aria-label', playing ? '暂停动画' : '播放动画');
        ui.button.setAttribute('aria-pressed', String(!playing));
        ui.icon.setAttribute('d', playing ? 'M4 3h3v10H4zm5 0h3v10H9z' : 'M4 2.5L13 8l-9 5.5z');
        ui.status.textContent = playing ? '正在海岸骑行' : '歇一会儿，等风来';
        ui.dot.style.background = playing ? 'var(--green)' : 'var(--orange)';
        lastTime = null;
        if (frameId !== null) cancelAnimationFrame(frameId);
        frameId = playing && !document.hidden ? requestAnimationFrame(tick) : null;
      }
      ui.button.addEventListener('click', () => { playing = !playing; syncPlayback(); });
      ui.range.addEventListener('input', () => {
        speed = Number(ui.range.value);
        ui.value.textContent = `${speed.toFixed(2)}×`;
        ui.range.setAttribute('aria-valuetext', `${speed} 倍速`);
      });
      document.addEventListener('keydown', event => {
        const target = event.target;
        if (event.code !== 'Space' || event.repeat || target.closest('button, input, textarea, select, [contenteditable="true"]')) return;
        event.preventDefault();
        playing = !playing;
        syncPlayback();
      });
      document.addEventListener('visibilitychange', syncPlayback);
      const onMotionChange = event => {
        if (event.matches) { playing = false; syncPlayback(); }
      };
      if (reducedMotion.addEventListener) reducedMotion.addEventListener('change', onMotionChange);
      else if (reducedMotion.addListener) reducedMotion.addListener(onMotionChange);
      render();
      syncPlayback();
    })();
  </script>
