  <script>
    (() => {
      const $ = id => document.getElementById(id);
      const button = $('motion-toggle');
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      let running = !reducedMotion.matches;
      let time = 0;
      let previous = null;
      let frame = null;
      const crankX = 575, crankY = 514, radius = 22;

      function legPath(hipX, hipY, footX, footY) {
        // Two fixed-length segments give each leg a natural bend as the pedal moves.
        const dx = footX - hipX, dy = footY - hipY;
        const distance = Math.hypot(dx, dy) || 1;
        const upper = 87, lower = 82;
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const bend = Math.sqrt(Math.max(0, upper * upper - along * along));
        const kneeX = hipX + dx * along / distance + dy * bend / distance;
        const kneeY = hipY + dy * along / distance - dx * bend / distance;
        return `M${hipX} ${hipY} L${kneeX.toFixed(1)} ${kneeY.toFixed(1)} L${footX.toFixed(1)} ${footY.toFixed(1)}`;
      }

      function render() {
        const angle = time / 2400 * Math.PI * 2 - Math.PI / 2;
        const x = Math.cos(angle) * radius, y = Math.sin(angle) * radius;
        const ax = crankX + x, ay = crankY + y, bx = crankX - x, by = crankY - y;
        const wheelAngle = time / 2400 * 360 * 2.15;
        $('rear-spokes').setAttribute('transform', `rotate(${wheelAngle})`);
        $('front-spokes').setAttribute('transform', `rotate(${wheelAngle})`);
        $('crank').setAttribute('d', `M${bx} ${by} L${ax} ${ay}`);
        $('rear-pedal').setAttribute('d', `M${bx - 8} ${by} h16`);
        $('front-pedal').setAttribute('d', `M${ax - 8} ${ay} h16`);
        $('far-leg').setAttribute('d', legPath(534, 384, bx, by - 3));
        $('near-leg').setAttribute('d', legPath(552, 385, ax, ay - 3));
        $('far-foot').setAttribute('d', `M${bx - 8} ${by - 4} h24`);
        $('near-foot').setAttribute('d', `M${ax - 8} ${ay - 4} h24`);
      }

      function tick(now) {
        if (!running) return;
        if (previous !== null) time += Math.min(now - previous, 50);
        previous = now;
        render();
        frame = requestAnimationFrame(tick);
      }

      function updateButton() {
        document.body.dataset.paused = String(!running);
        const text = running ? '暂停动画' : '播放动画';
        button.setAttribute('aria-label', text);
        button.querySelector('.button-label').textContent = text;
        button.querySelector('.button-icon').textContent = running ? 'Ⅱ' : '▶';
      }

      button.addEventListener('click', () => {
        running = !running;
        previous = null;
        if (running) frame = requestAnimationFrame(tick);
        else if (frame !== null) cancelAnimationFrame(frame);
        updateButton();
      });
      document.addEventListener('visibilitychange', () => { previous = null; });
      render();
      updateButton();
      if (running) frame = requestAnimationFrame(tick);
    })();
  </script>
