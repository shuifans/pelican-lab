<script>
(function () {
  var $ = function (id) { return document.getElementById(id); };
  var svg = $('scene');
  var BBX = 432, BBY = 400, CRANK = 34, L = 90;
  var hipN = { x: 392, y: 266 }, hipF = { x: 386, y: 264 };

  function knee(hx, hy, fx, fy) {
    var dx = fx - hx, dy = fy - hy;
    var d = Math.min(Math.hypot(dx, dy), 2 * L - 0.5);
    var a = Math.acos(Math.min(1, Math.max(-1, d / (2 * L))));
    var b = Math.atan2(dy, dx) - a;
    return [hx + L * Math.cos(b), hy + L * Math.sin(b)];
  }

  function legPath(hx, hy, px, py) {
    var fx = px - 2, fy = py - 5;
    var k = knee(hx, hy, fx, fy);
    return 'M' + hx.toFixed(1) + ' ' + hy.toFixed(1) +
      ' L' + k[0].toFixed(1) + ' ' + k[1].toFixed(1) +
      ' L' + fx.toFixed(1) + ' ' + fy.toFixed(1);
  }

  function render(t) {
    var th = t / 1600 * 2 * Math.PI;
    var deg = th * 180 / Math.PI;
    var wheel = t / 1400 * 360;
    $('spokesR').setAttribute('transform', 'rotate(' + wheel + ' 320 396)');
    $('spokesF').setAttribute('transform', 'rotate(' + wheel + ' 560 396)');
    $('crank').setAttribute('transform', 'rotate(' + deg + ' 432 400)');
    $('farArm').setAttribute('transform', 'rotate(' + deg + ' 432 400)');

    var bob = 2.2 * Math.sin(2 * th);
    $('rider').setAttribute('transform', 'translate(0 ' + bob.toFixed(2) + ')');

    var pnx = BBX + CRANK * Math.cos(th), pny = BBY + CRANK * Math.sin(th);
    var pfx = BBX - CRANK * Math.cos(th), pfy = BBY - CRANK * Math.sin(th);
    $('legNear').setAttribute('d', legPath(hipN.x, hipN.y + bob, pnx, pny));
    $('legFar').setAttribute('d', legPath(hipF.x, hipF.y + bob, pfx, pfy));
    $('pedalNear').setAttribute('transform', 'translate(' + pnx.toFixed(1) + ' ' + pny.toFixed(1) + ')');
    $('pedalFar').setAttribute('transform', 'translate(' + pfx.toFixed(1) + ' ' + pfy.toFixed(1) + ')');
  }

  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    svg.pauseAnimations();
    render(400);
    return;
  }
  var t0 = performance.now();
  requestAnimationFrame(function loop(now) {
    render(now - t0);
    requestAnimationFrame(loop);
  });
})();
</script>
