      function solveKneeIK(hx, hy, ax, ay, L1, L2) {
        const dx = ax - hx;
        const dy = ay - hy;
        let d = Math.hypot(dx, dy);
        const minD = Math.abs(L1 - L2) + 1;
        const maxD = L1 + L2 - 1;
        if (d < minD) d = minD;
        if (d > maxD) d = maxD;

        const cosAlpha = (L1 * L1 + d * d - L2 * L2) / (2 * L1 * d);
        const clampedCos = Math.max(-1, Math.min(1, cosAlpha));
        const alpha = Math.acos(clampedCos);
        const baseAngle = Math.atan2(dy, dx);

        // 减去 alpha 使膝盖朝前方 (+x, 车头方向) 自然弯曲
        const thighAngle = baseAngle - alpha;
        return {
          x: hx + L1 * Math.cos(thighAngle),
          y: hy + L1 * Math.sin(thighAngle)
        };
      }
