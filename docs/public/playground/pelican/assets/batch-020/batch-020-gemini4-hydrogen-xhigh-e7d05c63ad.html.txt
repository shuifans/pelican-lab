      // 2-Link 逆向运动学（膝盖自然朝车把方向前方弯曲）
      function solveLegIK(hipX, hipY, footX, footY, L1, L2) {
        const dx = footX - hipX;
        const dy = footY - hipY;
        const minD = Math.abs(L1 - L2) + 2;
        const maxD = L1 + L2 - 2;
        const d = Math.max(minD, Math.min(maxD, Math.hypot(dx, dy)));
        const baseAngle = Math.atan2(dy, dx);
        const cosAlpha = Math.max(-1, Math.min(1, (L1 * L1 + d * d - L2 * L2) / (2 * L1 * d)));
        const alpha = Math.acos(cosAlpha);
        const kneeAngle = baseAngle - alpha;
        return {
          kneeX: hipX + L1 * Math.cos(kneeAngle),
          kneeY: hipY + L1 * Math.sin(kneeAngle)
        };
      }

      // 构建自然羽毛大腿轮廓路径
      function buildThighPath(hx, hy, kx, ky) {
        const dx = kx - hx;
        const dy = ky - hy;
        const len = Math.hypot(dx, dy) || 1;
        const nx = -dy / len;
        const ny = dx / len;
        const rHip = 17;
        const rKnee = 6.2;
        const mx = (hx + kx) * 0.5;
        const my = (hy + ky) * 0.5;

        const p1x = hx + nx * rHip;
        const p1y = hy + ny * rHip;
        const p2x = kx + nx * rKnee;
        const p2y = ky + ny * rKnee;
        const p3x = kx - nx * rKnee;
        const p3y = ky - ny * rKnee;
        const p4x = hx - nx * rHip;
        const p4y = hy - ny * rHip;

        return `M ${p1x.toFixed(1)} ${p1y.toFixed(1)}
                Q ${(mx + nx * 13).toFixed(1)} ${(my + ny * 13).toFixed(1)}, ${p2x.toFixed(1)} ${p2y.toFixed(1)}
                A ${rKnee} ${rKnee} 0 0 1 ${p3x.toFixed(1)} ${p3y.toFixed(1)}
                Q ${(mx - nx * 12).toFixed(1)} ${(my - ny * 12).toFixed(1)}, ${p4x.toFixed(1)} ${p4y.toFixed(1)}
                A ${rHip} ${rHip} 0 0 1 ${p1x.toFixed(1)} ${p1y.toFixed(1)} Z`;
      }

      let audioCtx = null;
      function getAudioCtx() {
        if (!audioCtx && (window.AudioContext || window.webkitAudioContext)) {
          audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
        if (audioCtx && audioCtx.state === "suspended") {
          audioCtx.resume();
        }
        return audioCtx;

      function playBellSound() {
        const ctx = getAudioCtx();
        if (!ctx) return;
        const now = ctx.currentTime;
        [0, 0.14].forEach((delay) => {
          [2120, 4240, 3180].forEach((freq, idx) => {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = "sine";
            osc.frequency.setValueAtTime(freq, now + delay);
            const peak = idx === 0 ? 0.14 : idx === 1 ? 0.06 : 0.03;
            gain.gain.setValueAtTime(0.0001, now + delay);
            gain.gain.exponentialRampToValueAtTime(peak, now + delay + 0.008);
            gain.gain.exponentialRampToValueAtTime(0.0001, now + delay + 0.42);
            osc.connect(gain);
            gain.connect(ctx.destination);
            osc.start(now + delay);
            osc.stop(now + delay + 0.45);
          });
        });
      }

      function playPelicanCallSound() {
        const ctx = getAudioCtx();
        if (!ctx) return;
        const now = ctx.currentTime;
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = "triangle";
        osc.frequency.setValueAtTime(280, now);
        osc.frequency.exponentialRampToValueAtTime(430, now + 0.11);
        osc.frequency.exponentialRampToValueAtTime(320, now + 0.28);
        gain.gain.setValueAtTime(0.001, now);
        gain.gain.exponentialRampToValueAtTime(0.11, now + 0.03);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.32);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start(now);
        osc.stop(now + 0.35);
      }

          const tilt = item.isBoat ? Math.cos(elapsed * 2.0 + item.phase) * 2.2 : 0;
          item.node.setAttribute("transform", `translate(${x.toFixed(1)}, ${(item.y + bob).toFixed(1)}) rotate(${tilt.toFixed(2)})`);
        });

        waveNodes.forEach((w) => {
          const x = wrapPos(w.baseX - worldDist * 0.22 - elapsed * 12, w.period, -120);
          w.node.setAttribute("transform", `translate(${x.toFixed(1)}, ${w.y})`);
        });

        midItems.forEach((m) => {
          const x = wrapPos(m.baseX - worldDist * 0.52, m.period, -120);
          m.node.setAttribute("transform", `translate(${x.toFixed(1)}, ${m.y}) scale(${m.scale})`);
        });

        fencePosts.forEach((p) => {
          const x = wrapPos(p.baseX - worldDist * 0.58, p.period, -120);
          p.node.setAttribute("transform", `translate(${x.toFixed(1)}, 0)`);
        });

        dashNodes.forEach((d) => {
          const x = wrapPos(d.baseX - worldDist, dashPeriod * 9, -120);
          d.node.setAttribute("x", x.toFixed(1));
        });

        const windOpacity = Math.min(0.45, 0.12 + (speedMultiplier - 0.4) * 0.16);
        windNodes.forEach((w) => {
          const wx = wrapPos(w.baseX - (worldDist * 1.3 + elapsed * 180) * w.speedMul, 1440, -160);
          w.node.setAttribute("transform", `translate(${wx.toFixed(1)}, ${w.y})`);
          w.node.setAttribute("opacity", isPlaying ? windOpacity.toFixed(2) : "0.06");
        });

        fgTufts.forEach((f) => {
          const fx = wrapPos(f.baseX - worldDist * 1.18, f.period, -140);
          f.node.setAttribute("transform", `translate(${fx.toFixed(1)}, ${f.y})`);
        });
      }

      // 动画主循环
      function tick(now) {
        const dt = Math.min(0.05, Math.max(0.001, (now - lastTime) / 1000));
        lastTime = now;
        renderFrame(dt);
        requestAnimationFrame(tick);
      }

      // 控件事件绑定
      const togglePlayBtn = document.getElementById("togglePlayBtn");
      const playIcon = document.getElementById("playIcon");
      const playLabel = document.getElementById("playLabel");
      const speedSlider = document.getElementById("speedSlider");
      const speedValue = document.getElementById("speedValue");
      const bellBtn = document.getElementById("bellBtn");
      const squawkBtn = document.getElementById("squawkBtn");
      const themeBtn = document.getElementById("themeBtn");

      function togglePlay() {
        isPlaying = !isPlaying;
        playIcon.textContent = isPlaying ? "⏸" : "▶";
        playLabel.textContent = isPlaying ? "暂停骑行" : "继续骑行";
      }

      function toggleTheme() {
        isSunset = !isSunset;
        document.body.classList.toggle("sunset-mode", isSunset);
        themeBtn.textContent = isSunset ? "☀️ 切换晴昼" : "🌅 切换日落";

        document.getElementById("skyStop0").setAttribute("stop-color", isSunset ? "#384170" : "#6AC0DF");
        document.getElementById("skyStop1").setAttribute("stop-color", isSunset ? "#D16D7A" : "#BCE8F6");
        document.getElementById("skyStop2").setAttribute("stop-color", isSunset ? "#FFC278" : "#FDF2D2");
        document.getElementById("sunGlow0").setAttribute("stop-color", isSunset ? "#FFE49E" : "#FFFBE0");
        document.getElementById("sunGlow1").setAttribute("stop-color", isSunset ? "#FF8A5B" : "#FFE28A");
        document.getElementById("sun-disc").setAttribute("fill", isSunset ? "#FFCC70" : "#FFF7CA");
        document.getElementById("sun-group").setAttribute("transform", isSunset ? "translate(950, 225)" : "translate(970, 112)");
        document.getElementById("oceanStop0").setAttribute("stop-color", isSunset ? "#645C8A" : "#3BA4C9");
        document.getElementById("oceanStop1").setAttribute("stop-color", isSunset ? "#2E3A5E" : "#227196");
        document.getElementById("hillStop0").setAttribute("stop-color", isSunset ? "#5D7D69" : "#87BA8A");
        document.getElementById("hillStop1").setAttribute("stop-color", isSunset ? "#3D5749" : "#61946B");
        document.getElementById("headlamp-beam").setAttribute("opacity", isSunset ? "0.68" : "0.22");
        lighthouseBeam.setAttribute("opacity", isSunset ? "0.38" : "0");
      }

      togglePlayBtn.addEventListener("click", togglePlay);
      speedSlider.addEventListener("input", (e) => {
        speedMultiplier = parseFloat(e.target.value);
        speedValue.textContent = `${speedMultiplier.toFixed(1)}x`;
      });
      bellBtn.addEventListener("click", triggerBell);
      squawkBtn.addEventListener("click", triggerSquawk);
      themeBtn.addEventListener("click", toggleTheme);

      document.getElementById("bike-bell").addEventListener("click", triggerBell);
      pelicanUpper.addEventListener("click", triggerSquawk);

      window.addEventListener("keydown", (e) => {
        if (e.code === "Space") {
          e.preventDefault();
          togglePlay();
        } else if (e.key === "b" || e.key === "B") {
          triggerBell();
        } else if (e.key === "g" || e.key === "G") {
          triggerSquawk();
        }
      });

      // 首帧立即初始化，然后进入 60fps 循环
      renderFrame(0.016);
      requestAnimationFrame(tick);
    })();
  </script>
