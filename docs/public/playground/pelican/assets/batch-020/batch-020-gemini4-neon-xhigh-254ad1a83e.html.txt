      function solveLegIK(hx, hy, ax, ay, L1, L2) {
        const dx = ax - hx;
        const dy = ay - hy;
        let dist = Math.hypot(dx, dy);
        const minD = Math.abs(L1 - L2) + 1;
        const maxD = L1 + L2 - 1;
        if (dist < minD) dist = minD;
        if (dist > maxD) dist = maxD;

        const kneeAngle = baseAngle - alpha;
        const kx = hx + L1 * Math.cos(kneeAngle);
        const ky = hy + L1 * Math.sin(kneeAngle);

        return { kx, ky, thighAngle: kneeAngle };
      }

            nextBlinkAt = time + 2.8 + Math.random() * 2.6;
          }
          if (blinkTimer > 0) blinkTimer = Math.max(0, blinkTimer - dt);
        }

        // Allow bell/squawk animation to finish even if triggered while paused
        if (squawkTimer > 0) {
          squawkTimer = Math.max(0, squawkTimer - rawDt);
        }

        // ------------------------------------------------------------
        // 1. Parallax Environment Scrolling
        // ------------------------------------------------------------
        const cloudX = -((roadScroll * 0.045 + time * 10) % 1200);
        const horizonX = -((roadScroll * 0.14) % 1200);
        const midX = -((roadScroll * 0.48) % 1200);
        const roadX = -(roadScroll % 200);
        const fgX = -((roadScroll * 1.18) % 600);

        els.clouds.setAttribute('transform', `translate(${cloudX.toFixed(1)}, 0)`);
        els.horizon.setAttribute('transform', `translate(${horizonX.toFixed(1)}, 0)`);
        els.midground.setAttribute('transform', `translate(${midX.toFixed(1)}, 0)`);
        els.road.setAttribute('transform', `translate(${roadX.toFixed(1)}, 0)`);
        els.foreground.setAttribute('transform', `translate(${fgX.toFixed(1)}, 0)`);

        // Distant gliding seagulls
        const gull1X = 1040 - ((time * 28 + 180) % 1120);
        const gull1Y = 112 + Math.sin(time * 2.1) * 8;
        const g1Flap = Math.sin(time * 6.5) * 6;
        els.gull1.setAttribute('transform', `translate(${gull1X.toFixed(1)}, ${gull1Y.toFixed(1)})`);
        els.gull1.setAttribute('d', `M 0 0 Q -8 ${(-6 + g1Flap).toFixed(1)} -16 ${(g1Flap * 1.1).toFixed(1)} M 0 0 Q 8 ${(-6 + g1Flap).toFixed(1)} 16 ${(g1Flap * 1.1).toFixed(1)}`);

        const gull2X = 1040 - ((time * 22 + 540) % 1120);
        const gull2Y = 148 + Math.cos(time * 1.7) * 6;
        const g2Flap = Math.sin(time * 5.8 + 1.4) * 5;
        els.gull2.setAttribute('transform', `translate(${gull2X.toFixed(1)}, ${gull2Y.toFixed(1)})`);
        els.gull2.setAttribute('d', `M 0 0 Q -6 ${(-5 + g2Flap).toFixed(1)} -12 ${(g2Flap * 1.1).toFixed(1)} M 0 0 Q 6 ${(-5 + g2Flap).toFixed(1)} 12 ${(g2Flap * 1.1).toFixed(1)}`);

        const gull3X = 1040 - ((time * 32 + 860) % 1120);
        const gull3Y = 92 + Math.sin(time * 2.4 + 2.2) * 7;
        const g3Flap = Math.sin(time * 7.2 + 2.6) * 5.5;
        els.gull3.setAttribute('transform', `translate(${gull3X.toFixed(1)}, ${gull3Y.toFixed(1)})`);
        els.gull3.setAttribute('d', `M 0 0 Q -7 ${(-5 + g3Flap).toFixed(1)} -14 ${(g3Flap * 1.1).toFixed(1)} M 0 0 Q 7 ${(-5 + g3Flap).toFixed(1)} 14 ${(g3Flap * 1.1).toFixed(1)}`);

        // ------------------------------------------------------------
        // 2. Wheels, Sprocket, Chainring & Chain
        // ------------------------------------------------------------
        const wheelDeg = (wheelAngle * 180) / Math.PI;
        const crankDeg = (crankAngle * 180) / Math.PI;

        els.rearWheelSpin.setAttribute('transform', `rotate(${wheelDeg.toFixed(1)})`);
        els.frontWheelSpin.setAttribute('transform', `rotate(${wheelDeg.toFixed(1)})`);
        els.rearSprocketSpin.setAttribute('transform', `rotate(${wheelDeg.toFixed(1)})`);
        els.chainringSpin.setAttribute('transform', `rotate(${crankDeg.toFixed(1)})`);
        els.chainLinks.setAttribute('stroke-dashoffset', (-chainOffset).toFixed(2));

        // ------------------------------------------------------------
        // 3. Pelican Cycling Cadence Bob, Lean & Head Stabilization
        // ------------------------------------------------------------
        const bobY = Math.sin(2 * crankAngle - 0.5) * (2.6 + currentSpeed * 1.3);
        const sprintLean = Math.max(0, (currentSpeed - 0.8) * 4.2);
        const leanDeg = sprintLean + Math.sin(crankAngle) * 1.5;
        const leanRad = (leanDeg * Math.PI) / 180;

        // Subtle shadow breathing in sync with cadence
        els.bikeShadow.setAttribute('rx', (188 - bobY * 0.7).toFixed(1));

        // Apply torso transform around saddle pivot
        els.rider.setAttribute(
          'transform',
          `translate(0, ${bobY.toFixed(2)}) rotate(${leanDeg.toFixed(2)}, ${PIVOT.x}, ${PIVOT.y})`
        );

        // Counter-bob & playful tilt on head
        const sqProgress = squawkTimer > 0 ? 1 - squawkTimer / SQUAWK_DURATION : 0;
        const squawkEnv = squawkTimer > 0 ? Math.sin(sqProgress * Math.PI) : 0;

        const headNod = -leanDeg * 0.45 + Math.sin(2 * crankAngle - 1.3) * 2.2 - squawkEnv * 7;
        const headBobY = -bobY * 0.35;
        els.head.setAttribute('transform', `translate(502, ${(124 + headBobY).toFixed(2)}) rotate(${headNod.toFixed(2)})`);

        // Crest & tail wind flutter
        const crestFlutter = Math.sin(time * (9 + currentSpeed * 6)) * (3 + currentSpeed * 2.5);
        els.crest.setAttribute('transform', `rotate(${crestFlutter.toFixed(1)}, -18, -8)`);

        const tailWag = Math.sin(2 * crankAngle - 1.0) * 4.5 + Math.sin(time * 8) * 1.5;
        els.tail.setAttribute('transform', `translate(364, 240) rotate(${tailWag.toFixed(1)})`);

        // Iconic Beak & Gular Pouch Bounce
        const pouchJiggle = Math.sin(2 * crankAngle - 1.4) * (2.2 + currentSpeed * 1.2) + squawkEnv * 9;
        const upperBeakRot = -squawkEnv * 19;
        const lowerBeakRot = squawkEnv * 8;

        els.beakUpper.setAttribute('transform', `translate(18, 4) rotate(${upperBeakRot.toFixed(1)})`);
        els.beakLower.setAttribute('transform', `translate(18, 4) rotate(${lowerBeakRot.toFixed(1)})`);
        els.mouthInterior.setAttribute('opacity', Math.min(1, squawkEnv * 2.2).toFixed(2));

        const pouchBellyY = (58 + pouchJiggle).toFixed(1);
        const pouchMidY = (44 + pouchJiggle * 0.85).toFixed(1);
        els.pouchPath.setAttribute(
          'd',
          `M -22 14 C -8 ${(46 + pouchJiggle * 0.5).toFixed(1)}, 36 ${pouchBellyY}, 72 ${pouchMidY} C 96 ${(34 + pouchJiggle * 0.35).toFixed(1)}, 110 18, 116 6 L 0 2 Z`
        );

        // Eye Expression States (Happy squint during squawk > Blink > Normal)
        if (squawkTimer > 0.12) {
          els.eyeNormal.setAttribute('visibility', 'hidden');
          els.eyeBlink.setAttribute('visibility', 'hidden');
          els.eyeHappy.setAttribute('visibility', 'visible');
        } else if (blinkTimer > 0) {
          els.eyeNormal.setAttribute('visibility', 'hidden');
          els.eyeBlink.setAttribute('visibility', 'visible');
          els.eyeHappy.setAttribute('visibility', 'hidden');
        } else {
          els.eyeNormal.setAttribute('visibility', 'visible');
          els.eyeBlink.setAttribute('visibility', 'hidden');
          els.eyeHappy.setAttribute('visibility', 'hidden');
        }

        // Scarf billowing in wind
        updateScarf(time, currentSpeed);

        // ------------------------------------------------------------
        // 4. Articulated Wings Locked to Handlebar Grips
        // ------------------------------------------------------------
        const nearShoulderWorld = transformRiderPoint(NEAR_SHOULDER_BASE, bobY, leanRad);
        const farShoulderWorld = transformRiderPoint(FAR_SHOULDER_BASE, bobY, leanRad);
        updateWing(nearShoulderWorld, GRIP_NEAR, true);
        updateWing(farShoulderWorld, GRIP_FAR, false);

        // ------------------------------------------------------------
        // 5. Two-Bone IK Legs & Pedals (Near & Far, 180 deg apart)
        // ------------------------------------------------------------
        const nearHip = transformRiderPoint(NEAR_HIP_BASE, bobY, leanRad);
        const farHip = transformRiderPoint(FAR_HIP_BASE, bobY, leanRad);

        updateLegRig(crankAngle, nearHip, true);
        updateLegRig(crankAngle + Math.PI, farHip, false);

        // ------------------------------------------------------------
        // 6. Basket Fish Bounce, Bell Ring & Wind Speed Streaks
        // ------------------------------------------------------------
        const fishBounceY = Math.sin(2 * crankAngle - 0.9) * 1.6 - Math.abs(Math.sin(sqProgress * Math.PI * 2)) * (squawkTimer > 0 ? 11 : 0);
        const fishTilt = Math.sin(time * 6) * 2.5;
        els.basketFish.setAttribute('transform', `translate(0, ${fishBounceY.toFixed(1)}) rotate(${fishTilt.toFixed(1)}, 30, 14)`);

        if (squawkTimer > 0) {
          const bellShake = Math.sin(sqProgress * Math.PI * 10) * 10 * (1 - sqProgress);
          els.bikeBell.setAttribute('transform', `translate(563, 250) rotate(${bellShake.toFixed(1)})`);

          els.fxBell.setAttribute('opacity', (Math.sin(sqProgress * Math.PI) * 0.95).toFixed(2));
          els.bellRing1.setAttribute('r', (12 + sqProgress * 34).toFixed(1));
          els.bellRing2.setAttribute('r', (20 + sqProgress * 50).toFixed(1));
          els.musicNote.setAttribute(
            'transform',
            `translate(${(650 + sqProgress * 28).toFixed(1)}, ${(112 - sqProgress * 46).toFixed(1)}) rotate(${(Math.sin(sqProgress * 8) * 12).toFixed(1)})`
          );
          els.squawkLines.setAttribute(
            'transform',
            `translate(${(sqProgress * 12).toFixed(1)}, ${(-sqProgress * 6).toFixed(1)})`
          );
        } else {
          els.bikeBell.setAttribute('transform', 'translate(563, 250)');
          els.fxBell.setAttribute('opacity', '0');
        }

        // Dynamic Wind Speed Streaks when cruising fast / sprinting
        const windAlpha = Math.max(0, Math.min(0.65, (currentSpeed - 0.85) * 0.72));
        els.windStreaks.setAttribute('opacity', windAlpha.toFixed(2));
        if (windAlpha > 0.01) {
          const offsets = [0, 320, 610, 840];
          const ys = [118, 194, 276, 156];
          const lens = [92, 115, 84, 76];
          els.windLines.forEach((line, i) => {
            const wx = 1080 - ((roadScroll * 1.45 + offsets[i]) % 1240);
            line.setAttribute('x1', wx.toFixed(1));
            line.setAttribute('x2', (wx + lens[i] * currentSpeed * 0.7).toFixed(1));
            line.setAttribute('y1', ys[i]);
            line.setAttribute('y2', ys[i]);
          });
        }

        requestAnimationFrame(render);
      }

      // Kick off animation loop
      requestAnimationFrame(render);
    })();
  </script>
</body>
