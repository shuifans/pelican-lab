<script>
(function () {
  'use strict';
  const NS = 'http://www.w3.org/2000/svg';
  const $ = id => document.getElementById(id);
  const set = (el, attr, value) => el.setAttribute(attr, value);

  // 生成辐条
  document.querySelectorAll('.spokes').forEach(g => {
    const cx = +g.dataset.cx, cy = +g.dataset.cy, r = +g.dataset.r, n = 12;
    for (let i = 0; i < n; i++) {
      const a = i * 2 * Math.PI / n;
      const line = document.createElementNS(NS, 'line');
      set(line, 'x1', cx + Math.cos(a) * 6);
      set(line, 'y1', cy + Math.sin(a) * 6);
      set(line, 'x2', cx + Math.cos(a) * r);
      set(line, 'y2', cy + Math.sin(a) * r);
      g.appendChild(line);
    }
  });

  // 几何常量
  const C = { x: 450, y: 398 };       // 五通 / 牙盘中心
  const CRANK = 34;                   // 曲柄长度
  const HIP = { x: 424, y: 282 };     // 髋关节（相对车身）
  const LEG = 82;                     // 大腿 / 小腿等长
  const TAU = Math.PI * 2;

  const layers = [
    { el: $('cloudLayer'),    tile: 960, v: 13  },
    { el: $('hillFarLayer'),  tile: 960, v: 26  },
    { el: $('hillNearLayer'), tile: 960, v: 48  },
    { el: $('bushLayer'),     tile: 480, v: 92  },
    { el: $('dashLayer'),     tile: 160, v: 210 }
  ];

  const winds = [$('wind1'), $('wind2'), $('wind3'), $('wind4')];
  const windLen = [38, 52, 30, 46];
  const birds = [$('bird1'), $('bird2')];

  // 两点 IK：等长两段，膝盖朝前（+x）
  function knee(H, P, L) {
    const dx = P.x - H.x, dy = P.y - H.y;
    const d = Math.max(1, Math.hypot(dx, dy));
    const clamped = Math.min(d, 2 * L - 0.01);
    const h = Math.sqrt(Math.max(0, L * L - (clamped / 2) ** 2));
    return {
      x: (H.x + P.x) / 2 + (dy / d) * h,
      y: (H.y + P.y) / 2 - (dx / d) * h
    };
  }

  function render(t) {
    // 曲柄慢、车轮快，符合真实传动比；用同一相位保证踩踏与前进同步
    const crankA = t * 2.5 + 0.5;
    const wheelDeg = (t * 5.25) * 180 / Math.PI;

    set($('wheelBack'), 'transform', `rotate(${wheelDeg} 330 398)`);
    set($('wheelFront'), 'transform', `rotate(${wheelDeg} 610 398)`);
    set($('crankRot'), 'transform', `rotate(${crankA * 180 / Math.PI} 450 398)`);

    function foot(a) {
      return { x: C.x + Math.cos(a) * CRANK, y: C.y + Math.sin(a) * CRANK };
    }
    const pNear = foot(crankA);
    const pFar = foot(crankA + Math.PI);

    const kNear = knee(HIP, pNear, LEG);
    const kFar = knee(HIP, pFar, LEG);
    set($('legNear'), 'd', `M${HIP.x} ${HIP.y} L${kNear.x.toFixed(1)} ${kNear.y.toFixed(1)} L${pNear.x.toFixed(1)} ${pNear.y.toFixed(1)}`);
    set($('legFar'), 'd', `M${HIP.x} ${HIP.y} L${kFar.x.toFixed(1)} ${kFar.y.toFixed(1)} L${pFar.x.toFixed(1)} ${pFar.y.toFixed(1)}`);

    set($('footNear'), 'transform', `translate(${pNear.x.toFixed(1)} ${pNear.y.toFixed(1)})`);
    set($('footFar'), 'transform', `translate(${pFar.x.toFixed(1)} ${pFar.y.toFixed(1)})`);
    set($('pedalNear'), 'transform', `translate(${pNear.x.toFixed(1)} ${pNear.y.toFixed(1)})`);
    set($('pedalFar'), 'transform', `translate(${pFar.x.toFixed(1)} ${pFar.y.toFixed(1)})`);

    // 每踩两下起伏一次，身体轻微前倾摆动
    const bob = 2.2 * Math.sin(crankA * 2);
    set($('rider'), 'transform', `translate(0 ${bob.toFixed(2)})`);
    set($('pelican'), 'transform', `rotate(${(0.7 * Math.sin(crankA)).toFixed(2)} 424 282)`);
    set($('billGroup'), 'transform', `rotate(${(1.6 * Math.sin(crankA * 0.8 + 0.6)).toFixed(2)} 550 152)`);

    // 围巾迎风飘动
    const s = Math.sin(crankA * 1.3) * 8;
    set($('scarf'), 'd',
      `M500 190 C476 182 452 ${(194 + s).toFixed(1)} 414 ${(200 + s * 1.2).toFixed(1)} ` +
      `L422 ${(214 + s * 1.2).toFixed(1)} C456 ${(208 + s).toFixed(1)} 482 202 504 202 Z`);

    // 视差背景
    layers.forEach(L => {
      const off = (t * L.v) % L.tile;
      set(L.el, 'transform', `translate(${-off.toFixed(2)} 0)`);
    });

    // 风线：从骑手身后向左飘出并淡出
    winds.forEach((w, i) => {
      const p = (t * 190 + i * 88) % 340;
      const x = 396 - p;
      set(w, 'x1', x.toFixed(1));
      set(w, 'x2', (x + windLen[i]).toFixed(1));
      set(w, 'opacity', (0.55 * (1 - p / 340)).toFixed(2));
    });

    // 远处的小鸟
    birds.forEach((b, i) => {
      const x = 1000 - ((t * 42 + i * 300) % 1250);
      const y = 118 + i * 40 + Math.sin(t * 2 + i) * 3;
      set(b, 'transform', `translate(${x.toFixed(1)} ${y.toFixed(1)})`);
    });
  }

  let t = 0;
  let last = performance.now();
  let playing = !window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  let speed = 1;

  const playBtn = $('playBtn');
  const speedRange = $('speedRange');
  const speedLabel = $('speedLabel');

  function syncButton() { playBtn.textContent = playing ? '暂停' : '播放'; }
  function toggle() { playing = !playing; syncButton(); }
  syncButton();

  playBtn.addEventListener('click', toggle);
  $('replayBtn').addEventListener('click', () => { t = 0; render(t); });
  speedRange.addEventListener('input', () => {
    speed = +speedRange.value;
    speedLabel.textContent = speed.toFixed(1) + '×';
  });
  document.addEventListener('keydown', e => {
    if (e.code === 'Space' && e.target !== speedRange) { e.preventDefault(); toggle(); }
  });

  render(0);
  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    if (playing) t += dt * speed;
    render(t);
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
