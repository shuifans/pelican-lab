  <script>
    const scene = document.querySelector('.scene');
    const playButton = document.getElementById('play-button');
    const playLabel = document.getElementById('play-label');
    const speedButton = document.getElementById('speed-button');
    const speedLabel = document.getElementById('speed-label');
    const pelican = document.getElementById('pelican');
    const shadow = document.getElementById('bike-shadow');
    const nearLeg = document.getElementById('near-leg');
    const farLeg = document.getElementById('far-leg');
    const nearFoot = document.getElementById('near-foot');
    const farFoot = document.getElementById('far-foot');
    const toes = document.getElementById('near-toes');
    const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');

    const crank = { x: 570, y: 505, radius: 34 };
    const speeds = [1, 1.5, 0.65];
    let speedIndex = 0;
    let speed = speeds[speedIndex];
    let running = !reducedMotion.matches;
    let phase = -0.5;
    let lastTime = performance.now();

    function legPath(hip, foot, upper, lower) {
      const dx = foot.x - hip.x;
      const dy = foot.y - hip.y;
      const distance = Math.max(1, Math.min(Math.hypot(dx, dy), upper + lower - 0.1));
      const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
      const height = Math.sqrt(Math.max(0, upper * upper - along * along));
      const ux = dx / Math.hypot(dx, dy);
      const uy = dy / Math.hypot(dx, dy);
      const knee = {
        x: hip.x + along * ux + height * uy,
        y: hip.y + along * uy - height * ux
      };
      return `M${hip.x.toFixed(1)} ${hip.y.toFixed(1)} L${knee.x.toFixed(1)} ${knee.y.toFixed(1)} L${foot.x.toFixed(1)} ${foot.y.toFixed(1)}`;
    }

    function render() {
      const bob = Math.sin(phase * 2) * 2.2;
      const tilt = Math.sin(phase) * 0.45;
      const nearPedal = {
        x: crank.x + Math.cos(phase) * crank.radius,
        y: crank.y + Math.sin(phase) * crank.radius
      };
      const farPedal = {
        x: crank.x - Math.cos(phase) * crank.radius,
        y: crank.y - Math.sin(phase) * crank.radius
      };

      pelican.setAttribute('transform', `translate(0 ${bob.toFixed(2)}) rotate(${tilt.toFixed(2)} 550 345)`);
      shadow.setAttribute('rx', (247 + Math.sin(phase * 2) * 2).toFixed(1));
      nearLeg.setAttribute('d', legPath({ x: 570, y: 345 + bob }, nearPedal, 88, 94));
      farLeg.setAttribute('d', legPath({ x: 545, y: 345 + bob }, farPedal, 88, 94));
      nearFoot.setAttribute('d', `M${nearPedal.x - 10} ${nearPedal.y} h31`);
      farFoot.setAttribute('d', `M${farPedal.x - 10} ${farPedal.y} h31`);
      toes.setAttribute('d', `M${nearPedal.x + 10} ${nearPedal.y + 5} h15m-7-6 9 1`);
    }

    function updatePlayback() {
      scene.classList.toggle('paused', !running);
      playButton.setAttribute('aria-pressed', String(!running));
      playButton.setAttribute('aria-label', running ? '暂停动画' : '播放动画');
      playLabel.textContent = running ? '暂停' : '播放';
    }

    function tick(now) {
      const delta = Math.min((now - lastTime) / 1000, 0.05);
      lastTime = now;
      if (running) {
        phase += delta * speed * Math.PI * 2 / 1.15;
        render();
      }
      requestAnimationFrame(tick);
    }

    playButton.addEventListener('click', () => {
      running = !running;
      updatePlayback();
    });

    speedButton.addEventListener('click', () => {
      speedIndex = (speedIndex + 1) % speeds.length;
      speed = speeds[speedIndex];
      speedLabel.textContent = `速度 ${speed}×`;
      document.documentElement.style.setProperty('--wheel-time', `${1.15 / speed}s`);
      document.documentElement.style.setProperty('--road-time', `${2.4 / speed}s`);
      document.documentElement.style.setProperty('--cloud-time', `${55 / speed}s`);
      document.documentElement.style.setProperty('--wave-time', `${9 / speed}s`);
    });

    document.addEventListener('keydown', (event) => {
      if (event.code === 'Space' && event.target === document.body) {
        event.preventDefault();
        running = !running;
        updatePlayback();
      }
    });

    reducedMotion.addEventListener('change', (event) => {
      if (event.matches) {
        running = false;
        updatePlayback();
      }
    });

    updatePlayback();
    render();
    requestAnimationFrame(tick);
  </script>
</body>
</html>
