<script>
(() => {
  const $ = id => document.getElementById(id);
  const BB = { x: 110, y: 12 }, CRANK = 32, L1 = 80, L2 = 100, PERIOD = 1.6;
  const torso = $('torso'), ring = $('ring');
  const legs = [
    { phase: Math.PI, crank: $('farCrank'),  thighO: $('farThighO'),  thigh: $('farThigh'),  shinO: $('farShinO'),  shin: $('farShin'),  foot: $('farFoot') },
    { phase: 0,       crank: $('nearCrank'), thighO: $('nearThighO'), thigh: $('nearThigh'), shinO: $('nearShinO'), shin: $('nearShin'), foot: $('nearFoot') }
  ];

  function solveKnee(hip, tx, ty) {
    const dx = tx - hip.x, dy = ty - hip.y;
    const d = Math.min(L1 + L2 - 0.01, Math.max(Math.abs(L1 - L2) + 0.01, Math.hypot(dx, dy)));
    const a = Math.acos((L1 * L1 + d * d - L2 * L2) / (2 * L1 * d));
    const k = Math.atan2(dy, dx) - a;
    return { x: hip.x + L1 * Math.cos(k), y: hip.y + L1 * Math.sin(k) };
  }

  function draw(t) {
    const ang = t / PERIOD * Math.PI * 2;
    const bob = Math.sin(ang * 2) * 1.8;
    torso.setAttribute('transform', `translate(0 ${bob.toFixed(2)})`);
    ring.setAttribute('transform', `rotate(${(ang * 180 / Math.PI % 360).toFixed(1)} 110 12)`);

    for (const l of legs) {
      const a = ang + l.phase;
      const px = BB.x + CRANK * Math.cos(a);
      const py = BB.y + CRANK * Math.sin(a);
      const hip = { x: 98, y: -120 + bob };
      const tx = px, ty = py - 8;
      const k = solveKnee(hip, tx, ty);
      const sx = k.x + (tx - k.x) / L2 * 7;
      const sy = k.y + (ty - k.y) / L2 * 7;

      const thighD = `M${hip.x} ${hip.y.toFixed(1)}L${k.x.toFixed(1)} ${k.y.toFixed(1)}`;
      const shinD = `M${sx.toFixed(1)} ${sy.toFixed(1)}L${tx.toFixed(1)} ${ty.toFixed(1)}`;
      l.thighO.setAttribute('d', thighD);
      l.thigh.setAttribute('d', thighD);
      l.shinO.setAttribute('d', shinD);
      l.shin.setAttribute('d', shinD);
      l.crank.setAttribute('x2', px.toFixed(1));
      l.crank.setAttribute('y2', py.toFixed(1));
      l.foot.setAttribute('transform', `translate(${px.toFixed(1)} ${py.toFixed(1)})`);
    }
  }

  if (matchMedia('(prefers-reduced-motion: reduce)').matches) { draw(0); return; }
  const start = performance.now();
  (function loop(now) {
    draw((now - start) / 1000);
    requestAnimationFrame(loop);
  })(start);
})();
</script>
