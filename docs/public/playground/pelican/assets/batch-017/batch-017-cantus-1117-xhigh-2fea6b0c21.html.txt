<script>
const $ = id => document.getElementById(id);
const BB = { x: 385, y: 335 };
const CRANK = 30, LEG = 84, WHEEL_R = 70;
const WHEEL_SPEED = 4.5;
const CRANK_SPEED = WHEEL_SPEED / 1.6;
const ROAD_SPEED = WHEEL_SPEED * WHEEL_R;

const wrap = (x, span) => ((x % span) + span) % span;

// 波数取整数，保证每 800px 首尾无缝衔接
function hillPath(base, waves) {
  let d = 'M0 400';
  for (let x = 0; x <= 1600; x += 20) {
    let y = base;
    for (const [amp, k, phase] of waves) y += amp * Math.sin(2 * Math.PI * k * x / 800 + phase);
    d += ` L${x} ${y.toFixed(1)}`;
  }
  return d + ' L1600 400 Z';
}
$('hillsFar').setAttribute('d', hillPath(300, [[26, 1, 0], [12, 3, 1.3]]));
$('hillsNear').setAttribute('d', hillPath(340, [[18, 2, 2], [8, 5, 0.5]]));

const scenery = sel => [...document.querySelectorAll(sel)].map(el => ({
  el, x: +el.dataset.x, y: +el.dataset.y, s: +el.dataset.s
}));
const clouds = scenery('#clouds use');
const trees = scenery('#trees use');
const headParts = document.querySelectorAll('.headMove');
const neckParts = document.querySelectorAll('.neck');

function setLeg(leg, foot, hip, ankle) {
  const dx = ankle.x - hip.x, dy = ankle.y - hip.y;
  const dist = Math.hypot(dx, dy);
  const half = Math.min(dist, LEG * 2 - 1) / 2;
  const h = Math.sqrt(LEG * LEG - half * half);
  const ux = dx / dist, uy = dy / dist;
  const kx = hip.x + ux * half + uy * h;
  const ky = hip.y + uy * half - ux * h;
  leg.setAttribute('points', `${hip.x},${hip.y} ${kx},${ky} ${ankle.x},${ankle.y}`);
  foot.setAttribute('transform', `translate(${ankle.x},${ankle.y})`);
}

function setSide(name, angle, hip) {
  const px = BB.x + CRANK * Math.cos(angle);
  const py = BB.y + CRANK * Math.sin(angle);
  const crank = $('crank' + name);
  crank.setAttribute('x2', px);
  crank.setAttribute('y2', py);
  $('pedal' + name).setAttribute('transform', `translate(${px},${py})`);
  setLeg($('leg' + name), $('foot' + name), hip, { x: px - 4, y: py - 9 });
}

function frame(ms) {
  const t = ms / 1000;
  const crank = t * CRANK_SPEED;
  const bob = Math.sin(crank * 2) * 1.5;

  $('spokes').setAttribute('transform', `rotate(${t * WHEEL_SPEED * 180 / Math.PI})`);
  $('chain').setAttribute('stroke-dashoffset', crank * 15);
  setSide('Far', crank + Math.PI, { x: 368, y: 197 + bob });
  setSide('Near', crank, { x: 362, y: 198 + bob });

  $('pelican').setAttribute('transform', `translate(0,${bob})`);
  const hx = Math.sin(crank * 2 + 0.8) * 2;
  const hy = Math.cos(crank * 2 + 0.8) * 1.5;
  headParts.forEach(el => el.setAttribute('transform', `translate(${hx},${hy})`));
  const neck = `M396 150 C430 128 392 108 ${426 + hx} ${82 + hy}`;
  neckParts.forEach(el => el.setAttribute('d', neck));
  const sag = 136 + Math.sin(t * 7) * 3;
  $('pouch').setAttribute('d', `M446 88 L562 100 Q530 ${sag} 480 ${sag - 8} Q452 118 446 88 Z`);

  $('dashes').setAttribute('stroke-dashoffset', wrap(t * ROAD_SPEED, 100));
  $('hillsFar').setAttribute('transform', `translate(${-wrap(t * 25, 800)},0)`);
  $('hillsNear').setAttribute('transform', `translate(${-wrap(t * 70, 800)},0)`);
  for (const c of clouds) {
    c.el.setAttribute('transform', `translate(${wrap(c.x - t * 14 * c.s, 1100) - 150},${c.y}) scale(${c.s})`);
  }
  for (const tr of trees) {
    tr.el.setAttribute('transform', `translate(${wrap(tr.x - t * ROAD_SPEED * 0.9, 1300) - 150},377) scale(${tr.s})`);
  }

  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
</script>
