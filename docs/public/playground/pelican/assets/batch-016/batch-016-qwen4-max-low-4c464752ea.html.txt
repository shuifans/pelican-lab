  { t:0.70, skyTop:'#472b63', skyMid:'#e0526e', skyBot:'#ffb35c', seaA:'#7a4988', seaB:'#4e3160', sand:'#d99a6c', road:'#4c4258', palm:'#3a2f52', cloud:'#ffc9a3', sun:'#ff8c5a', amb:'#5e2a4a', ambOp:0.10, stars:0.15 },
  { t:1.00, skyTop:'#0b1035', skyMid:'#1b2a5e', skyBot:'#2e3f77', seaA:'#1a2c58', seaB:'#0e1a3c', sand:'#4a4668', road:'#2c2a3e', palm:'#1c2138', cloud:'#3a4a78', sun:'#f2ecd4', amb:'#0a1030', ambOp:0.22, stars:1 }
];
function env(t){
  var i=0;
  while(i<KEYS.length-2 && t>KEYS[i+1].t) i++;
  var a=KEYS[i], b=KEYS[i+1], f=clamp((t-a.t)/(b.t-a.t),0,1), o={};
  for(var k in a){
    if(k==='t') continue;
    if(typeof a[k]==='number') o[k]=a[k]+(b[k]-a[k])*f;
    else o[k]=mix(a[k],b[k],f);
  }
  return o;
}

var GROUND=560, REARX=470, FRONTX=750, BBX=590, BBY=522, CRANKLEN=34;

function ik(hx,hy,tx,ty){
  var dx=tx-hx, dy=ty-hy, d=Math.sqrt(dx*dx+dy*dy);
  d=clamp(d, Math.abs(L1-L2)+1, L1+L2-1);
  var a=Math.atan2(dy,dx);
  var cos=(L1*L1+d*d-L2*L2)/(2*L1*d);
  var ka=a-Math.acos(clamp(cos,-1,1));
  return [hx+L1*Math.cos(ka), hy+L1*Math.sin(ka)];
}

/* ---------- 音效 ---------- */
var actx=null;
function ding(){
  if(!soundOn) return;
  try{
    if(!actx) actx=new (window.AudioContext||window.webkitAudioContext)();
    if(actx.state==='suspended') actx.resume();
    var t0=actx.currentTime;
    [1568,2093].forEach(function(f,idx){
      var o=actx.createOscillator(), g=actx.createGain();
      o.type='sine'; o.frequency.value=f;
      g.gain.setValueAtTime(idx===0?0.28:0.14,t0);
      g.gain.exponentialRampToValueAtTime(0.0001,t0+0.9);
      o.connect(g); g.connect(actx.destination);
      o.start(t0); o.stop(t0+1);
    });
  }catch(e){}
}

/* ---------- 主循环 ---------- */
function frame(ts){
  if(lastTs===null) lastTs=ts;
  var dt=clamp((ts-lastTs)/1000,0,0.05);
  lastTs=ts;
  if(!paused){
    time+=dt;
    var px=S.speed*6;                 /* px/s */
    dist+=px*dt;
    crank+=px*dt*0.021*(S.cadence/100);
    if(bellT>0) bellT=Math.max(0,bellT-dt*1.4);
  }
  render();
  hudAcc+=dt;
  if(hudAcc>0.15){ hudAcc=0; updateHud(); }
  requestAnimationFrame(frame);
}
