            osc.frequency.setValueAtTime(idx === 0 ? 1568 : 2093, now + delay);
            osc2.frequency.setValueAtTime(idx === 0 ? 3136 : 4186, now + delay);
            gain.gain.setValueAtTime(0.14, now + delay);
            gain.gain.exponentialRampToValueAtTime(0.0008, now + delay + 0.65);
            osc.connect(gain);
            osc2.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start(now + delay);
            osc2.start(now + delay);
            osc.stop(now + delay + 0.66);
            osc2.stop(now + delay + 0.66);
          });
        } catch (_) {}
      }

      function playPopSound(freqStart, freqEnd, dur = 0.16) {
        try {
          ensureAudio();
          if (!audioCtx) return;
          const now = audioCtx.currentTime;
          const osc = audioCtx.createOscillator();
          const gain = audioCtx.createGain();
          osc.type = 'sine';
          osc.frequency.setValueAtTime(freqStart, now);
          osc.frequency.exponentialRampToValueAtTime(freqEnd, now + dur);
          gain.gain.setValueAtTime(0.11, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + dur);
          osc.connect(gain);
          gain.connect(audioCtx.destination);
          osc.start(now);
          osc.stop(now + dur + 0.02);
        } catch (_) {}
      }

      // --- 2-Bone Inverse Kinematics Helper ---
      function solveIK(hx, hy, tx, ty, l1, l2, bendSign = -1) {
        const dx = tx - hx;
        const dy = ty - hy;
        let d = Math.hypot(dx, dy);
        const minD = Math.abs(l1 - l2) + 2;
        const maxD = l1 + l2 - 2;
        d = Math.max(minD, Math.min(maxD, d));

        const cosA = (l1 * l1 + d * d - l2 * l2) / (2 * l1 * d);
        const alpha = Math.acos(Math.max(-1, Math.min(1, cosA)));
        const base = Math.atan2(dy, dx);
        const angle1 = base + bendSign * alpha;

        return {
          jx: hx + l1 * Math.cos(angle1),
          jy: hy + l1 * Math.sin(angle1),
          angle1
        };
      }

      // Build Fluffy Feathered Thigh Cuff (Open top merges seamlessly into belly)
      function buildThighFluff(hx, hy, kx, ky, openTop = false) {
        const ux = (kx - hx);
        const uy = (ky - hy);
        const len = Math.hypot(ux, uy) || 1;
        const nx = -uy / len;
        const ny = ux / len;
        const ex = hx + ux * 0.48;
        const ey = hy + uy * 0.48;
        const contour = `M ${hx + nx * 15},${hy + ny * 15}
                Q ${(hx + ex) * 0.5 + nx * 20},${(hy + ey) * 0.5 + ny * 20} ${ex + nx * 10},${ey + ny * 10}
                Q ${ex + ux * 0.06 + nx * 4},${ey + uy * 0.06 + ny * 4} ${ex + ux * 0.04},${ey + uy * 0.04}
                Q ${ex + ux * 0.05 - nx * 5},${ey + uy * 0.05 - ny * 5} ${ex - nx * 10},${ey - ny * 10}
                Q ${(hx + ex) * 0.5 - nx * 17},${(hy + ey) * 0.5 - ny * 17} ${hx - nx * 13},${hy - ny * 13}`;
        return openTop ? contour : `${contour} Z`;
      }

            const vx = targetX - fishState.startX;

            }
          }

          // Spring-damper simulation for flexible gular pouch bounce
          const pedalDrive = Math.cos(bouncePhase) * 28 * speedMult;
          const springForce = -115 * pouchOffset - 9.5 * pouchVel + pedalDrive;
          pouchVel += springForce * dt;
          pouchOffset += pouchVel * dt;

          const totalDip = Math.max(-8, Math.min(34, pouchOffset + extraPouchStretch));
          els.gularPouch.setAttribute(
            'd',
            `M8,6 C22,${(52 + totalDip).toFixed(1)} 68,${(66 + totalDip * 1.15).toFixed(1)} 106,${(40 + totalDip * 0.65).toFixed(1)} C120,${(30 + totalDip * 0.3).toFixed(1)} 128,18 134,12 L12,4 Z`
          );
          els.pouchFold1.setAttribute(
            'd',
            `M22,16 C38,${(44 + totalDip * 0.85).toFixed(1)} 72,${(50 + totalDip * 0.9).toFixed(1)} 104,${(28 + totalDip * 0.45).toFixed(1)}`
          );
          els.pouchFold2.setAttribute(
            'd',
            `M36,14 C50,${(32 + totalDip * 0.55).toFixed(1)} 76,${(36 + totalDip * 0.6).toFixed(1)} 108,${(20 + totalDip * 0.25).toFixed(1)}`
          );

          els.upperBeakGroup.setAttribute('transform', `rotate(${(-mouthOpenDeg * 0.65).toFixed(2)} 14 -2)`);
          els.lowerJawGroup.setAttribute('transform', `rotate(${(mouthOpenDeg * 0.45).toFixed(2)} 10 5)`);
          els.mouthInterior.setAttribute('opacity', mouthOpenDeg > 2 ? '1' : '0');

          const headPitch = -2.2 * Math.sin(bouncePhase) - mouthOpenDeg * 0.15;
          els.pelicanHeadGroup.setAttribute(
            'transform',
            `translate(${hx.toFixed(2)}, ${hy.toFixed(2)}) rotate(${headPitch.toFixed(2)})`
          );

          // 7. Eye Blinking & Happy Squint
          nextBlinkAt -= dt;
          if (nextBlinkAt <= 0) {
            blinkTimer = 0.14;
            nextBlinkAt = 2.8 + Math.random() * 3.2;
          }
          if (blinkTimer > 0) blinkTimer -= dt;

          els.pelicanPupil.setAttribute('cx', pupilX.toFixed(2));
          els.pelicanPupil.setAttribute('cy', pupilY.toFixed(2));
          if (happySquint || blinkTimer > 0) {
            els.pelicanEyelid.setAttribute('visibility', 'visible');
            els.pelicanPupil.setAttribute('visibility', 'hidden');
          } else {
            els.pelicanEyelid.setAttribute('visibility', 'hidden');
            els.pelicanPupil.setAttribute('visibility', 'visible');
          }

          // 8. Wings Gripping Handlebars (World-space IK from moving shoulders to fixed grips)
          const flutter = Math.sin(elapsed * 13 * Math.max(0.6, speedMult)) * 2.2;
          updateWingPaths(462, 196 + bodyDy, 556, 198, true, flutter);
          updateWingPaths(470, 190 + bodyDy, 546, 189, false, flutter);

          // 9. Bicycle Bell Ring Vibration
          if (bellTimer > 0) {
            bellTimer = Math.max(0, bellTimer - dt);
            const shake = Math.sin(elapsed * 55) * 8 * (bellTimer / 0.75);
            els.bikeBell.setAttribute('transform', `translate(579, 187) rotate(${shake.toFixed(1)})`);
            els.bellWaves.setAttribute('opacity', Math.min(1, bellTimer * 2).toFixed(2));
          } else {
            els.bikeBell.setAttribute('transform', 'translate(579, 187)');
            els.bellWaves.setAttribute('opacity', '0');
          }
        }

        requestAnimationFrame(renderFrame);
      }

      requestAnimationFrame(renderFrame);
    })();
