<script>
  const byId = id => document.getElementById(id);
  const svgNamespace = 'http://www.w3.org/2000/svg';
  function add(parent, tag, attributes) {
    const element = document.createElementNS(svgNamespace, tag);
    for (const [name, value] of Object.entries(attributes)) element.setAttribute(name, value);
    parent.appendChild(element);
    return element;
  }
  const spokes = document.querySelector('.spokes');
  for (let i = 0; i < 12; i++) {
    add(spokes, 'path', { d: 'M-69.5 0h139', transform: 'rotate(' + i * 15 + ')' });
  }
  add(spokes, 'path', { d: 'M0-69v6', stroke: '#294749', 'stroke-width': 3 });
  function makeLeg(id, far) {
    const group = byId(id);
    const outline = add(group, 'path', {fill:'none', stroke:'#2a4546', 'stroke-width':11});
    const leg = add(group, 'path', {fill:'none', stroke:far ? '#ba7951' : '#e8a05b', 'stroke-width':5.5});
    const crank = add(group, 'path', {fill:'none', stroke:far ? '#58706b' : '#334c4c', 'stroke-width':6});
    const foot = add(group, 'g', {});
    add(foot, 'path', {d:'M-13 1h32', stroke:'#29474a', 'stroke-width':7});
    add(foot, 'path', {d:'M-8-13q2 6 9 6l13 1 6 4q-7 3-16 0-7 3-15-1z', fill:far ? '#d88f54' : '#eda757', stroke:'#2a4546', 'stroke-width':2.8});
    return {outline, leg, crank, foot};
  }
  const nearLeg = makeLeg('near-leg', false);
  const farLeg = makeLeg('far-leg', true);
  function pedal(parts, angle, far, bob) {
    const x = 555 + 32 * Math.cos(angle);
    const y = 521 + 32 * Math.sin(angle);
    const hipX = far ? 520 : 503;
    const hipY = 373 + bob;
    const ankleX = x - 8, ankleY = y - 11;
    // Solve a two-link leg so that each webbed foot stays on its pedal.
    const dx = ankleX - hipX, dy = ankleY - hipY;
    const distance = Math.hypot(dx, dy);
    const bend = Math.sqrt(Math.max(0, 94 * 94 - distance * distance / 4));
    const kneeX = (ankleX + hipX) / 2 + dy / distance * bend;
    const kneeY = (ankleY + hipY) / 2 - dx / distance * bend;
    const d = 'M' + hipX + ' ' + hipY + 'L' + kneeX + ' ' + kneeY + ' ' + ankleX + ' ' + ankleY;
    parts.outline.setAttribute('d', d);
    parts.leg.setAttribute('d', d);
    parts.crank.setAttribute('d', 'M555 521L' + x + ' ' + y);
    parts.foot.setAttribute('transform', 'translate(' + x + ' ' + y + ')');
  }
  function shift(id, x, y = 0) {
    byId(id).setAttribute('transform', 'translate(' + x + ' ' + y + ')');
  }
  function draw(time) {
    const angle = time * 3.3 + .55;
    const bob = 1.7 * Math.sin(angle * 2);
    const distance = time * 155;
    shift('distant', -(time * 8 % 1200));
    shift('ripples', -(time * 15 % 228));
    shift('verge', -(time * 84 % 1200));
    shift('road', -(distance % 210));
    shift('birds', 12 * Math.sin(time * .19), 3 * Math.sin(time * .55));
    shift('pelican', 0, bob);
    const rotation = 'rotate(' + (distance / 85.5 * 180 / Math.PI % 360) + ')';
    byId('rear-wheel').setAttribute('transform', rotation);
    byId('front-wheel').setAttribute('transform', rotation);
    pedal(farLeg, angle + Math.PI, true, bob);
    pedal(nearLeg, angle, false, bob);
    const flutter = Math.sin(time * 6) * 4;
    byId('scarf-tail').setAttribute('d', 'M546 231q-38-9-63 0-26 9-53 ' + (-1+flutter) + 'l14 15-13 8q35 6 61 ' + (-7-flutter) + ' 28-13 55-4z');
  }
  const motion = matchMedia('(prefers-reduced-motion: reduce)');
  let running = !motion.matches, elapsed = 0, previous = null, frame = 0;
  function tick(now) {
    if (previous !== null) elapsed += (now - previous) / 1000;
    previous = now;
    draw(elapsed);
    frame = requestAnimationFrame(tick);
  }
  function sync() {
    cancelAnimationFrame(frame);
    previous = null;
    const label = running ? '暂停动画' : '播放动画';
    byId('toggle').setAttribute('aria-label', label);
    byId('toggle').title = label;
    byId('symbol').setAttribute('d', running ? 'M4 3h4v14H4zm8 0h4v14h-4z' : 'M5 2.5 17 10 5 17.5z');
    if (running && !document.hidden) frame = requestAnimationFrame(tick);
  }
  byId('toggle').addEventListener('click', () => { running = !running; sync(); });
  motion.addEventListener('change', () => { running = !motion.matches; sync(); });
  document.addEventListener('visibilitychange', sync);
  draw(0);
  sync();
</script>
