  <script>
    const element = (id) => document.getElementById(id);
    const toggle = element('toggle');
    const toggleIcon = element('toggle-icon');
    const toggleLabel = element('toggle-label');
    const motionPreference = matchMedia('(prefers-reduced-motion: reduce)');
    const wheels = [element('rear-wheel'), element('front-wheel')];
    const chainring = element('chainring');
    const pelican = element('pelican');
    const eye = element('eye');
    const gull = element('gull');
    const width = 1120;
    const speed = 465;
    const crankRate = 4.45;
    const crank = { x: 496, y: 480, radius: 29 };
    const layers = [
      [element('cloud-scroll'), 8],
      [element('wave-scroll'), 23],
      [element('post-scroll'), 100],
      [element('road-scroll'), speed],
      [element('grass-scroll'), 550],
    ];
    const legs = [
      {
        paths: [element('far-leg')], crankArm: element('far-crank'),
        pedal: element('far-pedal'), foot: element('far-foot'),
        hipX: 455, hipY: 327, phase: Math.PI,
      },
      {
        paths: [element('near-leg'), element('near-leg-outline')],
        crankArm: element('near-crank'), pedal: element('near-pedal'),
        foot: element('near-foot'), hipX: 452, hipY: 323, phase: 0,
      },
    ];

    let elapsed = 0;
    let previous;
    let frame;
    let playing = false;

    function positionLeg(leg, angle, bob) {
      const pedalX = crank.x + crank.radius * Math.cos(angle + leg.phase);
      const pedalY = crank.y + crank.radius * Math.sin(angle + leg.phase);
      const hipY = leg.hipY + bob;
      const ankleX = pedalX - 9;
      const ankleY = pedalY - 9;
      const dx = ankleX - leg.hipX;
      const dy = ankleY - hipY;
      const reach = Math.hypot(dx, dy);
      // Find the forward bend of two equal-length leg segments.
      const bend = Math.sqrt(Math.max(0, 94 ** 2 - (reach / 2) ** 2));
      const kneeX = leg.hipX + dx / 2 + bend * dy / reach;
      const kneeY = hipY + dy / 2 - bend * dx / reach;
      const shape = `M${leg.hipX} ${hipY} ${kneeX} ${kneeY} ${ankleX} ${ankleY}`;

      for (const path of leg.paths) path.setAttribute('d', shape);
      leg.crankArm.setAttribute('d', `M${crank.x} ${crank.y} ${pedalX} ${pedalY}`);
      leg.pedal.setAttribute('d', `M${pedalX - 15} ${pedalY}h32`);
      leg.foot.setAttribute('x', pedalX - 2);
      leg.foot.setAttribute('y', pedalY - 10);
    }

    function draw(time) {
      const angle = time * crankRate - 0.62;
      const bob = 2.3 * Math.sin(2 * angle);
      const wheelAngle = (time * speed / 101) * 180 / Math.PI % 360;
      for (const wheel of wheels) wheel.setAttribute('transform', `rotate(${wheelAngle})`);
      for (const [layer, velocity] of layers) {
        layer.setAttribute('transform', `translate(${-(time * velocity % width)} 0)`);
      }
      chainring.setAttribute('transform', `translate(${crank.x} ${crank.y}) rotate(${angle * 180 / Math.PI % 360})`);
      pelican.setAttribute('transform', `translate(0 ${bob})`);
      gull.setAttribute('transform', `translate(${12 * Math.sin(time * .6)} ${4 * Math.sin(time * 1.4)})`);
      const blink = Math.max(0.12, Math.min(1, Math.abs(time % 4.7 - 4.45) * 13));
      eye.setAttribute('transform', `translate(582 136) scale(1 ${blink})`);
      for (const leg of legs) positionLeg(leg, angle, bob);
    }

    function animate(timestamp) {
      if (previous !== undefined) elapsed += Math.min(timestamp - previous, 48) / 1000;
      previous = timestamp;
      draw(elapsed);
      frame = requestAnimationFrame(animate);
    }

    function setPlaying(value) {
      playing = value;
      previous = undefined;
      cancelAnimationFrame(frame);
      if (playing) frame = requestAnimationFrame(animate);
      toggleLabel.textContent = playing ? '暂停' : '播放';
      toggleIcon.setAttribute('d', playing ? 'M4 3h3v10H4zm5 0h3v10H9z' : 'M5 2.7 13 8l-8 5.3Z');
    }

    draw(0);
    toggle.addEventListener('click', () => setPlaying(!playing));
    motionPreference.addEventListener('change', (event) => setPlaying(!event.matches));
    toggle.hidden = false;
    setPlaying(!motionPreference.matches);
  </script>
