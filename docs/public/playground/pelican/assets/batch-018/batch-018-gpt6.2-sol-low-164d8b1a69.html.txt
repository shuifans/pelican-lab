<script>
const byId = id => document.getElementById(id)
const toggle = byId('toggle')
const motion = matchMedia('(prefers-reduced-motion: reduce)')
let playing = !motion.matches
let elapsed = 0
let previous = null
let frame

function draw() {
  const angle = elapsed * Math.PI / 900
  const degrees = angle * 180 / Math.PI
  const dx = Math.cos(angle) * 25
  const dy = Math.sin(angle) * 25
  const bob = Math.sin(angle * 2) * 1.4
  byId('rear').setAttribute('transform', `translate(277 344) rotate(${degrees})`)
  byId('front').setAttribute('transform', `translate(614 344) rotate(${degrees})`)
  byId('crank').setAttribute('transform', `translate(418 344) rotate(${degrees})`)
  byId('bird').setAttribute('transform', `rotate(${bob / 2} 550 184)`)
  byId('leg').setAttribute('d', `M388 ${210 + bob} Q424 224 ${447 + dx * .44} ${263 + dy * .55} L${418 + dx} ${341 + dy}`)
  byId('farLeg').setAttribute('d', `M392 196 Q412 221 ${431 - dx * .44} ${252 - dy * .55} L${418 - dx} ${344 - dy}`)
  byId('foot').setAttribute('transform', `translate(${418 + dx} ${341 + dy})`)
  byId('sand').setAttribute('patternTransform', `translate(${-(elapsed * .27 % 320)} 0)`)
  byId('cloud').setAttribute('patternTransform', `translate(${-(elapsed * .021 % 570)} 0)`)
}

function tick(now) {
  if (previous !== null) elapsed += Math.min(now - previous, 50)
  previous = now
  draw()
  frame = requestAnimationFrame(tick)
}

function updatePlayback() {
  cancelAnimationFrame(frame)
  previous = null
  toggle.textContent = playing ? '暂停动画' : '播放动画'
  if (playing) frame = requestAnimationFrame(tick)
}

toggle.addEventListener('click', () => {
  playing = !playing
  updatePlayback()
})
motion.addEventListener('change', event => {
  playing = !event.matches
  updatePlayback()
})
draw()
updatePlayback()
</script>
