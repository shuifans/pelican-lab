<script>
  const byId = (id) => document.getElementById(id);
  const rearWheel = byId('rear-wheel');
  const frontWheel = byId('front-wheel');
  const bird = byId('bird');
  const scarf = byId('scarf-tail');
  const eye = byId('eye');
  const eyeGlint = byId('eye-glint');
  const crank = byId('crank');
  const toggle = byId('toggle');
  const layers = [
    { element: byId('water'), speed: 13 },
    { element: byId('grass'), speed: 72 },
    { element: byId('road'), speed: 185 }
  ];
  const near = {
    hipX: 510, hipY: 260, knee: 55,
    leg: byId('leg-near'), outline: byId('leg-near-outline'),
    toes: [byId('foot-near'), byId('toe-near')], pedal: byId('pedal-near')
  };
  const far = {
    hipX: 491, hipY: 264, knee: 29,
    leg: byId('leg-far'), outline: byId('leg-far-outline'),
    toes: [byId('foot-far')], pedal: byId('pedal-far')
  };

  function positionLeg(side, angle, bob) {
    const x = 523 + 26 * Math.cos(angle);
    const y = 374 + 26 * Math.sin(angle);
    const kneeX = (side.hipX + x) / 2 + side.knee;
    const kneeY = (side.hipY + y) / 2 - 11;
    const shape = `M${side.hipX} ${side.hipY + bob} Q${kneeX} ${kneeY} ${x} ${y - 5}`;
    side.leg.setAttribute('d', shape);
    side.outline.setAttribute('d', shape);
    for (const toe of side.toes) toe.setAttribute('d', `M${x - 9} ${y - 5}q12 2 22-2`);
    side.pedal.setAttribute('d', `M${x - 10} ${y}h20`);
    return [x, y];
  }

  function draw(time) {
    const angle = time * 2.8 + Math.PI / 4;
    const rotation = (time * 185 / 81 * 180 / Math.PI) % 360;
    const bob = 2.2 * Math.sin(2 * angle);
    const ripple = 6 * Math.sin(time * 5);
    const trailingRipple = 8 * Math.cos(time * 5 + .9);
    for (const { element, speed } of layers) {
      element.setAttribute('transform', `translate(${-(time * speed % 1100)} 0)`);
    }
    rearWheel.setAttribute('transform', `rotate(${rotation})`);
    frontWheel.setAttribute('transform', `rotate(${rotation})`);
    bird.setAttribute('transform', `translate(0 ${bob})`);
    scarf.setAttribute('d', `M552 204Q505 ${185 + ripple} 456 199T372 ${191 + trailingRipple}l17 21-19 13q49 ${14 - trailingRipple} 91-5t92-1Z`);
    const blink = Math.max(.12, Math.min(1, Math.abs(time % 5.6 - 4.8) / .11));
    eye.setAttribute('ry', 7.2 * blink);
    eyeGlint.setAttribute('opacity', blink > .65 ? 1 : 0);
    const [nearX, nearY] = positionLeg(near, angle, bob);
    const [farX, farY] = positionLeg(far, angle + Math.PI, bob);
    crank.setAttribute('d', `M${farX} ${farY}L${nearX} ${nearY}`);
  }

  const motion = matchMedia('(prefers-reduced-motion: reduce)');
  let playing = !motion.matches;
  let elapsed = 0;
  let previous;
  let frame;

  function tick(now) {
    if (previous !== undefined) elapsed += Math.min(now - previous, 50) / 1000;
    previous = now;
    draw(elapsed);
    frame = requestAnimationFrame(tick);
  }

  function syncPlayback() {
    cancelAnimationFrame(frame);
    previous = undefined;
    byId('control-label').textContent = playing ? '暂停动画' : '播放动画';
    byId('control-icon').setAttribute('d', playing ? 'M3 2h3v12H3zm7 0h3v12h-3z' : 'M4 2l10 6-10 6z');
    if (playing) frame = requestAnimationFrame(tick);
  }

  toggle.addEventListener('click', () => {
    playing = !playing;
    syncPlayback();
  });
  motion.addEventListener('change', (event) => {
    playing = !event.matches;
    syncPlayback();
  });
  draw(0);
  toggle.hidden = false;
  syncPlayback();
</script>
