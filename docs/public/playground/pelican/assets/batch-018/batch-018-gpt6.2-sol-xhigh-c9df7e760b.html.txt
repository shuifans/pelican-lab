  <script>
    (() => {
      const byId = id => document.getElementById(id);
      const scene = byId('scene');
      const play = byId('play');
      const speed = byId('speed');
      const playLabel = byId('play-label');
      const speedValue = byId('speed-value');
      const compact = matchMedia('(max-width: 640px)');
      const reducedMotion = matchMedia('(prefers-reduced-motion: reduce)');
      const speeds = [1, 1.5, 0.6];
      const bike = { x: 558, y: 483, crank: 39, tire: 98 };
      const cycle = Math.PI * 2;
      const crankRate = cycle / 2.6;
      const wheelRate = cycle / 1.86;
      const moving = Object.fromEntries([
        'rear-spokes', 'front-spokes', 'chainring', 'pelican', 'scarf-tail',
        'near-crank', 'far-crank', 'crank-highlight', 'road', 'waves', 'clouds',
        'gulls', 'left-grass', 'right-grass'
      ].map(id => [id, byId(id)]));

      const legs = [
        { hip: [536, 307], lengths: [105, 111], phase: Math.PI, side: 'far' },
        { hip: [510, 303], lengths: [112, 116], phase: 0, side: 'near' }
      ].map(leg => ({
        ...leg,
        foot: byId(leg.side + '-foot'),
        strokes: [byId(leg.side + '-leg'), byId(leg.side + '-leg-edge')]
      }));

      let paused = reducedMotion.matches;
      let speedIndex = 0;
      let elapsed = 0;
      let previous = null;
      let frame = null;

      const point = (x, y) => x.toFixed(2) + ' ' + y.toFixed(2);
      const degrees = radians => radians * 180 / Math.PI;
      const move = (id, transform) => moving[id].setAttribute('transform', transform);

      function rotatePoint(x, y, angle, originX, originY) {
        x -= originX;
        y -= originY;
        return [
          originX + x * Math.cos(angle) - y * Math.sin(angle),
          originY + x * Math.sin(angle) + y * Math.cos(angle)
        ];
      }

      function drawLeg(leg, crankAngle, bodyAngle) {
        const angle = crankAngle + leg.phase;
        const pedalX = bike.x + bike.crank * Math.cos(angle);
        const pedalY = bike.y + bike.crank * Math.sin(angle);
        const pitch = 0.12 * Math.sin(angle + 0.6);
        const [footX, footY] = rotatePoint(pedalX - 12, pedalY - 9, pitch, pedalX, pedalY);
        const [hipX, hipY] = rotatePoint(...leg.hip, bodyAngle, 686, 300);
        const dx = footX - hipX;
        const dy = footY - hipY;
        const distance = Math.hypot(dx, dy);
        const [thigh, shin] = leg.lengths;
        const along = (thigh ** 2 - shin ** 2 + distance ** 2) / (2 * distance);
        const outward = Math.sqrt(Math.max(0, thigh ** 2 - along ** 2));
        const kneeX = hipX + (dx * along + dy * outward) / distance;
        const kneeY = hipY + (dy * along - dx * outward) / distance;
        const legPath = 'M' + point(hipX, hipY) + 'L' + point(kneeX, kneeY) + 'L' + point(footX, footY);
        const crankPath = 'M' + point(bike.x, bike.y) + 'L' + point(pedalX, pedalY);

        for (const stroke of leg.strokes) stroke.setAttribute('d', legPath);
        moving[leg.side + '-crank'].setAttribute('d', crankPath);
        if (leg.side === 'near') moving['crank-highlight'].setAttribute('d', crankPath);
        leg.foot.setAttribute('transform', 'translate(' + point(pedalX, pedalY) + ') rotate(' + degrees(pitch) + ')');
      }

      function draw(time) {
        const crank = time * crankRate;
        const wheel = degrees(time * wheelRate % cycle);
        const bob = 0.018 * Math.sin(crank * 2);
        const breeze = Math.sin(time * 3.1);

        move('rear-spokes', 'rotate(' + wheel + ')');
        move('front-spokes', 'rotate(' + (wheel + 22) + ')');
        move('chainring', 'rotate(' + degrees(crank % cycle) + ' 558 483)');
        move('pelican', 'rotate(' + degrees(bob) + ' 686 300)');
        move('scarf-tail', 'rotate(' + (5 * breeze) + ' 548 224)');
        move('road', 'translate(' + (-((time * wheelRate * bike.tire) % 560)) + ' 0)');
        move('waves', 'translate(' + (-(time * 17 % 600)) + ' 0)');
        move('clouds', 'translate(' + (-24 * Math.sin(time * 0.13)) + ' 0)');
        move('gulls', 'translate(' + point(38 * Math.sin(time * 0.17), 6 * Math.sin(time * 0.44)) + ')');
        move('left-grass', 'rotate(' + (2.5 * breeze) + ')');
        move('right-grass', 'rotate(' + (3 * Math.sin(time * 3.1 + 0.8)) + ')');
        for (const leg of legs) drawLeg(leg, crank, bob);
      }

      function tick(now) {
        if (previous !== null) elapsed += Math.min(now - previous, 50) * speeds[speedIndex] / 1000;
        previous = now;
        draw(elapsed);
        frame = requestAnimationFrame(tick);
      }

      function updatePlayback() {
        play.setAttribute('aria-pressed', String(paused));
        playLabel.textContent = paused ? '播放' : '暂停';
        if (paused || document.hidden) {
          if (frame !== null) cancelAnimationFrame(frame);
          previous = null;
          frame = null;
        } else if (frame === null) {
          frame = requestAnimationFrame(tick);
        }
      }

      function togglePlayback() {
        paused = !paused;
        updatePlayback();
      }

      function resizeScene() {
        scene.setAttribute('viewBox', compact.matches ? '232 38 626 560' : '0 0 1120 606');
      }

      play.addEventListener('click', togglePlayback);
      speed.addEventListener('click', () => {
        speedIndex = (speedIndex + 1) % speeds.length;
        speedValue.textContent = speeds[speedIndex] + '×';
        speed.setAttribute('aria-label', '速度 ' + speeds[speedIndex] + ' 倍，点击切换');
      });
      document.addEventListener('keydown', event => {
        if (event.code !== 'Space' || event.repeat || event.altKey || event.ctrlKey || event.metaKey ||
            event.target.closest('button, a, input, textarea, select, [contenteditable]')) return;
        event.preventDefault();
        togglePlayback();
      });
      document.addEventListener('visibilitychange', updatePlayback);
      compact.addEventListener('change', resizeScene);
      reducedMotion.addEventListener('change', () => {
        paused = reducedMotion.matches;
        updatePlayback();
      });

      resizeScene();
      draw(0);
      updatePlayback();
    })();
  </script>
