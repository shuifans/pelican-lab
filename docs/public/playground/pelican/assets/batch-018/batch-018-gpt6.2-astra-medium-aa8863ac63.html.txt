<script>
const get=id=>document.getElementById(id);
let running=!matchMedia('(prefers-reduced-motion: reduce)').matches, t=0, last=0;
function leg(angle,far,bob){
  const x=483+24*Math.cos(angle), y=392+24*Math.sin(angle);
  const hipX=far?444:453,hipY=267+bob;
  // Two linked segments keep each webbed foot on its pedal.
  const dx=x-hipX,dy=y-hipY,d=Math.hypot(dx,dy);
  const bend=Math.sqrt(Math.max(0,81*81-d*d/4));
  const kx=(hipX+x)/2+dy/d*bend, ky=(hipY+y)/2-dx/d*bend;
  get(far?'farleg':'leg').setAttribute('d',`M${hipX} ${hipY} L${kx} ${ky} ${x-5} ${y-5}`);
  get(far?'farfoot':'foot').setAttribute('d',`M${x-8} ${y-8}q6-3 12 2l15 6-8 1-5-1-3 1-12-2Z`);
  if(!far){get('crank').setAttribute('d',`M${966-x} ${784-y}L${x} ${y}`);get('pedal').setAttribute('d',`M${x-9} ${y+3}h22`)}
}
function draw(){
  const a=t*3.2,bob=1.8*Math.sin(2*a);
  for(const id of ['front','rear'])get(id).setAttribute('transform',`rotate(${t*124})`);
  get('bird').setAttribute('transform',`translate(0 ${bob})`);
  leg(a+Math.PI,true,bob);leg(a,false,bob);
  get('dashes').setAttribute('transform',`translate(${-(t*137%250)} 0)`);
  get('plants').setAttribute('transform',`translate(${-(t*95%1000)} 0)`);
  get('road').setAttribute('patternTransform',`translate(${-(t*137%160)} 0)`);
}
function frame(now){if(!running)return;if(last)t+=Math.min((now-last)/1000,.06);last=now;draw();raf=requestAnimationFrame(frame)}
let raf=0;
function sync(){cancelAnimationFrame(raf);last=0;get('toggle').textContent=running?'Ⅱ　暂停动画':'▶　播放动画';if(running)raf=requestAnimationFrame(frame)}
get('toggle').onclick=()=>{running=!running;sync()};draw();sync();
</script>
