<script>
(function () {
  const svg = document.getElementById('scene');
  const stage = document.getElementById('stage');
  const rearSpokes = document.getElementById('rearSpokes');
  const frontSpokes = document.getElementById('frontSpokes');
  const crank = document.getElementById('crank');
  const legBack = document.getElementById('legBack');
  const legFront = document.getElementById('legFront');
  const footBack = document.getElementById('footBack');
  const footFront = document.getElementById('footFront');
  const bird = document.getElementById('bird');
  const scarf = document.getElementById('scarf');
  const clouds = [...document.querySelectorAll('#clouds > g')];
  const dashes = document.getElementById('dashes');

  const BB = { x: 430, y: 392 };       // 中轴
  const R = 30;                        // 曲柄半径
  const HIP_BACK = { x: 396, y: 292 };
  const HIP_FRONT = { x: 418, y: 292 };
  const PERIOD = 1250;                 // 一次踩踏周期(ms)
  const cloudBase = clouds.map(c => parseFloat(c.getAttribute('transform').match(/translate\(([-\d.]+)/)[1]));

  let paused = false, pauseAt = 0, start = performance.now();

  function kneePoint(hip, foot, side) {
    const mx = (hip.x + foot.x) / 2, my = (hip.y + foot.y) / 2;
    const dx = foot.x - hip.x, dy = foot.y - hip.y;
    const len = Math.hypot(dx, dy) || 1;
    return { x: mx + (-dy / len) * 20 * side, y: my + (dx / len) * 20 * side };
  }

  function pedalPoint(a) {
    return { x: BB.x + Math.cos(a) * R, y: BB.y + Math.sin(a) * R };
  }

  function setLeg(path, footEl, hip, foot, side) {
    const k = kneePoint(hip, foot, side);
    path.setAttribute('d', `M ${hip.x} ${hip.y} L ${k.x.toFixed(1)} ${k.y.toFixed(1)} L ${foot.x.toFixed(1)} ${foot.y.toFixed(1)}`);
    footEl.setAttribute('cx', foot.x.toFixed(1));
    footEl.setAttribute('cy', foot.y.toFixed(1));
  }

  function frame(now) {
    if (!paused) {
      const t = now - start;
      const a = (t % PERIOD) / PERIOD * Math.PI * 2;
      const deg = a * 180 / Math.PI;

      // 车轮旋转（齿比约 2:1）
      const wheelDeg = deg * 2;
      rearSpokes.setAttribute('transform', `rotate(${wheelDeg.toFixed(1)} 250 392)`);
      frontSpokes.setAttribute('transform', `rotate(${wheelDeg.toFixed(1)} 610 392)`);

      // 曲柄与目标脚踏点保持同相位
      crank.setAttribute('transform', `rotate(${deg.toFixed(1)} ${BB.x} ${BB.y})`);
      const footA = pedalPoint(a);
      const footB = pedalPoint(a + Math.PI);
      setLeg(legBack, footBack, HIP_BACK, footB, -1);
      setLeg(legFront, footFront, HIP_FRONT, footA, 1);

      // 车身轻微起伏
      bird.setAttribute('transform', `translate(0 ${(Math.sin(a * 2) * 2.2).toFixed(2)})`);
      scarf.setAttribute('transform', `rotate(${(Math.sin(a * 3) * 6).toFixed(2)} 456 150)`);

      // 背景滚动
      clouds.forEach((c, i) => {
        const span = 1200;
        const x = ((cloudBase[i] - t * clouds[i].dataset.speed * 0.06) % span + span) % span - 120;
        const y = 90 + i * 25;
        c.setAttribute('transform', `translate(${x.toFixed(1)} ${y})`);
      });
      dashes.setAttribute('transform', `translate(${(-((t * 0.45) % 120)).toFixed(1)} 0)`);
    }
    requestAnimationFrame(frame);
  }

  stage.addEventListener('click', () => {
    paused = !paused;
    if (!paused) {
      const now = performance.now();
      start += now - pauseAt;
    } else {
      pauseAt = performance.now();
    }
  });

  requestAnimationFrame(frame);
})();
</script>
