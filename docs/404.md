<script setup>
import { onMounted } from 'vue'
import { withBase } from 'vitepress'

const knowledgeBase = 'https://shuifans.github.io/cloud-ai-knowledge'

onMounted(() => {
  const match = window.location.pathname.match(/\/(?:knowledge|cloud|ai|software|gaming|chronicle)(?:\/|$)/)
  if (!match || match.index === undefined) return
  const target = knowledgeBase + window.location.pathname.slice(match.index) + window.location.search + window.location.hash
  window.location.replace(target)
})
</script>

<div class="vp-doc" style="max-width: 760px; margin: 0 auto; padding: 32px 24px;">
  <h1>页面不存在</h1>
  <p>这个地址没有对应页面。知识内容已迁移到独立站点；若未自动跳转，可以从<a :href="knowledgeBase + '/'">云与 AI 技术知识库首页</a>进入。</p>
  <p>返回 <a :href="withBase('/')">Pelican Lab 首页</a>，或查看 <a :href="withBase('/benchmarks/pelican-2d/')">鹈鹕骑车 2D</a>。</p>
</div>
